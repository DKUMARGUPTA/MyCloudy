    91→interface ZipEntry {
    92→  name: string;
    93→  method: number;
    94→  data: Buffer;
    95→}
    96→
    97→/**
    98→ * node:zlib ZIP oracle — walks the local file headers IN ORDER and
    99→ * inflates every entry itself (fflate writes RAW deflate, hence
   100→ * inflateRawSync). Order is preserved, so "mimetype first" is testable.
   101→ */
   102→function readZipWithZlib(bytes: Uint8Array): ZipEntry[] {
   103→  const buf = Buffer.from(bytes);
   104→  const out: ZipEntry[] = [];
   105→  let off = 0;
   106→  while (off + 4 <= buf.length && buf.readUInt32LE(off) === 0x04034b50) {
   107→    const method = buf.readUInt16LE(off + 8);
   108→    const compSize = buf.readUInt32LE(off + 18);
   109→    const nameLen = buf.readUInt16LE(off + 26);
   110→    const extraLen = buf.readUInt16LE(off + 28);
   111→    const name = buf.slice(off + 30, off + 30 + nameLen).toString("utf8");
   112→    const dataStart = off + 30 + nameLen + extraLen;
   113→    const raw = buf.slice(dataStart, dataStart + compSize);
   114→    if (method === 0) {
   115→      out.push({ name, method, data: raw });
   116→    } else if (method === 8) {
   117→      out.push({ name, method, data: inflateRawSync(raw) });
   118→    } else {
   119→      throw new Error(`unsupported zip method ${method}`);
   120→    }
   121→    off = dataStart + compSize;
   122→  }
   123→  if (out.length === 0) throw new Error("no zip entries parsed");
   124→  return out;
   125→}

   132→/** Minimal XML well-formedness: balanced tags, no raw < in text. */
   133→function xmlWellFormed(xml: string): boolean {
   134→  const src = xml
   135→    .replace(/<\?[\s\S]*?\?>/g, "")
   136→    .replace(/<!--[\s\S]*?-->/g, "")
   137→    .replace(/<!DOCTYPE[^>]*>/gi, "");
   138→  const tagRe = /<(\/?)([A-Za-z_:][\w.:-]*)((?:\s+[\w.:-]+\s*=\s*"[^"]*")*)\s*(\/?)>/g;
   139→  const stack: string[] = [];
   140→  let lastIndex = 0;
   141→  let match: RegExpExecArray | null;
   142→  while ((match = tagRe.exec(src)) !== null) {
   143→    if (src.slice(lastIndex, match.index).includes("<")) return false;
   144→    lastIndex = tagRe.lastIndex;
   145→    const [full, close, name, , selfClose] = match;
   146→    if (selfClose === "/" || full.startsWith("<?") || full.startsWith("<!")) {
   147→      continue;
   148→    }
   149→    if (close === "/") {
   150→      if (stack.pop() !== name) return false;
   151→    } else {
   152→      stack.push(name);
   153→    }
   154→  }
   155→  if (src.slice(lastIndex).includes("<")) return false;
   156→  return stack.length === 0;
   157→}

   506→async function extractFixture(bytes: Uint8Array): Promise<PdfTextItemInput[][]> {
   507→  const task = getDocument({ data: bytes.slice(), useSystemFonts: true });
   508→  const pdf = await task.promise;
   509→  const pages: PdfTextItemInput[][] = [];
   510→  for (let number = 1; number <= pdf.numPages; number += 1) {
   511→    const page = await pdf.getPage(number);
   512→    try {
   513→      const content = await page.getTextContent();
   514→      pages.push(content.items as PdfTextItemInput[]);
   515→    } finally {
   516→      page.cleanup();
   517→    }
   518→  }
   519→  await pdf.cleanup();
   520→  await task.destroy();
   521→  return pages;
   522→}

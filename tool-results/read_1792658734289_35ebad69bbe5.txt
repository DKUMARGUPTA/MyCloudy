     1→/**
     2→ * Publisher to PDF model (RB4-e) — pure model. No DOM, no pdf-lib, no
     3→ * worker APIs: the same parsePub feeds the surface's live file summary
     4→ * AND the worker's render path, so the preview numbers can never
     5→ * disagree with the output.
     6→ *
     7→ * HONEST REALITY (documented everywhere): Microsoft Publisher's .pub is
     8→ * an UNDOCUMENTED OLE2 Compound File Binary container. No open-source
     9→ * complete parser exists. This tool is therefore an honest TEXT SALVAGE:
    10→ *
    11→ *  1. A clean-room OLE2/CFB container reader (header signature
    12→ *     D0 CF 11 E0 A1 B1 1A E1, 512-byte sectors for v3, DIFAT → FAT →
    13→ *     sector chains, 128-byte directory entries with UTF-16LE names,
    14→ *     mini-FAT + the root's mini stream for streams < 4096 bytes — both
    15→ *     paths implemented, with a normal-FAT fallback for nonstandard
    16→ *     writers).
    17→ *  2. Stream triage: every stream is scanned for long printable
    18→ *     UTF-16LE runs (the .pub convention stores story text as UTF-16);
    19→ *     single-byte ASCII runs are the fallback. Paragraph breaks come
    20→ *     from CR/LF inside runs and from NUL/non-text gaps between runs.
    21→ *  3. Reflow: paragraphs → blocks with an HONEST heading heuristic —
    22→ *     font/size/style data is not reliably extractable from the binary
    23→ *     story records, so headings are guessed from ALL-CAPS / Title-Case
    24→ *     short lines (and short Devanagari lines without a danda) at a
    25→ *     chosen base size.
    26→ *
    27→ * What this means for the user: the TEXT of the publication is carried
    28→ * over as clean, selectable, re-flowed PDF text; complex layouts,
    29→ * images, shapes, columns and exact page composition are NOT (mirrored
    30→ * in config.ts + the surface). For full fidelity the better path is
    31→ * exporting a PDF from Publisher itself — the FAQ says so.
    32→ */
    33→

    81→const OLE_MAGIC = [0xd0, 0xcf, 0x11, 0xe0, 0xa1, 0xb1, 0x1a, 0xe1];

    88→/** true when the bytes start with the OLE2 compound-file signature. */
    89→export function isOle2File(head: Uint8Array): boolean {
    90→  if (head.length < 8) return false;
    91→  for (let i = 0; i < 8; i += 1) {
    92→    if (head[i] !== OLE_MAGIC[i]) return false;
    93→  }
    94→  return true;
    95→}

   101→const WINANSI_EXTRA = new Set([
   102→  0x20ac, 0x201a, 0x0192, 0x201e, 0x2026, 0x2020, 0x2021, 0x02c6, 0x2030,
   103→  0x0160, 0x2039, 0x0152, 0x017d, 0x2018, 0x2019, 0x201c, 0x201d, 0x2022,
   104→  0x2013, 0x2014, 0x02dc, 0x2122, 0x0161, 0x203a, 0x0153, 0x017e, 0x0178,
   105→]);

   107→/** true when every char is encodable in WinAnsi (CP1252) minus controls. */
   108→export function isWinAnsiText(s: string): boolean {
   109→  for (let i = 0; i < s.length; i += 1) {
   110→    const c = s.charCodeAt(i);
   111→    if (c >= 0x20 && c <= 0x7e) continue;
   112→    if (c >= 0xa0 && c <= 0xff) continue;
   113→    if (WINANSI_EXTRA.has(c)) continue;
   114→    return false;
   115→  }
   116→  return true;
   117→}

   119→/* ------------------------------------------------------------------ */
   120→/* Text-run scanning (UTF-16LE primary, ASCII fallback)                */
   121→/* ------------------------------------------------------------------ */
   122→
   123→/**
   124→ * Code units that participate in a UTF-16LE text run: printable ASCII,
   125→ * Latin-1 supplement, WinAnsi punctuation, Devanagari, space/tab and
   126→ * CR/LF (paragraph breaks INSIDE a story — they are kept in the run and
   127→ * split later). NUL never qualifies, so NUL-separated records break the
   128→ * run exactly the way .pub story chunks do.
   129→ */
   130→function isTextCodeUnit(cp: number): boolean {
   131→  if (cp === 0x09 || cp === 0x0a || cp === 0x0d || cp === 0x20) return true;
   132→  if (cp >= 0x21 && cp <= 0x7e) return true;
   133→  if (cp >= 0xa0 && cp <= 0xff) return true;
   134→  if (cp >= 0x0900 && cp <= 0x097f) return true;
   135→  if (WINANSI_EXTRA.has(cp)) return true;
   136→  if (cp === 0x20b9 || cp === 0x200b || cp === 0x200c || cp === 0x200d) {
   137→    return true; // ₹ + zero-width joiners Devanagari needs
   138→  }
   139→  return false;
   140→}
   141→
   142→function isTextAsciiByte(b: number): boolean {
   143→  if (b === 0x09 || b === 0x0a || b === 0x0d) return true;
   144→  return b >= 0x20 && b <= 0x7e;
   145→}

   147→interface TextRun {
   148→  offset: number;
   149→  text: string;
   150→}
   151→
   152→function unitsToString(units: number[]): string {
   153→  let out = "";
   154→  for (let i = 0; i < units.length; i += 1024) {
   155→    out += String.fromCharCode(...units.slice(i, i + 1024));
   156→  }
   157→  return out;
   158→}
   159→
   160→/** Scans one byte alignment (0 or 1) for printable UTF-16LE runs. */
   161→function scanUtf16Runs(bytes: Uint8Array, align: 0 | 1): TextRun[] {
   162→  const runs: TextRun[] = [];
   163→  const n = bytes.length;
   164→  let i = align;
   165→  while (i + 1 < n) {
   166→    if (!isTextCodeUnit(bytes[i]! | (bytes[i + 1]! << 8))) {
   167→      i += 2;
   168→      continue;
   169→    }
   170→    const offset = i;
   171→    const units: number[] = [];
   172→    while (i + 1 < n) {
   173→      const cp = bytes[i]! | (bytes[i + 1]! << 8);
   174→      if (!isTextCodeUnit(cp)) break;
   175→      units.push(cp);
   176→      i += 2;
   177→    }
   178→    if (units.length >= MIN_RUN_CHARS) {
   179→      runs.push({ offset, text: unitsToString(units) });
   180→    }
   181→  }
   182→  return runs;
   183→}
   184→
   185→/** Scans for printable single-byte ASCII runs (the honest fallback). */
   186→function scanAsciiRuns(bytes: Uint8Array): TextRun[] {
   187→  const runs: TextRun[] = [];
   188→  const n = bytes.length;
   189→  let i = 0;
   190→  while (i < n) {
   191→    if (!isTextAsciiByte(bytes[i]!)) {
   192→      i += 1;
   193→      continue;
   194→    }
   195→    const offset = i;
   196→    let end = i;
   197→    while (end < n && isTextAsciiByte(bytes[end]!)) end += 1;
   198→    if (end - offset >= MIN_RUN_CHARS) {
   199→      const units: number[] = [];
   200→      for (let k = offset; k < end; k += 1) units.push(bytes[k]!);
   201→      runs.push({ offset, text: unitsToString(units) });
   202→    }
   203→    i = end;
   204→  }
   205→  return runs;
   206→}
   207→
   208→/** CR/LF splits + whitespace normalization + empty-drop. */
   209→function runsToParagraphs(runs: TextRun[]): string[] {
   210→  const out: string[] = [];
   211→  for (const run of runs) {
   212→    for (const piece of run.text.split(/[\r\n]+/)) {
   213→      const t = piece.replace(/[\t ]+/g, " ").trim();
   214→      if (t.length > 0) out.push(t);
   215→    }
   216→  }
   217→  return out;
   218→}
   219→
   220→function runsCharCount(runs: TextRun[]): number {
   221→  let n = 0;
   222→  for (const r of runs) n += r.text.length;
   223→  return n;
   224→}

   226→/** One stream's honest verdict: which encoding won, and its paragraphs. */
   227→export interface StreamText {
   228→  paragraphs: string[];
   229→  chars: number;
   230→  encoding: "utf16" | "ascii" | "none";
   231→}
   232→
   233→export function extractStreamText(data: Uint8Array): StreamText {
   234→  const u16 = [...scanUtf16Runs(data, 0), ...scanUtf16Runs(data, 1)].sort(
   235→    (a, b) => a.offset - b.offset
   236→  );
   237→  const u16Chars = runsCharCount(u16);
   238→  if (u16Chars >= MIN_STREAM_CHARS) {
   239→    return {
   240→      paragraphs: runsToParagraphs(u16),
   241→      chars: u16Chars,
   242→      encoding: "utf16",
   243→    };
   244→  }
   245→  const a8 = scanAsciiRuns(data);
   246→  const a8Chars = runsCharCount(a8);
   247→  if (a8Chars >= MIN_STREAM_CHARS) {
   248→    return {
   249→      paragraphs: runsToParagraphs(a8),
   250→      chars: a8Chars,
   251→      encoding: "ascii",
   252→    };
   253→  }
   254→  return { paragraphs: [], chars: Math.max(u16Chars, a8Chars), encoding: "none" };
   255→}

   284→interface ChainResult {
   285→  sectors: number[];
   286→  /** true when the walk ended on ENDOFCHAIN. */
   287→  ended: boolean;
   288→}
   289→
   290→function chainSectors(
   291→  fat: number[],
   292→  start: number,
   293→  totalSectors: number,
   294→  limit: number
   295→): ChainResult {
   296→  const sectors: number[] = [];
   297→  const seen = new Set<number>();
   298→  let s = start;
   299→  let guard = 0;
   300→  while (guard <= limit + 1) {
   301→    guard += 1;
   302→    if (s === ENDOFCHAIN) return { sectors, ended: true };
   303→    if (
   304→      s === FREESECT ||
   305→      s === FATSECT ||
   306→      s === DIFSECT ||
   307→      s < 0 ||
   308→      s >= totalSectors ||
   309→      seen.has(s)
   310→    ) {
   311→      return { sectors, ended: false };
   312→    }
   313→    seen.add(s);
   314→    sectors.push(s);
   315→    s = s < fat.length ? fat[s]! : FREESECT;
   316→  }
   317→  return { sectors, ended: false };
   318→}
   319→
   320→function concatSectors(
   321→  bytes: Uint8Array,
   322→  sectors: number[],
   323→  sectorSize: number,
   324→  size: number
   325→): Uint8Array {
   326→  const out = new Uint8Array(Math.min(size, sectors.length * sectorSize));
   327→  let filled = 0;
   328→  for (const s of sectors) {
   329→    if (filled >= out.length) break;
   330→    const off = HEADER_LEN + s * sectorSize;
   331→    const avail = Math.max(0, Math.min(sectorSize, bytes.length - off));
   332→    if (avail <= 0) break;
   333→    const take = Math.min(avail, out.length - filled);
   334→    out.set(bytes.subarray(off, off + take), filled);
   335→    filled += take;
   336→  }
   337→  return out;
   338→}

   340→/** Throws the honest non-CFB classification (zip / text / generic). */
   341→function throwNotCfb(bytes: Uint8Array): never {
   342→  if (
   343→    bytes.length >= 4 &&
   344→    bytes[0] === 0x50 &&
   345→    bytes[1] === 0x4b &&
   346→    (bytes[2] === 0x03 || bytes[2] === 0x05 || bytes[2] === 0x07)
   347→  ) {
   348→    throw new PubError("errZipFile", "zip signature");
   349→  }
   350→  if (bytes.length >= 16) {
   351→    const window = Math.min(bytes.length, 2048);
   352→    let texty = 0;
   353→    let binaryish = 0;
   354→    for (let i = 0; i < window; i += 1) {
   355→      const b = bytes[i]!;
   356→      if (
   357→        b === 0x09 ||
   358→        b === 0x0a ||
   359→        b === 0x0d ||
   360→        (b >= 0x20 && b <= 0x7e) ||
   361→        b >= 0x80
   362→      ) {
   363→        texty += 1; // UTF-8 multibyte counts as text
   364→      } else {
   365→        binaryish += 1;
   366→      }
   367→    }
   368→    if (binaryish === 0 && texty / window >= 0.7) {
   369→      throw new PubError("errTextFile", "plain text");
   370→    }
   371→  }
   372→  throw new PubError("errNotPub", "no OLE2 signature");
   373→}
   374→
   375→/**
   376→ * Clean-room OLE2/CFB container reader: header → DIFAT → FAT → directory
   377→ * (UTF-16LE names) → mini-FAT + root mini stream → stream bytes. Never
   378→ * returns a half-parsed file without saying so (streamFailures); only
   379→ * structural breakage (bad header, unreadable FAT, broken directory
   380→ * chain, no root) throws errCorrupt.
   381→ */
   382→export function parseCfb(bytes: Uint8Array): CfbFile {
   383→  if (bytes.length < HEADER_LEN || !isOle2File(bytes)) {
   384→    throwNotCfb(bytes);
   385→  }
   386→  const dv = new DataView(bytes.buffer, bytes.byteOffset, bytes.byteLength);
   387→
   388→  const major = dv.getUint16(0x1a, true);
   389→  const byteOrder = dv.getUint16(0x1c, true);
   390→  const sectorShift = dv.getUint16(0x1e, true);
   391→  let miniShift = dv.getUint16(0x20, true);
   392→
   393→  if (byteOrder !== 0xfffe) {
   394→    throw new PubError("errCorrupt", "bad byte order");
   395→  }
   396→  if (major === 3 ? sectorShift !== 9 : major !== 4 || sectorShift !== 12) {
   397→    throw new PubError("errCorrupt", `v${major} sector shift ${sectorShift}`);
   398→  }
   399→  if (miniShift < 6 || miniShift > sectorShift) miniShift = 6;
   400→  const sectorSize = 1 << sectorShift;
   401→  const miniSectorSize = 1 << miniShift;
   402→  const totalSectors = Math.floor((bytes.length - HEADER_LEN) / sectorSize);
   403→  if (totalSectors < 1) {
   404→    throw new PubError("errCorrupt", "no sectors after the header");
   405→  }
   406→
   407→  /* ---------- DIFAT (FAT sector list) ---------- */
   408→  const fatSectorNums: number[] = [];
   409→  for (let i = 0; i < 109; i += 1) {
   410→    const v = dv.getUint32(0x4c + i * 4, true);
   411→    if (v === FREESECT || v === ENDOFCHAIN) break;
   412→    fatSectorNums.push(v);
   413→    if (fatSectorNums.length > MAX_FAT_SECTORS) {
   414→      throw new PubError("errCorrupt", "DIFAT overflow");
   415→    }
   416→  }
   417→  let nextDifat = dv.getUint32(0x44, true);
   418→  let difatGuard = 0;
   419→  while (nextDifat !== ENDOFCHAIN && nextDifat !== FREESECT) {
   420→    difatGuard += 1;
   421→    if (
   422→      difatGuard > MAX_DIFAT_SECTORS ||
   423→      nextDifat < 0 ||
   424→      nextDifat >= totalSectors
   425→    ) {
   426→      throw new PubError("errCorrupt", "DIFAT chain broken");
   427→    }
   428→    const off = HEADER_LEN + nextDifat * sectorSize;
   429→    if (off + sectorSize > bytes.length) {
   430→      throw new PubError("errCorrupt", "truncated DIFAT sector");
   431→    }
   432→    const per = sectorSize / 4 - 1;
   433→    for (let i = 0; i < per; i += 1) {
   434→      const v = dv.getUint32(off + i * 4, true);
   435→      if (v === FREESECT || v === ENDOFCHAIN) continue;
   436→      fatSectorNums.push(v);
   437→      if (fatSectorNums.length > MAX_FAT_SECTORS) {
   438→        throw new PubError("errCorrupt", "DIFAT overflow");
   439→      }
   440→    }
   441→    nextDifat = dv.getUint32(off + per * 4, true);
   442→  }
   443→
   444→  /* ---------- FAT (flat next-sector table) ---------- */
   445→  const fat: number[] = [];
   446→  for (const fs of fatSectorNums) {
   447→    if (fs < 0 || fs >= totalSectors) {
   448→      throw new PubError("errCorrupt", "FAT sector out of range");
   449→    }
   450→    const off = HEADER_LEN + fs * sectorSize;
   451→    const avail = Math.max(0, Math.min(sectorSize, bytes.length - off));
   452→    const entries = Math.floor(avail / 4);
   453→    for (let i = 0; i < entries; i += 1) {
   454→      fat.push(dv.getUint32(off + i * 4, true));
   455→    }
   456→  }
   457→  if (fat.length === 0) {
   458→    throw new PubError("errCorrupt", "no FAT");
   459→  }
   460→
   461→  /* ---------- Directory chain ---------- */
   462→  const firstDir = dv.getUint32(0x30, true);
   463→  const dirWalk = chainSectors(fat, firstDir, totalSectors, totalSectors + 2);
   464→  if (dirWalk.sectors.length === 0 || !dirWalk.ended) {
   465→    throw new PubError("errCorrupt", "directory chain broken");
   466→  }
   467→  const dirBytes = concatSectors(
   468→    bytes,
   469→    dirWalk.sectors,
   470→    sectorSize,
   471→    dirWalk.sectors.length * sectorSize
   472→  );
   473→  const entryCount = Math.floor(dirBytes.length / 128);
   474→  if (entryCount < 1) {
   475→    throw new PubError("errCorrupt", "no directory entries");
   476→  }
   477→
   478→  /* ---------- Entry structs ---------- */
   479→  interface RawEntry {
   480→    id: number;
   481→    type: number;
   482→    name: string;
   483→    start: number;
   484→    size: number;
   485→  }
   486→  const raw: RawEntry[] = [];
   487→  for (let id = 0; id < entryCount; id += 1) {
   488→    const off = id * 128;
   489→    const type = dirBytes[off + 0x42]!;
   490→    if (type !== 1 && type !== 2 && type !== 5) continue; // 0/unallocated/reserved
   491→    const nameLen = dirBytes[off + 0x40]! | (dirBytes[off + 0x41]! << 8);
   492→    let name = "";
   493→    if (nameLen >= 4 && nameLen <= 64) {
   494→      const chars = Math.floor((nameLen - 2) / 2); // minus the null terminator
   495→      for (let c = 0; c < chars; c += 1) {
   496→        name += String.fromCharCode(
   497→          dirBytes[off + c * 2]! | (dirBytes[off + c * 2 + 1]! << 8)
   498→        );
   499→      }
   500→    }
   501→    let size = 0;
   502→    for (let b = 0; b < 8; b += 1) {
   503→      size += dirBytes[off + 0x78 + b]! * 2 ** (8 * b);
   504→    }
   505→    raw.push({
   506→      id,
   507→      type,
   508→      name,
   509→      start:
   510→        dirBytes[off + 0x74]! +
   511→        dirBytes[off + 0x75]! * 0x100 +
   512→        dirBytes[off + 0x76]! * 0x10000 +
   513→        dirBytes[off + 0x77]! * 0x1000000,
   514→      size,
   515→    });
   516→  }
   517→
   518→  const root = raw.find((e) => e.type === 5);
   519→  if (!root) {
   520→    throw new PubError("errCorrupt", "no root entry");
   521→  }
   522→
   523→  /* ---------- Mini FAT + the root's mini stream ---------- */
   524→  const miniCutoff = dv.getUint32(0x38, true) || 4096;
   525→  const firstMiniFat = dv.getUint32(0x3c, true);
   526→  const miniFat: number[] = [];
   527→  if (firstMiniFat !== ENDOFCHAIN && firstMiniFat !== FREESECT) {
   528→    const walk = chainSectors(fat, firstMiniFat, totalSectors, totalSectors + 2);
   529→    for (const s of walk.sectors) {
   530→      const off = HEADER_LEN + s * sectorSize;
   531→      const avail = Math.max(0, Math.min(sectorSize, bytes.length - off));
   532→      const entries = Math.floor(avail / 4);
   533→      for (let i = 0; i < entries; i += 1) {
   534→        miniFat.push(dv.getUint32(off + i * 4, true));
   535→      }
   536→    }
   537→  }
   538→  const rootSize = Math.min(root.size, bytes.length);
   539→  let miniStream: Uint8Array = new Uint8Array(0);
   540→  if (rootSize > 0) {
   541→    const walk = chainSectors(fat, root.start, totalSectors, totalSectors + 2);
   542→    miniStream = concatSectors(bytes, walk.sectors, sectorSize, rootSize);
   543→  }
   544→
   545→  /** Mini-FAT read with the honest normal-FAT fallback for odd writers. */
   546→  const readStreamBytes = (
   547→    start: number,
   548→    size: number
   549→  ): { data: Uint8Array; complete: boolean; path: "mini" | "normal" } => {
   550→    if (size <= 0) {
   551→      return { data: new Uint8Array(0), complete: true, path: "normal" };
   552→    }
   553→    if (size < miniCutoff && miniStream.length > 0 && miniFat.length > 0) {
   554→      const need = Math.ceil(size / miniSectorSize);
   555→      const sectors: number[] = [];
   556→      const seen = new Set<number>();
   557→      let s = start;
   558→      let complete = false;
   559→      const cap = Math.min(need + 1, miniFat.length + 2);
   560→      let guard = 0;
   561→      while (guard <= cap + 1) {
   562→        guard += 1;
   563→        if (s === ENDOFCHAIN) {
   564→          complete = sectors.length >= need;
   565→          break;
   566→        }
   567→        if (s === FREESECT || s < 0 || seen.has(s)) break;
   568→        const off = s * miniSectorSize;
   569→        if (off + miniSectorSize > miniStream.length + miniSectorSize) break;
   570→        seen.add(s);
   571→        sectors.push(s);
   572→        if (sectors.length >= need) {
   573→          complete = true;
   574→          break;
   575→        }
   576→        s = miniFat[s] ?? FREESECT;
   577→      }
   578→      if (sectors.length > 0) {
   579→        const data = new Uint8Array(
   580→          Math.min(size, sectors.length * miniSectorSize)
   581→        );
   582→        let filled = 0;
   583→        for (const s2 of sectors) {
   584→          if (filled >= data.length) break;
   585→          const off = s2 * miniSectorSize;
   586→          const avail = Math.max(
   587→            0,
   588→            Math.min(miniSectorSize, miniStream.length - off)
   589→          );
   590→          if (avail <= 0) break;
   591→          const take = Math.min(avail, data.length - filled);
   592→          data.set(miniStream.subarray(off, off + take), filled);
   593→          filled += take;
   594→        }
   595→        return { data, complete, path: "mini" };
   596→      }
   597→      // zero mini sectors gathered — try the normal FAT (nonstandard but
   598→      // seen in the wild); falls through to the normal read below.
   599→    }
   600→    const need = Math.ceil(size / sectorSize);
   601→    const walk = chainSectors(
   602→      fat,
   603→      start,
   604→      totalSectors,
   605→      Math.min(need, totalSectors + 2)
   606→    );
   607→    const data = concatSectors(bytes, walk.sectors, sectorSize, size);
   608→    return {
   609→      data,
   610→      complete: walk.sectors.length >= need,
   611→      path: "normal",
   612→    };
   613→  };
   614→
   615→  /* ---------- Streams (directory id order — see file header) ---------- */
   616→  const streams: CfbStreamEntry[] = [];
   617→  let streamFailures = 0;
   618→  for (const e of raw) {
   619→    if (e.type !== 2) continue;
   620→    const size = Math.min(e.size, bytes.length);
   621→    const { data, complete, path } = readStreamBytes(e.start, size);
   622→    if (!complete && size > 0) streamFailures += 1;
   623→    streams.push({
   624→      id: e.id,
   625→      name: e.name || `stream-${e.id}`,
   626→      size,
   627→      data,
   628→      complete,
   629→      path,
   630→    });
   631→  }
   632→
   633→  return {
   634→    sectorSize,
   635→    miniSectorSize,
   636→    rootName: root.name,
   637→    streams,
   638→    streamFailures,
   639→  };
   640→}

   685→/* ------------------------------------------------------------------ */
   686→/* hindiNorm — the pinned lost-session lesson                           */
   687→/* ------------------------------------------------------------------ */
   688→
   689→const DEVA_NORM_RE = /[\u0900-\u097F]/;
   690→
   691→/**
   692→ * Noto subset CMaps can inject stray ASCII quotes (") between
   693→ * Devanagari words. Strip `"` — but ONLY from strings that actually
   694→ * contain Devanagari; English quotes are left completely alone.
   695→ */
   696→export function hindiNorm(s: string): string {
   697→  return DEVA_NORM_RE.test(s) ? s.split('"').join("") : s;
   698→}

   700→/* ------------------------------------------------------------------ */
   701→/* Heading heuristic (honest: sizes are NOT reliably in the binary)     */
   702→/* ------------------------------------------------------------------ */
   703→
   704→const DEVA_RE = /[\u0900-\u097F]/;
   705→
   706→/**
   707→ * 1 = ALL-CAPS short line, 2 = Title-Case short line (or a short
   708→ * Devanagari line without a danda — Hindi sentences end with ।).
   709→ * undefined = body text. The first heading-like paragraph of the whole
   710→ * document is upgraded to a title by the assembler.
   711→ */
   712→export function classifyHeading(text: string): 1 | 2 | undefined {
   713→  const t = text.trim();
   714→  if (t.length === 0 || t.length > 64) return undefined;
   715→  if (t.endsWith(".")) return undefined; // a sentence, not a heading
   716→  const words = t.split(/\s+/).filter(Boolean);
   717→  if (words.length === 0) return undefined;
   718→  if (/[A-Z]/.test(t) && t === t.toUpperCase()) return 1; // ALL CAPS
   719→  const titleCased = words.every((w) => /^[A-Za-z]/.test(w) ? /^[A-Z]/.test(w) : true);
   720→  if (titleCased && words.length >= 2 && !/^[a-z]/.test(t)) return 2;
   721→  if (DEVA_RE.test(t) && !/[।?!]/.test(t) && words.length >= 2) return 2;
   722→  return undefined;
   723→}

   729→export interface PubPara {
   730→  text: string;
   731→  /** 0 = title (first heading-like para), 1 = ALL-CAPS, 2 = Title-Case. */
   732→  heading?: 0 | 1 | 2;
   733→}

   735→export interface ParsedPub {
   736→  paras: PubPara[];
   737→  paraCount: number;
   738→  wordCount: number;
   739→  charCount: number;
   740→  streamsScanned: number;
   741→  streamsUsed: number;
   742→  streamsFailed: number;
   743→  /** text-bearing streams (name/chars/encoding/path) for the preview. */
   744→  stories: PubStoryStream[];
   745→  /** true when any hard cap was hit — the surface shows an honest note. */
   746→  truncated: boolean;
   747→}
   748→
   749→/**
   750→ * Full honest pipeline: container → stories → reflowed paragraphs.
   751→ * Throws PubError for every non-CFB / structural case; a container
   752→ * with no usable text returns an EMPTY document (the caller decides
   753→ * errNoText vs errStreamRead from streamsFailed).
   754→ */
   755→export function parsePub(bytes: Uint8Array): ParsedPub {
   756→  const cfb = parseCfb(bytes);
   757→  const stories = extractStories(cfb);
   758→
   759→  const paras: PubPara[] = [];
   760→  let chars = 0;
   761→  let truncated = false;
   762→  outer: for (const st of stories.streams) {
   763→    for (const text of st.paragraphs) {
   764→      if (paras.length >= MAX_PARAGRAPHS || chars >= MAX_TOTAL_CHARS) {
   765→        truncated = true;
   766→        break outer;
   767→      }
   768→      const t = hindiNorm(text.slice(0, MAX_PARA_TEXT)).trim();
   769→      if (t.length === 0) continue;
   770→      // consecutive-identical noise (story thumbnails) collapses
   771→      if (paras.length > 0 && paras[paras.length - 1]!.text === t) continue;
   772→      paras.push({ text: t });
   773→      chars += t.length;
   774→    }
   775→  }
   776→
   777→  for (let i = 0; i < paras.length; i += 1) {
   778→    const h = classifyHeading(paras[i]!.text);
   779→    paras[i]!.heading = i === 0 && h !== undefined ? 0 : h;
   780→  }
   781→
   782→  let words = 0;
   783→  for (const p of paras) {
   784→    words += p.text.split(/\s+/).filter(Boolean).length;
   785→  }
   786→  return {
   787→    paras,
   788→    paraCount: paras.length,
   789→    wordCount: words,
   790→    charCount: chars,
   791→    streamsScanned: stories.streamsScanned,
   792→    streamsUsed: stories.streams.length,
   793→    streamsFailed: stories.streamsFailed,
   794→    stories: stories.streams,
   795→    truncated,
   796→  };
   797→}

   807→/* ------------------------------------------------------------------ */
   808→/* Page estimate (same geometry constants as render.ts)                  */
   809→/* ------------------------------------------------------------------ */
   810→
   811→export const PAGE_SIZES: Record<"a4" | "letter", [number, number]> = {
   812→  a4: [595.28, 841.89],
   813→  letter: [612, 792],
   814→};
   815→
   816→export const MARGIN_PT: Record<"normal" | "narrow", number> = {
   817→  normal: 56,
   818→  narrow: 28,
   819→};
   820→
   821→export interface PubEstimateOpts {
   822→  pageSize: "a4" | "letter";
   823→  margin: "normal" | "narrow";
   824→  baseSize: number;
   825→}
   826→
   827→/** Heading size multipliers — MUST mirror render.ts exactly. */
   828→export const HEADING_SCALE: Record<0 | 1 | 2, number> = {
   829→  0: 1.9,
   830→  1: 1.5,
   831→  2: 1.22,
   832→};
   833→
   834→/** Honest ± page estimate from the same geometry the renderer uses. */
   835→export function estimatePages(doc: ParsedPub, opts: PubEstimateOpts): number {
   836→  if (doc.paras.length === 0) return 0;
   837→  const [pw, ph] = PAGE_SIZES[opts.pageSize];
   838→  const margin = MARGIN_PT[opts.margin];
   839→  const contentW = pw - margin * 2;
   840→  const contentH = ph - margin * 2;
   841→  const lhFactor = 1.43;
   842→  let totalHeight = 0;
   843→  for (const p of doc.paras) {
   844→    const size =
   845→      p.heading !== undefined
   846→        ? opts.baseSize * HEADING_SCALE[p.heading]
   847→        : opts.baseSize;
   848→    const lh = size * lhFactor;
   849→    const avgCharW = size * 0.5;
   850→    const charsPerLine = Math.max(20, Math.floor(contentW / avgCharW));
   851→    const lines = Math.max(1, Math.ceil(p.text.length / charsPerLine));
   852→    totalHeight += lines * lh + opts.baseSize * 0.3;
   853→  }
   854→  return Math.max(1, Math.ceil(totalHeight / contentH));
   855→}

   857→/* ------------------------------------------------------------------ */
   858→/* Output file name (house convention + localized fallback)             */
   859→/* ------------------------------------------------------------------ */
   860→
   861→function cleanName(s: string): string {
   862→  let out = "";
   863→  for (const ch of s) {
   864→    const c = ch.codePointAt(0) ?? 32;
   865→    out += c >= 0x20 && c !== 0x7f ? ch : "_";
   866→  }
   867→  return out;
   868→}

   870→/**
   871→ * "Newsletter.pub" → "Newsletter.pdf"; a nameless file falls back to
   872→ * the localized word ("document" / "दस्तावेज़"). Unicode survives;
   873→ * path separators and unsafe chars do not.
   874→ */
   875→export function outputPdfName(fileName: string, fileWord: string): string {
   876→  const base = cleanName(fileName.replace(/\.(pub|publisher)$/i, ""))
   877→    .trim()
   878→    .slice(0, 120);
   879→  const fallback = cleanName(fileWord).trim().slice(0, 60) || "document";
   880→  return `${base || fallback}.pdf`;
   881→}

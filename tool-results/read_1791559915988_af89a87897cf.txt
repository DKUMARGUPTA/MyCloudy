    64→/**
    65→ * Fill out PDF forms live surface (PRD §8.2 "Fill out"). Scans the AcroForm
    66→ * with pdf-lib (read-only, on-device), renders the current page with
    67→ * pdf.js and overlays a highlight box on every widget — click one to jump
    68→ * to its input. The field form below lists every field in reading order
    69→ * (Tab flows naturally), grouped visually by a page chip. Values live in
    70→ * the shell options as "f:<fileId>::<name>" keys so replacing the file
    71→ * can never leak answers into a different form (the worker only applies
    72→ * keys carrying the current file id).
    73→ */
    74→
    75→const PREVIEW_MAX_DIM = 560;
    76→const ZOOM_STEPS = [1, 1.5, 2] as const;
    77→/** Above this many fields the list scrolls instead of growing the page. */
    78→const LIST_SCROLL_LIMIT = 12;
    79→
    80→const FIELD_KEY_ID = "fkey";
    81→const fieldKey = (fileId: string, name: string): string =>
    82→  `f:${fileId}::${name}`;
    83→
    84→function originalString(field: FillFieldScan): string {
    85→  if (field.kind === "text") {
    86→    return typeof field.value === "string" ? field.value : "";
    87→  }
    88→  if (field.kind === "optionlist") return field.selected.join("\n");
    89→  return typeof field.value === "string" ? field.value : "";
    90→}
    91→
    92→function originalBool(field: FillFieldScan): boolean {
    93→  return field.kind === "checkbox" && field.value === true;
    94→}
    95→
    96→/** The value the input shows (shell override or the file's own value). */
    97→function currentValue(
    98→  field: FillFieldScan,
    99→  values: Record<string, string | number | boolean>,
   100→  fileId: string
   101→): string | boolean {
   102→  const raw = values[fieldKey(fileId, field.name)];
   103→  if (field.kind === "checkbox") {
   104→    return typeof raw === "boolean" ? raw : originalBool(field);
   105→  }
   106→  return typeof raw === "string" ? raw : originalString(field);
   107→}
   108→
   109→/** Did the user actually change this field (vs. the file's own value)? */
   110→function isChanged(
   111→  field: FillFieldScan,
   112→  values: Record<string, string | number | boolean>,
   113→  fileId: string
   114→): boolean {
   115→  const raw = values[fieldKey(fileId, field.name)];
   116→  if (raw === undefined) return false;
   117→  if (field.kind === "checkbox") {
   118→    return typeof raw === "boolean" && raw !== originalBool(field);
   119→  }
   120→  const text = typeof raw === "string" ? raw : String(raw);
   121→  return text !== originalString(field);
   122→}
   123→
   124→/** Does the field hold a meaningful answer right now? */
   125→function isFilled(
   126→  field: FillFieldScan,
   127→  values: Record<string, string | number | boolean>,
   128→  fileId: string
   129→): boolean {
   130→  const value = currentValue(field, values, fileId);
   131→  return typeof value === "boolean" ? value : value.trim() !== "";
   132→}
   133→
   134→function isDateField(field: FillFieldScan): boolean {
   135→  if (field.kind !== "text") return false;
   136→  const label = field.label.toLowerCase();
   137→  if (/(?:^|[^a-z])(date|dob)(?![a-z])/.test(label)) return true;
   138→  return label.includes("तिथि") || label.includes("जन्म");
   139→}
   140→
   141→/** dd/mm/yyyy (India norm) from a native date input's yyyy-mm-dd. */
   142→function isoToDateText(iso: string): string {
   143→  const match = /^(\d{4})-(\d{2})-(\d{2})$/.exec(iso);
   144→  if (!match) return "";
   145→  return `${match[3]}/${match[2]}/${match[1]}`;
   146→}

   149→  text: "fill.typeText",
   150→  checkbox: "fill.typeCheckbox",
   151→  radio: "fill.typeRadio",
   152→  dropdown: "fill.typeDropdown",
   153→  optionlist: "fill.typeOptionlist",
   154→};

   156→interface PageHighlight {
   157→  field: FillFieldScan;
   158→  index: number;
   159→  left: number;
   160→  top: number;
   161→  w: number;

   165→export function FillSurface({
   166→  files,
   167→  values,
   168→  onChange,
   169→  setFileOrder,
   170→  setRunBlocked,
   171→}: ToolSurfaceProps): React.JSX.Element | null {
   172→  const { t } = useI18n();
   173→  const { toast } = useToast();
   174→  const uid = useId();
   175→  const fileMeta = files[0];
   176→  const file = fileMeta?.file ?? null;
   177→  const fileId = fileMeta?.id ?? "";
   178→
   179→  // --- Page preview state (declared before the scan effect uses it) --
   180→  const [selectedPage, setSelectedPage] = useState(0);
   181→  const [zoom, setZoom] = useState<number>(1);
   182→  const [base, setBase] = useState<{
   183→    file: File;
   184→    pageIndex: number;
   185→    bitmap: ImageBitmap;
   186→    dispW: number;
   187→    dispH: number;
   188→  } | null>(null);
   189→  const [failure, setFailure] = useState<{
   190→    file: File;
   191→    pageIndex: number;
   192→  } | null>(null);
   193→  const [focusedField, setFocusedField] = useState<string | null>(null);
   194→
   195→  // --- Form scan (pdf-lib, read only) --------------------------------
   196→  const [scanState, setScanState] = useState<{
   197→    file: File;
   198→    scan: FormScan | null;
   199→    error: boolean;
   200→  } | null>(null);
   201→
   202→  useEffect(() => {
   203→    if (!file) return;
   204→    let cancelled = false;
   205→    void (async () => {
   206→      const bytes = await file.arrayBuffer();
   207→      const scan = await scanFormFields(bytes);
   208→      if (cancelled) return;
   209→      setScanState({ file, scan, error: false });
   210→      setSelectedPage((page) =>
   211→        Math.min(page, Math.max(0, scan.pageCount - 1))
   212→      );
   213→    })().catch(() => {
   214→      if (!cancelled) setScanState({ file, scan: null, error: true });
   215→    });
   216→    return () => {
   217→      cancelled = true;
   218→    };
   219→  }, [file]);
   220→
   221→  const active = scanState !== null && scanState.file === file ? scanState : null;
   222→  const scan = active?.scan ?? null;
   223→  const scanError = active?.error === true;
   224→  const fields = scan?.fields ?? [];
   225→  const pageCount = scan?.pageCount ?? 0;
   226→
   227→  // Stamp the current file id into the options once per file — the worker
   228→  // only applies "f:<fileId>::" keys, so answers can never leak into a
   229→  // different file after a replacement.
   230→  useEffect(() => {
   231→    if (fileId !== "") onChange(FIELD_KEY_ID, fileId);
   232→  }, [fileId, onChange]);
   233→
   234→  useEffect(() => {
   235→    if (!file || scan === null) return;
   236→    let cancelled = false;
   237→    void (async () => {
   238→      const [thumb, pixel] = await Promise.all([
   239→        renderPdfThumbnail(file, {
   240→          pageIndex: selectedPage,
   241→          maxDim: PREVIEW_MAX_DIM,
   242→        }),
   243→        getPdfPagePixelSize(file, 1, selectedPage),
   244→      ]);
   245→      if (cancelled) return;
   246→      const bitmap = await createImageBitmap(thumb.blob);
   247→      if (cancelled) return;
   248→      setFailure(null);
   249→      setBase({
   250→        file,
   251→        pageIndex: selectedPage,
   252→        bitmap,
   253→        dispW: pixel.width,
   254→        dispH: pixel.height,
   255→      });
   256→    })().catch(() => {
   257→      if (!cancelled) {
   258→        setFailure({ file, pageIndex: selectedPage });
   259→        setBase(null);
   260→      }
   261→    });
   262→    return () => {
   263→      cancelled = true;
   264→    };
   265→  }, [file, scan, selectedPage]);
   266→
   267→  const canvasRef = useRef<HTMLCanvasElement>(null);
   268→  useEffect(() => {
   269→    const canvas = canvasRef.current;
   270→    if (!canvas || !base) return;
   271→    const ctx = canvas.getContext("2d");
   272→    if (!ctx) return;
   273→    canvas.width = base.bitmap.width;
   274→    canvas.height = base.bitmap.height;
   275→    ctx.clearRect(0, 0, canvas.width, canvas.height);
   276→    ctx.drawImage(base.bitmap, 0, 0);
   277→  }, [base]);
   278→
   279→  // --- Auto-fill profile ----------------------------------------------
   280→  // Lazy initializer: one idempotent localStorage read (client only — the
   281→  // surface mounts after a file drop, never during SSR).
   282→  const [profile, setProfile] = useState<FillProfile>(() =>
   283→    typeof window === "undefined" ? { ...EMPTY_PROFILE } : loadFillProfile()
   284→  );
   285→  const [profileOpen, setProfileOpen] = useState(false);
   286→
   287→  const runAutoFill = useCallback((): void => {
   288→    if (!profileHasAny(profile)) {
   289→      setProfileOpen(true);
   290→      toast({ title: t("fill.autofillEmpty") });
   291→      return;
   292→    }
   293→    let count = 0;
   294→    for (const field of fields) {
   295→      if (field.kind !== "text" || field.readOnly) continue;
   296→      // Label-only matching (never the qualified name — path segments
   297→      // like "applicant." would false-match every field under them).
   298→      const value = matchProfileValue(field.label, profile);
   299→      if (value === "" || value === currentValue(field, values, fileId)) {
   300→        continue;
   301→      }
   302→      onChange(fieldKey(fileId, field.name), value);
   303→      count += 1;
   304→    }
   305→    toast({
   306→      title:
   307→        count > 0
   308→          ? t("fill.autofillDone", { count })
   309→          : t("fill.autofillNoMatch"),
   310→    });
   311→  }, [fields, fileId, profile, t, toast, onChange, values]);
   312→
   313→  // --- Derived state ---------------------------------------------------
   314→  const filledCount = fields.filter(
   315→    (field) => isChanged(field, values, fileId)
   316→  ).length;
   317→  const requiredLeft = fields.filter(
   318→    (field) =>
   319→      field.required && !field.readOnly && !isFilled(field, values, fileId)
   320→  ).length;
   321→
   322→  const blockedKey =
   323→    !file
   324→      ? null
   325→      : scanError
   326→        ? "fill.scanError"
   327→        : scan === null
   328→          ? "fill.ctaScanning"
   329→          : fields.length === 0
   330→            ? "fill.ctaNoFields"
   331→            : filledCount === 0
   332→              ? "fill.ctaNoChanges"
   333→              : null;
   334→  useEffect(() => {
   335→    setRunBlocked(blockedKey);
   336→  }, [blockedKey, setRunBlocked]);
   337→
   338→  const markedPages = useMemo(() => {
   339→    const set = new Set<number>();
   340→    for (const field of fields) {
   341→      for (const widget of field.widgets) set.add(widget.page);
   342→    }
   343→    return set;
   344→  }, [fields]);
   345→
   346→  const current =
   347→    base !== null && base.file === file && base.pageIndex === selectedPage
   348→      ? base
   349→      : null;
   350→  const failed =
   351→    failure !== null &&
   352→    failure.file === file &&
   353→    failure.pageIndex === selectedPage &&
   354→    current === null;
   355→
   356→  const pageInfo =
   357→    scan !== null && selectedPage < scan.pages.length
   358→      ? scan.pages[selectedPage]
   359→      : null;
   360→
   361→  const highlights = useMemo<PageHighlight[]>(() => {
   362→    if (scan === null || pageInfo === null) return [];
   363→    const out: PageHighlight[] = [];
   364→    scan.fields.forEach((field, index) => {
   365→      for (const widget of field.widgets) {
   366→        if (widget.page !== selectedPage) continue;
   367→        const disp = userRectToDisplay(
   368→          widget.rect,
   369→          pageInfo.rotation,
   370→          pageInfo.width,
   371→          pageInfo.height
   372→        );
   373→        if (disp.w <= 0 || disp.h <= 0) continue;
   374→        const size = displaySize(
   375→          pageInfo.rotation,
   376→          pageInfo.width,
   377→          pageInfo.height
   378→        );
   379→        out.push({
   380→          field,
   381→          index,
   382→          left: (disp.left / size.w) * 100,
   383→          top: (disp.top / size.h) * 100,
   384→          w: (disp.w / size.w) * 100,
   385→          h: (disp.h / size.h) * 100,
   386→        });
   387→      }
   388→    });
   389→    return out;
   390→  }, [scan, pageInfo, selectedPage]);
   391→
   392→  // --- Field interactions ----------------------------------------------
   393→  const rowId = useCallback(
   394→    (index: number): string => `${uid}-row-${index}`,
   395→    [uid]
   396→  );
   397→
   398→  const focusFieldRow = useCallback(
   399→    (field: FillFieldScan, index: number): void => {
   400→      setFocusedField(field.name);
   401→      const widget = field.widgets[0];
   402→      if (widget && widget.page !== selectedPage) setSelectedPage(widget.page);
   403→      window.setTimeout(() => {
   404→        const row = document.getElementById(rowId(index));
   405→        if (!row) return;
   406→        row.scrollIntoView({ behavior: "smooth", block: "center" });
   407→        row
   408→          .querySelector<HTMLElement>(
   409→            "input:not([type='date']):not([tabindex='-1']), textarea, button[role='checkbox'], button[role='radio'], button[role='combobox']"
   410→          )
   411→          ?.focus();
   412→      }, 60);
   413→    },
   414→    [rowId, selectedPage]
   415→  );
   416→
   417→  const handleInputFocus = useCallback(
   418→    (field: FillFieldScan): void => {
   419→      setFocusedField(field.name);
   420→      const widget = field.widgets[0];
   421→      if (widget && widget.page !== selectedPage) setSelectedPage(widget.page);
   422→    },
   423→    [selectedPage]
   424→  );
   425→
   426→  const removeFile = useCallback((): void => {
   427→    setFileOrder([]);
   428→  }, [setFileOrder]);
   429→
   430→  if (!file || !fileMeta) return null;
   431→
   432→  const loading = scan === null && !scanError;
   433→
   434→  return (
   435→    <div className="flex flex-col gap-4">
   436→      <PreviewFileHeader
   437→        file={file}
   438→        pageCount={pageCount}
   439→        onRemove={removeFile}
   440→      />
   441→
   442→      {loading ? (
   443→        <div
   444→          className="flex items-center gap-3 rounded-card border border-border bg-card px-4 py-8 text-sm text-muted-foreground"
   445→          role="status"
   446→        >
   447→          <Loader2 className="size-4 animate-spin" aria-hidden="true" />
   448→          {t("fill.scanning")}
   449→        </div>
   450→      ) : scanError ? (
   451→        <div
   452→          className="flex items-center gap-3 rounded-card border border-border bg-card px-4 py-8 text-sm text-muted-foreground"
   453→          role="alert"
   454→        >
   455→          <TriangleAlert
   456→            className="size-4 shrink-0 text-amber-600 dark:text-amber-400"
   457→            aria-hidden="true"
   458→          />
   459→          {t("fill.scanError")}
   460→        </div>
   461→      ) : fields.length === 0 ? (
   462→        <div className="rounded-card border border-border bg-card px-4 py-8">
   463→          <div className="flex items-start gap-3">
   464→            <span className="flex size-10 shrink-0 items-center justify-center rounded-btn bg-amber-500/10 text-amber-600 dark:text-amber-400">
   465→              <TriangleAlert className="size-5" aria-hidden="true" />
   466→            </span>
   467→            <div className="min-w-0">
   468→              <p className="text-sm font-semibold text-foreground">
   469→                {t("fill.noFieldsTitle")}
   470→              </p>
   471→              <p className="mt-1.5 text-sm leading-relaxed text-muted-foreground">
   472→                {t("fill.noFieldsBody")}
   473→              </p>
   474→              <p className="mt-1 text-xs leading-relaxed text-muted-foreground/80">
   475→                {t("fill.noFieldsEditor")}
   476→              </p>
   477→            </div>
   478→          </div>
   479→        </div>
   480→      ) : (
   481→        <>
   482→          {/* --- Live preview with field highlights --------------------- */}
   483→          <div className="overflow-hidden rounded-card border border-border bg-card">
   484→            <div className="flex flex-wrap items-center justify-between gap-2 border-b border-border/60 bg-muted/40 px-4 py-2.5">
   485→              <p className="flex items-center gap-1.5 text-xs font-semibold text-foreground">
   486→                <ClipboardCheck
   487→                  className="size-3.5 text-primary"
   488→                  aria-hidden="true"
   489→                />
   490→                {t("fill.previewTitle", { page: selectedPage + 1 })}
   491→              </p>
   492→              <PreviewZoomControl
   493→                steps={ZOOM_STEPS}
   494→                value={zoom}
   495→                onChange={setZoom}
   496→              />
   497→            </div>
   498→            <div
   499→              className={cn(
   500→                "relative flex max-w-full justify-center bg-muted/30 px-4 py-5",
   501→                zoom > 1
   502→                  ? "overflow-x-auto overflow-y-hidden scrollbar-thin"
   503→                  : "items-center overflow-hidden"
   504→              )}
   505→            >
   506→              {failed ? (
   507→                <div className="flex items-start gap-2.5 px-3 py-6 text-xs leading-relaxed text-muted-foreground">
   508→                  <TriangleAlert
   509→                    className="mt-0.5 size-4 shrink-0 text-amber-500"
   510→                    aria-hidden="true"
   511→                  />
   512→                  {t("wm.previewFailed")}
   513→                </div>
   514→              ) : current ? (
   515→                <div
   516→                  className={cn(
   517→                    "relative inline-block max-w-full",
   518→                    zoom > 1 && "max-w-none shrink-0"
   519→                  )}
   520→                >
   521→                  <canvas
   522→                    ref={canvasRef}
   523→                    style={
   524→                      zoom > 1
   525→                        ? {
   526→                            width: `${Math.round(current.bitmap.width * zoom)}px`,
   527→                          }
   528→                        : undefined
   529→                    }
   530→                    className="block h-auto w-auto max-h-[62vh] max-w-full rounded-md border border-border/60 shadow-sm"
   531→                    aria-label={t("fill.previewTitle", {
   532→                      page: selectedPage + 1,
   533→                    })}
   534→                    role="img"
   535→                  />
   536→                  {/* Field highlight boxes (percent-positioned — zoom-proof). */}
   537→                  {highlights.map((item, hi) => {
   538→                    const filled = isFilled(item.field, values, fileId);
   539→                    const focused = focusedField === item.field.name;
   540→                    return (
   541→                      <button
   542→                        key={`${item.field.name}-${hi}`}
   543→                        type="button"
   544→                        onClick={() => focusFieldRow(item.field, item.index)}
   545→                        onFocus={() => setFocusedField(item.field.name)}
   546→                        title={item.field.label}
   547→                        aria-label={t("fill.jumpToField", {
   548→                          field: item.field.label,
   549→                        })}
   550→                        className={cn(
   551→                          "absolute rounded-sm border-2 transition-colors",
   552→                          focused
   553→                            ? "border-primary bg-primary/30 ring-2 ring-primary/60"
   554→                            : filled
   555→                              ? "border-emerald-500 bg-emerald-500/20 hover:bg-emerald-500/30"
   556→                              : "border-primary/80 bg-primary/15 hover:bg-primary/25"
   557→                        )}
   558→                        style={{
   559→                          left: `${item.left}%`,
   560→                          top: `${item.top}%`,
   561→                          width: `${item.w}%`,
   562→                          height: `${item.h}%`,
   563→                        }}
   564→                      >
   565→                        {focused ? (
   566→                          <span className="sr-only">
   567→                            {t("fill.jumpToField", { field: item.field.label })}
   568→                          </span>
   569→                        ) : null}
   570→                      </button>
   571→                    );
   572→                  })}
   573→                </div>
   574→              ) : (
   575→                <div className="flex h-72 w-52 items-center justify-center rounded-md border border-dashed border-border">
   576→                  <Loader2
   577→                    className="size-5 animate-spin text-muted-foreground"
   578→                    aria-hidden="true"
   579→                  />
   580→                </div>
   581→              )}
   582→            </div>
   583→            <p className="border-t border-border/60 px-4 py-2 text-[10px] text-muted-foreground">
   584→              {t("fill.previewNote")}
   585→            </p>
   586→          </div>
   587→
   588→          <PreviewPageStrip
   589→            file={file}
   590→            pageCount={pageCount}
   591→            selected={selectedPage}
   592→            onSelect={setSelectedPage}
   593→            markedSet={markedPages}
   594→          />
   595→
   596→          {/* --- Auto-fill bar ------------------------------------------ */}
   597→          <div className="flex flex-wrap items-center gap-2 rounded-card border border-border bg-card px-4 py-3">
   598→            <Button
   599→              type="button"
   600→              variant="outline"
   601→              size="sm"
   602→              className="h-8 gap-1.5 rounded-btn px-3"
   603→              onClick={runAutoFill}
   604→            >

   687→                    onChange={(event) =>
   688→                      setProfile((prev) => ({
   689→                        ...prev,
   690→                        address: event.target.value,
   691→                      }))
   692→                    }
   693→                    className="h-8 text-sm"

   719→                >
   720→                  {t("fill.profileClear")}

   722→              </div>
   723→            </div>
   724→          ) : null}
   725→
   726→          {/* --- Field form (reading order = Tab order) ------------------ */}
   727→          <div className="overflow-hidden rounded-card border border-border bg-card">
   728→            <p className="flex items-center gap-1.5 border-b border-border/60 bg-muted/40 px-4 py-2.5 text-xs font-semibold text-foreground">
   729→              <ClipboardCheck
   730→                className="size-3.5 text-primary"

   732→              />
   733→              {t("fill.fieldsTitle")}
   734→            </p>
   735→            <div
   736→              className={cn(
   737→                "flex flex-col divide-y divide-border/60",
   738→                fields.length > LIST_SCROLL_LIMIT &&
   739→                  "max-h-[70vh] overflow-y-auto scrollbar-thin"
   740→              )}
   741→            >
   742→              {fields.map((field, index) => (
   743→                <FieldRow
   744→                  key={field.name}
   745→                  id={rowId(index)}
   746→                  field={field}
   747→                  value={currentValue(field, values, fileId)}
   748→                  changed={isChanged(field, values, fileId)}
   749→                  filled={isFilled(field, values, fileId)}
   750→                  focused={focusedField === field.name}
   751→                  showPageChip={pageCount > 1}
   752→                  onPageChipClick={() =>
   753→                    setSelectedPage(field.widgets[0]?.page ?? 0)
   754→                  }
   755→                  onFocus={handleInputFocus}
   756→                  onValueChange={(next) =>
   757→                    onChange(fieldKey(fileId, field.name), next)
   758→                  }
   759→                />
   760→              ))}
   761→            </div>
   762→          </div>
   763→        </>
   764→      )}
   765→    </div>
   766→  );

   769→/** One field row: label + badges on the left, the input on the right. */
   770→function FieldRow({
   771→  id,
   772→  field,
   773→  value,
   774→  changed,
   775→  filled,
   776→  focused,
   777→  showPageChip,
   778→  onPageChipClick,
   779→  onFocus,
   780→  onValueChange,
   781→}: {
   782→  id: string;
   783→  field: FillFieldScan;
   784→  value: string | boolean;
   785→  changed: boolean;
   786→  filled: boolean;
   787→  focused: boolean;
   788→  showPageChip: boolean;
   789→  onPageChipClick: () => void;
   790→  onFocus: (field: FillFieldScan) => void;
   791→  onValueChange: (next: string | boolean) => void;
   792→}): React.JSX.Element {
   793→  const { t } = useI18n();
   794→  const uid = useId();
   795→  const focusField = (): void => onFocus(field);
   796→
   797→  // Values flow straight through onValueChange (the shell owns the state).
   798→  // Rapid same-frame double toggles are an automation-only artifact —
   799→  // human interactions always land in separate frames (P3.4 lesson).
   800→
   801→  return (
   802→    <div
   803→      id={id}
   804→      className={cn(
   805→        "grid grid-cols-1 gap-2 px-4 py-3 transition-colors sm:grid-cols-[minmax(0,13rem)_minmax(0,1fr)] sm:gap-4",
   806→        focused && "bg-primary/[0.05]"
   807→      )}
   808→    >
   809→      <div className="min-w-0">
   810→        <div className="flex flex-wrap items-center gap-1.5">

   812→            <button
   813→              type="button"
   814→              onClick={onPageChipClick}
   815→              className="rounded-chip border border-border bg-muted/50 px-1.5 py-0.5 text-[10px] font-semibold tabular-nums text-muted-foreground transition-colors hover:border-primary/40 hover:text-foreground"
   816→              title={t("fill.gotoPage")}
   817→            >
   818→              {t("fill.pageLabel", { page: (field.widgets[0]?.page ?? 0) + 1 })}
   819→            </button>
   820→          ) : null}
   821→          <Label
   822→            htmlFor={`${uid}-input`}
   823→            className="min-w-0 break-words text-sm font-medium leading-snug text-foreground"
   824→          >
   825→            {field.label}

   829→            <span

   831→              title={t("fill.required")}
   832→            >
   833→              {t("fill.required")}
   834→            </span>
   835→          ) : null}
   836→          {field.readOnly ? (
   837→            <span
   838→              className="flex items-center gap-1 rounded-chip border border-border bg-muted/50 px-1.5 py-0.5 text-[10px] font-medium text-muted-foreground"
   839→              title={t("fill.readonly")}
   840→            >
   841→              <Lock className="size-2.5" aria-hidden="true" />
   842→              {t("fill.readonly")}
   843→            </span>
   844→          ) : null}
   845→          {field.kind !== "text" && field.kind !== "checkbox" ? (
   846→            <span className="rounded-chip border border-border bg-muted/40 px-1.5 py-0.5 text-[10px] font-medium text-muted-foreground">
   847→              {t(KIND_BADGE_KEYS[field.kind])}
   848→            </span>
   849→          ) : null}
   850→        </div>
   851→        {field.maxLength !== null && field.kind === "text" ? (
   852→          <p className="mt-1 text-[10px] tabular-nums text-muted-foreground">
   853→            {t("fill.maxChars", { count: field.maxLength })}

   857→
   858→      <div className="min-w-0">
   859→        {field.kind === "text" ? (

   861→            <Textarea
   862→              id={`${uid}-input`}
   863→              value={typeof value === "string" ? value : ""}
   864→              maxLength={field.maxLength ?? undefined}
   865→              disabled={field.readOnly}
   866→              rows={3}
   867→              onFocus={focusField}
   868→              onChange={(event) => onValueChange(event.target.value)}
   869→              className={cn(
   870→                "min-h-0 resize-y text-sm",
   871→                changed && "border-primary/50"
   872→              )}
   873→            />
   874→          ) : (
   875→            <span className="relative block">
   876→              <Input
   877→                id={`${uid}-input`}
   878→                value={typeof value === "string" ? value : ""}
   879→                maxLength={field.maxLength ?? undefined}
   880→                disabled={field.readOnly}
   881→                onFocus={focusField}
   882→                onChange={(event) => onValueChange(event.target.value)}
   883→                className={cn(
   884→                  "text-sm",
   885→                  isDateField(field) && "pr-9",
   886→                  changed && "border-primary/50"
   887→                )}
   888→              />
   889→              {isDateField(field) && !field.readOnly ? (
   890→                <label
   891→                  className="absolute right-1.5 top-1/2 flex size-7 -translate-y-1/2 cursor-pointer items-center justify-center rounded-md text-muted-foreground transition-colors hover:bg-muted hover:text-foreground"

   902→                      if (text !== "") onValueChange(text);

   905→                </label>

   907→            </span>

   910→          <label
   911→            htmlFor={`${uid}-input`}
   912→            className="flex min-h-9 cursor-pointer items-center gap-2.5 text-sm text-foreground"

   914→            <Checkbox
   915→              id={`${uid}-input`}
   916→              checked={value === true}
   917→              disabled={field.readOnly}
   918→              onFocus={focusField}

   920→            />

   922→              {t("fill.checkboxLabel")}

   940→                    id={`${uid}-opt-${option}`}
   941→                    value={option}

   943→                  />
   944→                  <span className="text-foreground">{option}</span>

   946→              ))}
   947→            </RadioGroup>

   949→              <button
   950→                type="button"
   951→                onClick={() => onValueChange("")}

   953→              >
   954→                {t("fill.clearSelection")}

   959→          field.options.length === 0 ? (
   960→            <p className="min-h-9 py-2 text-sm text-muted-foreground">

   965→              value={
   966→                typeof value === "string" && value !== "" ? value : "__none__"
   967→              }
   968→              disabled={field.readOnly}
   969→              onValueChange={(next) =>
   970→                onValueChange(next === "__none__" ? "" : next)
   971→              }
   972→            >
   973→              <SelectTrigger
   974→                id={`${uid}-input`}
   975→                onFocus={focusField}
   976→                className={cn(
   977→                  "h-9 w-full text-sm",

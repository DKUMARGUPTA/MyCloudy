     1→/**
     2→ * Edit PDF — PURE object model (PRD §8.2.1 flagship editor, sub-phases
     3→ * 8.1 shell + 8.2 objects + 8.5 export).
     4→ *
     5→ * Everything here is framework-free and bun-testable:
     6→ *  - the EditorObject union (text / image / shapes / freehand / whiteout)
     7→ *  - the PDF-user-space ↔ display-space transforms for all four /Rotate
     8→ *    values (mirroring pdf.js PageViewport — the make-pdf-fillable recipe,
     9→ *    copied here so the SURFACE never imports pdf-lib transitively)
    10→ *  - object factories with sane defaults + clamping/sanitising for the
    11→ *    worker (never trust the wire blob)
    12→ *  - z-order operations, hit-testing, bbox math, freehand normalisation
    13→ *  - the coalescing history helper (snapshot stack semantics)
    14→ *
    15→ * Coordinate convention (same as make-pdf-fillable): every object rect is
    16→ * in PDF USER-SPACE points, y measured from the BOTTOM (pdf-lib
    17→ * convention). `page` is 0-based. Display rects (y from top) are derived
    18→ * per page through the PageView (crop box + /Rotate).
    19→ */

   114→// ---------------------------------------------------------------------------
   115→// PageView + display transforms (pdf.js-mirrored, /Rotate-aware)

   118→export interface PageView {
   119→  /** pdf.js page.view: [x0, y0, x1, y1] of the CROP box, user space. */
   120→  x0: number;
   121→  y0: number;
   122→  x1: number;
   123→  y1: number;
   124→  /** /Rotate 0 | 90 | 180 | 270 (normalized). */
   125→  rotate: number;
   126→}
   127→
   128→/** Display (visible) page size in points, /Rotate applied. */
   129→export function displaySize(view: PageView): { w: number; h: number } {
   130→  const uw = view.x1 - view.x0;
   131→  const uh = view.y1 - view.y0;
   132→  return view.rotate === 90 || view.rotate === 270
   133→    ? { w: uh, h: uw }
   134→    : { w: uw, h: uh };
   135→}
   136→
   137→/** User-space point → display point (y down, origin top-left). */
   138→export function pdfToDisplayPoint(
   139→  px: number,
   140→  py: number,
   141→  view: PageView
   142→): { x: number; y: number } {
   143→  switch (view.rotate) {
   144→    case 90:
   145→      return { x: py - view.y0, y: px - view.x0 };
   146→    case 180:
   147→      return { x: view.x1 - px, y: py - view.y0 };
   148→    case 270:
   149→      return { x: view.y1 - py, y: view.x1 - px };
   150→    default:
   151→      return { x: px - view.x0, y: view.y1 - py };
   152→  }
   153→}
   154→
   155→/** Display point → user-space point (inverse of pdfToDisplayPoint). */
   156→export function displayToPdfPoint(
   157→  dx: number,
   158→  dy: number,
   159→  view: PageView
   160→): { x: number; y: number } {
   161→  switch (view.rotate) {
   162→    case 90:
   163→      return { x: dy + view.x0, y: dx + view.y0 };
   164→    case 180:
   165→      return { x: view.x1 - dx, y: view.y0 + dy };
   166→    case 270:
   167→      return { x: view.x1 - dy, y: view.y1 - dx };
   168→    default:
   169→      return { x: dx + view.x0, y: view.y1 - dy };
   170→  }
   171→}
   172→
   173→/** PDF rect (y from bottom) → display rect (y from top), /Rotate-aware. */
   174→export function pdfRectToDisplay(rect: ObjRect, view: PageView): ObjRect {
   175→  const corners = [
   176→    pdfToDisplayPoint(rect.x, rect.y, view),
   177→    pdfToDisplayPoint(rect.x + rect.w, rect.y + rect.h, view),
   178→  ];
   179→  const xs = corners.map((c) => c.x);
   180→  const ys = corners.map((c) => c.y);
   181→  const x = Math.min(...xs);
   182→  const y = Math.min(...ys);
   183→  return { x, y, w: Math.max(...xs) - x, h: Math.max(...ys) - y };
   184→}
   185→
   186→/** Display rect (y from top) → PDF rect (y from bottom), /Rotate-aware. */
   187→export function displayRectToPdf(rect: ObjRect, view: PageView): ObjRect {
   188→  const corners = [
   189→    displayToPdfPoint(rect.x, rect.y, view),
   190→    displayToPdfPoint(rect.x + rect.w, rect.y + rect.h, view),
   191→  ];
   192→  const xs = corners.map((c) => c.x);
   193→  const ys = corners.map((c) => c.y);
   194→  const x = Math.min(...xs);
   195→  const y = Math.min(...ys);
   196→  return { x, y, w: Math.max(...xs) - x, h: Math.max(...ys) - y };
   197→}

   230→/** #rgb / #rrggbb → pdf-lib rgb() components. White for garbage. */
   231→export function hexToRgb(hex: string): {
   232→  r: number;
   233→  g: number;
   234→  b: number;
   235→} {
   236→  const m = /^#?([0-9a-f]{6}|[0-9a-f]{3})$/i.exec(hex.trim());
   237→  if (!m) return { r: 1, g: 1, b: 1 };
   238→  let h = m[1];
   239→  if (h.length === 3) h = h[0] + h[0] + h[1] + h[1] + h[2] + h[2];
   240→  const num = Number.parseInt(h, 16);
   241→  return {
   242→    r: ((num >> 16) & 255) / 255,
   243→    g: ((num >> 8) & 255) / 255,
   244→    b: (num & 255) / 255,
   245→  };
   246→}

   378→/**
   379→ * Converts raw display-space sample points (pt) into a bbox + fractional
   380→ * point list. Points outside the page are NOT clamped (the caller clamps
   381→ * the gesture to the overlay).
   382→ */
   383→export function normaliseDraw(
   384→  pts: { x: number; y: number }[]
   385→): { rect: ObjRect; fracs: number[][] } | null {
   386→  if (pts.length < MIN_DRAW_POINTS) return null;
   387→  let minX = Infinity;
   388→  let minY = Infinity;
   389→  let maxX = -Infinity;
   390→  let maxY = -Infinity;
   391→  for (const p of pts) {
   392→    minX = Math.min(minX, p.x);
   393→    minY = Math.min(minY, p.y);
   394→    maxX = Math.max(maxX, p.x);
   395→    maxY = Math.max(maxY, p.y);
   396→  }
   397→  const w = maxX - minX;
   398→  const h = maxY - minY;
   399→  // Degenerate (a dot / a straight horizontal or vertical line) — keep a
   400→  // minimal bbox so the stroke still renders with its width.
   401→  const bw = Math.max(w, 2);
   402→  const bh = Math.max(h, 2);
   403→  const fracs = pts.map((p) => [
   404→    Number(((p.x - minX) / bw).toFixed(4)),
   405→    Number(((p.y - minY) / bh).toFixed(4)),
   406→  ]);
   407→  return {
   408→    rect: { x: minX, y: minY, w: bw, h: bh },
   409→    fracs,
   410→  };
   411→}

   423→// ---------------------------------------------------------------------------
   424→// Hit-testing + z-order (display-space, axis-aligned)

   427→export type HitEdge = "nw" | "ne" | "sw" | "se";
   428→
   429→export function hitRect(disp: ObjRect, px: number, py: number): boolean {
   430→  return (
   431→    px >= disp.x &&
   432→    px <= disp.x + disp.w &&
   433→    py >= disp.y &&
   434→    py <= disp.y + disp.h
   435→  );
   436→}
   437→
   438→/** Corner handle the point is over (within `tol` pt), if any. */
   439→export function hitHandle(
   440→  disp: ObjRect,
   441→  px: number,
   442→  py: number,
   443→  tol: number
   444→): HitEdge | null {
   445→  const corners: { edge: HitEdge; cx: number; cy: number }[] = [
   446→    { edge: "nw", cx: disp.x, cy: disp.y },
   447→    { edge: "ne", cx: disp.x + disp.w, cy: disp.y },
   448→    { edge: "sw", cx: disp.x, cy: disp.y + disp.h },
   449→    { edge: "se", cx: disp.x + disp.w, cy: disp.y + disp.h },
   450→  ];
   451→  for (const c of corners) {
   452→    if (Math.abs(px - c.cx) <= tol && Math.abs(py - c.cy) <= tol) return c.edge;
   453→  }
   454→  return null;
   455→}

   457→/** Topmost (last in array) object at the display point. */
   458→export function pickObject(
   459→  objects: EditorObject[],
   460→  view: PageView,
   461→  px: number,
   462→  py: number
   463→): EditorObject | null {
   464→  for (let i = objects.length - 1; i >= 0; i -= 1) {
   465→    const obj = objects[i];
   466→    if (obj.hidden) continue;
   467→    if (!hitRect(pdfRectToDisplay(obj, view), px, py)) continue;
   468→    return obj;
   469→  }
   470→  return null;
   471→}

   473→/** Resize a rect from one corner, keeping w/h ≥ min. Returns display rect. */
   474→export function resizeRect(
   475→  orig: ObjRect,
   476→  edge: HitEdge,
   477→  x2: number,
   478→  y2: number,
   479→  min: number,
   480→  keepAspect: boolean,
   481→  aspect: number
   482→): ObjRect {
   483→  let { x, y, w, h } = orig;
   484→  if (edge === "se") {
   485→    w = Math.max(min, x2 - x);
   486→    h = Math.max(min, y2 - y);
   487→    if (keepAspect) {
   488→      w = Math.max(min, Math.min(w, h * aspect));
   489→      h = w / aspect;
   490→    }
   491→  } else if (edge === "ne") {
   492→    const ny = Math.min(y + h - min, y2);
   493→    w = Math.max(min, x2 - x);
   494→    h = Math.max(min, y + h - ny);
   495→    y = ny;
   496→    if (keepAspect) {
   497→      w = Math.max(min, Math.min(w, h * aspect));
   498→      h = w / aspect;
   499→      y = orig.y + orig.h - h;
   500→    }
   501→  } else if (edge === "sw") {
   502→    const nx = Math.min(x + w - min, x2);
   503→    w = Math.max(min, x + w - nx);
   504→    x = nx;
   505→    h = Math.max(min, y2 - y);
   506→    if (keepAspect) {
   507→      h = Math.max(min, Math.min(h, w / aspect));
   508→      w = h * aspect;
   509→      x = orig.x + orig.w - w;
   510→    }
   511→  } else {
   512→    const nx = Math.min(x + w - min, x2);
   513→    const ny = Math.min(y + h - min, y2);
   514→    w = Math.max(min, x + w - nx);
   515→    h = Math.max(min, y + h - ny);
   516→    x = nx;
   517→    y = ny;
   518→    if (keepAspect) {
   519→      w = Math.max(min, Math.min(w, h * aspect));
   520→      h = w / aspect;
   521→      x = orig.x + orig.w - w;
   522→      y = orig.y + orig.h - h;
   523→    }
   524→  }
   525→  return { x, y, w, h };
   526→}

   550→export function raiseOne(objects: EditorObject[], id: string): EditorObject[] {
   551→  const idx = objects.findIndex((o) => o.id === id);
   552→  if (idx < 0 || idx === objects.length - 1) return objects;
   553→  const next = [...objects];
   554→  const tmp = next[idx];
   555→  next[idx] = next[idx + 1];
   556→  next[idx + 1] = tmp;
   557→  return next;
   558→}
   559→
   560→export function lowerOne(objects: EditorObject[], id: string): EditorObject[] {
   561→  const idx = objects.findIndex((o) => o.id === id);
   562→  if (idx <= 0) return objects;
   563→  const next = [...objects];
   564→  const tmp = next[idx];
   565→  next[idx] = next[idx - 1];
   566→  next[idx - 1] = tmp;
   567→  return next;
   568→}

   570→export function duplicateObject(
   571→  objects: EditorObject[],
   572→  id: string
   573→): { objects: EditorObject[]; copy: EditorObject | null } {
   574→  const idx = objects.findIndex((o) => o.id === id);
   575→  if (idx < 0) return { objects, copy: null };
   576→  const src = objects[idx];
   577→  const copy: EditorObject = {
   578→    ...src,
   579→    id: nextObjId("c"),
   580→    x: roundPt(src.x + 12),
   581→    y: roundPt(Math.max(0, src.y - 12)),
   582→  };
   583→  const next = [...objects];
   584→  next.splice(idx + 1, 0, copy);
   585→  return { objects: next, copy };
   586→}

   594→export function pushSnapshot(

   596→  snapshot: string,
   597→  future: string[]
   598→): { past: string[]; future: string[] } {

   601→  return { past: nextPast, future: [] };

   604→/** Undo: pops past into future, returns the restored snapshot. */
   605→export function undoStep(
   606→  past: string[],
   607→  future: string[],
   608→  current: string
   609→): { past: string[]; future: string[]; snapshot: string | null } {
   610→  if (past.length === 0) return { past, future, snapshot: null };
   611→  const nextPast = [...past];
   612→  const snapshot = nextPast.pop() as string;
   613→  return { past: nextPast, future: [current, ...future], snapshot };
   614→}
   615→
   616→/** Redo: pops future into past, returns the restored snapshot. */
   617→export function redoStep(
   618→  past: string[],
   619→  future: string[],
   620→  current: string
   621→): { past: string[]; future: string[]; snapshot: string | null } {
   622→  if (future.length === 0) return { past, future, snapshot: null };
   623→  const [snapshot, ...rest] = future;
   624→  return { past: [...past, current], future: rest, snapshot };
   625→}

   631→function num(v: unknown, dflt: number): number {
   632→  return typeof v === "number" && Number.isFinite(v) ? v : dflt;
   633→}
   634→
   635→function str(v: unknown, dflt: string): string {
   636→  return typeof v === "string" ? v : dflt;
   637→}
   638→
   639→function bool(v: unknown, dflt: boolean): boolean {
   640→  return typeof v === "boolean" ? v : dflt;
   641→}

   643→function color(v: unknown, dflt: string): string {
   644→  return typeof v === "string" && /^#[0-9a-f]{3}([0-9a-f]{3})?$/i.test(v)
   645→    ? v
   646→    : dflt;
   647→}

   649→function optColor(v: unknown): string | null {
   650→  if (v === null) return null;
   651→  return color(v, "#000000");

   656→  "text",
   657→  "image",

   666→/** Parses + clamps one raw object from the wire blob. null = drop it. */
   667→export function sanitizeObject(
   668→  raw: unknown,
   669→  pageCount: number,
   670→  knownAssets: ReadonlySet<string>
   671→): EditorObject | null {
   672→  if (typeof raw !== "object" || raw === null) return null;
   673→  const o = raw as Record<string, unknown>;
   674→  const kind = KINDS.includes(o.kind as ObjKind) ? (o.kind as ObjKind) : null;
   675→  if (kind === null) return null;
   676→  const page = Math.min(
   677→    Math.max(0, Math.trunc(num(o.page, 0))),
   678→    Math.max(0, pageCount - 1)
   679→  );
   680→  const base = {
   681→    id: str(o.id, nextObjId("x")),
   682→    page,
   683→    x: num(o.x, 72),
   684→    y: num(o.y, 72),
   685→    w: Math.max(MIN_OBJ_PT, num(o.w, 120)),
   686→    h: Math.max(MIN_OBJ_PT, num(o.h, 24)),
   687→    opacity: clampN(num(o.opacity, 1), 0.05, 1),
   688→    locked: bool(o.locked, false),
   689→    hidden: bool(o.hidden, false),
   690→  };
   691→  switch (kind) {
   692→    case "text":
   693→      return {
   694→        ...base,
   695→        kind,
   696→        text: str(o.text, ""),
   697→        font: FONTS.includes(o.font as EditorFontId)
   698→          ? (o.font as EditorFontId)
   699→          : "helvetica",
   700→        size: clampN(num(o.size, 12), 4, 200),
   701→        color: color(o.color, "#111827"),
   702→        bold: bool(o.bold, false),
   703→        italic: bool(o.italic, false),
   704→        underline: bool(o.underline, false),
   705→        align: o.align === "center" || o.align === "right" ? o.align : "left",
   706→        bg: optColor(o.bg),
   707→      };
   708→    case "image": {
   709→      const key = str(o.assetKey, "");
   710→      if (!knownAssets.has(key)) return null;
   711→      return { ...base, kind, assetKey: key };
   712→    }
   713→    case "rect":
   714→    case "ellipse":
   715→    case "line":
   716→    case "arrow":
   717→      return {
   718→        ...base,
   719→        kind,
   720→        fill: optColor(o.fill),
   721→        stroke: optColor(o.stroke),
   722→        strokeW: clampN(num(o.strokeW, 2), 0.25, 40),
   723→        dash: bool(o.dash, false),
   724→        flip: bool(o.flip, false),
   725→      };
   726→    case "draw": {
   727→      const rawPts = Array.isArray(o.pts) ? o.pts : [];
   728→      const pts = rawPts
   729→        .filter(
   730→          (p): p is number[] =>
   731→            Array.isArray(p) &&
   732→            p.length === 2 &&
   733→            typeof p[0] === "number" &&
   734→            typeof p[1] === "number"
   735→        )
   736→        .map((p) => [clampN(p[0], 0, 1), clampN(p[1], 0, 1)]);
   737→      if (pts.length < 2) return null;
   738→      return {
   739→        ...base,
   740→        kind,
   741→        pts,
   742→        stroke: color(o.stroke, "#dc2626"),
   743→        strokeW: clampN(num(o.strokeW, 2.5), 0.25, 40),
   744→      };
   745→    }
   746→    case "whiteout":
   747→      return { ...base, kind, fill: color(o.fill, "#ffffff") };
   748→  }
   749→}

   751→/** Parses the full objects array from the wire blob. */
   752→export function sanitizeObjects(
   753→  raw: unknown,
   754→  pageCount: number,
   755→  knownAssets: ReadonlySet<string>
   756→): EditorObject[] {
   757→  if (!Array.isArray(raw)) return [];
   758→  const out: EditorObject[] = [];
   759→  for (const item of raw) {
   760→    const obj = sanitizeObject(item, pageCount, knownAssets);
   761→    if (obj !== null && !obj.hidden) out.push(obj);
   762→  }
   763→  return out;
   764→}

   890→function normaliseFontName(raw: string): string {
   891→  // Strip a subset prefix ("ABCDEF+ArialMT") and punctuation, lowercase.
   892→  const plus = raw.indexOf("+");
   893→  const base = plus >= 0 ? raw.slice(plus + 1) : raw;
   894→  return base.replace(/[\s_-]+/g, "").toLowerCase();
   895→}

   897→/**
   898→ * Maps an original PDF font onto the editor's embedded fonts (PRD §8.2:
   899→ * "font match — standard fonts, embedded font agar subset allow kare, warna
   900→ * closest font suggest"). Exact only for the standard-14 families (and the
   901→ * Devanagari font we embed verbatim); everything else maps to the closest
   902→ * family by name table, class flags or CSS fallback family.
   903→ */
   904→export function matchFont(
   905→  rawName: string,
   906→  hints: FontMatchHints
   907→): FontMatch {
   908→  const norm = normaliseFontName(rawName);
   909→  const bold =
   910→    hints.bold === true ||
   911→    (hints.bold !== false && /bold|black|heavy|semibold|demibold/i.test(rawName));
   912→  const italic =
   913→    hints.italic === true ||
   914→    (hints.italic !== false && /italic|oblique/i.test(rawName));
   915→
   916→  // Devanagari text needs the Devanagari pipeline regardless of the name.
   917→  if (hints.hasDeva === true) {
   918→    return {
   919→      font: "devanagari",
   920→      bold,
   921→      italic,
   922→      exact: norm === "notosansdevanagari",
   923→      original: rawName,
   924→    };
   925→  }
   926→
   927→  let font: EditorFontId | null = null;
   928→  if (norm !== "" && FAMILY_BY_NAME[norm] !== undefined) {
   929→    font = FAMILY_BY_NAME[norm];
   930→  } else if (hints.mono === true || hints.cssFamily === "monospace") {
   931→    font = "courier";
   932→  } else if (hints.serif === true || hints.cssFamily === "serif") {
   933→    font = "times";
   934→  } else if (norm.includes("devanagari")) {
   935→    font = "devanagari";
   936→  } else if (
   937→    norm.includes("mono") ||
   938→    norm.includes("courier") ||
   939→    norm.includes("consolas") ||
   940→    norm.includes("menlo")
   941→  ) {
   942→    font = "courier";
   943→  } else if (
   944→    norm.includes("serif") ||
   945→    norm.includes("roman") ||
   946→    norm.includes("times") ||
   947→    norm.includes("georgia") ||
   948→    norm.includes("garamond")
   949→  ) {
   950→    font = "times";
   951→  } else if (
   952→    norm.includes("sans") ||
   953→    norm.includes("helvetica") ||
   954→    norm.includes("arial") ||
   955→    norm.includes("verdana") ||
   956→    norm.includes("tahoma") ||
   957→    norm.includes("segoe")
   958→  ) {
   959→    font = "helvetica";
   960→  } else if (font === null) {
   961→    font = "helvetica";
   962→  }
   963→
   964→  return {
   965→    font,
   966→    bold,
   967→    italic,
   968→    exact: EXACT_NAMES.has(norm),
   969→    original: rawName,
   970→  };
   971→}

   973→/** Vertical metrics used to expand a piece into a comfortable cover rect. */
   974→export const PIECE_ASCENT = 0.85;
   975→export const PIECE_DESCENT = 0.25;
   976→export const PIECE_PAD = 1.5;
   977→
   978→/**
   979→ * The DISPLAY rect a piece's cover/text objects occupy: the glyph band
   980→ * (ascent above the baseline, descent below) plus a small pad, widened by
   981→ * 1pt on each side so ascenders/descenders never peek out.
   982→ */
   983→export function pieceRect(piece: {
   984→  x: number;
   985→  xEnd: number;
   986→  baseline: number;
   987→  h: number;
   988→}): ObjRect {
   989→  const top = piece.baseline - PIECE_ASCENT * piece.h - PIECE_PAD;
   990→  const bottom = piece.baseline + PIECE_DESCENT * piece.h + PIECE_PAD;
   991→  return {
   992→    x: piece.x - 1,
   993→    y: top,
   994→    w: Math.max(8, piece.xEnd - piece.x + 2),
   995→    h: Math.max(10, bottom - top),
   996→  };
   997→}

  1033→/** A visual text line: adjacent same-baseline pieces merged with spaces. */
  1034→export interface PieceLine {
  1035→  page: number;
  1036→  /** Indices into the page's piece array (x order). */
  1037→  pieces: number[];
  1038→  text: string;
  1039→  x: number;
  1040→  xEnd: number;
  1041→  baseline: number;
  1042→  h: number;
  1043→}
  1044→
  1045→/**
  1046→ * Groups a page's pieces into visual lines: baseline bucket (Δ ≤ 0.45 ×
  1047→ * the larger em) + horizontal gap ≤ 1.2 × em joins as a space, pieces sorted
  1048→ * by x. The same-baseline rule keeps columns separate (P4.3 insight).
  1049→ */
  1050→export function groupPiecesIntoLines(
  1051→  page: number,
  1052→  pieces: HarvestPiece[]
  1053→): PieceLine[] {
  1054→  const sorted = [...pieces].sort((a, b) => a.x - b.x);
  1055→  const lines: PieceLine[] = [];
  1056→  for (const p of sorted) {
  1057→    const line = lines.find((l) => {
  1058→      if (Math.abs(l.baseline - p.baseline) > 0.45 * Math.max(l.h, p.h)) {
  1059→        return false;
  1060→      }
  1061→      // Gap between the line's right edge and this piece (x-sorted so ≥ 0).
  1062→      return p.x - l.xEnd <= 1.2 * Math.max(l.h, p.h);
  1063→    });
  1064→    if (line) {
  1065→      line.pieces.push(p.index);
  1066→      line.xEnd = Math.max(line.xEnd, p.xEnd);
  1067→      line.text = `${line.text} ${p.str}`;
  1068→      line.h = Math.max(line.h, p.h);
  1069→    } else {
  1070→      lines.push({
  1071→        page,
  1072→        pieces: [p.index],
  1073→        text: p.str,
  1074→        x: p.x,
  1075→        xEnd: p.xEnd,
  1076→        baseline: p.baseline,
  1077→        h: p.h,
  1078→      });
  1079→    }
  1080→  }
  1081→  return lines;
  1082→}

  1093→/** Literal (non-regex) occurrences of the query across all lines. */
  1094→export function planLineReplacements(
  1095→  lines: PieceLine[],
  1096→  query: string,
  1097→  matchCase: boolean
  1098→): LineMatch[] {
  1099→  const out: LineMatch[] = [];
  1100→  const needle = matchCase ? query : query.toLowerCase();
  1101→  if (needle.length === 0) return out;
  1102→  for (let i = 0; i < lines.length; i += 1) {
  1103→    const line = lines[i];
  1104→    const hay = matchCase ? line.text : line.text.toLowerCase();
  1105→    let from = 0;
  1106→    for (;;) {
  1107→      const at = hay.indexOf(needle, from);
  1108→      if (at < 0) break;
  1109→      out.push({ page: line.page, line: i, start: at, end: at + needle.length });
  1110→      from = at + needle.length;
  1111→    }
  1112→  }
  1113→  return out;
  1114→}

  1116→/**
  1117→ * Builds the whiteout + retype pair for one VISUAL LINE (find & replace):
  1118→ * the cover spans the union of the line's piece rects, the retype uses the
  1119→ * dominant piece's font match and the substituted text. One history group
  1120→ * per call is the caller's responsibility.
  1121→ */
  1122→export function buildReplaceLineObjects(
  1123→  page: number,
  1124→  view: PageView,
  1125→  line: PieceLine,
  1126→  pagePieces: HarvestPiece[],
  1127→  newText: string,
  1128→  bgHex: string,
  1129→  inkHex: string
  1130→): { cover: WhiteoutObj; text: TextObj } {
  1131→  let x0 = Infinity;
  1132→  let y0 = Infinity;
  1133→  let x1 = -Infinity;
  1134→  let y1 = -Infinity;
  1135→  for (const idx of line.pieces) {
  1136→    const p = pagePieces[idx];
  1137→    if (!p) continue;
  1138→    const r = pieceRect(p);
  1139→    x0 = Math.min(x0, r.x);
  1140→    y0 = Math.min(y0, r.y);
  1141→    x1 = Math.max(x1, r.x + r.w);
  1142→    y1 = Math.max(y1, r.y + r.h);
  1143→  }
  1144→  if (!Number.isFinite(x0) || !Number.isFinite(y0)) {
  1145→    x0 = line.x;
  1146→    y0 = line.baseline - line.h;
  1147→    x1 = line.xEnd;
  1148→    y1 = line.baseline;
  1149→  }
  1150→  const disp = { x: x0, y: y0, w: Math.max(8, x1 - x0), h: Math.max(10, y1 - y0) };
  1151→  const pdf = displayRectToPdf(disp, view);
  1152→  const cover = makeWhiteout(page, pdf);
  1153→  cover.fill = bgHex;
  1154→  // Dominant piece = largest em (ties: first in x order).
  1155→  let dom = pagePieces[line.pieces[0]] ?? null;
  1156→  for (const idx of line.pieces) {
  1157→    const p = pagePieces[idx];
  1158→    if (p && dom !== null && p.h > dom.h) dom = p;
  1159→    else if (p && dom === null) dom = p;
  1160→  }
  1161→  const text = makeText(page, pdf, {
  1162→    text: newText,
  1163→    font: dom !== null ? dom.font.font : "helvetica",
  1164→    size: clampN(dom !== null ? dom.h : 12, 4, 200),
  1165→    color: inkHex,
  1166→    bold: dom !== null ? dom.font.bold : false,
  1167→    italic: dom !== null ? dom.font.italic : false,
  1168→    align: "left",
  1169→  });
  1170→  return { cover, text };
  1171→}
  1172→
  1173→/**
  1174→ * Line text with the given occurrence ranges substituted (literal
  1175→ * semantics — split/join so regex metacharacters in the query are safe).
  1176→ */
  1177→export function substituteOccurrences(
  1178→  text: string,
  1179→  ranges: { start: number; end: number }[],
  1180→  replacement: string
  1181→): string {
  1182→  if (ranges.length === 0) return text;
  1183→  const sorted = [...ranges].sort((a, b) => a.start - b.start);
  1184→  let out = "";
  1185→  let cursor = 0;
  1186→  for (const r of sorted) {
  1187→    if (r.start < cursor) continue; // overlap guard
  1188→    out += text.slice(cursor, r.start) + replacement;
  1189→    cursor = r.end;
  1190→  }
  1191→  return out + text.slice(cursor);
  1192→}

  1204→export function luminance(r: number, g: number, b: number): number {
  1205→  return (0.2126 * r + 0.7152 * g + 0.0722 * b) / 255;

  1208→/** Median of an RGBA pixel run (ignores alpha — pages render opaque). */
  1209→export function medianRgb(pixels: number[]): { r: number; g: number; b: number } {
  1210→  if (pixels.length === 0) return { r: 255, g: 255, b: 255 };
  1211→  const by = (ch: 0 | 1 | 2): number => {
  1212→    const vals = pixels.filter((_, i) => i % 4 === ch).sort((a, b) => a - b);
  1213→    return vals[Math.floor(vals.length / 2)] ?? 255;
  1214→  };
  1215→  return { r: by(0), g: by(1), b: by(2) };
  1216→}

  1218→/**
  1219→ * Picks the text ink colour from the pixels under a piece: the darkest
  1220→ * cluster on a light background, the brightest on a dark one. Returns null
  1221→ * when the contrast is too weak to trust (caller falls back to a default).
  1222→ */
  1223→export function pickInkFromPixels(
  1224→  pixels: number[],
  1225→  bgLum: number
  1226→): { r: number; g: number; b: number } | null {
  1227→  if (pixels.length < 16) return null;
  1228→  const lums = pixels
  1229→    .filter((_, i) => i % 4 === 3 && pixels[i] > 40)
  1230→    .length;
  1231→  if (lums < 4) return null;
  1232→  const entries: { lum: number; r: number; g: number; b: number }[] = [];
  1233→  for (let i = 0; i + 3 < pixels.length; i += 4) {
  1234→    const { r, g, b } = { r: pixels[i], g: pixels[i + 1], b: pixels[i + 2] };
  1235→    entries.push({ lum: luminance(r, g, b), r, g, b });
  1236→  }
  1237→  entries.sort((a, b) => a.lum - b.lum);
  1238→  const dark = entries[Math.floor(entries.length * 0.12)];
  1239→  const bright = entries[Math.floor(entries.length * 0.88)];
  1240→  const pick = bgLum > 0.5 ? dark : bright;
  1241→  const contrast = Math.abs(pick.lum - bgLum);
  1242→  if (contrast < 0.16) return null;
  1243→  return { r: pick.r, g: pick.g, b: pick.b };
  1244→}

  1246→/** Greedy word-wrap that hard-breaks overlong words (surface CSS + worker). */
  1247→export function wrapText(
  1248→  text: string,
  1249→  maxWidth: number,
  1250→  measure: (s: string) => number
  1251→): string[] {
  1252→  const lines: string[] = [];
  1253→  for (const para of text.split("\n")) {
  1254→    if (para.length === 0) {
  1255→      lines.push("");
  1256→      continue;
  1257→    }
  1258→    let current = "";
  1259→    for (const word of para.split(/ +/)) {
  1260→      const candidate = current.length === 0 ? word : `${current} ${word}`;
  1261→      if (measure(candidate) <= maxWidth) {
  1262→        current = candidate;
  1263→        continue;
  1264→      }
  1265→      if (current.length > 0) {
  1266→        lines.push(current);
  1267→        current = "";
  1268→      }
  1269→      if (measure(word) <= maxWidth) {
  1270→        current = word;
  1271→      } else {
  1272→        // Hard-break the overlong word by characters (a single character
  1273→        // wider than maxWidth is still emitted — best effort).
  1274→        let chunk = "";
  1275→        for (const ch of word) {
  1276→          if (measure(chunk + ch) > maxWidth && chunk.length > 0) {
  1277→            lines.push(chunk);
  1278→            chunk = ch;
  1279→          } else {
  1280→            chunk += ch;
  1281→          }
  1282→        }
  1283→        current = chunk;
  1284→      }
  1285→    }
  1286→    lines.push(current);
  1287→  }
  1288→  return lines;
  1289→}

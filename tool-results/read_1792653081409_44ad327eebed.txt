     1→/**
     2→ * HTML to PDF (RB2-d — the 74th tool) — pure model. No DOM, no
     3→ * DOMParser (workers have none), no pdf-lib: the same parseHtml feeds
     4→ * the surface's live summary AND the worker's render path, so the
     5→ * preview numbers can never disagree with the output.
     6→ *
     7→ *  - tokenizeHtml: a hand-rolled linear tokenizer — tags (`<tag
     8→ *    attr="v">`, `</tag>`, `<br/>`), `<!-- comments -->` and
     9→ *    `<!doctype …>` / `<?…?>` skipped wholesale, and `<script>…</script>`
    10→ *    plus `<style>…</style>` dropped CONTENT AND ALL (case-insensitive
    11→ *    close scan). Text runs get entity decoding (a small named table +
    12→ *    `&#2361;` / `&#x0939;` numeric forms).
    13→ *  - HtmlAssembler: a tolerant tag-stack parser (never throws): void
    14→ *    elements auto-close (br, hr, img, input, meta, link…), `<p>text<p>
    15→ *    next` auto-closes, stray close tags are ignored, skipped subtrees
    16→ *    (svg, form controls, buttons, iframes, templates) drop text.
    17→ *    Produces the typed IR: heading/para/list-item/quote/rule/pre/table
    18→ *    blocks with inline runs {bold, italic, underline, mono, link}.
    19→ *  - parseHtmlRun: the worker-boundary coercion (parseWordRun pattern)
    20→ *    — clamp arrays, strip control chars, cap every length. A tampered
    21→ *    payload can never reach the renderer as a wrong type.
    22→ *  - needsNoto / isWinAnsiText: pure-WinAnsi content → non-embedded
    23→ *    base-14 Helvetica; ANY non-WinAnsi char (Devanagari, ₹ …) → the
    24→ *    self-hosted Noto Sans Devanagari subset (fontkit).
    25→ *  - decodeHtmlBytes / detectCharsetLabel: UTF-8 default, BOM + <meta
    26→ *    charset> aware (windows-1252 and friends via TextDecoder labels).
    27→ *
    28→ * Honest limits (mirrored in config.ts): CSS layout/styling is NOT
    29→ * applied — this is content extraction, not a browser engine; scripts,
    30→ * SVG and images are not carried beyond alt text in brackets; complex
    31→ * nested tables approximate (inner tables hoist as their own grid);
    32→ * Hindi bold renders regular (the Noto OFL face ships Regular only).
    33→ */
    34→

   112→const WINANSI_EXTRA = new Set([
   113→  0x20ac, 0x201a, 0x0192, 0x201e, 0x2026, 0x2020, 0x2021, 0x02c6, 0x2030,
   114→  0x0160, 0x2039, 0x0152, 0x017d, 0x2018, 0x2019, 0x201c, 0x201d, 0x2022,
   115→  0x2013, 0x2014, 0x02dc, 0x2122, 0x0161, 0x203a, 0x0153, 0x017e, 0x0178,
   116→]);
   117→
   118→/** true when every char is encodable in WinAnsi (CP1252) minus controls.
   119→ * `\n` and `\t` are layout-consumed (the renderer splits on them before
   120→ * drawText — the ppt tool's pinned lesson) so they pass even though they
   121→ * are control codes. */
   122→export function isWinAnsiText(s: string): boolean {
   123→  for (let i = 0; i < s.length; i++) {
   124→    const c = s.charCodeAt(i);
   125→    if (c === 0x0a /* \n */ || c === 0x09 /* \t */) continue;
   126→    if (c >= 0x20 && c <= 0x7e) continue;
   127→    if (c >= 0xa0 && c <= 0xff) continue;
   128→    if (WINANSI_EXTRA.has(c)) continue;
   129→    return false;
   130→  }
   131→  return true;
   132→}

   134→/** true when any drawn text needs the Devanagari (non-WinAnsi) font. */
   135→export function needsNoto(doc: ParsedHtml): boolean {
   136→  for (const b of doc.blocks) {
   137→    if (b.kind === "pre") {
   138→      if (!isWinAnsiText(b.text)) return true;
   139→    } else if (b.kind === "table") {
   140→      for (const row of b.grid) {
   141→        for (const cell of row) if (!isWinAnsiText(cell)) return true;
   142→      }
   143→    } else if (b.kind !== "rule") {
   144→      for (const r of b.runs) if (!isWinAnsiText(r.text)) return true;
   145→    }
   146→  }
   147→  return false;
   148→}

   209→const ENTITY_RE = /&(#[xX][0-9a-fA-F]+|#[0-9]+|[a-zA-Z][a-zA-Z0-9]*);/g;
   210→
   211→function decodeEntities(s: string): string {
   212→  if (!s.includes("&")) return s;
   213→  return s.replace(ENTITY_RE, (match, body: string) => {
   214→    if (body.charCodeAt(0) === 35 /* # */) {
   215→      const hex = body[1] === "x" || body[1] === "X";
   216→      const code = Number.parseInt(body.slice(hex ? 2 : 1), hex ? 16 : 10);
   217→      if (!Number.isFinite(code) || code < 0 || code > 0x10ffff) return match;
   218→      try {
   219→        return String.fromCodePoint(code);
   220→      } catch {
   221→        return match;
   222→      }
   223→    }
   224→    return NAMED_ENTITIES[body] ?? match;
   225→  });
   226→}

   232→function latin1Of(bytes: Uint8Array, max: number): string {
   233→  let out = "";
   234→  const n = Math.min(bytes.length, max);
   235→  for (let i = 0; i < n; i++) out += String.fromCharCode(bytes[i]);
   236→  return out;

   239→/** Reads `<meta charset=…>` / `content="text/html; charset=…"` from a head. */
   240→export function detectCharsetLabel(headLatin1: string): string {
   241→  const re = /<meta[^>]*?charset\s*=\s*["']?\s*([a-zA-Z0-9_.:-]+)/i;
   242→  const m = re.exec(headLatin1);
   243→  const label = m ? m[1].toLowerCase() : "";
   244→  if (!label) return "utf-8";
   245→  try {
   246→    new TextDecoder(label); // throws RangeError on unsupported labels
   247→    return label;
   248→  } catch {
   249→    return "utf-8";
   250→  }
   251→}
   252→
   253→/**
   254→ * Bytes → HTML source: UTF-8/UTF-16 BOM honoured, `<meta charset>`
   255→ * respected (windows-1252 and every other TextDecoder label), malformed
   256→ * sequences tolerated. Pure string work — usable on any thread.
   257→ */
   258→export function decodeHtmlBytes(bytes: Uint8Array): string {
   259→  if (
   260→    bytes.length >= 3 &&
   261→    bytes[0] === 0xef &&
   262→    bytes[1] === 0xbb &&
   263→    bytes[2] === 0xbf
   264→  ) {
   265→    return new TextDecoder("utf-8", { fatal: false }).decode(bytes.subarray(3));
   266→  }
   267→  if (bytes.length >= 2 && bytes[0] === 0xff && bytes[1] === 0xfe) {
   268→    return new TextDecoder("utf-16le", { fatal: false }).decode(bytes.subarray(2));
   269→  }
   270→  if (bytes.length >= 2 && bytes[0] === 0xfe && bytes[1] === 0xff) {
   271→    return new TextDecoder("utf-16be", { fatal: false }).decode(bytes.subarray(2));
   272→  }
   273→  const label = detectCharsetLabel(latin1Of(bytes, 2048));
   274→  try {
   275→    return new TextDecoder(label, { fatal: false }).decode(bytes);
   276→  } catch {
   277→    return new TextDecoder("utf-8", { fatal: false }).decode(bytes);
   278→  }
   279→}

   281→/**
   282→ * Honest sniff for the drop zone / worker gate: the source must contain
   283→ * a real tag open (`</?[a-zA-Z]…`) or a doctype. A plain-text file
   284→ * without any tag shape is rejected (documented choice) — the model
   285→ * itself would still treat bare text as one paragraph.
   286→ */
   287→export function looksLikeHtml(source: string): boolean {
   288→  const head = source.slice(0, 4096);
   289→  if (/<!doctype/i.test(head)) return true;
   290→  return /<\/?[a-zA-Z][a-zA-Z0-9:_-]*(\s|\/?>|>)/.test(head);
   291→}

   297→export interface HtmlTokenHandlers {
   298→  open(name: string, attrs: Record<string, string>, selfClosing: boolean): void;
   299→  close(name: string): void;
   300→  /** entities already decoded; raw (pre needs un-collapsed text). */

   303→
   304→const TAG_NAME_STOP = new Set([
   305→  " ",
   306→  "\t",
   307→  "\r",
   308→  "\n",
   309→  "\f",
   310→  "/",
   311→  ">",
   312→  "<",
   313→  "=",
   314→]);
   315→const ATTR_NAME_STOP = new Set([" ", "\t", "\r", "\n", "\f", "/", ">", "="]);
   316→
   317→function isWs(ch: string): boolean {
   318→  return ch === " " || ch === "\t" || ch === "\r" || ch === "\n" || ch === "\f";
   319→}
   320→
   321→/**
   322→ * Case-insensitive "</name" scan at word boundary (the char after the
   323→ * name must be whitespace, "/", or ">") — treats EOF as not found.
   324→ */
   325→function indexOfCloseTag(s: string, tagName: string, from: number): number {
   326→  const need = tagName.length + 2; // "</" + name
   327→  outer: for (let i = from; i + need <= s.length; i++) {
   328→    if (s.charCodeAt(i) !== 60 /* < */ || s.charCodeAt(i + 1) !== 47 /* / */) {
   329→      continue;
   330→    }
   331→    for (let k = 0; k < tagName.length; k++) {
   332→      let c = s.charCodeAt(i + 2 + k);
   333→      if (c >= 65 && c <= 90) c += 32;
   334→      if (c !== tagName.charCodeAt(k)) continue outer;
   335→    }
   336→    const after = s[i + need];
   337→    if (
   338→      after === undefined ||
   339→      after === ">" ||
   340→      after === "/" ||
   341→      isWs(after)
   342→    ) {
   343→      return i;
   344→    }
   345→  }
   346→  return -1;
   347→}
   348→
   349→/**
   350→ * Tolerant linear tokenizer: real elements emit open/close (self-closing
   351→ * non-void elements still emit open — the HTML5 rule, the slash is
   352→ * ignored outside foreign content), comments / doctypes / processing
   353→ * instructions are skipped, and `<script>` / `<style>` elements are
   354→ * dropped WHOLESALE (content included). Everything else between tags is
   355→ * text with entity decoding. Malformed input degrades instead of
   356→ * throwing; script/style tokens are never emitted.
   357→ */
   358→export function tokenizeHtml(source: string, h: HtmlTokenHandlers): void {
   359→  const n = source.length;
   360→  let i = 0;
   361→  while (i < n) {
   362→    const lt = source.indexOf("<", i);
   363→    if (lt < 0) {
   364→      if (i < n) h.text(decodeEntities(source.slice(i)));
   365→      break;
   366→    }
   367→    if (lt > i) h.text(decodeEntities(source.slice(i, lt)));
   368→
   369→    const c1 = source.charCodeAt(lt + 1);
   370→    if (c1 === 33 /* ! */) {
   371→      if (source.startsWith("<!--", lt)) {
   372→        const end = source.indexOf("-->", lt);
   373→        i = end < 0 ? n : end + 3;
   374→        continue;
   375→      }
   376→      const end = source.indexOf(">", lt); // doctype / CDATA-ish
   377→      i = end < 0 ? n : end + 1;
   378→      continue;
   379→    }
   380→    if (c1 === 63 /* ? */) {
   381→      const end = source.indexOf(">", lt);
   382→      i = end < 0 ? n : end + 1;
   383→      continue;
   384→    }
   385→    if (c1 === 47 /* / */) {
   386→      let j = lt + 2;
   387→      const nameStart = j;
   388→      while (j < n && !TAG_NAME_STOP.has(source[j])) j += 1;
   389→      const name = source.slice(nameStart, j).toLowerCase();
   390→      const end = source.indexOf(">", j);
   391→      i = end < 0 ? n : end + 1;
   392→      if (name) h.close(name);
   393→      continue;
   394→    }
   395→    if (!(c1 >= 97 && c1 <= 122) && !(c1 >= 65 && c1 <= 90)) {
   396→      // "<" that isn't a tag at all — emit it as text and move on.
   397→      h.text("<");
   398→      i = lt + 1;
   399→      continue;
   400→    }
   401→
   402→    // open tag: scan the name
   403→    let j = lt + 1;
   404→    const nameStart = j;
   405→    while (j < n && !TAG_NAME_STOP.has(source[j])) j += 1;
   406→    const name = source.slice(nameStart, j).toLowerCase();
   407→
   408→    // attributes until ">"
   409→    const attrs: Record<string, string> = {};
   410→    let selfClosing = false;
   411→    let guard = 0;
   412→    while (j < n && guard < 20_000) {
   413→      guard += 1;
   414→      while (j < n && isWs(source[j])) j += 1;
   415→      const c = source.charCodeAt(j);
   416→      if (c === 62 /* > */) {
   417→        j += 1;
   418→        break;
   419→      }
   420→      if (c === 47 /* / */) {
   421→        if (source.charCodeAt(j + 1) === 62 /* > */) selfClosing = true;
   422→        j += 1;
   423→        continue;
   424→      }
   425→      if (j >= n) break;
   426→      const aStart = j;
   427→      while (j < n && !ATTR_NAME_STOP.has(source[j])) j += 1;
   428→      const aName = source.slice(aStart, j).toLowerCase();
   429→      while (j < n && isWs(source[j])) j += 1;
   430→      let aVal = "";
   431→      if (source.charCodeAt(j) === 61 /* = */) {
   432→        j += 1;
   433→        while (j < n && isWs(source[j])) j += 1;
   434→        const q = source.charCodeAt(j);
   435→        if (q === 34 /* " */ || q === 39 /* ' */) {
   436→          j += 1;
   437→          const vStart = j;
   438→          while (j < n && source.charCodeAt(j) !== q) j += 1;
   439→          aVal = source.slice(vStart, j);
   440→          j += 1;
   441→        } else {
   442→          const vStart = j;
   443→          while (j < n && !isWs(source[j]) && source.charCodeAt(j) !== 62) {
   444→            j += 1;
   445→          }
   446→          aVal = source.slice(vStart, j);
   447→        }
   448→      }
   449→      if (aName) attrs[aName] = decodeEntities(aVal);
   450→    }
   451→
   452→    if (name === "script" || name === "style") {
   453→      // raw-text element: skip content AND the closing tag entirely
   454→      const close = indexOfCloseTag(source, name, j);
   455→      if (close < 0) break;
   456→      const gt = source.indexOf(">", close);
   457→      i = gt < 0 ? n : gt + 1;
   458→      continue;
   459→    }
   460→
   461→    h.open(name, attrs, selfClosing);
   462→    i = j;
   463→  }
   464→}

   497→/** HTML whitespace collapse — the explicit set, so &nbsp; (\u00A0) survives. */
   498→function collapseWs(s: string): string {
   499→  let out = "";
   500→  let inWs = false;
   501→  for (let i = 0; i < s.length; i++) {
   502→    const ch = s[i];
   503→    if (ch === " " || ch === "\t" || ch === "\r" || ch === "\n" || ch === "\f") {
   504→      if (!inWs && out.length > 0) out += " ";
   505→      inWs = true;
   506→    } else {
   507→      out += ch;
   508→      inWs = false;
   509→    }
   510→  }
   511→  return out;
   512→}

   545→  runs: HtmlRun[];
   546→  /** 1-based stack depth of the frame that owns this collector. */
   547→  owner: number;
   548→  textLen: number;
   549→}
   550→
   551→interface TableFrame {
   552→  rows: string[][];
   553→  row: string[] | null;
   554→  cell: string | null;
   555→  inHeaderCell: boolean;
   556→  headerRow: boolean;
   557→  firstRow: boolean;
   558→}
   559→
   560→interface ListFrame {
   561→  ordered: boolean;
   562→  counter: number;
   563→}
   564→
   565→/* ------------------------------------------------------------------ */
   566→/* Assembler                                                            */
   567→/* ------------------------------------------------------------------ */
   568→
   569→class HtmlAssembler {
   570→  readonly blocks: HtmlBlock[] = [];
   571→  title = "";
   572→  truncated = false;
   573→
   574→  /* --- walker state --- */
   575→  private stack: string[] = [];
   576→  private skipDepth = -1;
   577→  private inHead = false;
   578→  private quoteDepth = 0;
   579→  private titleDepth = -1;
   580→  private titleText = "";
   581→
   582→  /* --- pre blocks --- */
   583→  private preDepth = 0;
   584→  private preText = "";
   585→
   586→  /* --- inline style state --- */
   587→  private boldDepth = 0;
   588→  private italicDepth = 0;
   589→  private underlineDepth = 0;
   590→  private monoDepth = 0;
   591→  private linkStack: string[] = [];
   592→
   593→  /* --- collector / lists / tables --- */
   594→  private collector: Collector | null = null;
   595→  private listStack: ListFrame[] = [];
   596→  private tableStack: TableFrame[] = [];
   597→
   598→  private pushBlock(b: HtmlBlock): void {
   599→    if (this.blocks.length >= MAX_BLOCKS) {
   600→      this.truncated = true;
   601→      return;
   602→    }
   603→    this.blocks.push(b);
   604→  }
   605→
   606→  private style(): RunStyle {
   607→    const link = this.linkStack[this.linkStack.length - 1];
   608→    return {
   609→      bold: this.boldDepth > 0,
   610→      italic: this.italicDepth > 0,
   611→      underline: this.underlineDepth > 0,
   612→      mono: this.monoDepth > 0,
   613→      link: link === "" || link === undefined ? undefined : link,
   614→    };
   615→  }
   616→
   617→  /** the open td/th frame, if any. */
   618→  private openCell(): TableFrame | null {
   619→    const top = this.tableStack[this.tableStack.length - 1];
   620→    return top !== undefined && top.cell !== null ? top : null;
   621→  }
   622→
   623→  private ensureCollector(): Collector {
   624→    if (this.collector === null) {
   625→      // implicit paragraph: div/body text without a <p>
   626→      this.collector = {
   627→        kind: this.quoteDepth > 0 ? "quote" : "para",
   628→        level: 1,
   629→        depth: 0,
   630→        ordered: false,
   631→        number: 0,
   632→        runs: [],
   633→        owner: this.stack.length,
   634→        textLen: 0,
   635→      };
   636→    }
   637→    return this.collector;
   638→  }
   639→
   640→  private appendCellText(frame: TableFrame, raw: string): void {
   641→    const t = collapseWs(raw);
   642→    if (t === "") return;
   643→    if (frame.cell === null) frame.cell = "";
   644→    if (frame.cell === "") {
   645→      frame.cell = t;
   646→    } else if (frame.cell.endsWith(" ") || t.startsWith(" ")) {
   647→      frame.cell += t;
   648→    } else {
   649→      frame.cell += " " + t;
   650→    }
   651→  }
   652→
   653→  private appendRunText(raw: string): void {
   654→    let t = hindiNorm(collapseWs(raw));
   655→    if (t === "") return;
   656→    const style = this.style();
   657→    const col = this.ensureCollector();
   658→    const last = col.runs[col.runs.length - 1];
   659→    if (last && last.text.endsWith(" ") && t.startsWith(" ")) t = t.slice(1);
   660→    if (col.textLen >= MAX_PARA_TEXT) {
   661→      this.truncated = true;
   662→      return;
   663→    }
   664→    if (t.length > MAX_RUN_TEXT) {
   665→      t = t.slice(0, MAX_RUN_TEXT);
   666→      this.truncated = true;
   667→    }
   668→    col.textLen += t.length;
   669→    if (last && runMatchesStyle(last, style)) {
   670→      last.text += t;
   671→    } else if (col.runs.length < MAX_RUNS_PER_PARA) {
   672→      col.runs.push({
   673→        text: t,
   674→        bold: style.bold,
   675→        italic: style.italic,
   676→        underline: style.underline,
   677→        mono: style.mono,
   678→        link: style.link,
   679→      });
   680→    } else {
   681→      this.truncated = true;
   682→    }
   683→  }
   684→
   685→  /** append a soft-break run (<br>) to the active collector. */
   686→  private appendBreak(): void {
   687→    const style = this.style();
   688→    const col = this.ensureCollector();
   689→    const last = col.runs[col.runs.length - 1];
   690→    if (last && runMatchesStyle(last, style)) {
   691→      last.text += "\n";
   692→    } else if (col.runs.length < MAX_RUNS_PER_PARA) {
   693→      col.runs.push({
   694→        text: "\n",
   695→        bold: style.bold,
   696→        italic: style.italic,
   697→        underline: style.underline,
   698→        mono: style.mono,
   699→        link: style.link,
   700→      });
   701→    }
   702→  }
   703→
   704→  /** flush the active collector into the block list. */
   705→  private flushCollector(): void {
   706→    const col = this.collector;
   707→    this.collector = null;
   708→    if (!col) return;
   709→    const runs = normalizeRuns(col.runs);
   710→    if (runs.length === 0) return;
   711→    switch (col.kind) {
   712→      case "heading":
   713→        this.pushBlock({ kind: "heading", level: col.level, runs });
   714→        break;
   715→      case "quote":
   716→        this.pushBlock({ kind: "quote", runs });
   717→        break;
   718→      case "list-item":
   719→        this.pushBlock({
   720→          kind: "list-item",
   721→          depth: Math.min(8, Math.max(0, col.depth)),
   722→          ordered: col.ordered,
   723→          number: col.number,
   724→          runs,
   725→        });
   726→        break;
   727→      default:
   728→        this.pushBlock({ kind: "para", runs });
   729→        break;
   730→    }
   731→  }
   732→
   733→  /** flush when the collector's owning frame has been popped. */
   734→  private flushIfOrphaned(): void {
   735→    if (this.collector && this.collector.owner > this.stack.length) {
   736→      this.flushCollector();
   737→    }
   738→  }
   739→
   740→  private finalizeTitle(): void {
   741→    if (this.titleDepth !== -1) {
   742→      this.title = cleanText(this.titleText.trim(), MAX_TITLE_TEXT);
   743→      this.titleDepth = -1;
   744→      this.titleText = "";
   745→    }
   746→  }
   747→
   748→  readonly handlers: HtmlTokenHandlers = {
   749→    open: (name, attrs, selfClosing) => {
   750→      const isVoid = VOID_ELEMENTS.has(name);
   751→      if (isVoid) {
   752→        if (selfClosing) {
   753→          /* "<br/>" === "<br>" */
   754→        }
   755→      } else if (selfClosing && SKIP_ELEMENTS.has(name)) {
   756→        // foreign-content self-closing ("<svg/>") — an empty element
   757→        return;
   758→      } else {
   759→        // HTML5 rule: the slash on non-void elements is IGNORED
   760→        this.stack.push(name);
   761→      }
   762→      const depth = this.stack.length;
   763→
   764→      if (this.skipDepth !== -1) return;
   765→
   766→      if (SKIP_ELEMENTS.has(name) && !isVoid) {
   767→        this.skipDepth = depth;
   768→        return;
   769→      }
   770→
   771→      switch (name) {
   772→        case "html":
   773→        case "body":
   774→          this.inHead = false;
   775→          break;
   776→        case "head":
   777→          this.inHead = true;
   778→          break;
   779→        case "title":
   780→          this.titleDepth = depth;
   781→          this.titleText = "";
   782→          break;
   783→        case "br":
   784→          if (this.preDepth > 0) {
   785→            this.preText += "\n";
   786→          } else if (this.openCell()) {
   787→            /* cell breaks collapse to the space the joiner adds */
   788→          } else {
   789→            this.appendBreak();
   790→          }
   791→          break;
   792→        case "hr":
   793→          this.flushCollector();
   794→          this.pushBlock({ kind: "rule" });
   795→          break;
   796→        case "img": {
   797→          const alt = typeof attrs.alt === "string" ? collapseWs(attrs.alt).trim() : "";
   798→          if (alt && this.preDepth === 0 && !this.openCell()) {
   799→            const col = this.ensureCollector();
   800→            if (col.runs.length < MAX_RUNS_PER_PARA) {
   801→              col.runs.push({
   802→                text: `[${alt.slice(0, 200)}]`,
   803→                bold: false,
   804→                italic: true,
   805→                underline: false,
   806→                mono: false,
   807→              });
   808→            }
   809→          }
   810→          break;
   811→        }
   812→        case "a":
   813→          this.linkStack.push(String(attrs.href ?? "").slice(0, 500));
   814→          break;
   815→        case "b":
   816→        case "strong":
   817→          this.boldDepth += 1;
   818→          break;
   819→        case "i":
   820→        case "em":
   821→        case "cite":
   822→        case "var":
   823→          this.italicDepth += 1;
   824→          break;
   825→        case "u":
   826→        case "ins":
   827→          this.underlineDepth += 1;
   828→          break;
   829→        case "code":
   830→        case "kbd":
   831→        case "samp":
   832→        case "tt":
   833→          this.monoDepth += 1;
   834→          break;
   835→        case "ul":
   836→        case "ol": {
   837→          this.flushCollector();
   838→          const start = Number.parseInt(String(attrs.start ?? ""), 10);
   839→          this.listStack.push({
   840→            ordered: name === "ol",
   841→            // first <li> increments to the printed number
   842→            counter: Number.isFinite(start) && start > -999 ? start - 1 : 0,
   843→          });
   844→          break;
   845→        }
   846→        case "li": {
   847→          const list = this.listStack[this.listStack.length - 1];
   848→          if (list) list.counter += 1;
   849→          this.flushCollector();
   850→          this.collector = {
   851→            kind: "list-item",
   852→            level: 1,
   853→            depth: Math.max(0, this.listStack.length - 1),
   854→            ordered: list ? list.ordered : false,
   855→            number: list ? list.counter : 1,
   856→            runs: [],
   857→            owner: depth,
   858→            textLen: 0,
   859→          };
   860→          break;
   861→        }
   862→        case "dt":
   863→        case "dd": {
   864→          this.flushCollector();
   865→          this.collector = {
   866→            kind: "list-item",
   867→            level: 1,
   868→            depth: 0,
   869→            ordered: false,
   870→            number: 0,
   871→            runs: [],
   872→            owner: depth,
   873→            textLen: 0,
   874→          };
   875→          break;
   876→        }
   877→        case "p":
   878→        case "h1":
   879→        case "h2":
   880→        case "h3":
   881→        case "h4":
   882→        case "h5":
   883→        case "h6": {
   884→          if (this.openCell()) break; // paragraphs inside cells → cell text
   885→          const col = this.collector;
   886→          if (
   887→            col &&
   888→            col.kind === "list-item" &&
   889→            this.stack.length >= col.owner &&
   890→            name === "p"
   891→          ) {
   892→            // paragraph inside the list item that owns the collector —
   893→            // merge (soft break between them, the markdown approach)
   894→            const last = col.runs[col.runs.length - 1];
   895→            if (last && !last.text.endsWith("\n")) last.text += "\n";
   896→            break;
   897→          }
   898→          this.flushCollector();
   899→          this.collector = {
   900→            kind:
   901→              HEADING_TAGS[name] !== undefined
   902→                ? "heading"
   903→                : this.quoteDepth > 0
   904→                  ? "quote"
   905→                  : "para",
   906→            level: HEADING_TAGS[name] ?? 1,
   907→            depth: 0,
   908→            ordered: false,
   909→            number: 0,
   910→            runs: [],
   911→            owner: depth,
   912→            textLen: 0,
   913→          };
   914→          break;
   915→        }
   916→        case "blockquote":
   917→          this.flushCollector();
   918→          this.quoteDepth += 1;
   919→          break;
   920→        case "pre":
   921→          this.flushCollector();
   922→          this.preDepth += 1;
   923→          if (this.preDepth === 1) this.preText = "";
   924→          break;
   925→        case "table":
   926→          this.flushCollector();
   927→          this.tableStack.push({
   928→            rows: [],
   929→            row: null,
   930→            cell: null,
   931→            inHeaderCell: false,
   932→            headerRow: false,
   933→            firstRow: true,
   934→          });
   935→          break;
   936→        case "tr": {
   937→          const top = this.tableStack[this.tableStack.length - 1];
   938→          if (top) {
   939→            top.row = [];
   940→            top.inHeaderCell = false;
   941→          }
   942→          break;
   943→        }
   944→        case "td":
   945→        case "th": {
   946→          const top = this.tableStack[this.tableStack.length - 1];
   947→          if (top) {
   948→            if (top.row === null) top.row = []; // stray cell: implicit row
   949→            top.cell = "";
   950→            top.inHeaderCell = name === "th";
   951→          }
   952→          break;
   953→        }
   954→        default:
   955→          if (BLOCK_ELEMENTS.has(name)) this.flushCollector();
   956→          break;
   957→      }
   958→    },
   959→
   960→    text: (value) => {
   961→      if (value === "") return;
   962→      if (this.skipDepth !== -1) return;
   963→
   964→      if (this.titleDepth !== -1) {
   965→        this.titleText += collapseWs(value);
   966→        return;
   967→      }
   968→      if (this.inHead) return;
   969→      if (this.preDepth > 0) {
   970→        if (this.preText.length < MAX_PRE_TEXT) {
   971→          this.preText = (this.preText + hindiNorm(value)).slice(0, MAX_PRE_TEXT);
   972→        } else {
   973→          this.truncated = true;
   974→        }
   975→        return;
   976→      }
   977→      const cellFrame = this.openCell();
   978→      if (cellFrame) {
   979→        this.appendCellText(cellFrame, value);
   980→        return;
   981→      }
   982→      this.appendRunText(value);
   983→    },
   984→
   985→    close: (name) => {
   986→      const k = this.stack.lastIndexOf(name);
   987→      if (k < 0) return; // stray close — tolerate
   988→
   989→      if (this.skipDepth !== -1) {
   990→        if (k === this.skipDepth - 1) {
   991→          // the skip root itself closes
   992→          this.stack.length = k;
   993→          this.skipDepth = -1;
   994→          this.flushIfOrphaned();
   995→          return;
   996→        }
   997→        if (k >= this.skipDepth) {
   998→          // inner close inside the skipped subtree — no side effects
   999→          this.stack.length = k;
  1000→          return;
  1001→        }
  1002→        // auto-close popped past the skip root (unclosed skip element)
  1003→        this.stack.length = k;
  1004→        this.skipDepth = -1;
  1005→        this.applyClose(name);
  1006→        this.flushIfOrphaned();
  1007→        return;
  1008→      }
  1009→
  1010→      this.stack.length = k; // pop name + auto-close anything above it
  1011→      this.applyClose(name);
  1012→      this.flushIfOrphaned();
  1013→    },
  1014→  };
  1015→
  1016→  /** close-tag effects (skip-aware callers only). */
  1017→  private applyClose(name: string): void {
  1018→    switch (name) {
  1019→      case "head":
  1020→        this.inHead = false;
  1021→        this.finalizeTitle();
  1022→        break;
  1023→      case "title":
  1024→        this.finalizeTitle();
  1025→        break;
  1026→      case "a":
  1027→        if (this.linkStack.length > 0) this.linkStack.pop();
  1028→        break;
  1029→      case "b":
  1030→      case "strong":
  1031→        this.boldDepth = Math.max(0, this.boldDepth - 1);
  1032→        break;
  1033→      case "i":
  1034→      case "em":
  1035→      case "cite":
  1036→      case "var":
  1037→        this.italicDepth = Math.max(0, this.italicDepth - 1);
  1038→        break;
  1039→      case "u":
  1040→      case "ins":
  1041→        this.underlineDepth = Math.max(0, this.underlineDepth - 1);
  1042→        break;
  1043→      case "code":
  1044→      case "kbd":
  1045→      case "samp":
  1046→      case "tt":
  1047→        this.monoDepth = Math.max(0, this.monoDepth - 1);
  1048→        break;
  1049→      case "blockquote":
  1050→        this.quoteDepth = Math.max(0, this.quoteDepth - 1);
  1051→        break;
  1052→      case "pre":
  1053→        this.preDepth = Math.max(0, this.preDepth - 1);
  1054→        if (this.preDepth === 0) {
  1055→          let text = this.preText
  1056→            .replace(/\r\n?/g, "\n")
  1057→            .replace(/^\n/, "")
  1058→            .replace(/[ \t]+$/, "");
  1059→          if (text.length > MAX_PRE_TEXT) {
  1060→            text = text.slice(0, MAX_PRE_TEXT);
  1061→            this.truncated = true;
  1062→          }
  1063→          if (text.trim().length > 0) this.pushBlock({ kind: "pre", text });
  1064→          this.preText = "";
  1065→        }
  1066→        break;
  1067→      case "td":
  1068→      case "th": {
  1069→        const top = this.tableStack[this.tableStack.length - 1];
  1070→        if (top && top.cell !== null) {
  1071→          const cell = hindiNorm(cleanText(top.cell.trim(), MAX_CELL_TEXT));
  1072→          if (top.row && top.row.length < MAX_TABLE_COLS) {
  1073→            top.row.push(cell);
  1074→            if (top.inHeaderCell && top.firstRow) top.headerRow = true;
  1075→          } else if (top.row) {
  1076→            this.truncated = true;
  1077→          }
  1078→          top.cell = null;
  1079→          top.inHeaderCell = false;
  1080→        }
  1081→        break;
  1082→      }
  1083→      case "tr": {
  1084→        const top = this.tableStack[this.tableStack.length - 1];
  1085→        if (top && top.row !== null) {
  1086→          if (top.rows.length < MAX_TABLE_ROWS) {
  1087→            if (top.row.length > 0) top.rows.push(top.row);
  1088→          } else {
  1089→            this.truncated = true;
  1090→          }
  1091→          top.firstRow = false;
  1092→          top.row = null;
  1093→        }
  1094→        break;
  1095→      }
  1096→      case "table": {
  1097→        const top = this.tableStack.pop();
  1098→        if (!top) break;
  1099→        // A nested table (closed while an outer one is open) hoists as
  1100→        // its own grid before the outer table completes — approximate,
  1101→        // documented in the FAQ.
  1102→        if (top.rows.length > 0) {
  1103→          this.pushBlock({
  1104→            kind: "table",
  1105→            grid: top.rows,
  1106→            headerRow: top.headerRow || undefined,
  1107→          });
  1108→        }
  1109→        break;
  1110→      }
  1111→      case "ul":
  1112→      case "ol":
  1113→        if (this.listStack.length > 0) this.listStack.pop();
  1114→        break;
  1115→      default:
  1116→        break;
  1117→    }
  1118→  }
  1119→
  1120→  finish(): ParsedHtml {
  1121→    this.finalizeTitle();
  1122→    this.flushCollector();
  1123→    return { title: this.title, blocks: this.blocks, truncated: this.truncated };
  1124→  }
  1125→}
  1126→
  1127→/** merge same-style neighbours, trim paragraph edges, drop empties. */
  1128→function normalizeRuns(runs: HtmlRun[]): HtmlRun[] {
  1129→  const out: HtmlRun[] = [];
  1130→  for (const r of runs) {
  1131→    if (r.text === "") continue;
  1132→    const last = out[out.length - 1];
  1133→    if (last && runMatchesStyle(last, r)) {
  1134→      last.text += r.text;
  1135→    } else {
  1136→      out.push({ ...r });
  1137→    }
  1138→  }
  1139→  if (out.length > 0) {
  1140→    const first = out[0];
  1141→    first.text = first.text.replace(/^[ \t]+/, "");
  1142→    const last = out[out.length - 1];
  1143→    last.text = last.text
  1144→      .replace(/[ \t]+$/, "")
  1145→      .replace(/\n+$/, "");
  1146→  }
  1147→  return out.filter((r) => r.text !== "");
  1148→}

  1150→/* ------------------------------------------------------------------ */
  1151→/* parseHtml                                                            */
  1152→/* ------------------------------------------------------------------ */
  1153→
  1154→/**
  1155→ * Tolerant HTML → typed IR. Never throws: garbage degrades to fewer
  1156→ * blocks. The title comes from <title>; script/style/comments/doctype
  1157→ * never contribute text.
  1158→ */
  1159→export function parseHtml(source: string): ParsedHtml {
  1160→  const asm = new HtmlAssembler();
  1161→  try {
  1162→    tokenizeHtml(source, asm.handlers);
  1163→  } catch {
  1164→    return emptyDoc();
  1165→  }
  1166→  return asm.finish();
  1167→}

  1172→
  1173→/** Honest stats derived from the blocks alone (parse + coercion agree). */
  1174→export function stats(doc: ParsedHtml): HtmlDocStats & { title: string } {
  1175→  let headings = 0;
  1176→  let paras = 0;
  1177→  let listItems = 0;
  1178→  let quotes = 0;
  1179→  let preBlocks = 0;
  1180→  let tables = 0;
  1181→  let words = 0;
  1182→  let chars = 0;
  1183→  const count = (s: string): void => {
  1184→    chars += s.length;
  1185→    words += s.split(/\s+/).filter(Boolean).length;
  1186→  };
  1187→  for (const b of doc.blocks) {
  1188→    switch (b.kind) {
  1189→      case "heading":
  1190→        headings += 1;
  1191→        for (const r of b.runs) count(r.text);
  1192→        break;
  1193→      case "para":
  1194→        paras += 1;
  1195→        for (const r of b.runs) count(r.text);
  1196→        break;
  1197→      case "list-item":
  1198→        listItems += 1;
  1199→        for (const r of b.runs) count(r.text);
  1200→        break;
  1201→      case "quote":
  1202→        quotes += 1;
  1203→        for (const r of b.runs) count(r.text);
  1204→        break;
  1205→      case "pre":
  1206→        preBlocks += 1;
  1207→        count(b.text);
  1208→        break;
  1209→      case "table":
  1210→        tables += 1;
  1211→        for (const row of b.grid) for (const cell of row) count(cell);
  1212→        break;
  1213→      default:
  1214→        break;
  1215→    }
  1216→  }
  1217→  return {
  1218→    title: doc.title,
  1219→    headings,
  1220→    paras,
  1221→    listItems,
  1222→    quotes,
  1223→    preBlocks,
  1224→    tables,
  1225→    blockCount: doc.blocks.length,
  1226→    wordCount: words,
  1227→    charCount: chars,
  1228→  };
  1229→}

  1231→/** Heading pt sizes at the 11 pt default (h1 22 → h6 12, scaled by base). */
  1232→export const HEADING_SIZES: readonly number[] = [22, 17, 14.5, 13, 12.3, 12];
  1233→
  1234→/**
  1235→ * Pure-math page estimate (the surface badge): the render geometry with
  1236→ * an average-glyph-width wrap model — honest "≈" numbers, never a lie.
  1237→ */
  1238→export function estimatePages(doc: ParsedHtml, baseFontSize = 11): number {
  1239→  const contentW = 595.28 - 112;
  1240→  const usable = 841.89 - 112;
  1241→  const cpl = Math.max(20, contentW / (baseFontSize * 0.5));
  1242→  const lh = baseFontSize * 1.36;
  1243→  let total = 0;
  1244→  for (const b of doc.blocks) {
  1245→    switch (b.kind) {
  1246→      case "heading": {
  1247→        const size = HEADING_SIZES[b.level - 1] * (baseFontSize / 11);
  1248→        const len = b.runs.reduce((a, r) => a + r.text.length, 0);
  1249→        total += Math.max(1, Math.ceil(len / cpl)) * size * 1.3 + size * 0.8;
  1250→        break;
  1251→      }
  1252→      case "para":
  1253→      case "quote": {
  1254→        const len = b.runs.reduce(
  1255→          (a, r) => a + r.text.replace(/\n/g, " ").length,
  1256→          0
  1257→        );
  1258→        const breaks = b.runs.reduce((a, r) => a + (r.text.split("\n").length - 1), 0);
  1259→        total += Math.max(1 + breaks, Math.ceil(len / cpl)) * lh + 4;
  1260→        break;
  1261→      }
  1262→      case "list-item": {
  1263→        const len = b.runs.reduce((a, r) => a + r.text.length, 0);
  1264→        total += Math.max(1, Math.ceil(len / (cpl - 4))) * lh + 2;
  1265→        break;
  1266→      }
  1267→      case "rule":
  1268→        total += 14;
  1269→        break;
  1270→      case "pre": {
  1271→        const lines = b.text.split("\n").length;
  1272→        total += lines * (baseFontSize * 0.9 * 1.35) + 12;
  1273→        break;
  1274→      }
  1275→      case "table": {
  1276→        for (const row of b.grid) {
  1277→          let cellLines = 1;
  1278→          for (const cell of row) {
  1279→            cellLines = Math.max(cellLines, Math.ceil(cell.length / 14));
  1280→          }
  1281→          total += cellLines * (baseFontSize * 0.82 * 1.35) + 6;
  1282→        }
  1283→        total += 8;
  1284→        break;
  1285→      }
  1286→      default:
  1287→        break;
  1288→    }
  1289→  }
  1290→  return Math.max(1, Math.ceil(total / usable));
  1291→}

  1297→function coerceHeadingLevel(v: unknown): 1 | 2 | 3 | 4 | 5 | 6 | undefined {
  1298→  const n = typeof v === "number" ? v : Number.parseInt(String(v), 10);
  1299→  if (!Number.isFinite(n)) return undefined;
  1300→  if (n >= 1 && n <= 6) return n as 1 | 2 | 3 | 4 | 5 | 6;
  1301→  return undefined;
  1302→}
  1303→
  1304→function coerceDepth(v: unknown): number {
  1305→  const n = typeof v === "number" ? v : Number.parseInt(String(v), 10);
  1306→  if (!Number.isFinite(n) || n < 0) return 0;
  1307→  return Math.min(8, Math.floor(n));
  1308→}
  1309→
  1310→function coerceRuns(raw: unknown): HtmlRun[] {
  1311→  const list = Array.isArray(raw) ? raw : [];
  1312→  const runs: HtmlRun[] = [];
  1313→  let textLen = 0;
  1314→  for (const ro of list) {
  1315→    if (runs.length >= MAX_RUNS_PER_PARA) break;
  1316→    if (typeof ro !== "object" || ro === null) continue;
  1317→    const r = ro as Record<string, unknown>;
  1318→    let text = hindiNorm(cleanText(r.text, MAX_RUN_TEXT));
  1319→    if (textLen + text.length > MAX_PARA_TEXT) {
  1320→      text = text.slice(0, Math.max(0, MAX_PARA_TEXT - textLen));
  1321→    }
  1322→    if (text === "") continue;
  1323→    textLen += text.length;
  1324→    const link =
  1325→      typeof r.link === "string" && r.link !== "" ? r.link.slice(0, 500) : undefined;
  1326→    runs.push({
  1327→      text,
  1328→      bold: boolOf(r.bold),
  1329→      italic: boolOf(r.italic),
  1330→      underline: boolOf(r.underline),
  1331→      mono: boolOf(r.mono),
  1332→      link,
  1333→    });
  1334→  }
  1335→  return runs;
  1336→}
  1337→
  1338→/** truthy coercion for JSON-ish boundaries: true, 1 and 1.0 count;
  1339→ * strings and everything else fall back to false. */
  1340→function boolOf(v: unknown): boolean {
  1341→  return v === true || (typeof v === "number" && Number.isFinite(v) && v !== 0);
  1342→}
  1343→
  1344→/**
  1345→ * Full defensive coercion of a serialized ParsedHtml at the worker
  1346→ * boundary: garbage → empty doc, arrays clamped, every text capped,
  1347→ * unknown enum values dropped.
  1348→ */
  1349→export function parseHtmlRun(raw: string): ParsedHtml {
  1350→  let parsed: unknown;
  1351→  try {
  1352→    parsed = JSON.parse(raw);
  1353→  } catch {
  1354→    return emptyDoc();
  1355→  }
  1356→  const o = (typeof parsed === "object" && parsed !== null ? parsed : {}) as Record<
  1357→    string,
  1358→    unknown
  1359→  >;
  1360→  if (!Array.isArray(o.blocks)) return emptyDoc();
  1361→
  1362→  let truncated = o.truncated === true;
  1363→  const blocks: HtmlBlock[] = [];
  1364→  for (const bo of o.blocks) {
  1365→    if (blocks.length >= MAX_BLOCKS) {
  1366→      truncated = true;
  1367→      break;
  1368→    }
  1369→    if (typeof bo !== "object" || bo === null) continue;
  1370→    const b = bo as Record<string, unknown>;
  1371→    const kind = b.kind;
  1372→
  1373→    if (kind === "rule") {
  1374→      blocks.push({ kind: "rule" });
  1375→    } else if (kind === "pre") {
  1376→      blocks.push({ kind: "pre", text: hindiNorm(cleanText(b.text, MAX_PRE_TEXT)) });
  1377→    } else if (kind === "table") {
  1378→      const rawRows = Array.isArray(b.grid) ? b.grid : [];
  1379→      const rows: string[][] = [];
  1380→      for (const ro of rawRows) {
  1381→        if (rows.length >= MAX_TABLE_ROWS) {
  1382→          truncated = true;
  1383→          break;
  1384→        }
  1385→        if (!Array.isArray(ro)) continue;
  1386→        const cells: string[] = [];
  1387→        for (const co of ro) {
  1388→          if (cells.length >= MAX_TABLE_COLS) {
  1389→            truncated = true;
  1390→            break;
  1391→          }
  1392→          cells.push(hindiNorm(cleanText(co, MAX_CELL_TEXT)));
  1393→        }
  1394→        if (cells.length > 0) rows.push(cells);
  1395→      }
  1396→      if (rows.length > 0) {
  1397→        blocks.push({
  1398→          kind: "table",
  1399→          grid: rows,
  1400→          headerRow: b.headerRow === true || undefined,
  1401→        });
  1402→      }
  1403→    } else if (
  1404→      kind === "heading" ||
  1405→      kind === "para" ||
  1406→      kind === "quote" ||
  1407→      kind === "list-item"
  1408→    ) {
  1409→      const runs = coerceRuns(b.runs);
  1410→      if (kind === "heading") {
  1411→        const level = coerceHeadingLevel(b.level);
  1412→        if (level !== undefined) blocks.push({ kind: "heading", level, runs });
  1413→        else blocks.push({ kind: "para", runs });
  1414→      } else if (kind === "para") {
  1415→        blocks.push({ kind: "para", runs });
  1416→      } else if (kind === "quote") {
  1417→        blocks.push({ kind: "quote", runs });
  1418→      } else {
  1419→        const numberRaw =
  1420→          typeof b.number === "number" ? b.number : Number.parseInt(String(b.number), 10);
  1421→        blocks.push({
  1422→          kind: "list-item",
  1423→          depth: coerceDepth(b.depth),
  1424→          ordered: b.ordered === true,
  1425→          number: Number.isFinite(numberRaw)
  1426→            ? Math.min(999_999, Math.max(0, numberRaw))
  1427→            : 1,
  1428→          runs,
  1429→        });
  1430→      }
  1431→    }
  1432→  }
  1433→
  1434→  const title = cleanText(o.title, MAX_TITLE_TEXT);
  1435→  return { title, blocks, truncated };
  1436→}

  1442→/**
  1443→ * Base name for the download: the <title> when available, else the
  1444→ * input file's name; a blank result falls back to the localized word
  1445→ * ("Webpage" / "वेबपेज"). Unicode survives; path separators and unsafe

  1448→export function outputBaseName(name: string, lang: "en" | "hi"): string {

  1454→    .trim()
  1455→    .slice(0, 120);
  1456→  return base.length > 0 ? base : fallback;
  1457→}

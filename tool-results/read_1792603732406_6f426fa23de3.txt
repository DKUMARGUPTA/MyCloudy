   921→          if (sp && para) {
   922→            if (sp.run) sp.run.text += "\n";
   923→            else
   924→              sp.run = {
   925→                text: "\n",
   926→                bold: false,
   927→                italic: false,
   928→                sizePt: undefined,
   929→                color: undefined,
   930→              };
   931→          }
   932→          return;
   933→        }
   934→        case "t": {
   935→          const sp = this.topSp();
   936→          if (sp && sp.run) this.textTarget = true;
   937→          return;
   938→        }
   939→        default:
   940→          return;
   941→      }
   942→    },
   943→
   944→    text: (value) => {
   945→      if (!this.textTarget) return;
   946→      const run = this.topSp()?.run ?? null;
   947→      if (run) run.text += hindiNorm(value);
   948→    },
   949→
   950→    close: (name) => {
   951→      const pre = prefixOf(name);
   952→      const local = localName(name);
   953→
   954→      if (pre === "p") {
   955→        if (local === "sp") {
   956→          this.endShape();
   957→        } else if (local === "grpSp" || local === "spTree") {
   958→          this.endGroup();
   959→        } else if (
   960→          local === "pic" ||
   961→          local === "graphicFrame" ||
   962→          local === "contentPart"
   963→        ) {
   964→          this.stack.pop();
   965→        }
   966→        return;
   967→      }
   968→
   969→      if (pre !== "a") return;
   970→      switch (local) {
   971→        case "xfrm":
   972→          this.xfrmData = null;
   973→          return;
   974→        case "txBody": {
   975→          const sp = this.topSp();
   976→          if (sp) {
   977→            if (sp.para) this.endPara();
   978→            sp.inTxBody = false;
   979→          }
   980→          return;
   981→        }
   982→        case "p":
   983→          this.endPara();
   984→          return;
   985→        case "r":
   986→        case "fld":
   987→          this.endRun();
   988→          return;
   989→        case "rPr": {
   990→          const sp = this.topSp();
   991→          if (sp) sp.inRPr = false;
   992→          return;
   993→        }
   994→        case "t":
   995→          this.textTarget = false;
   996→          return;
   997→        default:
   998→          return;
   999→      }
  1000→    },
  1001→  };
  1002→
  1003→  finish(): PptSlide {
  1004→    return { shapes: this.shapes, notes: "" };
  1005→  }
  1006→}

  1038→function decodeUtf8(b: Uint8Array): string {
  1039→  let s = new TextDecoder("utf-8", { fatal: false }).decode(b);
  1040→  if (s.charCodeAt(0) === 0xfeff) s = s.slice(1);
  1041→  return s;

  1044→/**
  1045→ * Unzips a .pptx (OOXML ZIP) and extracts the slide IR. Throws
  1046→ * PptxParseError with an honest code for everything the renderer can't
  1047→ * consume (legacy .ppt, non-ZIP, no presentation part, no slides).
  1048→ */
  1049→export function parsePptx(bytes: Uint8Array): ParsedPptx {
  1050→  if (bytes.length > MAX_PPTX_BYTES) throw new PptxParseError("pptx-too-large");
  1051→  if (isLegacyPptBytes(bytes.slice(0, 8))) {
  1052→    throw new PptxParseError("pptx-legacy");
  1053→  }
  1054→  if (!isZipBytes(bytes.slice(0, 4))) throw new PptxParseError("pptx-not-zip");
  1055→
  1056→  let entries: Unzipped;
  1057→  try {
  1058→    entries = unzipSync(bytes);
  1059→  } catch {
  1060→    throw new PptxParseError("pptx-not-zip");
  1061→  }
  1062→
  1063→  const presBytes = entries["ppt/presentation.xml"];
  1064→  if (!presBytes || presBytes.length === 0) {
  1065→    throw new PptxParseError("pptx-corrupt");
  1066→  }
  1067→  const pres = parsePresentationXml(decodeUtf8(presBytes));
  1068→
  1069→  const relsBytes = entries["ppt/_rels/presentation.xml.rels"];
  1070→  const rels = relsBytes ? parseRelsXml(decodeUtf8(relsBytes)) : {};
  1071→
  1072→  const seen = new Set<string>();
  1073→  const slideParts: string[] = [];
  1074→  for (const rid of pres.rIds) {
  1075→    const rel = rels[rid];
  1076→    if (!rel || !rel.type.endsWith("/slide")) continue;
  1077→    const part = resolvePart("ppt", rel.target);
  1078→    const data = entries[part];
  1079→    if (!data || data.length === 0 || seen.has(part)) continue;
  1080→    seen.add(part);
  1081→    slideParts.push(part);
  1082→  }
  1083→  if (slideParts.length === 0) throw new PptxParseError("pptx-empty");
  1084→
  1085→  const truncated = slideParts.length > MAX_SLIDES;
  1086→  const parts = slideParts.slice(0, MAX_SLIDES);
  1087→  const slideW = emuToPt(pres.cx);
  1088→  const slideH = emuToPt(pres.cy);
  1089→
  1090→  const slides: PptSlide[] = [];
  1091→  for (const part of parts) {
  1092→    const asm = new SlideAssembler(slideW, slideH);
  1093→    try {
  1094→      parseXmlSax(decodeUtf8(entries[part] as Uint8Array), asm.handlers);
  1095→    } catch {
  1096→      /* a damaged slide degrades to empty — the deck keeps going */
  1097→    }
  1098→    asm.truncated = asm.truncated || truncated;
  1099→    const slide = asm.finish();
  1100→
  1101→    // speaker notes: this slide's .rels → notesSlide part
  1102→    const slash = part.lastIndexOf("/");
  1103→    const dir = slash >= 0 ? part.slice(0, slash) : "";
  1104→    const file = slash >= 0 ? part.slice(slash + 1) : part;
  1105→    const notesRelsBytes = entries[`${dir}/_rels/${file}.rels`];
  1106→    if (notesRelsBytes) {
  1107→      const nrels = parseRelsXml(decodeUtf8(notesRelsBytes));
  1108→      for (const rel of Object.values(nrels)) {
  1109→        if (!rel.type.endsWith("/notesSlide")) continue;
  1110→        const notesPart = resolvePart(dir, rel.target);
  1111→        const notesBytes = entries[notesPart];
  1112→        if (notesBytes && notesBytes.length > 0) {
  1113→          try {
  1114→            slide.notes = parseNotesText(decodeUtf8(notesBytes));
  1115→          } catch {
  1116→            slide.notes = "";
  1117→          }
  1118→          break;
  1119→        }
  1120→      }
  1121→    }
  1122→    slides.push(slide);
  1123→  }
  1124→
  1125→  return { slides, slideW, slideH, truncated };
  1126→}

  1132→function cleanText(v: unknown, max: number): string {
  1133→  if (typeof v !== "string") return "";
  1134→  return v
  1135→    .replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F\u007F]/g, "")
  1136→    .slice(0, max);

  1139→function coercePt(v: unknown, lo: number, hi: number): number {
  1140→  const n = typeof v === "number" ? v : Number.parseFloat(String(v));
  1141→  if (!Number.isFinite(n)) return 0;
  1142→  return Math.min(hi, Math.max(lo, n));
  1143→}
  1144→
  1145→function coerceSize(v: unknown): number | undefined {
  1146→  const n = typeof v === "number" ? v : Number.NaN;
  1147→  if (!Number.isFinite(n)) return undefined;
  1148→  return n >= 4 && n <= 144 ? n : undefined;
  1149→}
  1150→
  1151→function coerceColor(v: unknown): string | undefined {
  1152→  return typeof v === "string" && COLOR_RE.test(v) ? v.toUpperCase() : undefined;
  1153→}
  1154→
  1155→function coerceAlign(v: unknown): "left" | "center" | "right" | undefined {
  1156→  return v === "left" || v === "center" || v === "right" ? v : undefined;
  1157→}
  1158→
  1159→function coerceNotes(v: unknown): string {
  1160→  return cleanText(hindiNorm(typeof v === "string" ? v : ""), MAX_NOTES_CHARS)
  1161→    .replace(/\r/g, "")
  1162→    .replace(/\n{3,}/g, "\n\n")
  1163→    .trim();
  1164→}

  1166→/**
  1167→ * Full defensive coercion of a serialized ParsedPptx at the worker
  1168→ * boundary: garbage → empty deck, arrays clamped, every text capped,
  1169→ * unknown enum values dropped.
  1170→ */
  1171→export function parsePptxRun(raw: string): ParsedPptx {
  1172→  let parsed: unknown;
  1173→  try {
  1174→    parsed = JSON.parse(raw);
  1175→  } catch {
  1176→    return { slides: [], slideW: 960, slideH: 540, truncated: false };
  1177→  }
  1178→  const o = (typeof parsed === "object" && parsed !== null ? parsed : {}) as Record<
  1179→    string,
  1180→    unknown
  1181→  >;
  1182→  const rawSlides = Array.isArray(o.slides) ? o.slides : [];
  1183→  const slideW = coercePt(o.slideW, 32, 4032) || 960;
  1184→  const slideH = coercePt(o.slideH, 32, 4032) || 540;
  1185→  let truncated = o.truncated === true || rawSlides.length > MAX_SLIDES;
  1186→
  1187→  const slides: PptSlide[] = [];
  1188→  for (const so of rawSlides.slice(0, MAX_SLIDES)) {
  1189→    if (typeof so !== "object" || so === null) continue;
  1190→    const s = so as Record<string, unknown>;
  1191→    const rawShapes = Array.isArray(s.shapes) ? s.shapes : [];
  1192→    const shapes: PptShape[] = [];
  1193→    for (const ho of rawShapes.slice(0, MAX_SHAPES_PER_SLIDE)) {
  1194→      if (typeof ho !== "object" || ho === null) continue;
  1195→      const h = ho as Record<string, unknown>;
  1196→      const rawParas = Array.isArray(h.paras) ? h.paras : [];
  1197→      const paras: PptPara[] = [];
  1198→      for (const po of rawParas.slice(0, MAX_PARAS_PER_SHAPE)) {
  1199→        if (typeof po !== "object" || po === null) continue;
  1200→        const p = po as Record<string, unknown>;
  1201→        const rawRuns = Array.isArray(p.runs) ? p.runs : [];
  1202→        const runs: PptRun[] = [];
  1203→        for (const ro of rawRuns.slice(0, MAX_RUNS_PER_PARA)) {
  1204→          if (typeof ro !== "object" || ro === null) continue;
  1205→          const r = ro as Record<string, unknown>;
  1206→          const text = hindiNorm(cleanText(r.text, MAX_RUN_TEXT));
  1207→          if (text === "") continue;
  1208→          runs.push({
  1209→            text,
  1210→            bold: r.bold === true,
  1211→            italic: r.italic === true,
  1212→            sizePt: coerceSize(r.sizePt),
  1213→            color: coerceColor(r.color),
  1214→          });
  1215→        }
  1216→        paras.push({
  1217→          runs,
  1218→          align: coerceAlign(p.align),
  1219→          bullet: p.bullet === true || undefined,
  1220→        });
  1221→      }
  1222→      shapes.push({
  1223→        x: coercePt(h.x, -1440, 4320),
  1224→        y: coercePt(h.y, -1440, 4320),
  1225→        w: coercePt(h.w, 0, 4320),
  1226→        h: coercePt(h.h, 0, 4320),
  1227→        paras,
  1228→        isTitle: h.isTitle === true,
  1229→      });
  1230→    }
  1231→    if (rawShapes.length > MAX_SHAPES_PER_SLIDE) truncated = true;
  1232→    slides.push({ shapes, notes: coerceNotes(s.notes) });
  1233→  }
  1234→
  1235→  return { slides, slideW, slideH, truncated };
  1236→}

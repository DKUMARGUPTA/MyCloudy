    87→/**
    88→ * ZIP oracle — walks the central directory and inflates each entry
    89→ * with node:zlib (raw deflate), completely independent of fflate.
    90→ */
    91→function unzipOracle(bytes: Uint8Array): Map<string, Uint8Array> {
    92→  const buf = Buffer.from(bytes);
    93→  const eocd = buf.lastIndexOf(Buffer.from([0x50, 0x4b, 0x05, 0x06]));
    94→  if (eocd < 0) throw new Error("no EOCD");
    95→  const count = buf.readUInt16LE(eocd + 10);
    96→  const cdOff = buf.readUInt32LE(eocd + 16);
    97→  let p = cdOff;
    98→  const out = new Map<string, Uint8Array>();
    99→  for (let i = 0; i < count; i++) {
   100→    if (buf.readUInt32LE(p) !== 0x02014b50) throw new Error("bad CD signature");
   101→    const method = buf.readUInt16LE(p + 10);
   102→    const compSize = buf.readUInt32LE(p + 20);
   103→    const nameLen = buf.readUInt16LE(p + 28);
   104→    const extraLen = buf.readUInt16LE(p + 30);
   105→    const commentLen = buf.readUInt16LE(p + 32);
   106→    const localOff = buf.readUInt32LE(p + 42);
   107→    const name = buf.slice(p + 46, p + 46 + nameLen).toString("utf8");
   108→    if (buf.readUInt32LE(localOff) !== 0x04034b50) {
   109→      throw new Error("bad local header signature");
   110→    }
   111→    const lNameLen = buf.readUInt16LE(localOff + 26);
   112→    const lExtraLen = buf.readUInt16LE(localOff + 28);
   113→    const dataStart = localOff + 30 + lNameLen + lExtraLen;
   114→    const comp = buf.slice(dataStart, dataStart + compSize);
   115→    const data = method === 0 ? comp : inflateRawSync(comp);
   116→    out.set(name, new Uint8Array(data));
   117→    p += 46 + nameLen + extraLen + commentLen;
   118→  }
   119→  return out;
   120→}

   127→/** Minimal XML well-formedness: balanced tags, no raw < in text. */
   128→function xmlWellFormed(xml: string): boolean {
   129→  const src = xml
   130→    .replace(/<\?[\s\S]*?\?>/g, "")
   131→    .replace(/<!--[\s\S]*?-->/g, "")
   132→    .replace(/<!DOCTYPE[^>]*>/gi, "");
   133→  const tagRe = /<(\/?)([A-Za-z_:][\w.:-]*)((?:\s+[\w.:-]+\s*=\s*"[^"]*")*)\s*(\/?)>/g;
   134→  const stack: string[] = [];
   135→  let lastIndex = 0;
   136→  let match: RegExpExecArray | null;
   137→  while ((match = tagRe.exec(src)) !== null) {
   138→    if (src.slice(lastIndex, match.index).includes("<")) return false;
   139→    lastIndex = tagRe.lastIndex;
   140→    const [full, close, name, , selfClose] = match;
   141→    if (selfClose === "/" || full.startsWith("<?") || full.startsWith("<!")) {
   142→      continue;
   143→    }
   144→    if (close === "/") {
   145→      if (stack.pop() !== name) return false;
   146→    } else {
   147→      stack.push(name);
   148→    }
   149→  }
   150→  if (src.slice(lastIndex).includes("<")) return false;
   151→  return stack.length === 0;
   152→}

   170→/* Minimal real PNG (RGB, no filter) — built with node:zlib. */
   171→const CRC_TABLE = new Uint32Array(256);
   172→for (let n = 0; n < 256; n += 1) {
   173→  let c = n;
   174→  for (let k = 0; k < 8; k += 1) c = c & 1 ? 0xedb88320 ^ (c >>> 1) : c >>> 1;
   175→  CRC_TABLE[n] = c >>> 0;
   176→}
   177→function crc32(bytes: Uint8Array): number {
   178→  let c = ~0;
   179→  for (let i = 0; i < bytes.length; i += 1) {
   180→    c = CRC_TABLE[(c ^ bytes[i]) & 0xff] ^ (c >>> 8);
   181→  }
   182→  return ~c >>> 0;
   183→}
   184→function pngChunk(type: string, data: Uint8Array): Uint8Array {
   185→  const out = new Uint8Array(12 + data.length);
   186→  const dv = new DataView(out.buffer);
   187→  dv.setUint32(0, data.length);
   188→  out.set(new TextEncoder().encode(type), 4);
   189→  out.set(data, 8);
   190→  dv.setUint32(8 + data.length, crc32(out.subarray(4, 8 + data.length)));
   191→  return out;
   192→}
   193→function tinyPng(w: number, h: number, gray: number): Uint8Array {
   194→  const sig = new Uint8Array([0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a]);
   195→  const ihdr = new Uint8Array(13);
   196→  const dv = new DataView(ihdr.buffer);
   197→  dv.setUint32(0, w);
   198→  dv.setUint32(4, h);
   199→  ihdr[8] = 8; // bit depth
   200→  ihdr[9] = 2; // color type: truecolor RGB
   201→  const raw: number[] = [];
   202→  for (let y = 0; y < h; y += 1) {
   203→    raw.push(0); // filter: none
   204→    for (let x = 0; x < w; x += 1) raw.push(gray, gray, gray);
   205→  }
   206→  const idat = new Uint8Array(deflateSync(new Uint8Array(raw)));
   207→  const parts = [
   208→    sig,
   209→    pngChunk("IHDR", ihdr),
   210→    pngChunk("IDAT", idat),
   211→    pngChunk("IEND", new Uint8Array(0)),
   212→  ];
   213→  const total = parts.reduce((n, p) => n + p.length, 0);
   214→  const out = new Uint8Array(total);
   215→  let o = 0;
   216→  for (const p of parts) {
   217→    out.set(p, o);
   218→    o += p.length;
   219→  }
   220→  return out;
   221→}

   517→async function extractFixture(bytes: Uint8Array): Promise<Extracted> {
   518→  const task = getDocument({ data: bytes.slice(), useSystemFonts: true });
   519→  const pdf = await task.promise;
   520→  const pages: PdfTextItemInput[][] = [];
   521→  const widths: number[] = [];
   522→  const heights: number[] = [];
   523→  for (let number = 1; number <= pdf.numPages; number += 1) {
   524→    const page = await pdf.getPage(number);

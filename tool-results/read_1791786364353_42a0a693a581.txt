    79→/**
    80→ * ToolShell (PRD §6.3): the universal 4-state tool template.
    81→ * Landing (upload + SEO copy) → Workspace (preview + options) →
    82→ * Processing (real worker progress) → Result (download/share/chain).
    83→ * Every enabled tool renders from its registry entry + config — this
    84→ * component is the ONLY tool page in the app.
    85→ */
    86→
    87→type ShellPhase = "landing" | "workspace" | "processing" | "result";
    88→
    89→let fileSeq = 0;
    90→function nextFileId(): string {
    91→  fileSeq += 1;
    92→  return `wf-${Date.now().toString(36)}-${fileSeq}`;
    93→}
    94→
    95→export function ToolShell({ tool, config }: { tool: ToolDef; config: ToolConfig }) {
    96→  const { t, locale } = useI18n();
    97→  const { toast } = useToast();
    98→  const category = getCategory(tool.category);
    99→  // Viewer tools (PRD §6.5) never process anything: no run CTA, no options
   100→  // aside, no processing/result phases — the workspace surface IS the tool.
   101→  const viewerOnly = config.viewerOnly === true;
   102→  // Surface-first tools (PRD §2.6 pattern extension): the registered surface
   103→  // is the primary entry point — it renders even with ZERO files and owns
   104→  // its own inputs (camera capture, generator, …).
   105→  const noInput = config.noInput === true;
   106→
   107→  // One-time chain handoff peek — set by a previous tool's result view or
   108→  // the home Smart Drop Zone right before navigating here. Consumed via
   109→  // the lazy initializer (pure read); cleared in the effect below.
   110→  const [chainIn] = useState<{
   111→    fromToolId: string;
   112→    items: WorkspaceFile[];
   113→  } | null>(() => {
   114→    const handoff = peekChainHandoff();
   115→    if (!handoff || handoff.items.length === 0) return null;
   116→    return {
   117→      fromToolId: handoff.fromToolId,
   118→      items: handoff.items.map((item) => ({
   119→        id: item.id,
   120→        file: item.file,
   121→      })),
   122→    };
   123→  });
   124→
   125→  const [phase, setPhase] = useState<ShellPhase>(() => (chainIn ? "workspace" : "landing"));
   126→  const [files, setFiles] = useState<WorkspaceFile[]>(() => chainIn?.items ?? []);
   127→  const [options, setOptions] = useState<OptionValues>(() =>
   128→    defaultOptionValues(config.options)
   129→  );
   130→  const [progress, setProgress] = useState<RunProgress>({
   131→    pct: 0,
   132→    stage: "reading",
   133→  });
   134→  const [outputs, setOutputs] = useState<ToolOutput[]>([]);
   135→  const [inputBytes, setInputBytes] = useState(0);
   136→  const [runSummary, setRunSummary] = useState<RunDetail | null>(null);
   137→  const [optionsOpen, setOptionsOpen] = useState(false);
   138→  const [helpOpen, setHelpOpen] = useState(false);
   139→
   140→  // Chain handoff consumed once at mount (lazy initializer = pure read);
   141→  // clearing the module store is an external-system sync in this effect.
   142→  const chainFrom: ToolDef | null = chainIn
   143→    ? (getTool(chainIn.fromToolId) ?? null)
   144→    : null;
   145→  useEffect(() => {
   146→    if (chainIn) clearChainHandoff();
   147→  }, [chainIn]);
   148→
   149→  // Input kind drives the DropZone sniffing + accept filter + labels.
   150→  const acceptKind =
   151→    tool.accepts === "image" || tool.accepts === "pdf"
   152→      ? tool.accepts
   153→      : "any";
   154→  const selectLabelKey =
   155→    tool.accepts === "image"
   156→      ? "tool.selectImageFiles"
   157→      : tool.accepts === "pdf"
   158→        ? "tool.selectPdfFiles"
   159→        : "tool.selectAnyFiles";
   160→  const addMoreLabelKey =
   161→    tool.accepts === "image"
   162→      ? "tool.addMoreImages"
   163→      : tool.accepts === "pdf"
   164→        ? "tool.addMoreFiles"
   165→        : "tool.addMoreAny";
   166→  const recentKind = tool.accepts === "image" ? "image" : "pdf";
   167→
   168→  // Chain/SmartDrop files get the same treatment as dropped files (PRD
   169→  // §6.1): OPFS working copy + a metadata-only recent entry with thumbnail.
   170→  // External-system sync only — no React state is touched here.
   171→  useEffect(() => {
   172→    if (!chainIn) return;
   173→    for (const item of chainIn.items) {
   174→      void writeWorkingFile(item.id, item.file).catch(() => undefined);
   175→      void makeFileThumbnail(item.file, 160)
   176→        .then((blob) =>
   177→          addRecentFile({
   178→            id: item.id,
   179→            kind: recentKind as "pdf" | "image",
   180→            label: item.file.name,
   181→            size: item.file.size,
   182→            toolId: tool.id,
   183→            thumbnail: blob ?? undefined,
   184→          })
   185→        )
   186→        .catch(() => undefined);
   187→    }
   188→  }, [chainIn, recentKind, tool.id]);
   189→
   190→  const runHandleRef = useRef<RunHandle | null>(null);
   191→
   192→  // Optional custom options panel (PRD §2.6 Panel.tsx pattern). Direct
   193→  // map lookup — a stable module-level reference, never created in render.
   194→  const CustomPanel = TOOL_PANELS[tool.id];
   195→
   196→  // Optional workspace surface (PageGrid family): rendered in the main
   197→  // column instead of the FileList; can gate the CTA via runBlocked.
   198→  const WorkspaceSurface = TOOL_SURFACES[tool.id];
   199→
   200→  // Surface-driven CTA gating (PRD §8.1 safety: "sab pages delete ho to
   201→  // Apply disabled" / extract needs a selection). Event-context only —
   202→  // surfaces call this from their action handlers, never during render.
   203→  const [runBlockedKey, setRunBlockedKey] = useState<string | null>(null);
   204→  const handleRunBlocked = useCallback((reasonKey: string | null) => {
   205→    setRunBlockedKey(reasonKey);
   206→  }, []);
   207→  const runDisabled =
   208→    files.length < config.minFiles || runBlockedKey !== null;
   209→
   210→  const handleOptionChange = useCallback(
   211→    (id: string, value: string | number | boolean) => {
   212→      setOptions((prev) => ({ ...prev, [id]: value }));
   213→    },
   214→    []
   215→  );
   216→
   217→  const relatedTools = useMemo(
   218→    () =>
   219→      toolsByCategory(tool.category)
   220→        .filter((item) => item.id !== tool.id)
   221→        .slice(0, 6),
   222→    [tool.category, tool.id]
   223→  );
   224→
   225→  const chainTargets = useMemo(() => {
   226→    // PRD §6.4 "Ab aage kya?" — natural next steps. Image outputs chain
   227→    // to image-accepting tools (the PDF default set would be a type lie);
   228→    // configs can override with their own curated list.
   229→    const suggestionIds =
   230→      config.chainToolIds ?? [
   231→        "compress-pdf",
   232→        "protect-pdf",
   233→        "sign-pdf",
   234→        "split-pdf",
   235→        "edit-pdf",
   236→      ];
   237→    const resolved = suggestionIds
   238→      .map((id) => getTool(id))
   239→      .filter((item): item is ToolDef => item !== undefined && item.id !== tool.id);
   240→    return resolved.slice(0, 5);
   241→  }, [config.chainToolIds, tool.id]);
   242→
   243→  // Working copies are cleared when the tool session ends (PRD §2.4).
   244→  useEffect(
   245→    () => () => {
   246→      void clearWorkingFiles().catch(() => undefined);
   247→    },
   248→    []
   249→  );
   250→
   251→  const addFiles = useCallback(
   252→    (incoming: File[]) => {
   253→      if (incoming.length === 0) return;
   254→      const room = config.maxFiles - files.length;
   255→      if (room <= 0) {
   256→        toast({
   257→          title: t("dropzone.errTooManyTitle"),
   258→          description: t("dropzone.errTooManyDesc", {
   259→            count: config.maxFiles,
   260→          }),
   261→          variant: "destructive",
   262→        });
   263→        return;
   264→      }
   265→      const accepted = incoming.slice(0, room);
   266→      const items = accepted.map((file) => ({ id: nextFileId(), file }));
   267→
   268→      // Side effects run ONCE here — never inside the setFiles updater
   269→      // (React StrictMode double-invokes updaters; they must stay pure).
   270→      for (const item of items) {
   271→        void writeWorkingFile(item.id, item.file).catch(() => undefined);
   272→        // Recent entry: metadata + thumbnail only, never the bytes.
   273→        void makeFileThumbnail(item.file, 160)
   274→          .then(async (blob) => {
   275→            await addRecentFile({
   276→              id: item.id,
   277→              kind: recentKind as "pdf" | "image",
   278→              label: item.file.name,
   279→              size: item.file.size,
   280→              toolId: tool.id,
   281→              thumbnail: blob ?? undefined,
   282→            });
   283→          })
   284→          .catch(() => undefined);
   285→      }
   286→
   287→      setFiles((prev) => [...prev, ...items]);
   288→      setPhase("workspace");
   289→    },
   290→    [config.maxFiles, files.length, recentKind, t, toast, tool.id]
   291→  );
   292→
   293→  const removeFile = useCallback((id: string) => {
   294→    setFiles((prev) => prev.filter((item) => item.id !== id));
   295→  }, []);
   296→
   297→  const moveFile = useCallback((from: number, to: number) => {
   298→    setFiles((prev) => {
   299→      if (to < 0 || to >= prev.length) return prev;
   300→      const next = [...prev];
   301→      const [moved] = next.splice(from, 1);
   302→      next.splice(to, 0, moved);
   303→      return next;
   304→    });
   305→  }, []);
   306→
   307→  const clearAll = useCallback(() => {
   308→    setFiles([]);
   309→    void clearWorkingFiles().catch(() => undefined);
   310→    setPhase("landing");
   311→  }, []);
   312→
   313→  // Stable function ref from the config (module constant) — keeps the
   314→  // React Compiler's dependency inference happy.
   315→  const collectAssets = config.collectAssets;
   316→
   317→  const startRun = useCallback(() => {
   318→    if (files.length < config.minFiles || runBlockedKey !== null) return;
   319→    trackToolUse(tool.id);
   320→    setInputBytes(files.reduce((sum, item) => sum + item.file.size, 0));
   321→    setProgress({ pct: 0, stage: "reading" });
   322→    setPhase("processing");
   323→    // Extra in-memory assets (watermark logo etc.) ride the worker message.
   324→    const assets = collectAssets?.() ?? undefined;
   325→    runHandleRef.current = toolRunner.run(
   326→      tool.id,
   327→      options,
   328→      files.map((item) => item.file),
   329→      setProgress,
   330→      files.map((item) => item.id),
   331→      assets
   332→    );
   333→    runHandleRef.current.promise
   334→      .then((result) => {
   335→        runHandleRef.current = null;
   336→        // Working copies of the INPUTS die with the run (PRD §2.4);
   337→        // results live in memory + can chain into the next tool.
   338→        setOutputs(result.outputs);
   339→        setRunSummary(result.detail ?? null);
   340→        setPhase("result");
   341→        void clearWorkingFiles().catch(() => undefined);
   342→      })
   343→      .catch((error: unknown) => {
   344→        runHandleRef.current = null;
   345→        if (error instanceof RunCancelledError) {
   346→          setPhase("workspace");
   347→          toast({ title: t("tool.cancelledToast") });
   348→          return;
   349→        }
   350→        setPhase("workspace");
   351→        const code =
   352→          error instanceof ToolRunError ? error.code : "tool.errorGeneric";
   353→        toast({
   354→          title: t("tool.errorTitle"),
   355→          description: t(code, {
   356→            detail: error instanceof ToolRunError ? error.detail ?? "" : "",
   357→          }),
   358→          variant: "destructive",
   359→        });
   360→      });
   361→  }, [collectAssets, config.minFiles, files, options, runBlockedKey, t, toast, tool.id]);
   362→
   363→  const cancelRun = useCallback(() => {
   364→    runHandleRef.current?.cancel();
   365→  }, []);
   366→
   367→  const handleChain = useCallback(
   368→    (target: ToolDef, resultFiles: File[]) => {
   369→      if (!target.enabled) {
   370→        toast({
   371→          title: t("tools.comingToastTitle", { tool: target.name[locale] }),
   372→          description: t("tools.comingToastDesc", {
   373→            phase: target.phase,
   374→          }),
   375→        });
   376→        return;
   377→      }
   378→      setChainHandoff({
   379→        fromToolId: tool.id,
   380→        items: resultFiles.map((file) => ({ id: nextFileId(), file })),
   381→      });
   382→      navigateToHash(ROUTES.tool(target.id));
   383→    },
   384→    [locale, t, toast, tool.id]
   385→  );
   386→
   387→  // Viewer surfaces chain the current file straight into another tool
   388→  // (they have no result phase of their own). Same enabled/coming-soon
   389→  // behavior as result-view chains.
   390→  const surfaceChain = useCallback(
   391→    (targetId: string) => {
   392→      const target = getTool(targetId);
   393→      if (!target) return;
   394→      handleChain(target, files.map((item) => item.file));
   395→    },
   396→    [files, handleChain]
   397→  );
   398→
   399→  const removeFromDevice = useCallback(() => {
   400→    void clearWorkingFiles().catch(() => undefined);
   401→    // Remove this run's recent entries by their file metadata (label+size).
   402→    void removeRecentFilesByMeta(
   403→      files.map((item) => ({ label: item.file.name, size: item.file.size }))
   404→    ).catch(() => undefined);
   405→    setOutputs([]);
   406→    setFiles([]);
   407→    setPhase("landing");
   408→    toast({ title: t("result.removedToast") });
   409→  }, [files, t, toast]);
   410→
   411→  const totalSize = files.reduce((sum, item) => sum + item.file.size, 0);
   412→  const ctaLabel = `${tool.name[locale]} ${t("tool.ctaArrow")}`;
   413→  const Icon = tool.icon;
   414→
   415→  return (
   416→    <div className="mx-auto w-full max-w-6xl px-4 pb-28 sm:pb-24">
   417→      {/* Breadcrumb (PRD §6.3): Home > Category > Tool */}
   418→      <nav aria-label={t("a11y.breadcrumb")} className="pt-4">
   419→        <ol className="flex flex-wrap items-center gap-1.5 text-xs text-muted-foreground">
   420→          <li>
   421→            <a
   422→              href={ROUTES.home}
   423→              className="inline-flex items-center gap-1 rounded-btn outline-none hover:text-foreground focus-visible:ring-[3px] focus-visible:ring-ring/50"
   424→            >
   425→              <Home className="size-3" aria-hidden="true" />
   426→              {t("nav.home")}
   427→            </a>
   428→          </li>
   429→          <li aria-hidden="true">/</li>
   430→          <li>
   431→            <a
   432→              href={ROUTES.category(tool.category)}
   433→              className="rounded-btn outline-none hover:text-foreground focus-visible:ring-[3px] focus-visible:ring-ring/50"
   434→            >
   435→              {category.name[locale]}
   436→            </a>
   437→          </li>
   438→          <li aria-hidden="true">/</li>
   439→          <li aria-current="page" className="font-medium text-foreground">
   440→            {tool.name[locale]}
   441→          </li>
   442→        </ol>
   443→      </nav>
   444→
   445→      {phase === "landing" ? (
   446→        <section className="flex flex-col gap-8 pt-6">
   447→          <header className="flex flex-col gap-3">
   448→            <div className="flex items-center gap-3">
   449→              <span
   450→                className={cn(
   451→                  "flex size-11 items-center justify-center rounded-btn",
   452→                  "bg-cat-organize/10 text-cat-organize"
   453→                )}
   454→                aria-hidden="true"
   455→              >
   456→                <Icon className="size-5" />
   457→              </span>
   458→              <h1 className="text-2xl font-bold tracking-tight text-foreground sm:text-3xl">
   459→                {tool.name[locale]}
   460→              </h1>
   461→            </div>
   462→            <p className="max-w-2xl text-sm text-muted-foreground sm:text-base">
   463→              {tool.description[locale]}
   464→            </p>
   465→            <div className="flex flex-wrap gap-2">
   466→              <Badge variant={tool.mode === "D" ? "private" : "server"} className="rounded-chip">
   467→                <Lock className="size-3" aria-hidden="true" />
   468→                {tool.mode === "D"
   469→                  ? t("tools.privateBadge")
   470→                  : t("tools.serverBadge")}
   471→              </Badge>
   472→              <Badge variant="outline" className="rounded-chip">
   473→                {t("tool.freeBadge")}
   474→              </Badge>
   475→            </div>
   476→          </header>
   477→
   478→          {/* Surface-first tools open straight into the workspace — the
   479→              surface owns its inputs (camera, generator). Still show the
   480→              DropZone when files are accepted, for gallery-upload flows. */}
   481→          {noInput ? (
   482→            <div className="flex flex-col items-center gap-4">
   483→              <Button
   484→                type="button"
   485→                size="touch"
   486→                className="rounded-btn px-8 text-base font-semibold"
   487→                onClick={() => setPhase("workspace")}
   488→              >
   489→                {ctaLabel}
   490→                <ArrowRight className="size-5" aria-hidden="true" />
   491→              </Button>
   492→              {config.maxFiles > 0 ? (
   493→                <DropZone
   494→                  labelKey={selectLabelKey}
   495→                  multiple={config.maxFiles > 1}
   496→                  maxFiles={config.maxFiles}
   497→                  acceptKind={acceptKind}
   498→                  onAccept={addFiles}
   499→                  currentCount={files.length}
   500→                />
   501→              ) : null}
   502→            </div>
   503→          ) : (
   504→            <DropZone
   505→              labelKey={selectLabelKey}
   506→              multiple={config.maxFiles > 1}
   507→              maxFiles={config.maxFiles}
   508→              acceptKind={acceptKind}
   509→              onAccept={addFiles}
   510→              currentCount={files.length}
   511→            />
   512→          )}
   513→
   514→          <div className="grid gap-8 lg:grid-cols-[2fr_1fr]">
   515→            <div className="flex flex-col gap-8">
   516→              {/* How it works */}
   517→              <section aria-labelledby="how-heading">
   518→                <h2 id="how-heading" className="flex items-center gap-2 text-sm font-semibold text-foreground">
   519→                  <ListChecks className="size-4 text-primary" aria-hidden="true" />
   520→                  {t("tool.howTitle")}
   521→                </h2>
   522→                <ol className="mt-3 grid gap-3 sm:grid-cols-3">
   523→                  {(config.howStepKeys ??
   524→                    (viewerOnly
   525→                      ? (["view.how1", "view.how2", "view.how3"] as const)
   526→                      : (["tool.how1", "tool.how2", "tool.how3"] as const))
   527→                  ).map((key, index) => (
   528→                    <li
   529→                      key={key}
   530→                      className="rounded-card border border-border bg-card p-4"
   531→                    >
   532→                      <span className="flex size-7 items-center justify-center rounded-full bg-primary/10 text-xs font-bold text-primary">
   533→                        {index + 1}
   534→                      </span>
   535→                      <p className="mt-2 text-xs leading-relaxed text-muted-foreground">
   536→                        {t(key)}
   537→                      </p>
   538→                    </li>
   539→                  ))}
   540→                </ol>
   541→              </section>
   542→
   543→              {/* FAQ */}
   544→              <section aria-labelledby="faq-heading">
   545→                <h2 id="faq-heading" className="text-sm font-semibold text-foreground">
   546→                  {t("tool.faqTitle")}
   547→                </h2>
   548→                <Accordion type="single" collapsible className="mt-3">
   549→                  {config.faq.map((item, index) => (
   550→                    <AccordionItem key={index} value={`faq-${index}`}>
   551→                      <AccordionTrigger className="text-left text-sm">
   552→                        {item.q[locale]}
   553→                      </AccordionTrigger>
   554→                      <AccordionContent className="text-xs leading-relaxed text-muted-foreground">
   555→                        {item.a[locale]}
   556→                      </AccordionContent>
   557→                    </AccordionItem>
   558→                  ))}
   559→                </Accordion>
   560→              </section>
   561→            </div>
   562→
   563→            <aside className="flex flex-col gap-8">
   564→              {/* Features */}
   565→              <section aria-labelledby="features-heading">
   566→                <h2 id="features-heading" className="flex items-center gap-2 text-sm font-semibold text-foreground">
   567→                  <Sparkles className="size-4 text-primary" aria-hidden="true" />
   568→                  {t("tool.featuresTitle")}
   569→                </h2>
   570→                <ul className="mt-3 flex flex-col gap-2.5">
   571→                  {config.features.map((feature, index) => (
   572→                    <li
   573→                      key={index}
   574→                      className="flex gap-2 rounded-card border border-border bg-card p-3 text-xs leading-relaxed text-muted-foreground"
   575→                    >
   576→                      <span
   577→                        aria-hidden="true"
   578→                        className="mt-0.5 size-1.5 shrink-0 rounded-full bg-primary"
   579→                      />
   580→                      {feature[locale]}
   581→                    </li>
   582→                  ))}
   583→                </ul>
   584→              </section>
   585→
   586→              {/* Related */}
   587→              <section aria-labelledby="related-heading">
   588→                <h2 id="related-heading" className="text-sm font-semibold text-foreground">
   589→                  {t("tool.relatedTitle")}
   590→                </h2>
   591→                <ul className="mt-3 flex flex-col gap-1.5">
   592→                  {relatedTools.map((item) => {
   593→                    const RelatedIcon = item.icon;
   594→                    return (
   595→                      <li key={item.id}>
   596→                        <a
   597→                          href={item.enabled ? ROUTES.tool(item.id) : ROUTES.category(item.category)}
   598→                          className="flex min-h-11 items-center gap-2.5 rounded-btn px-2 text-sm outline-none transition-colors hover:bg-accent focus-visible:ring-[3px] focus-visible:ring-ring/50"
   599→                        >
   600→                          <RelatedIcon className="size-4 text-muted-foreground" aria-hidden="true" />
   601→                          <span className="truncate text-foreground">
   602→                            {item.name[locale]}
   603→                          </span>
   604→                          {!item.enabled ? (
   605→                            <Badge variant="outline" className="ml-auto rounded-chip text-[10px] text-muted-foreground">
   606→                              {t("tools.soonBadge")}
   607→                            </Badge>
   608→                          ) : null}
   609→                        </a>
   610→                      </li>
   611→                    );
   612→                  })}
   613→                </ul>
   614→              </section>
   615→            </aside>
   616→          </div>
   617→        </section>
   618→      ) : null}
   619→
   620→      {phase === "workspace" ? (
   621→        <section className="pt-6">
   622→          {/* Workspace toolbar (PRD §6.3 state 2) */}
   623→          <div className="flex flex-wrap items-center gap-2">
   624→            <Button
   625→              type="button"
   626→              variant="ghost"
   627→              size="sm"
   628→              onClick={() => setPhase("landing")}
   629→              className="rounded-btn"
   630→            >
   631→              <ArrowLeft className="size-4" aria-hidden="true" />
   632→              {t("tool.back")}
   633→            </Button>
   634→            <span className="ml-auto flex items-center gap-2">
   635→              <Button
   636→                type="button"
   637→                variant="outline"
   638→                size="sm"
   639→                className="rounded-btn"
   640→                onClick={() => setHelpOpen(true)}
   641→                aria-label={t("tool.helpLabel")}
   642→              >
   643→                <CircleHelp className="size-4" aria-hidden="true" />
   644→                <span className="hidden sm:inline">{t("tool.helpLabel")}</span>
   645→              </Button>
   646→              <Button
   647→                type="button"
   648→                variant="outline"
   649→                size="sm"
   650→                className="rounded-btn"
   651→                onClick={clearAll}
   652→              >
   653→                <Trash2 className="size-4" aria-hidden="true" />
   654→                {t("tool.clearAll")}
   655→              </Button>
   656→            </span>
   657→          </div>
   658→
   659→          <div
   660→            className={cn(
   661→              "mt-4 grid grid-cols-1 gap-6",
   662→              !viewerOnly && "lg:grid-cols-[minmax(0,1fr)_320px]"
   663→            )}
   664→          >
   665→            {/* Main column: the surface owns file presentation when one is
   666→                registered (PageGrid family); otherwise the classic
   667→                FileList + add-more DropZone (PRD §6.3). Viewer tools go
   668→                full-width — their surface is the entire product (§6.5). */}
   669→            <div className="flex flex-col gap-3">
   670→              {WorkspaceSurface && (files.length > 0 || noInput) ? (
   671→                <WorkspaceSurface
   672→                  files={files}
   673→                  values={options}
   674→                  onChange={handleOptionChange}
   675→                  setFileOrder={setFiles}
   676→                  setRunBlocked={handleRunBlocked}
   677→                  addFiles={addFiles}
   678→                  onChain={surfaceChain}
   679→                />
   680→              ) : (
   681→                <>
   682→                  <FileList
   683→                    items={files}
   684→                    onMove={moveFile}
   685→                    onRemove={removeFile}
   686→                  />
   687→                  <DropZone
   688→                    labelKey={addMoreLabelKey}
   689→                    multiple={config.maxFiles > 1}
   690→                    maxFiles={config.maxFiles}
   691→                    acceptKind={acceptKind}
   692→                    onAccept={addFiles}
   693→                    currentCount={files.length}
   694→                    className="border-dashed py-4"
   695→                  />
   696→                </>
   697→              )}
   698→            </div>
   699→
   700→            {/* Options: desktop side panel / mobile bottom sheet.
   701→                Height budget: sticky header (top-20) + sticky CTA bar at the
   702→                bottom — so the panel never slides under the CTA bar.
   703→                Viewer tools have no options panel — full-width surface. */}
   704→            {!viewerOnly ? (
   705→            <aside className="hidden lg:block">
   706→              <div className="sticky top-20 mb-20 flex max-h-[calc(100vh-11.5rem)] flex-col gap-5 overflow-y-auto scrollbar-thin rounded-card border border-border bg-card p-5">
   707→                <div>
   708→                  <h2 className="text-sm font-semibold text-foreground">
   709→                    {t("tool.optionsTitle")}
   710→                  </h2>
   711→                  {CustomPanel ? (
   712→                    <div className="mt-4">
   713→                      <CustomPanel
   714→                        files={files}
   715→                        values={options}
   716→                        onChange={handleOptionChange}
   717→                        setFileOrder={setFiles}
   718→                      />
   719→                    </div>
   720→                  ) : null}
   721→                  <OptionsPanel
   722→                    fields={config.options}
   723→                    values={options}
   724→                    onChange={handleOptionChange}
   725→                    className="mt-4"
   726→                  />
   727→                </div>
   728→                <p className="border-t border-border pt-3 text-[11px] text-muted-foreground">
   729→                  {files.length > 0 && files.length < config.minFiles
   730→                    ? t("tool.summaryMinFiles", { count: config.minFiles })
   731→                    : t("tool.summary", {
   732→                        count: files.length,
   733→                        size: formatBytes(totalSize),
   734→                      })}
   735→                </p>
   736→              </div>
   737→            </aside>
   738→            ) : null}
   739→          </div>
   740→
   741→          {/* Sticky CTA (PRD §6.3): count + estimated size + action.
   742→              Hidden for viewer tools — the surface owns its actions. */}
   743→          {!viewerOnly ? (
   744→          <div className="fixed inset-x-0 bottom-0 z-40 border-t border-border bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/85">
   745→            <div className="mx-auto flex max-w-6xl items-center gap-3 px-4 py-3">
   746→              <div className="min-w-0 flex-1">
   747→                <p
   748→                  className={cn(
   749→                    "flex items-center gap-1.5 truncate text-xs",
   750→                    runBlockedKey !== null
   751→                      ? "font-medium text-amber-600 dark:text-amber-400"
   752→                      : files.length > 0 && files.length < config.minFiles
   753→                        ? "font-medium text-primary"
   754→                        : "text-muted-foreground"
   755→                  )}
   756→                >
   757→                  {runBlockedKey !== null ? (
   758→                    <TriangleAlert
   759→                      className="size-3.5 shrink-0"
   760→                      aria-hidden="true"
   761→                    />
   762→                  ) : null}
   763→                  <span className="truncate">
   764→                    {runBlockedKey !== null
   765→                      ? t(runBlockedKey)
   766→                      : files.length > 0 && files.length < config.minFiles
   767→                        ? t("tool.summaryMinFiles", { count: config.minFiles })
   768→                        : t("tool.summary", {
   769→                            count: files.length,
   770→                            size: formatBytes(totalSize),
   771→                          })}
   772→                  </span>
   773→                </p>
   774→                {chainFrom ? (
   775→                  <p className="truncate text-[11px] text-primary">
   776→                    {t("tool.chainFrom", { tool: chainFrom.name[locale] })}
   777→                  </p>
   778→                ) : null}
   779→              </div>
   780→              <Button
   781→                type="button"
   782→                variant="outline"
   783→                size="touch"
   784→                className="rounded-btn lg:hidden"
   785→                onClick={() => setOptionsOpen(true)}
   786→              >
   787→                {t("tool.optionsButton")}
   788→              </Button>
   789→              <Button
   790→                type="button"
   791→                size="touch"
   792→                className="rounded-btn font-semibold"
   793→                onClick={startRun}
   794→                disabled={runDisabled}
   795→              >
   796→                {ctaLabel}
   797→                <ArrowRight className="size-4" aria-hidden="true" />
   798→              </Button>
   799→            </div>
   800→          </div>
   801→          ) : null}
   802→
   803→          {!viewerOnly ? (
   804→          <BottomSheet
   805→            open={optionsOpen}
   806→            onOpenChange={setOptionsOpen}
   807→            title={t("tool.optionsTitle")}
   808→            description={tool.description[locale]}
   809→            footer={
   810→              <Button
   811→                type="button"
   812→                size="touch"
   813→                className="w-full rounded-btn font-semibold"
   814→                onClick={() => {
   815→                  setOptionsOpen(false);
   816→                  startRun();
   817→                }}
   818→                disabled={runDisabled}
   819→              >
   820→                {ctaLabel}
   821→                <ArrowRight className="size-4" aria-hidden="true" />
   822→              </Button>
   823→            }
   824→          >
   825→            {CustomPanel ? (
   826→              <div className="mb-4">
   827→                <CustomPanel
   828→                  files={files}
   829→                  values={options}
   830→                  onChange={handleOptionChange}
   831→                  setFileOrder={setFiles}
   832→                />
   833→              </div>
   834→            ) : null}
   835→            <OptionsPanel
   836→              fields={config.options}
   837→              values={options}
   838→              onChange={handleOptionChange}
   839→            />
   840→          </BottomSheet>
   841→          ) : null}
   842→        </section>
   843→      ) : null}
   844→
   845→      {phase === "processing" ? (
   846→        <ProcessingView
   847→          toolName={tool.name[locale]}
   848→          progress={progress}
   849→          onCancel={cancelRun}
   850→        />
   851→      ) : null}
   852→
   853→      {phase === "result" ? (
   854→        <ResultView
   855→          tool={tool}
   856→          outputs={outputs}
   857→          inputBytes={inputBytes}
   858→          runSummary={runSummary}
   859→          chainFromName={chainFrom ? chainFrom.name[locale] : undefined}
   860→          chainTargets={chainTargets}
   861→          onChain={handleChain}
   862→          onRemoveFromDevice={removeFromDevice}
   863→        />
   864→      ) : null}
   865→
   866→      {/* Help dialog */}
   867→      <Modal
   868→        open={helpOpen}
   869→        onOpenChange={setHelpOpen}
   870→        title={t("tool.helpTitle")}
   871→        footer={
   872→          <Button
   873→            type="button"
   874→            variant="default"
   875→            size="touch"
   876→            className="w-full rounded-btn sm:w-auto"
   877→            onClick={() => setHelpOpen(false)}
   878→          >
   879→            {t("pro.action")}
   880→          </Button>
   881→        }
   882→      >
   883→        <div className="flex flex-col gap-3 text-sm text-muted-foreground">
   884→          <p className="flex gap-2">
   885→            <FilePlus2 className="mt-0.5 size-4 shrink-0 text-primary" aria-hidden="true" />
   886→            {t("tool.how2")}
   887→          </p>
   888→          <p>{t("tool.helpPrivacy")}</p>
   889→        </div>
   890→      </Modal>
   891→    </div>
   892→  );
   893→}

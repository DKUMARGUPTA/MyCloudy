    77→/**
    78→ * ToolShell (PRD §6.3): the universal 4-state tool template.
    79→ * Landing (upload + SEO copy) → Workspace (preview + options) →
    80→ * Processing (real worker progress) → Result (download/share/chain).
    81→ * Every enabled tool renders from its registry entry + config — this
    82→ * component is the ONLY tool page in the app.
    83→ */
    84→
    85→type ShellPhase = "landing" | "workspace" | "processing" | "result";
    86→
    87→let fileSeq = 0;
    88→function nextFileId(): string {
    89→  fileSeq += 1;
    90→  return `wf-${Date.now().toString(36)}-${fileSeq}`;
    91→}
    92→
    93→export function ToolShell({ tool, config }: { tool: ToolDef; config: ToolConfig }) {
    94→  const { t, locale } = useI18n();
    95→  const { toast } = useToast();
    96→  const category = getCategory(tool.category);
    97→
    98→  // One-time chain handoff peek — set by a previous tool's result view or
    99→  // the home Smart Drop Zone right before navigating here. Consumed via
   100→  // the lazy initializer (pure read); cleared in the effect below.
   101→  const [chainIn] = useState<{
   102→    fromToolId: string;
   103→    items: WorkspaceFile[];
   104→  } | null>(() => {
   105→    const handoff = peekChainHandoff();
   106→    if (!handoff || handoff.items.length === 0) return null;
   107→    return {
   108→      fromToolId: handoff.fromToolId,
   109→      items: handoff.items.map((item) => ({
   110→        id: item.id,
   111→        file: item.file,
   112→      })),
   113→    };
   114→  });
   115→
   116→  const [phase, setPhase] = useState<ShellPhase>(() => (chainIn ? "workspace" : "landing"));
   117→  const [files, setFiles] = useState<WorkspaceFile[]>(() => chainIn?.items ?? []);
   118→  const [options, setOptions] = useState<OptionValues>(() =>
   119→    defaultOptionValues(config.options)
   120→  );
   121→  const [progress, setProgress] = useState<RunProgress>({
   122→    pct: 0,
   123→    stage: "reading",
   124→  });
   125→  const [outputs, setOutputs] = useState<ToolOutput[]>([]);
   126→  const [inputBytes, setInputBytes] = useState(0);
   127→  const [runSummary, setRunSummary] = useState<RunDetail | null>(null);
   128→  const [optionsOpen, setOptionsOpen] = useState(false);
   129→  const [helpOpen, setHelpOpen] = useState(false);
   130→
   131→  // Chain handoff consumed once at mount (lazy initializer = pure read);
   132→  // clearing the module store is an external-system sync in this effect.
   133→  const chainFrom: ToolDef | null = chainIn
   134→    ? (getTool(chainIn.fromToolId) ?? null)
   135→    : null;
   136→  useEffect(() => {
   137→    if (chainIn) clearChainHandoff();
   138→  }, [chainIn]);
   139→
   140→  // Chain/SmartDrop files get the same treatment as dropped files (PRD
   141→  // §6.1): OPFS working copy + a metadata-only recent entry with thumbnail.
   142→  // External-system sync only — no React state is touched here.
   143→  useEffect(() => {
   144→    if (!chainIn) return;
   145→    for (const item of chainIn.items) {
   146→      void writeWorkingFile(item.id, item.file).catch(() => undefined);
   147→      void renderPdfThumbnail(item.file, { maxDim: 160 })
   148→        .then(({ blob }) =>
   149→          addRecentFile({
   150→            id: item.id,
   151→            kind: "pdf",
   152→            label: item.file.name,
   153→            size: item.file.size,
   154→            toolId: tool.id,
   155→            thumbnail: blob,
   156→          })
   157→        )
   158→        .catch(() => undefined);
   159→    }
   160→  }, [chainIn, tool.id]);
   161→
   162→  const runHandleRef = useRef<RunHandle | null>(null);
   163→
   164→  // Optional custom options panel (PRD §2.6 Panel.tsx pattern). Direct
   165→  // map lookup — a stable module-level reference, never created in render.
   166→  const CustomPanel = TOOL_PANELS[tool.id];
   167→
   168→  const handleOptionChange = useCallback(
   169→    (id: string, value: string | number | boolean) => {
   170→      setOptions((prev) => ({ ...prev, [id]: value }));
   171→    },
   172→    []
   173→  );
   174→
   175→  const relatedTools = useMemo(
   176→    () =>
   177→      toolsByCategory(tool.category)
   178→        .filter((item) => item.id !== tool.id)
   179→        .slice(0, 6),
   180→    [tool.category, tool.id]
   181→  );
   182→
   183→  const chainTargets = useMemo(() => {
   184→    // PRD §6.4 "Ab aage kya?" — the natural next steps for a PDF result.
   185→    const suggestionIds = [
   186→      "compress-pdf",
   187→      "protect-pdf",
   188→      "sign-pdf",
   189→      "split-pdf",
   190→      "edit-pdf",
   191→    ];
   192→    const resolved = suggestionIds
   193→      .map((id) => getTool(id))
   194→      .filter((item): item is ToolDef => item !== undefined && item.id !== tool.id);
   195→    return resolved.slice(0, 5);
   196→  }, [tool.id]);
   197→
   198→  // Working copies are cleared when the tool session ends (PRD §2.4).
   199→  useEffect(
   200→    () => () => {
   201→      void clearWorkingFiles().catch(() => undefined);
   202→    },
   203→    []
   204→  );
   205→
   206→  const addFiles = useCallback(
   207→    (incoming: File[]) => {
   208→      if (incoming.length === 0) return;
   209→      const room = config.maxFiles - files.length;
   210→      if (room <= 0) {
   211→        toast({
   212→          title: t("dropzone.errTooManyTitle"),
   213→          description: t("dropzone.errTooManyDesc", {
   214→            count: config.maxFiles,
   215→          }),
   216→          variant: "destructive",
   217→        });
   218→        return;
   219→      }
   220→      const accepted = incoming.slice(0, room);
   221→      const items = accepted.map((file) => ({ id: nextFileId(), file }));
   222→
   223→      // Side effects run ONCE here — never inside the setFiles updater
   224→      // (React StrictMode double-invokes updaters; they must stay pure).
   225→      for (const item of items) {
   226→        void writeWorkingFile(item.id, item.file).catch(() => undefined);
   227→        // Recent entry: metadata + thumbnail only, never the bytes.
   228→        void renderPdfThumbnail(item.file, { maxDim: 160 })
   229→          .then(async ({ blob }) => {
   230→            await addRecentFile({
   231→              id: item.id,
   232→              kind: "pdf" as const,
   233→              label: item.file.name,
   234→              size: item.file.size,
   235→              toolId: tool.id,
   236→              thumbnail: blob,
   237→            });
   238→          })
   239→          .catch(() => undefined);
   240→      }
   241→
   242→      setFiles((prev) => [...prev, ...items]);
   243→      setPhase("workspace");
   244→    },
   245→    [config.maxFiles, files.length, t, toast, tool.id]
   246→  );
   247→
   248→  const removeFile = useCallback((id: string) => {
   249→    setFiles((prev) => prev.filter((item) => item.id !== id));
   250→  }, []);
   251→
   252→  const moveFile = useCallback((from: number, to: number) => {
   253→    setFiles((prev) => {
   254→      if (to < 0 || to >= prev.length) return prev;
   255→      const next = [...prev];
   256→      const [moved] = next.splice(from, 1);
   257→      next.splice(to, 0, moved);
   258→      return next;
   259→    });
   260→  }, []);
   261→
   262→  const clearAll = useCallback(() => {
   263→    setFiles([]);
   264→    void clearWorkingFiles().catch(() => undefined);
   265→    setPhase("landing");
   266→  }, []);
   267→
   268→  const startRun = useCallback(() => {
   269→    if (files.length < config.minFiles) return;
   270→    trackToolUse(tool.id);
   271→    setInputBytes(files.reduce((sum, item) => sum + item.file.size, 0));
   272→    setProgress({ pct: 0, stage: "reading" });
   273→    setPhase("processing");
   274→    runHandleRef.current = toolRunner.run(
   275→      tool.id,
   276→      options,
   277→      files.map((item) => item.file),
   278→      setProgress,
   279→      files.map((item) => item.id)
   280→    );
   281→    runHandleRef.current.promise
   282→      .then((result) => {
   283→        runHandleRef.current = null;
   284→        // Working copies of the INPUTS die with the run (PRD §2.4);
   285→        // results live in memory + can chain into the next tool.
   286→        setOutputs(result.outputs);
   287→        setRunSummary(result.detail ?? null);
   288→        setPhase("result");
   289→        void clearWorkingFiles().catch(() => undefined);
   290→      })
   291→      .catch((error: unknown) => {
   292→        runHandleRef.current = null;
   293→        if (error instanceof RunCancelledError) {
   294→          setPhase("workspace");
   295→          toast({ title: t("tool.cancelledToast") });
   296→          return;
   297→        }
   298→        setPhase("workspace");
   299→        const code =
   300→          error instanceof ToolRunError ? error.code : "tool.errorGeneric";
   301→        toast({
   302→          title: t("tool.errorTitle"),
   303→          description: t(code, {
   304→            detail: error instanceof ToolRunError ? error.detail ?? "" : "",
   305→          }),
   306→          variant: "destructive",
   307→        });
   308→      });
   309→  }, [config.minFiles, files, options, t, toast, tool.id]);
   310→
   311→  const cancelRun = useCallback(() => {
   312→    runHandleRef.current?.cancel();
   313→  }, []);
   314→
   315→  const handleChain = useCallback(
   316→    (target: ToolDef, resultFiles: File[]) => {
   317→      if (!target.enabled) {
   318→        toast({
   319→          title: t("tools.comingToastTitle", { tool: target.name[locale] }),
   320→          description: t("tools.comingToastDesc", {
   321→            phase: target.phase,
   322→          }),
   323→        });
   324→        return;
   325→      }
   326→      setChainHandoff({
   327→        fromToolId: tool.id,
   328→        items: resultFiles.map((file) => ({ id: nextFileId(), file })),
   329→      });
   330→      navigateToHash(ROUTES.tool(target.id));
   331→    },
   332→    [locale, t, toast, tool.id]
   333→  );
   334→
   335→  const removeFromDevice = useCallback(() => {
   336→    void clearWorkingFiles().catch(() => undefined);
   337→    // Remove this run's recent entries by their file metadata (label+size).
   338→    void removeRecentFilesByMeta(
   339→      files.map((item) => ({ label: item.file.name, size: item.file.size }))
   340→    ).catch(() => undefined);
   341→    setOutputs([]);
   342→    setFiles([]);
   343→    setPhase("landing");
   344→    toast({ title: t("result.removedToast") });
   345→  }, [files, t, toast]);
   346→
   347→  const totalSize = files.reduce((sum, item) => sum + item.file.size, 0);
   348→  const ctaLabel = `${tool.name[locale]} ${t("tool.ctaArrow")}`;
   349→  const Icon = tool.icon;
   350→
   351→  return (
   352→    <div className="mx-auto w-full max-w-6xl px-4 pb-28 sm:pb-24">
   353→      {/* Breadcrumb (PRD §6.3): Home > Category > Tool */}
   354→      <nav aria-label={t("a11y.breadcrumb")} className="pt-4">
   355→        <ol className="flex flex-wrap items-center gap-1.5 text-xs text-muted-foreground">
   356→          <li>
   357→            <a
   358→              href={ROUTES.home}
   359→              className="inline-flex items-center gap-1 rounded-btn outline-none hover:text-foreground focus-visible:ring-[3px] focus-visible:ring-ring/50"
   360→            >
   361→              <Home className="size-3" aria-hidden="true" />
   362→              {t("nav.home")}
   363→            </a>
   364→          </li>
   365→          <li aria-hidden="true">/</li>
   366→          <li>
   367→            <a
   368→              href={ROUTES.category(tool.category)}
   369→              className="rounded-btn outline-none hover:text-foreground focus-visible:ring-[3px] focus-visible:ring-ring/50"
   370→            >
   371→              {category.name[locale]}
   372→            </a>
   373→          </li>
   374→          <li aria-hidden="true">/</li>
   375→          <li aria-current="page" className="font-medium text-foreground">
   376→            {tool.name[locale]}
   377→          </li>
   378→        </ol>
   379→      </nav>
   380→
   381→      {phase === "landing" ? (
   382→        <section className="flex flex-col gap-8 pt-6">
   383→          <header className="flex flex-col gap-3">
   384→            <div className="flex items-center gap-3">
   385→              <span
   386→                className={cn(
   387→                  "flex size-11 items-center justify-center rounded-btn",
   388→                  "bg-cat-organize/10 text-cat-organize"
   389→                )}
   390→                aria-hidden="true"
   391→              >
   392→                <Icon className="size-5" />
   393→              </span>
   394→              <h1 className="text-2xl font-bold tracking-tight text-foreground sm:text-3xl">
   395→                {tool.name[locale]}
   396→              </h1>
   397→            </div>
   398→            <p className="max-w-2xl text-sm text-muted-foreground sm:text-base">
   399→              {tool.description[locale]}
   400→            </p>
   401→            <div className="flex flex-wrap gap-2">
   402→              <Badge variant={tool.mode === "D" ? "private" : "server"} className="rounded-chip">
   403→                <Lock className="size-3" aria-hidden="true" />
   404→                {tool.mode === "D"
   405→                  ? t("tools.privateBadge")
   406→                  : t("tools.serverBadge")}
   407→              </Badge>
   408→              <Badge variant="outline" className="rounded-chip">
   409→                {t("tool.freeBadge")}
   410→              </Badge>
   411→            </div>
   412→          </header>
   413→
   414→          <DropZone
   415→            labelKey="tool.selectPdfFiles"
   416→            multiple={config.maxFiles > 1}
   417→            maxFiles={config.maxFiles}
   418→            acceptKind="pdf"
   419→            onAccept={addFiles}
   420→            currentCount={files.length}
   421→          />
   422→
   423→          <div className="grid gap-8 lg:grid-cols-[2fr_1fr]">
   424→            <div className="flex flex-col gap-8">
   425→              {/* How it works */}
   426→              <section aria-labelledby="how-heading">
   427→                <h2 id="how-heading" className="flex items-center gap-2 text-sm font-semibold text-foreground">
   428→                  <ListChecks className="size-4 text-primary" aria-hidden="true" />
   429→                  {t("tool.howTitle")}
   430→                </h2>
   431→                <ol className="mt-3 grid gap-3 sm:grid-cols-3">
   432→                  {["tool.how1", "tool.how2", "tool.how3"].map((key, index) => (
   433→                    <li
   434→                      key={key}
   435→                      className="rounded-card border border-border bg-card p-4"
   436→                    >
   437→                      <span className="flex size-7 items-center justify-center rounded-full bg-primary/10 text-xs font-bold text-primary">
   438→                        {index + 1}
   439→                      </span>
   440→                      <p className="mt-2 text-xs leading-relaxed text-muted-foreground">
   441→                        {t(key)}
   442→                      </p>
   443→                    </li>
   444→                  ))}
   445→                </ol>
   446→              </section>
   447→
   448→              {/* FAQ */}
   449→              <section aria-labelledby="faq-heading">
   450→                <h2 id="faq-heading" className="text-sm font-semibold text-foreground">
   451→                  {t("tool.faqTitle")}
   452→                </h2>
   453→                <Accordion type="single" collapsible className="mt-3">
   454→                  {config.faq.map((item, index) => (
   455→                    <AccordionItem key={index} value={`faq-${index}`}>
   456→                      <AccordionTrigger className="text-left text-sm">
   457→                        {item.q[locale]}
   458→                      </AccordionTrigger>
   459→                      <AccordionContent className="text-xs leading-relaxed text-muted-foreground">
   460→                        {item.a[locale]}
   461→                      </AccordionContent>
   462→                    </AccordionItem>
   463→                  ))}
   464→                </Accordion>
   465→              </section>
   466→            </div>
   467→
   468→            <aside className="flex flex-col gap-8">
   469→              {/* Features */}
   470→              <section aria-labelledby="features-heading">
   471→                <h2 id="features-heading" className="flex items-center gap-2 text-sm font-semibold text-foreground">
   472→                  <Sparkles className="size-4 text-primary" aria-hidden="true" />
   473→                  {t("tool.featuresTitle")}
   474→                </h2>
   475→                <ul className="mt-3 flex flex-col gap-2.5">
   476→                  {config.features.map((feature, index) => (
   477→                    <li
   478→                      key={index}
   479→                      className="flex gap-2 rounded-card border border-border bg-card p-3 text-xs leading-relaxed text-muted-foreground"
   480→                    >
   481→                      <span
   482→                        aria-hidden="true"
   483→                        className="mt-0.5 size-1.5 shrink-0 rounded-full bg-primary"
   484→                      />
   485→                      {feature[locale]}
   486→                    </li>
   487→                  ))}
   488→                </ul>
   489→              </section>
   490→
   491→              {/* Related */}
   492→              <section aria-labelledby="related-heading">
   493→                <h2 id="related-heading" className="text-sm font-semibold text-foreground">
   494→                  {t("tool.relatedTitle")}
   495→                </h2>
   496→                <ul className="mt-3 flex flex-col gap-1.5">
   497→                  {relatedTools.map((item) => {
   498→                    const RelatedIcon = item.icon;
   499→                    return (
   500→                      <li key={item.id}>
   501→                        <a
   502→                          href={item.enabled ? ROUTES.tool(item.id) : ROUTES.category(item.category)}
   503→                          className="flex min-h-11 items-center gap-2.5 rounded-btn px-2 text-sm outline-none transition-colors hover:bg-accent focus-visible:ring-[3px] focus-visible:ring-ring/50"
   504→                        >
   505→                          <RelatedIcon className="size-4 text-muted-foreground" aria-hidden="true" />
   506→                          <span className="truncate text-foreground">
   507→                            {item.name[locale]}
   508→                          </span>
   509→                          {!item.enabled ? (
   510→                            <Badge variant="outline" className="ml-auto rounded-chip text-[10px] text-muted-foreground">
   511→                              {t("tools.soonBadge")}
   512→                            </Badge>
   513→                          ) : null}
   514→                        </a>
   515→                      </li>
   516→                    );
   517→                  })}
   518→                </ul>
   519→              </section>
   520→            </aside>
   521→          </div>
   522→        </section>
   523→      ) : null}
   524→
   525→      {phase === "workspace" ? (
   526→        <section className="pt-6">
   527→          {/* Workspace toolbar (PRD §6.3 state 2) */}
   528→          <div className="flex flex-wrap items-center gap-2">
   529→            <Button
   530→              type="button"
   531→              variant="ghost"
   532→              size="sm"
   533→              onClick={() => setPhase("landing")}
   534→              className="rounded-btn"
   535→            >
   536→              <ArrowLeft className="size-4" aria-hidden="true" />
   537→              {t("tool.back")}
   538→            </Button>
   539→            <span className="ml-auto flex items-center gap-2">
   540→              <Button
   541→                type="button"
   542→                variant="outline"
   543→                size="sm"
   544→                className="rounded-btn"
   545→                onClick={() => setHelpOpen(true)}
   546→                aria-label={t("tool.helpLabel")}
   547→              >
   548→                <CircleHelp className="size-4" aria-hidden="true" />
   549→                <span className="hidden sm:inline">{t("tool.helpLabel")}</span>
   550→              </Button>
   551→              <Button
   552→                type="button"
   553→                variant="outline"
   554→                size="sm"
   555→                className="rounded-btn"
   556→                onClick={clearAll}
   557→              >
   558→                <Trash2 className="size-4" aria-hidden="true" />
   559→                {t("tool.clearAll")}
   560→              </Button>
   561→            </span>
   562→          </div>
   563→
   564→          <div className="mt-4 grid gap-6 lg:grid-cols-[1fr_320px]">
   565→            {/* Preview list (drag-reorder via dnd-kit; PRD §8.1) */}
   566→            <div className="flex flex-col gap-3">
   567→              <FileList items={files} onMove={moveFile} onRemove={removeFile} />
   568→              <DropZone
   569→                labelKey="tool.addMoreFiles"
   570→                multiple={config.maxFiles > 1}
   571→                maxFiles={config.maxFiles}
   572→                acceptKind="pdf"
   573→                onAccept={addFiles}
   574→                currentCount={files.length}
   575→                className="border-dashed py-4"

   577→            </div>

   584→                <div>

   586→                    {t("tool.optionsTitle")}
   587→                  </h2>
   588→                  {CustomPanel ? (
   589→                    <div className="mt-4">
   590→                      <CustomPanel
   591→                        files={files}
   592→                        values={options}
   593→                        onChange={handleOptionChange}
   594→                        setFileOrder={setFiles}
   595→                      />
   596→                    </div>
   597→                  ) : null}
   598→                  <OptionsPanel
   599→                    fields={config.options}
   600→                    values={options}
   601→                    onChange={handleOptionChange}
   602→                    className="mt-4"
   603→                  />
   604→                </div>
   605→                <p className="border-t border-border pt-3 text-[11px] text-muted-foreground">
   606→                  {files.length > 0 && files.length < config.minFiles
   607→                    ? t("tool.summaryMinFiles", { count: config.minFiles })
   608→                    : t("tool.summary", {
   609→                        count: files.length,
   610→                        size: formatBytes(totalSize),
   611→                      })}
   612→                </p>
   613→              </div>
   614→            </aside>
   615→          </div>
   616→
   617→          {/* Sticky CTA (PRD §6.3): count + estimated size + action */}
   618→          <div className="fixed inset-x-0 bottom-0 z-40 border-t border-border bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/85">
   619→            <div className="mx-auto flex max-w-6xl items-center gap-3 px-4 py-3">
   620→              <div className="min-w-0 flex-1">
   621→                <p
   622→                  className={cn(
   623→                    "truncate text-xs",
   624→                    files.length > 0 && files.length < config.minFiles
   625→                      ? "font-medium text-primary"
   626→                      : "text-muted-foreground"
   627→                  )}
   628→                >
   629→                  {files.length > 0 && files.length < config.minFiles
   630→                    ? t("tool.summaryMinFiles", { count: config.minFiles })
   631→                    : t("tool.summary", {
   632→                        count: files.length,
   633→                        size: formatBytes(totalSize),
   634→                      })}
   635→                </p>
   636→                {chainFrom ? (

   638→                    {t("tool.chainFrom", { tool: chainFrom.name[locale] })}

   644→                variant="outline"
   645→                size="touch"
   646→                className="rounded-btn lg:hidden"
   647→                onClick={() => setOptionsOpen(true)}
   648→              >
   649→                {t("tool.optionsButton")}
   650→              </Button>
   651→              <Button
   652→                type="button"
   653→                size="touch"
   654→                className="rounded-btn font-semibold"
   655→                onClick={startRun}
   656→                disabled={files.length < config.minFiles}
   657→              >
   658→                {ctaLabel}
   659→                <ArrowRight className="size-4" aria-hidden="true" />
   660→              </Button>
   661→            </div>
   662→          </div>
   663→
   664→          <BottomSheet
   665→            open={optionsOpen}
   666→            onOpenChange={setOptionsOpen}
   667→            title={t("tool.optionsTitle")}
   668→            description={tool.description[locale]}
   669→            footer={
   670→              <Button
   671→                type="button"
   672→                size="touch"
   673→                className="w-full rounded-btn font-semibold"
   674→                onClick={() => {
   675→                  setOptionsOpen(false);
   676→                  startRun();
   677→                }}
   678→                disabled={files.length < config.minFiles}
   679→              >
   680→                {ctaLabel}
   681→                <ArrowRight className="size-4" aria-hidden="true" />
   682→              </Button>
   683→            }
   684→          >
   685→            {CustomPanel ? (
   686→              <div className="mb-4">
   687→                <CustomPanel
   688→                  files={files}
   689→                  values={options}
   690→                  onChange={handleOptionChange}
   691→                  setFileOrder={setFiles}
   692→                />
   693→              </div>
   694→            ) : null}
   695→            <OptionsPanel
   696→              fields={config.options}
   697→              values={options}
   698→              onChange={handleOptionChange}
   699→            />
   700→          </BottomSheet>
   701→        </section>
   702→      ) : null}
   703→
   704→      {phase === "processing" ? (
   705→        <ProcessingView
   706→          toolName={tool.name[locale]}
   707→          progress={progress}
   708→          onCancel={cancelRun}
   709→        />
   710→      ) : null}
   711→
   712→      {phase === "result" ? (
   713→        <ResultView
   714→          tool={tool}
   715→          outputs={outputs}
   716→          inputBytes={inputBytes}
   717→          runSummary={runSummary}
   718→          chainFromName={chainFrom ? chainFrom.name[locale] : undefined}
   719→          chainTargets={chainTargets}
   720→          onChain={handleChain}
   721→          onRemoveFromDevice={removeFromDevice}
   722→        />
   723→      ) : null}
   724→
   725→      {/* Help dialog */}
   726→      <Modal
   727→        open={helpOpen}
   728→        onOpenChange={setHelpOpen}
   729→        title={t("tool.helpTitle")}
   730→        footer={
   731→          <Button
   732→            type="button"
   733→            variant="default"
   734→            size="touch"
   735→            className="w-full rounded-btn sm:w-auto"
   736→            onClick={() => setHelpOpen(false)}
   737→          >
   738→            {t("pro.action")}
   739→          </Button>
   740→        }
   741→      >
   742→        <div className="flex flex-col gap-3 text-sm text-muted-foreground">
   743→          <p className="flex gap-2">
   744→            <FilePlus2 className="mt-0.5 size-4 shrink-0 text-primary" aria-hidden="true" />
   745→            {t("tool.how2")}
   746→          </p>
   747→          <p>{t("tool.helpPrivacy")}</p>
   748→        </div>
   749→      </Modal>
   750→    </div>
   751→  );

    64→/** Minimal XML well-formedness: balanced tags, quoted attrs, no raw < in text. */
    65→function xmlWellFormed(xml: string): boolean {
    66→  // The prolog / comments / DOCTYPE are not elements — strip them so
    67→  // the tag scanner only sees real tags (<?xml …?> has an illegal name).
    68→  const src = xml
    69→    .replace(/<\?[\s\S]*?\?>/g, "")
    70→    .replace(/<!--[\s\S]*?-->/g, "")
    71→    .replace(/<!DOCTYPE[^>]*>/gi, "");
    72→  const tagRe = /<(\/?)([A-Za-z_:][\w.:-]*)((?:\s+[\w.:-]+\s*=\s*"[^"]*")*)\s*(\/?)>/g;
    73→  const stack: string[] = [];
    74→  let lastIndex = 0;
    75→  let match: RegExpExecArray | null;
    76→  while ((match = tagRe.exec(src)) !== null) {
    77→    if (src.slice(lastIndex, match.index).includes("<")) return false;
    78→    lastIndex = tagRe.lastIndex;
    79→    const [full, close, name, , selfClose] = match;
    80→    if (selfClose === "/" || full.startsWith("<?") || full.startsWith("<!")) {
    81→      continue;
    82→    }
    83→    if (close === "/") {
    84→      if (stack.pop() !== name) return false;
    85→    } else {
    86→      stack.push(name);
    87→    }
    88→  }
    89→  if (src.slice(lastIndex).includes("<")) return false;
    90→  return stack.length === 0;
    91→}

    93→function countOf(haystack: string, needle: string): number {
    94→  return haystack.split(needle).length - 1;
    95→}

    97→/** Synthetic pdf.js item factory for the pure-model section. */
    98→function item(
    99→  str: string,
   100→  x: number,
   101→  y: number,
   102→  size: number,
   103→  opts: { fontName?: string; bold?: boolean; width?: number } = {}
   104→): PdfTextItemInput {
   105→  return {
   106→    str,
   107→    transform: [size, 0, 0, size, x, y],
   108→    width: opts.width ?? str.length * size * 0.5,
   109→    fontName: opts.fontName ?? "F",
   110→    bold: opts.bold,
   111→  };
   112→}

     1→/**
     2→ * Create fillable PDF form — pure model (P3-FORM, the 65th tool).
     3→ *
     4→ * The ZERO-MISTAKE core of the visual builder. Everything here is pure
     5→ * (no DOM, no storage, no pdf-lib) and the smoke script
     6→ * (scripts/smoke-create-fillable-pdf-form.ts) re-proves every function
     7→ * with hand-computed vectors:
     8→ *
     9→ *  - CANVAS MODEL: elements live in CSS-px space on an A4 page at 96 dpi
    10→ *    (page = 595.28pt × 841.89pt ÷ 0.75 = 793.71×1122.52 px, margin 40 px,
    11→ *    8 px grid). px→pt is an exact ×0.75 with a y-flip:
    12→ *    PDF y = (PAGE_H_PX − y − h) × 0.75.
    13→ *  - NAMES: field names are formula-safe slugs ([a-z0-9_], never a leading
    14→ *    digit, never a hyphen — a hyphen would be ambiguous with the minus
    15→ *    operator). Collisions dedupe with an "_2" suffix (the make-pdf-
    16→ *    fillable "Name 2" pattern, hyphen-free so `total_2 * price` stays one
    17→ *    token).
    18→ *  - FORMULA COMPILER: tokenizer → recursive-descent parser → validation
    19→ *    (unknown identifier with near-match, non-number reference, duplicate
    20→ *    target, empty/syntax errors, self/cycle detection with the exact
    21→ *    path) → stable topological order → per-target JavaScript emit.
    22→ *  - PARSE: parseFormDoc coerces a serialized form at the worker boundary
    23→ *    (unknown kinds dropped, geometry clamped to the page, NaN guards,
    24→ *    name dedupe, ASCII-safe export values).
    25→ */
    26→
    27→/* ------------------------------------------------------------------ */
    28→/* Canvas geometry (px @96dpi on A4; ×0.75 → pt)                        */
    29→/* ------------------------------------------------------------------ */
    30→
    31→export const PT_PER_PX = 0.75;
    32→export const PAGE_W_PT = 595.28;
    33→export const PAGE_H_PT = 841.89;
    34→export const PAGE_W_PX = PAGE_W_PT / PT_PER_PX; // 793.7066…
    35→export const PAGE_H_PX = PAGE_H_PT / PT_PER_PX; // 1122.52
    36→export const MARGIN_PX = 40;
    37→export const GRID_PX = 8;
    38→
    39→/** Display px rect (x,y from TOP) → PDF pt rect (y from BOTTOM). */
    40→export function pxToPtRect(rect: {
    41→  x: number;
    42→  y: number;
    43→  w: number;
    44→  h: number;
    45→}): { x: number; y: number; w: number; h: number } {
    46→  return {
    47→    x: rect.x * PT_PER_PX,
    48→    y: (PAGE_H_PX - rect.y - rect.h) * PT_PER_PX,
    49→    w: rect.w * PT_PER_PX,
    50→    h: rect.h * PT_PER_PX,
    51→  };
    52→}
    53→
    54→/** Display px point (x, y from top) → PDF pt point (y from bottom). */
    55→export function pxToPt(x: number, y: number): { x: number; y: number } {
    56→  return { x: x * PT_PER_PX, y: (PAGE_H_PX - y) * PT_PER_PX };
    57→}
    58→
    59→export function snap(v: number, grid = GRID_PX): number {
    60→  return Math.round(v / grid) * grid;
    61→}
    62→
    63→export function clampNum(v: unknown, min: number, max: number, fallback: number): number {
    64→  const n = typeof v === "number" ? v : Number(v);
    65→  if (!Number.isFinite(n)) return fallback;
    66→  return Math.min(max, Math.max(min, n));
    67→}

   187→/** The interactive box rect inside a field element (px, y from top). */
   188→export function boxRect(el: FieldEl): {
   189→  x: number;
   190→  y: number;
   191→  w: number;
   192→  h: number;
   193→} {
   194→  const lh = labelHeight(el);
   195→  if (el.kind === "checkbox") {
   196→    // Square centred in the area below the label (make-pdf-fillable pattern).
   197→    const size = Math.max(8, Math.floor(Math.min(el.w, el.h - lh)));
   198→    return {
   199→      x: el.x,
   200→      y: el.y + lh + Math.max(0, (el.h - lh - size) / 2),
   201→      w: size,
   202→      h: size,
   203→    };
   204→  }
   205→  return { x: el.x, y: el.y + lh, w: el.w, h: Math.max(12, el.h - lh) };
   206→}

   208→/** Radio row height derived from the element height and option count. */
   209→export function radioRowHeight(el: FieldEl): number {
   210→  const n = Math.max(1, el.options.filter((o) => o.trim() !== "").length || 1);
   211→  return Math.max(16, Math.round((el.h - labelHeight(el)) / n));
   212→}
   213→
   214→/** Radio circle side (px). */
   215→export function radioCircle(el: FieldEl): number {
   216→  return Math.max(8, Math.min(14, radioRowHeight(el) - 4));
   217→}

   219→/** Rows for a radio group: per-option circle rect (px, y from top). */
   220→export function radioRows(el: FieldEl): {
   221→  y: number;
   222→  h: number;
   223→  circle: number;
   224→  cy: number;
   225→}[] {
   226→  const lh = labelHeight(el);
   227→  const rowH = radioRowHeight(el);
   228→  const circle = radioCircle(el);
   229→  return el.options.map((_, i) => ({
   230→    y: el.y + lh + i * rowH,
   231→    h: rowH,
   232→    circle,
   233→    cy: el.y + lh + i * rowH + (rowH - circle) / 2,
   234→  }));
   235→}

   240→  return labelHeight(el) + Math.max(1, count) * rowH;

   247→export type Measurer = (text: string, fs: number) => number;
   248→
   249→/** Greedy word wrap with hard-break for over-long words; \n honored. */
   250→export function wrapText(
   251→  text: string,
   252→  maxW: number,
   253→  fs: number,
   254→  measure: Measurer,
   255→): string[] {
   256→  const out: string[] = [];
   257→  for (const para of String(text ?? "").split("\n")) {
   258→    if (para === "") {
   259→      out.push("");
   260→      continue;
   261→    }
   262→    let line = "";
   263→    for (const word of para.split(/\s+/).filter(Boolean)) {
   264→      const next = line ? line + " " + word : word;
   265→      if (measure(next, fs) <= maxW) {
   266→        line = next;
   267→        continue;
   268→      }
   269→      if (!line) {
   270→        // Single word wider than the box: hard-break it.
   271→        let chunk = "";
   272→        for (const ch of word) {
   273→          if (measure(chunk + ch, fs) > maxW && chunk) {
   274→            out.push(chunk);
   275→            chunk = ch;
   276→          } else {
   277→            chunk += ch;
   278→          }
   279→        }
   280→        line = chunk;
   281→        continue;
   282→      }
   283→      out.push(line);
   284→      line = word;
   285→    }
   286→    out.push(line);
   287→  }
   288→  return out.length ? out : [""];
   289→}

   291→/* ------------------------------------------------------------------ */
   292→/* Names: slug + dedupe + ASCII safety                                  */
   293→/* ------------------------------------------------------------------ */
   294→
   295→/**
   296→ * Formula-safe slug: lowercase, [a-z0-9_] only, no leading digit, never
   297→ * empty. Hyphens/spaces fold to "_" (a hyphen in a name would be
   298→ * ambiguous with the minus operator in expressions).
   299→ */
   300→export function slugify(s: string): string {
   301→  const base = String(s ?? "")
   302→    .toLowerCase()
   303→    .replace(/[^a-z0-9]+/g, "_")
   304→    .replace(/^_+|_+$/g, "")
   305→    .slice(0, 40);
   306→  if (base === "") return "field";
   307→  if (/^[0-9]/.test(base)) return "f_" + base;
   308→  return base;
   309→}
   310→
   311→/** Dedupe a slug against taken names: "qty", "qty_2", "qty_3"… */
   312→export function uniqueName(base: string, taken: Set<string>): string {
   313→  const clean = base.slice(0, 40) || "field";
   314→  if (!taken.has(clean)) {
   315→    taken.add(clean);
   316→    return clean;
   317→  }
   318→  for (let n = 2; n < 500; n += 1) {
   319→    const candidate = `${clean}_${n}`;
   320→    if (!taken.has(candidate)) {
   321→      taken.add(candidate);
   322→      return candidate;
   323→    }
   324→  }
   325→  const stamped = `${clean}_${Date.now()}`;
   326→  taken.add(stamped);
   327→  return stamped;
   328→}

   330→/** WinAnsi-safe ASCII for field-side strings (option export values). */
   331→export function asciiSafe(s: string): string {
   332→  return (
   333→    String(s ?? "")
   334→      .replace(/[\u2018\u2019]/g, "'")
   335→      .replace(/[\u201c\u201d]/g, '"')
   336→      .replace(/[\u2013\u2014]/g, "-")
   337→      .replace(/\u00a0/g, " ")
   338→      // eslint-disable-next-line no-control-regex
   339→      .replace(/[^\x20-\x7e]/g, "")
   340→      .trim()
   341→  );
   342→}
   343→
   344→/** True when asciiSafe would change the string (honest-warning signal). */
   345→export function needsAsciiTrim(s: string): boolean {
   346→  return asciiSafe(s) !== String(s ?? "").trim();
   347→}

   355→/* ------------------------------------------------------------------ */
   356→/* Column-flow placement (next free 8px grid cell, top-left scan)       */
   357→/* ------------------------------------------------------------------ */
   358→
   359→function overlaps(
   360→  a: { x: number; y: number; w: number; h: number },
   361→  b: { x: number; y: number; w: number; h: number },
   362→): boolean {
   363→  return a.x < b.x + b.w && b.x < a.x + a.w && a.y < b.y + b.h && b.y < a.y + a.h;
   364→}

   366→/**
   367→ * First free position scanning row by row from the top-left margin in
   368→ * 8px steps (column-flow). Falls back to the margin origin when the page
   369→ * is crowded — the user can drag from there.
   370→ */
   371→export function nextSlot(
   372→  elements: FormElement[],
   373→  w: number,
   374→  h: number,
   375→): { x: number; y: number } {
   376→  const maxX = PAGE_W_PX - MARGIN_PX;
   377→  const maxY = PAGE_H_PX - MARGIN_PX;
   378→  for (let y = MARGIN_PX; ; y += GRID_PX) {
   379→    const clampedY = Math.min(y, maxY);
   380→    for (let x = MARGIN_PX; ; x += GRID_PX) {
   381→      const clampedX = Math.min(x, maxX);
   382→      const probe = {
   383→        x: clampedX,
   384→        y: clampedY,
   385→        w: Math.min(w, maxX - clampedX),
   386→        h,
   387→      };
   388→      if (!elements.some((el) => overlaps(probe, el))) {
   389→        return { x: clampedX, y: clampedY };
   390→      }
   391→      if (x + GRID_PX > maxX) break;
   392→    }
   393→    if (y + GRID_PX > maxY) break;
   394→  }
   395→  return { x: MARGIN_PX, y: MARGIN_PX };
   396→}

   480→/* ------------------------------------------------------------------ */
   481→/* Formula compiler — tokenizer                                         */
   482→/* ------------------------------------------------------------------ */
   483→
   484→export type Tok =
   485→  | { t: "num"; v: string }
   486→  | { t: "id"; v: string }
   487→  | { t: "op"; v: "+" | "-" | "*" | "/" | "(" | ")" };
   488→
   489→export interface TokenizeResult {
   490→  toks: Tok[];
   491→  error: boolean;
   492→}
   493→
   494→export function tokenize(src: string): TokenizeResult {
   495→  const toks: Tok[] = [];
   496→  let i = 0;
   497→  while (i < src.length) {
   498→    const ch = src[i];
   499→    if (/\s/.test(ch)) {
   500→      i += 1;
   501→      continue;
   502→    }
   503→    if (/[0-9]/.test(ch) || (ch === "." && /[0-9]/.test(src[i + 1] ?? ""))) {
   504→      let j = i;
   505→      while (j < src.length && /[0-9]/.test(src[j])) j += 1;
   506→      if (src[j] === ".") {
   507→        j += 1;
   508→        while (j < src.length && /[0-9]/.test(src[j])) j += 1;
   509→      }
   510→      toks.push({ t: "num", v: src.slice(i, j) });
   511→      i = j;
   512→      continue;
   513→    }
   514→    if (/[A-Za-z_]/.test(ch)) {
   515→      let j = i;
   516→      while (j < src.length && /[A-Za-z0-9_]/.test(src[j])) j += 1;
   517→      toks.push({ t: "id", v: src.slice(i, j) });
   518→      i = j;
   519→      continue;
   520→    }
   521→    if (
   522→      ch === "+" ||
   523→      ch === "-" ||
   524→      ch === "*" ||
   525→      ch === "/" ||
   526→      ch === "(" ||
   527→      ch === ")"
   528→    ) {
   529→      toks.push({ t: "op", v: ch });
   530→      i += 1;
   531→      continue;
   532→    }
   533→    return { toks, error: true };
   534→  }
   535→  return { toks, error: false };
   536→}
   537→
   538→/* ------------------------------------------------------------------ */
   539→/* Formula compiler — parser (recursive descent)                        */
   540→/* ------------------------------------------------------------------ */
   541→
   542→/**
   543→ * Grammar:
   544→ *   expr   := term (("+" | "-") term)*
   545→ *   term   := unary (("*" | "/") unary)*
   546→ *   unary  := ("-" | "+") unary | primary
   547→ *   primary:= NUM | ID | "(" expr ")"
   548→ * Unary minus IS supported and survives into the emitted JS verbatim
   549→ * (e.g. `-qty * 2` → `- g("qty") * 2`).
   550→ */
   551→export function parseTokens(toks: Tok[]): boolean {
   552→  let pos = 0;
   553→  const peek = (): Tok | null => toks[pos] ?? null;
   554→  const eat = (): Tok | null => toks[pos++] ?? null;
   555→
   556→  const parsePrimary = (): boolean => {
   557→    const t = peek();
   558→    if (t === null) return false;
   559→    if (t.t === "num" || t.t === "id") {
   560→      eat();
   561→      return true;
   562→    }
   563→    if (t.t === "op" && t.v === "(") {
   564→      eat();
   565→      if (!parseExpr()) return false;
   566→      const close = peek();
   567→      if (close === null || close.t !== "op" || close.v !== ")") return false;
   568→      eat();
   569→      return true;
   570→    }
   571→    return false;
   572→  };
   573→
   574→  const parseUnary = (): boolean => {
   575→    const t = peek();
   576→    if (t !== null && t.t === "op" && (t.v === "-" || t.v === "+")) {
   577→      eat();
   578→      return parseUnary();
   579→    }
   580→    return parsePrimary();
   581→  };
   582→
   583→  const parseTerm = (): boolean => {
   584→    if (!parseUnary()) return false;
   585→    for (;;) {
   586→      const t = peek();
   587→      if (t !== null && t.t === "op" && (t.v === "*" || t.v === "/")) {
   588→        eat();
   589→        if (!parseUnary()) return false;
   590→      } else {
   591→        return true;
   592→      }
   593→    }
   594→  };
   595→
   596→  function parseExpr(): boolean {
   597→    if (!parseTerm()) return false;
   598→    for (;;) {
   599→      const t = peek();
   600→      if (t !== null && t.t === "op" && (t.v === "+" || t.v === "-")) {
   601→        eat();
   602→        if (!parseTerm()) return false;
   603→      } else {
   604→        return true;
   605→      }
   606→    }
   607→  }
   608→
   609→  if (toks.length === 0) return false;
   610→  if (!parseExpr()) return false;
   611→  return pos === toks.length; // trailing operator = syntax error
   612→}
   613→
   614→/** Identifiers referenced by a token stream (order preserved, deduped). */
   615→export function collectIds(toks: Tok[]): string[] {
   616→  const out: string[] = [];
   617→  for (const t of toks) if (t.t === "id" && !out.includes(t.v)) out.push(t.v);
   618→  return out;
   619→}
   620→
   621→/**
   622→ * Emit JS from the ORIGINAL token stream: every identifier becomes
   623→ * g("name"), everything else is verbatim — parens and precedence are
   624→ * preserved by construction.
   625→ */
   626→export function emitJs(toks: Tok[]): string {
   627→  return toks
   628→    .map((t) => (t.t === "id" ? `g(${JSON.stringify(t.v)})` : t.v))
   629→    .join(" ");
   630→}

   636→/**
   637→ * The per-target calculation script written into the field's /AA /C
   638→ * action (ISO 32000-1 §12.6.4). `this` at the outer level is the Doc in
   639→ * Acrobat's calculation context; the IIFE captures it once as `d` so the
   640→ * nested `g` helper can call `d.getField(name)` for every reference —
   641→ * missing fields read as 0 and non-numeric values read as 0, exactly like
   642→ * the pinned template. (Capturing `d` instead of using `this` inside `g`
   643→ * is the one deliberate deviation: a plainly-called `g` would otherwise
   644→ * lose its `this` and the script would not run — proven by executing the
   645→ * emitted string in the smoke battery.)
   646→ */
   647→export const CALC_PREFIX =
   648→  'event.value = (function(){var d=this;var g=function(n){var f=d.getField(n);if(!f){return 0;}var v=parseFloat(f.value);return isNaN(v)?0:v;};return (';
   649→export const CALC_SUFFIX = ');}).call(this);';
   650→
   651→export function calcScript(exprJs: string): string {
   652→  return CALC_PREFIX + exprJs + CALC_SUFFIX;
   653→}

   682→/** Levenshtein ≤ 2 (case-insensitive) → the closest known name. */
   683→export function nearMatch(name: string, known: readonly string[]): string {
   684→  const lower = name.toLowerCase();
   685→  let best = "";
   686→  let bestDist = 3;
   687→  for (const k of known) {
   688→    const d = editDistance(lower, k.toLowerCase());
   689→    if (d < bestDist) {
   690→      bestDist = d;
   691→      best = k;
   692→    }
   693→  }
   694→  return best;
   695→}
   696→
   697→function editDistance(a: string, b: string): number {
   698→  const m = a.length;
   699→  const n = b.length;
   700→  if (Math.abs(m - n) > 2) return 9;
   701→  const dp: number[][] = Array.from({ length: m + 1 }, () => Array(n + 1).fill(0));
   702→  for (let i = 0; i <= m; i++) dp[i][0] = i;
   703→  for (let j = 0; j <= n; j++) dp[0][j] = j;
   704→  for (let i = 1; i <= m; i++) {
   705→    for (let j = 1; j <= n; j++) {
   706→      dp[i][j] = Math.min(
   707→        dp[i - 1][j] + 1,
   708→        dp[i][j - 1] + 1,
   709→        dp[i - 1][j - 1] + (a[i - 1] === b[j - 1] ? 0 : 1),
   710→      );
   711→    }
   712→  }
   713→  return dp[m][n];
   714→}

   716→/** DFS cycle detection; returns the exact cycle path "a → b → a" or null. */
   717→export function findCycle(edges: Map<string, string[]>): string[] | null {
   718→  const state = new Map<string, 1 | 2>(); // 1 = on stack, 2 = done
   719→  const stack: string[] = [];
   720→  const visit = (node: string): string[] | null => {
   721→    const st = state.get(node);
   722→    if (st === 2) return null;
   723→    if (st === 1) {
   724→      const at = stack.indexOf(node);
   725→      return [...stack.slice(at), node];
   726→    }
   727→    state.set(node, 1);
   728→    stack.push(node);
   729→    for (const next of edges.get(node) ?? []) {
   730→      const cycle = visit(next);
   731→      if (cycle) return cycle;
   732→    }
   733→    stack.pop();
   734→    state.set(node, 2);
   735→    return null;
   736→  };
   737→  for (const node of edges.keys()) {
   738→    const cycle = visit(node);
   739→    if (cycle) return cycle;
   740→  }
   741→  return null;
   742→}

   744→/**
   745→ * Compile the rules list. Referencable names: number fields and other
   746→ * rule targets. Output order is a stable topological sort (definition
   747→ * order among ready nodes).
   748→ */
   749→export function compileRules(
   750→  rules: CalcRule[],
   751→  fields: readonly { name: string; kind: FieldKind }[],
   752→): CalcCompile {
   753→  const errors: CalcError[] = [];
   754→  const known = new Set(fields.map((f) => f.name));
   755→  const numeric = new Set(
   756→    fields.filter((f) => f.kind === "number").map((f) => f.name),
   757→  );
   758→
   759→  interface Parsed {
   760→    idx: number;
   761→    target: string;
   762→    exprJs: string;
   763→    ids: string[];
   764→  }
   765→  const parsed: Parsed[] = [];
   766→
   767→  rules.forEach((rule, idx) => {
   768→    const target = String(rule.target ?? "").trim();
   769→    if (target === "") {
   770→      errors.push({ rule: idx, key: "form.calcErrNoTarget" });
   771→      return;
   772→    }
   773→    if (!numeric.has(target)) {
   774→      errors.push({
   775→        rule: idx,
   776→        key: known.has(target) ? "form.calcErrTargetKind" : "form.calcErrTarget",
   777→        name: target,
   778→      });
   779→      return;
   780→    }
   781→    const { toks, error } = tokenize(String(rule.expr ?? ""));
   782→    if (error) {
   783→      errors.push({ rule: idx, key: "form.calcErrChar" });
   784→      return;
   785→    }
   786→    if (toks.length === 0) {
   787→      errors.push({ rule: idx, key: "form.calcErrEmpty" });
   788→      return;
   789→    }
   790→    if (!parseTokens(toks)) {
   791→      errors.push({ rule: idx, key: "form.calcErrSyntax" });
   792→      return;
   793→    }
   794→    const ids = collectIds(toks);
   795→    let ok = true;
   796→    for (const id of ids) {
   797→      if (!numeric.has(id)) {
   798→        if (known.has(id)) {
   799→          errors.push({ rule: idx, key: "form.calcErrNotNumber", name: id });
   800→        } else {
   801→          errors.push({
   802→            rule: idx,
   803→            key: "form.calcErrUnknown",
   804→            name: id,
   805→            near: nearMatch(id, [...numeric]),
   806→          });
   807→        }
   808→        ok = false;
   809→      }
   810→    }
   811→    if (!ok) return;
   812→    parsed.push({ idx, target, exprJs: emitJs(toks), ids });
   813→  });
   814→
   815→  // Duplicate targets.
   816→  const seen = new Map<string, number>();
   817→  for (const p of parsed) {
   818→    if (seen.has(p.target)) {
   819→      errors.push({
   820→        rule: p.idx,
   821→        key: "form.calcErrDup",
   822→        name: p.target,
   823→      });
   824→    } else {
   825→      seen.set(p.target, p.idx);
   826→    }
   827→  }
   828→
   829→  // Cycle detection (includes self-reference a = a + 1).
   830→  const edgeMap = new Map<string, string[]>();
   831→  for (const p of parsed) {
   832→    edgeMap.set(p.target, p.ids.filter((id) => numeric.has(id)));
   833→  }
   834→  const cycle = findCycle(edgeMap);
   835→  if (cycle) {
   836→    errors.push({ rule: -1, key: "form.calcErrCycle", detail: cycle.join(" → ") });
   837→  }
   838→
   839→  if (errors.length > 0) {
   840→    return { ok: false, errors, compiled: [] };
   841→  }
   842→
   843→  // Stable topological order (first ready rule in definition order wins).
   844→  const order: string[] = [];
   845→  const done = new Set<string>();
   846→  while (order.length < parsed.length) {
   847→    let progressed = false;
   848→    for (const p of parsed) {
   849→      if (done.has(p.target)) continue;
   850→      if (p.ids.every((id) => done.has(id) || !edgeMap.has(id))) {
   851→        order.push(p.target);
   852→        done.add(p.target);
   853→        progressed = true;
   854→      }
   855→    }
   856→    if (!progressed) break; // unreachable (cycles rejected above)
   857→  }
   858→
   859→  const compiled: CompiledRule[] = order.map((target) => {
   860→    const p = parsed.find((x) => x.target === target);
   861→    return { target, js: calcScript(p.exprJs), deps: p.ids };
   862→  });
   863→  return { ok: true, errors: [], compiled };
   864→}

   901→function coerceStatic(o: Record<string, unknown>, id: string): StaticEl | null {
   902→  const kind = o.kind;
   903→  if (
   904→    kind !== "heading" &&
   905→    kind !== "text" &&
   906→    kind !== "divider" &&
   907→    kind !== "box"
   908→  ) {
   909→    return null;
   910→  }
   911→  const x = num(o.x, MARGIN_PX, 0, PAGE_W_PX - 8);
   912→  const y = num(o.y, MARGIN_PX, 0, PAGE_H_PX - 8);
   913→  const w = num(o.w, 100, 8, PAGE_W_PX - x);
   914→  const h = num(o.h, 18, 8, PAGE_H_PX - y);
   915→  const fs = num(o.fs, kind === "heading" ? 16 : 12, MIN_FS, MAX_FS);
   916→  return { id, kind, x, y, w, h, fs, text: str(o.text, 4000) };
   917→}

   919→const KINDS = new Set<string>(FIELD_KINDS);
   920→
   921→function coerceField(
   922→  o: Record<string, unknown>,
   923→  id: string,
   924→  taken: Set<string>,
   925→): FieldEl | null {
   926→  const kind = o.kind;
   927→  if (typeof kind !== "string" || !KINDS.has(kind)) return null;
   928→  const k = kind as FieldKind;
   929→  const x = num(o.x, MARGIN_PX, 0, PAGE_W_PX - 8);
   930→  const y = num(o.y, MARGIN_PX, 0, PAGE_H_PX - 8);
   931→  const def = FIELD_DEFAULTS[k];
   932→  const w = num(o.w, def.w, 8, PAGE_W_PX - x);
   933→  const h = num(o.h, def.h, 8, PAGE_H_PX - y);
   934→  const fs = num(o.fs, 12, MIN_FS, MAX_FS);
   935→  const label = str(o.label, 200);
   936→  const name = uniqueName(slugify(String(o.name ?? "")), taken);
   937→  let options: string[] = [];
   938→  if (Array.isArray(o.options)) {
   939→    options = o.options
   940→      .filter((v): v is string => typeof v === "string")
   941→      .map((v) => v.trim().slice(0, 80))
   942→      .filter((v) => v !== "")
   943→      .slice(0, MAX_OPTIONS);
   944→  } else if (k === "radio" || k === "dropdown") {
   945→    options = def.options.slice();
   946→  }
   947→  const format = o.format === "rupee" || o.format === "percent" ? o.format : "plain";
   948→  const lh = label ? Math.round(fs * 1.5) : 0;
   949→  return {
   950→    id,
   951→    kind: k,
   952→    x,
   953→    y,
   954→    w,
   955→    h:
   956→      k === "radio"
   957→        ? Math.max(h, lh + Math.max(1, options.length) * 16)
   958→        : h,
   959→    fs,
   960→    label,
   961→    name,
   962→    nameMan: o.nameMan === true,
   963→    required: o.required === true,
   964→    maxLength:
   965→      k === "text" || k === "multiline"
   966→        ? Math.round(num(o.maxLength, 0, 0, 500))
   967→        : 0,
   968→    format: k === "number" ? format : "plain",
   969→    options: k === "radio" || k === "dropdown" ? options : [],
   970→  };
   971→}

   973→function coerceId(v: unknown, fallback: string): string {
   974→  const s = str(v, 24).replace(/[^a-zA-Z0-9_-]/g, "");
   975→  return s || fallback;
   976→}

   978→/**
   979→ * Parse a serialized FormDoc (used by the surface boot, the smoke script
   980→ * and the worker). Everything untrusted is clamped/validated — a
   981→ * tampered payload can never reach the PDF builder as a wrong type, an
   982→ * off-page rect or a duplicate field name. Unreadable JSON falls back to
   983→ * a single blank page.
   984→ */
   985→export function parseFormDoc(raw: unknown): FormDoc {
   986→  let parsed: unknown = raw;
   987→  if (typeof raw === "string" && raw !== "") {
   988→    try {
   989→      parsed = JSON.parse(raw);
   990→    } catch {
   991→      parsed = {};
   992→    }
   993→  }
   994→  if (typeof parsed === "string") parsed = {};
   995→  const o = (typeof parsed === "object" && parsed !== null ? parsed : {}) as Record<
   996→    string,
   997→    unknown
   998→  >;
   999→
  1000→  const pagesRaw = Array.isArray(o.pages) ? o.pages.slice(0, MAX_PAGES) : [];
  1001→  const taken = new Set<string>();
  1002→  let elSeq = 0;
  1003→  const pages: FormPage[] = pagesRaw.map((p, pi) => {
  1004→    const po = (typeof p === "object" && p !== null ? p : {}) as Record<
  1005→      string,
  1006→      unknown
  1007→    >;
  1008→    const elementsRaw = Array.isArray(po.elements)
  1009→      ? po.elements.slice(0, MAX_ELEMENTS_PER_PAGE)
  1010→      : [];
  1011→    const elements: FormElement[] = [];
  1012→    for (const e of elementsRaw) {
  1013→      const eo = (typeof e === "object" && e !== null ? e : {}) as Record<
  1014→        string,
  1015→        unknown
  1016→      >;
  1017→      elSeq += 1;
  1018→      const id = coerceId(eo.id, `e${elSeq}`);
  1019→      const el =
  1020→        eo.kind === "heading" ||
  1021→        eo.kind === "text" ||
  1022→        eo.kind === "divider" ||
  1023→        eo.kind === "box"
  1024→          ? coerceStatic(eo, id)
  1025→          : coerceField(eo, id, taken);
  1026→      if (el) elements.push(el);
  1027→    }
  1028→    return { id: coerceId(po.id, `p${pi + 1}`), elements };
  1029→  });
  1030→
  1031→  const rulesRaw = Array.isArray(o.rules) ? o.rules.slice(0, MAX_RULES) : [];
  1032→  let ruleSeq = 0;
  1033→  const rules: CalcRule[] = [];
  1034→  for (const r of rulesRaw) {
  1035→    const ro = (typeof r === "object" && r !== null ? r : {}) as Record<
  1036→      string,
  1037→      unknown
  1038→    >;
  1039→    ruleSeq += 1;
  1040→    rules.push({
  1041→      id: coerceId(ro.id, `r${ruleSeq}`),
  1042→      target: str(ro.target, 60).trim(),
  1043→      expr: str(ro.expr, 300),
  1044→    });
  1045→  }
  1046→
  1047→  return {
  1048→    title: str(o.title, 120),
  1049→    pages: pages.length ? pages : [{ id: "p1", elements: [] }],
  1050→    rules,
  1051→  };
  1052→}

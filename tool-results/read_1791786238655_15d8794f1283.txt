    12→/**
    13→ * Markdown → PDF engine (PRD §8.6) — PURE. `parseMarkdown` walks the mdast
    14→ * (mdast-util-from-markdown@2.0.2, MIT — chosen over "marked" because our
    15→ * layout engine needs the AST, not HTML) into a small document model;
    16→ * `layoutMd` wraps + paginates it into draw-ready runs and rects. The worker
    17→ * only maps fonts/colors and paints. Everything here is verifiable
    18→ * headlessly with bun (scripts/smoke-md2pdf.ts).
    19→ *
    20→ * Shared wrap/flow/script machinery lives in ../txt-to-pdf/text-layout.ts
    21→ * (file ownership keeps it there — imported here with permission).
    22→ *
    23→ * Supported mdast nodes: heading (1-6), paragraph, list (ordered/unordered,
    24→ * nested via listItem depth), listItem, code, inlineCode, strong, emphasis,
    25→ * link (blue text + numbered footnote + "Links" section at the end), break,
    26→ * blockquote (indent + bar + muted text), thematicBreak, text,
    27→ * linkReference/imageReference (resolved when a matching definition exists,
    28→ * literal "[label]" otherwise), image (honest "[alt]" text — no raster
    29→ * embedding in v1). Omissions documented in the tool FAQ + code: html
    30→ * nodes skipped, GFM tables not parsed by core mdast (they surface as
    31→ * plain paragraph text), no syntax coloring in code blocks.
    32→ *
    33→ * Honest v1 font limits: Devanagari (or any non-WinAnsi block) renders in
    34→ * the regular embedded Noto — no bold/italic/mono variants exist for it
    35→ * yet; links are footnote-style, not clickable annotations (pdf-lib has
    36→ * no high-level link annotation API on drawText — a v2 candidate).
    37→ */

    72→function pickNumber(
    73→  value: unknown,
    74→  allowed: number[],
    75→  fallback: number
    76→): number {
    77→  const n = typeof value === "number" ? value : Number(value);
    78→  return allowed.includes(n) ? n : fallback;
    79→}
    80→
    81→/** Coerces raw panel values into typed options. */
    82→export function resolveMdOptions(values: OptionValues): MdOptions {
    83→  const margin = Number(values.margin);
    84→  return {
    85→    pageSize: values.pageSize === "letter" ? "letter" : "a4",
    86→    margin: Number.isFinite(margin)
    87→      ? Math.min(40, Math.max(10, margin))
    88→      : 20,
    89→    bodySize: pickNumber(values.bodySize, [10, 11, 12], 11),
    90→    monoSize: pickNumber(values.monoSize, [8, 9, 10], 9),
    91→    theme: values.theme === "warm" ? "warm" : "clean",
    92→  };
    93→}

   136→function baseRun(text: string): MdInlineRun {
   137→  return {
   138→    text,
   139→    bold: false,
   140→    italic: false,
   141→    code: false,
   142→    link: false,
   143→    sup: false,
   144→  };
   145→}
   146→
   147→function sameStyle(a: MdInlineRun, b: MdInlineRun): boolean {
   148→  return (
   149→    a.bold === b.bold &&
   150→    a.italic === b.italic &&
   151→    a.code === b.code &&
   152→    a.link === b.link &&
   153→    a.sup === b.sup
   154→  );
   155→}
   156→
   157→/** Merges adjacent runs with identical style (fewer draw calls). */
   158→function mergeRuns(runs: MdInlineRun[]): MdInlineRun[] {
   159→  const out: MdInlineRun[] = [];
   160→  for (const run of runs) {
   161→    if (run.text.length === 0) continue;
   162→    const last = out[out.length - 1];
   163→    if (
   164→      last &&
   165→      sameStyle(last, run) &&
   166→      last.text !== "\n" &&
   167→      run.text !== "\n"
   168→    ) {
   169→      last.text += run.text;
   170→    } else {
   171→      out.push({ ...run });
   172→    }
   173→  }
   174→  return out;
   175→}

   177→interface InlineStyle {
   178→  bold: boolean;
   179→  italic: boolean;
   180→  link: boolean;
   181→}
   182→
   183→function walkPhrasing(
   184→  nodes: readonly PhrasingContent[],
   185→  style: InlineStyle,
   186→  ctx: ParseCtx,
   187→  out: MdInlineRun[]
   188→): void {
   189→  for (const node of nodes) {
   190→    switch (node.type) {
   191→      case "text":
   192→        out.push({
   193→          ...baseRun(node.value),
   194→          bold: style.bold,
   195→          italic: style.italic,
   196→          link: style.link,
   197→        });
   198→        break;
   199→      case "emphasis":
   200→        walkPhrasing(
   201→          node.children,
   202→          { ...style, italic: true },
   203→          ctx,
   204→          out
   205→        );
   206→        break;
   207→      case "strong":
   208→        walkPhrasing(node.children, { ...style, bold: true }, ctx, out);
   209→        break;
   210→      case "inlineCode":
   211→        out.push({ ...baseRun(node.value), code: true });
   212→        break;
   213→      case "break":
   214→        out.push(baseRun("\n"));
   215→        break;
   216→      case "link": {
   217→        walkPhrasing(node.children, { ...style, link: true }, ctx, out);
   218→        const n = ctx.links.push(node.url);
   219→        out.push({
   220→          ...baseRun(` [${n}]`),
   221→          link: true,
   222→          sup: true,
   223→        });
   224→        break;
   225→      }
   226→      case "linkReference": {
   227→        const url = ctx.defs.get(node.identifier);
   228→        if (url !== undefined) {
   229→          walkPhrasing(node.children, { ...style, link: true }, ctx, out);
   230→          const n = ctx.links.push(url);
   231→          out.push({ ...baseRun(` [${n}]`), link: true, sup: true });
   232→        } else {
   233→          out.push(baseRun(`[${node.label ?? node.identifier}]`));
   234→        }
   235→        break;
   236→      }
   237→      case "image":
   238→        out.push({ ...baseRun(`[${node.alt ?? ""}]`), italic: true });
   239→        break;
   240→      case "imageReference": {
   241→        const url = ctx.defs.get(node.identifier);
   242→        const alt = `[${node.alt ?? ""}]`;
   243→        if (url !== undefined) {
   244→          const n = ctx.links.push(url);
   245→          out.push({ ...baseRun(`${alt} [${n}]`), link: true, sup: true });
   246→        } else {
   247→          out.push({ ...baseRun(alt), italic: true });
   248→        }
   249→        break;
   250→      }
   251→      case "html":
   252→        // inline raw HTML — skipped (honest omission, see FAQ)
   253→        break;
   254→      default:
   255→        break;
   256→    }
   257→  }
   258→}
   259→
   260→function inlineRuns(
   261→  nodes: readonly PhrasingContent[],
   262→  ctx: ParseCtx
   263→): MdInlineRun[] {
   264→  const out: MdInlineRun[] = [];
   265→  walkPhrasing(nodes, { bold: false, italic: false, link: false }, ctx, out);
   266→  return mergeRuns(out);
   267→}
   268→
   269→/** Collects reference definitions anywhere in the document (label → url). */
   270→function collectDefinitions(
   271→  nodes: readonly RootContent[],
   272→  into: Map<string, string>
   273→): void {
   274→  for (const node of nodes) {
   275→    if (node.type === "definition") {
   276→      into.set(node.identifier, node.url);
   277→    } else if ("children" in node && Array.isArray(node.children)) {
   278→      collectDefinitions(
   279→        node.children as RootContent[],
   280→        into
   281→      );
   282→    }
   283→  }
   284→}
   285→
   286→function codeLines(value: string): string[] {
   287→  const cleaned = value.replace(/\r\n?/g, "\n").replace(/\t/g, "    ");
   288→  const lines = cleaned.split("\n");
   289→  // fenced blocks implicitly end with a newline — drop that phantom line
   290→  if (lines.length > 1 && lines[lines.length - 1] === "") lines.pop();
   291→  return lines;
   292→}
   293→
   294→function walkBlocks(
   295→  nodes: readonly BlockContent[] | readonly RootContent[],
   296→  ctx: ParseCtx,
   297→  out: MdBlock[],
   298→  depth: number
   299→): void {
   300→  for (const node of nodes) {
   301→    switch (node.type) {
   302→      case "heading":
   303→        out.push({
   304→          kind: "heading",
   305→          level: node.depth,
   306→          runs: inlineRuns(node.children, ctx),
   307→        });
   308→        break;
   309→      case "paragraph":
   310→        out.push({
   311→          kind: "paragraph",
   312→          runs: inlineRuns(node.children, ctx),
   313→        });
   314→        break;
   315→      case "code":
   316→        out.push({ kind: "code", lines: codeLines(node.value) });
   317→        break;
   318→      case "thematicBreak":
   319→        out.push({ kind: "rule" });
   320→        break;
   321→      case "blockquote":
   322→        out.push({
   323→          kind: "quote",
   324→          children: walkBlocksInto(node.children, ctx),
   325→        });
   326→        break;
   327→      case "list": {
   328→        const list = node as List;
   329→        const start = typeof list.start === "number" ? list.start : 1;
   330→        const hoisted: MdBlock[] = [];
   331→        list.children.forEach((item, index) => {
   332→          hoisted.push(...listItemBlocks(item, ctx, depth, list.ordered === true, start + index));
   333→        });
   334→        out.push(...hoisted);
   335→        break;
   336→      }
   337→      case "html":
   338→      case "definition":
   339→        // block raw HTML / reference definitions — skipped (see FAQ)
   340→        break;
   341→      default:
   342→        break;
   343→    }
   344→  }
   345→}
   346→
   347→/**
   348→ * One mdast listItem → its own block (phrasing content merged with hard
   349→ * breaks) + nested lists flattened as deeper item blocks. Block-level
   350→ * code/quote/heading inside an item is hoisted as a top-level block right
   351→ * after the list (a documented v1 simplification — rare in the wild).
   352→ */
   353→function listItemBlocks(
   354→  item: ListItem,
   355→  ctx: ParseCtx,
   356→  depth: number,
   357→  ordered: boolean,
   358→  number: number
   359→): MdBlock[] {
   360→  const out: MdBlock[] = [];
   361→  const runs: MdInlineRun[] = [];
   362→  const after: MdBlock[] = [];
   363→  for (const child of item.children) {
   364→    if (child.type === "paragraph") {
   365→      if (runs.length > 0) runs.push(baseRun("\n"));
   366→      runs.push(...inlineRuns(child.children, ctx));
   367→    } else if (child.type === "list") {
   368→      const list = child as List;
   369→      const start = typeof list.start === "number" ? list.start : 1;
   370→      list.children.forEach((nested, index) => {
   371→        after.push(
   372→          ...listItemBlocks(
   373→            nested,
   374→            ctx,
   375→            depth + 1,
   376→            list.ordered === true,
   377→            start + index
   378→          )
   379→        );
   380→      });
   381→    } else if (child.type === "blockquote") {
   382→      after.push({
   383→        kind: "quote",
   384→        children: walkBlocksInto(child.children, ctx),
   385→      });
   386→    } else if (child.type === "code") {
   387→      after.push({ kind: "code", lines: codeLines(child.value) });
   388→    } else if (child.type === "heading") {
   389→      after.push({
   390→        kind: "heading",
   391→        level: child.depth,
   392→        runs: inlineRuns(child.children, ctx),
   393→      });
   394→    } else if (child.type === "thematicBreak") {
   395→      after.push({ kind: "rule" });
   396→    }
   397→    // html/definition inside items: skipped like everywhere else
   398→  }
   399→  if (runs.length > 0) {
   400→    out.push({ kind: "listItem", depth, ordered, number, runs: mergeRuns(runs) });
   401→  }
   402→  out.push(...after);
   403→  return out;
   404→}
   405→
   406→function walkBlocksInto(
   407→  nodes: readonly BlockContent[],
   408→  ctx: ParseCtx
   409→): MdBlock[] {
   410→  const out: MdBlock[] = [];
   411→  walkBlocks(nodes, ctx, out, 0);
   412→  return out;
   413→}
   414→
   415→/** Parses markdown source into the document model + numbered links. */
   416→export function parseMarkdown(source: string): ParsedMarkdown {
   417→  const root = fromMarkdown(source);
   418→  const ctx: ParseCtx = { links: [], defs: new Map() };
   419→  collectDefinitions(root.children, ctx.defs);
   420→  const blocks: MdBlock[] = [];
   421→  walkBlocks(root.children, ctx, blocks, 0);
   422→  return { blocks, links: ctx.links };
   423→}

   480→function fontSpecFor(run: MdInlineRun): FontSpec {
   481→  if (run.code) return { family: "mono", bold: false, italic: false };
   482→  return { family: "sans", bold: run.bold, italic: run.italic };
   483→}

   485→function runsText(runs: readonly MdInlineRun[]): string {
   486→  return runs.map((run) => run.text).join("");
   487→}

   489→/** Rough advance of a block's first line — heading keep-with-next math. */
   490→function firstLineAdvance(
   491→  block: MdBlock,
   492→  bodyAdvance: number,
   493→  monoAdvance: number
   494→): number {
   495→  switch (block.kind) {
   496→    case "heading":
   497→      return (
   498→        HEADING_SIZES[Math.min(6, Math.max(1, block.level)) - 1] *
   499→        HEADING_LINE_HEIGHT
   500→      );
   501→    case "code":
   502→      return monoAdvance;
   503→    case "quote":
   504→    case "paragraph":
   505→    case "listItem":
   506→    case "rule":
   507→      return bodyAdvance;
   508→  }
   509→}

   511→export function layoutMd(
   512→  parsed: ParsedMarkdown,
   513→  options: MdOptions,
   514→  measure: MeasureRun
   515→): MdLayoutResult {
   516→  const { blocks, links } = parsed;
   517→  const size = PAGE_SIZES[options.pageSize];
   518→  const marginPt = options.margin * MM_TO_PT;
   519→  const geometry: PageGeometry = {
   520→    pageW: size.width,
   521→    pageH: size.height,
   522→    marginLeft: marginPt,
   523→    marginRight: marginPt,
   524→    marginTop: marginPt,
   525→    marginBottom: marginPt,
   526→  };
   527→  const flow = new PageFlow(geometry);
   528→  const pages: MdPageDraw[] = [
   529→    { runs: [], rects: [], warm: options.theme === "warm" },
   530→  ];
   531→  const ensurePage = (index: number): MdPageDraw => {
   532→    while (pages.length <= index) {
   533→      pages.push({ runs: [], rects: [], warm: options.theme === "warm" });
   534→    }
   535→    return pages[index];
   536→  };
   537→
   538→  const bodyAdvance = options.bodySize * BODY_LINE_HEIGHT;
   539→  const monoAdvance = options.monoSize * MONO_LINE_HEIGHT;
   540→  const monoSpec: FontSpec = { family: "mono", bold: false, italic: false };
   541→  const sansSpec: FontSpec = { family: "sans", bold: false, italic: false };
   542→
   543→  /** Open quote extents, innermost last: per-page line extents. */
   544→  const quoteStack: {
   545→    extents: { page: number; top: number; bottom: number }[];
   546→  }[] = [];
   547→
   548→  let indent = 0;
   549→  let muted = false;
   550→  let gapScale = 1;
   551→  let placedAny = false;
   552→
   553→  const contentLeft = (): number => geometry.marginLeft + indent;
   554→
   555→  const toWrapRuns = (runs: readonly MdInlineRun[]): WrapRun[] => {
   556→    const out: WrapRun[] = [];
   557→    for (const run of runs) {
   558→      const color: RunColor = run.link
   559→        ? "link"
   560→        : muted
   561→          ? "muted"
   562→          : "text";
   563→      out.push({
   564→        text: run.text,
   565→        spec: fontSpecFor(run),
   566→        color,
   567→        sup: run.sup,
   568→      });
   569→    }
   570→    return out;
   571→  };
   572→
   573→  /**
   574→   * Wraps one run group (hard-break segments flow without extra gaps) and
   575→   * places it with orphan/widow protection. Registers extents with every
   576→   * open quote so their bars size correctly across page breaks.
   577→   */
   578→  const placeRuns = (
   579→    runs: readonly MdInlineRun[],
   580→    x: number,
   581→    maxWidth: number,
   582→    fontSize: number,
   583→    advance: number
   584→  ): { page: number; top: number; bottom: number; baseline: number }[] => {
   585→    const script = blockScript(runsText(runs));
   586→    const segments = splitRunsAtBreaks(toWrapRuns(runs));
   587→    const allLines: WrapRun[][] = [];
   588→    for (const segment of segments) {
   589→      if (segment.length === 0) {
   590→        allLines.push([]);
   591→        continue;
   592→      }
   593→      allLines.push(
   594→        ...wrapRuns(segment, { maxWidth, size: fontSize, script, measure })
   595→      );
   596→    }
   597→    const placements = planLineGroup(flow, allLines.length, advance, fontSize);
   598→    const extents: {
   599→      page: number;
   600→      top: number;
   601→      bottom: number;
   602→      baseline: number;
   603→    }[] = [];
   604→    for (let i = 0; i < placements.length; i += 1) {
   605→      const place = placements[i];
   606→      const page = ensurePage(place.pageIndex);
   607→      page.runs.push(
   608→        ...positionRuns(allLines[i], x, place.baseline, fontSize, script, measure)
   609→      );
   610→      const extent = {
   611→        page: place.pageIndex,
   612→        top: place.top,
   613→        bottom: place.top + advance,
   614→        baseline: place.baseline,
   615→      };
   616→      extents.push(extent);
   617→      for (const open of quoteStack) open.extents.push(extent);
   618→    }
   619→    return extents;
   620→  };
   621→
   622→  const gapBefore = (gap: number): void => {
   623→    if (!placedAny) return;
   624→    flow.consumeGap(gap * gapScale);
   625→  };
   626→
   627→  const placeCodeBlock = (lines: string[]): void => {
   628→    if (lines.length === 0) return;
   629→    const blockX = contentLeft();
   630→    const height = lines.length * monoAdvance + 2 * CODE_PAD;
   631→    // ≤3-line blocks never split across pages (when a full page could hold
   632→    // them); longer blocks split without 1-line fragments (strictSingle)
   633→    if (
   634→      lines.length <= 3 &&
   635→      flow.pageHasContent &&
   636→      !flow.fits(height) &&
   637→      height <= flow.contentHeight
   638→    ) {
   639→      flow.breakPage();
   640→    }
   641→    gapBefore(bodyAdvance * BLOCK_GAP);
   642→
   643→    // Wrapped + positioned per source line (code wraps only overlong
   644→    // lines; indentation is preserved as an x offset per line)
   645→    const indentWidths = lines.map((line) => {
   646→      const indentMatch = /^[ ]*/.exec(line);
   647→      const indentText = indentMatch ? indentMatch[0] : "";
   648→      return {
   649→        text: indentText,
   650→        width: measure(indentText, monoSpec, options.monoSize, "std"),
   651→        body: line.slice(indentText.length).replace(/[ ]+$/, ""),
   652→      };
   653→    });
   654→    const maxIndent = indentWidths.reduce(
   655→      (max, entry) => Math.max(max, entry.width),
   656→      0
   657→    );
   658→    const wrapWidth = Math.max(
   659→      24,
   660→      flow.contentWidth - indent - 2 * CODE_PAD - maxIndent
   661→    );
   662→    const wrappedLines: { runs: WrapRun[]; script: ScriptFont; indent: number }[] =
   663→      indentWidths.map((entry) => {
   664→        const script = blockScript(entry.text + entry.body || " ");
   665→        const wrapped = wrapRuns([{ text: entry.body, spec: monoSpec }], {
   666→          maxWidth: wrapWidth,
   667→          size: options.monoSize,
   668→          script,
   669→          measure,
   670→        });
   671→        return {
   672→          runs: wrapped[0] ?? [],
   673→          script,
   674→          indent: entry.width,
   675→        };
   676→      });
   677→
   678→    const placements = planLineGroup(
   679→      flow,
   680→      wrappedLines.length,
   681→      monoAdvance,
   682→      options.monoSize,
   683→      true,
   684→      CODE_PAD
   685→    );
   686→    const fragmentByPage = new Map<number, { top: number; count: number }>();
   687→    for (let i = 0; i < placements.length; i += 1) {
   688→      const place = placements[i];
   689→      const fragment = fragmentByPage.get(place.pageIndex);
   690→      if (fragment) {
   691→        fragment.count += 1;
   692→      } else {
   693→        fragmentByPage.set(place.pageIndex, { top: place.top, count: 1 });
   694→      }
   695→      const line = wrappedLines[i];
   696→      const page = ensurePage(place.pageIndex);
   697→      page.runs.push(
   698→        ...positionRuns(
   699→          line.runs,
   700→          blockX + CODE_PAD + line.indent,
   701→          place.baseline,
   702→          options.monoSize,
   703→          line.script,
   704→          measure
   705→        )
   706→      );
   707→      const extent = {
   708→        page: place.pageIndex,
   709→        top: place.top,
   710→        bottom: place.top + monoAdvance,
   711→      };
   712→      for (const open of quoteStack) open.extents.push(extent);
   713→    }
   714→    for (const [pageIndex, fragment] of fragmentByPage) {
   715→      const page = ensurePage(pageIndex);
   716→      // no top padding when the fragment starts at the content top — the
   717→      // page margin already provides the gap there
   718→      const topPad = fragment.top > 0 ? CODE_PAD : 0;
   719→      const topY =
   720→        geometry.pageH - geometry.marginTop - fragment.top + topPad;
   721→      page.rects.push({
   722→        x: blockX,
   723→        y: topY - fragment.count * monoAdvance - topPad - CODE_PAD,
   724→        w: flow.contentWidth - indent,
   725→        h: fragment.count * monoAdvance + topPad + CODE_PAD,
   726→        kind: "codeBg",
   727→      });
   728→    }
   729→  };
   730→
   731→  const placeBlock = (
   732→    block: MdBlock,
   733→    next: MdBlock | null
   734→  ): void => {
   735→    const bodyGap = bodyAdvance * BLOCK_GAP;
   736→    switch (block.kind) {
   737→      case "heading": {
   738→        const level = Math.min(6, Math.max(1, block.level));
   739→        const fontSize = HEADING_SIZES[level - 1];
   740→        const advance = fontSize * HEADING_LINE_HEIGHT;
   741→        // keep-with-next: heading + ~2 lines of the following block, else
   742→        // the heading never dangles at the very bottom of a page
   743→        const nextAdvance = next
   744→          ? firstLineAdvance(next, bodyAdvance, monoAdvance)
   745→          : bodyAdvance;
   746→        const needed = advance + 2 * nextAdvance;
   747→        if (flow.pageHasContent && !flow.fits(needed)) flow.breakPage();
   748→        gapBefore(HEADING_SPACING[level - 1][0]);
   749→        const headingRuns: MdInlineRun[] = block.runs.map((run) => ({
   750→          ...run,
   751→          bold: true,
   752→          italic: level === 6,
   753→          code: false,
   754→        }));
   755→        placeRuns(
   756→          headingRuns,
   757→          contentLeft(),
   758→          flow.contentWidth - indent,
   759→          fontSize,
   760→          advance
   761→        );
   762→        gapBefore(HEADING_SPACING[level - 1][1]);
   763→        break;
   764→      }
   765→      case "paragraph": {
   766→        if (block.runs.length === 0) return;
   767→        gapBefore(bodyGap);
   768→        placeRuns(
   769→          block.runs,
   770→          contentLeft(),
   771→          flow.contentWidth - indent,
   772→          options.bodySize,
   773→          bodyAdvance
   774→        );
   775→        break;
   776→      }
   777→      case "code": {
   778→        placeCodeBlock(block.lines);
   779→        break;
   780→      }
   781→      case "rule": {
   782→        gapBefore(bodyGap);
   783→        if (flow.pageHasContent && !flow.fits(RULE_THICKNESS * 2)) {
   784→          flow.breakPage();
   785→        }
   786→        const page = ensurePage(flow.pageIndex);
   787→        const y = flow.contentTopY() - RULE_THICKNESS;
   788→        page.rects.push({
   789→          x: contentLeft(),
   790→          y,
   791→          w: flow.contentWidth - indent,
   792→          h: RULE_THICKNESS * 2,
   793→          kind: "rule",
   794→        });
   795→        flow.consume(RULE_THICKNESS * 2);
   796→        gapBefore(bodyGap * 0.5);
   797→        break;
   798→      }
   799→      case "quote": {
   800→        gapBefore(bodyGap);
   801→        indent += QUOTE_INDENT;
   802→        muted = true;
   803→        gapScale *= 0.7;
   804→        quoteStack.push({ extents: [] });
   805→        for (let i = 0; i < block.children.length; i += 1) {
   806→          placeBlock(
   807→            block.children[i],
   808→            block.children[i + 1] ?? null
   809→          );
   810→        }
   811→        const open = quoteStack.pop();
   812→        indent -= QUOTE_INDENT;
   813→        muted = false;
   814→        gapScale /= 0.7;
   815→        if (open && open.extents.length > 0) {
   816→          const byPage = new Map<number, { top: number; bottom: number }>();
   817→          for (const extent of open.extents) {
   818→            const span = byPage.get(extent.page);
   819→            if (span) {
   820→              span.top = Math.min(span.top, extent.top);
   821→              span.bottom = Math.max(span.bottom, extent.bottom);
   822→            } else {
   823→              byPage.set(extent.page, {
   824→                top: extent.top,
   825→                bottom: extent.bottom,
   826→              });
   827→            }
   828→          }
   829→          for (const [pageIndex, span] of byPage) {
   830→            const page = ensurePage(pageIndex);
   831→            const topY = geometry.pageH - geometry.marginTop - span.top;
   832→            page.rects.push({
   833→              x: contentLeft(),
   834→              y: topY - span.bottom - QUOTE_BAR_OVERHANG,
   835→              w: QUOTE_BAR_WIDTH,
   836→              h: span.bottom - span.top + 2 * QUOTE_BAR_OVERHANG,
   837→              kind: "quoteBar",
   838→            });
   839→          }
   840→        }
   841→        break;
   842→      }
   843→      case "listItem": {
   844→        gapBefore(bodyAdvance * ITEM_GAP);
   845→        const markerX = contentLeft() + block.depth * LIST_INDENT;
   846→        const marker = block.ordered
   847→          ? `${block.number}.`
   848→          : MARKERS[Math.min(MARKERS.length - 1, block.depth)];
   849→        const markerWidth = measure(
   850→          marker,
   851→          sansSpec,
   852→          options.bodySize,
   853→          "std"
   854→        );
   855→        const textX = markerX + markerWidth + MARKER_GAP;
   856→        const maxWidth = Math.max(
   857→          24,
   858→          geometry.marginLeft + flow.contentWidth - textX
   859→        );
   860→        const extents = placeRuns(
   861→          block.runs,
   862→          textX,
   863→          maxWidth,
   864→          options.bodySize,
   865→          bodyAdvance
   866→        );
   867→        if (extents.length > 0) {
   868→          const page = ensurePage(extents[0].page);
   869→          page.runs.push({
   870→            text: marker,
   871→            x: markerX,
   872→            baseline: extents[0].baseline,
   873→            size: options.bodySize,
   874→            spec: sansSpec,
   875→            script: "std",
   876→            color: muted ? "muted" : "text",
   877→          });
   878→        }
   879→        break;
   880→      }
   881→    }
   882→    if (block.kind !== "paragraph" || block.runs.length > 0) {
   883→      placedAny = true;
   884→    }
   885→  };
   886→
   887→  for (let i = 0; i < blocks.length; i += 1) {
   888→    placeBlock(blocks[i], blocks[i + 1] ?? null);
   889→  }
   890→
   891→  // Links section: bold "Links" heading + one numbered line per URL
   892→  if (links.length > 0) {
   893→    gapBefore(bodyAdvance * BLOCK_GAP);
   894→    placeRuns(
   895→      [{ ...baseRun("Links"), bold: true }],
   896→      contentLeft(),
   897→      flow.contentWidth - indent,
   898→      12,
   899→      12 * HEADING_LINE_HEIGHT
   900→    );
   901→    for (let n = 1; n <= links.length; n += 1) {
   902→      placeRuns(
   903→        [baseRun(`${n}. `), { ...baseRun(links[n - 1]), link: true }],
   904→        contentLeft(),
   905→        flow.contentWidth - indent,
   906→        options.bodySize,
   907→        bodyAdvance
   908→      );
   909→    }
   910→  }
   911→
   912→  return { pages, linkCount: links.length };
   913→}

   142→export interface PageView {

   144→  x0: number;

   151→
   152→/** Display (visible) page size in points, /Rotate applied. */
   153→export function displaySize(view: PageView): { w: number; h: number } {
   154→  const uw = view.x1 - view.x0;
   155→  const uh = view.y1 - view.y0;
   156→  return view.rotate === 90 || view.rotate === 270
   157→    ? { w: uh, h: uw }
   158→    : { w: uw, h: uh };
   159→}
   160→
   161→/** User-space point → display point (y down, origin top-left). */
   162→export function pdfToDisplayPoint(
   163→  px: number,
   164→  py: number,
   165→  view: PageView
   166→): { x: number; y: number } {
   167→  switch (view.rotate) {
   168→    case 90:
   169→      return { x: py - view.y0, y: px - view.x0 };
   170→    case 180:
   171→      return { x: view.x1 - px, y: py - view.y0 };
   172→    case 270:
   173→      return { x: view.y1 - py, y: view.x1 - px };
   174→    default:
   175→      return { x: px - view.x0, y: view.y1 - py };
   176→  }
   177→}
   178→
   179→/** Display point → user-space point (inverse of pdfToDisplayPoint). */
   180→export function displayToPdfPoint(
   181→  dx: number,
   182→  dy: number,
   183→  view: PageView
   184→): { x: number; y: number } {
   185→  switch (view.rotate) {
   186→    case 90:
   187→      return { x: dy + view.x0, y: dx + view.y0 };
   188→    case 180:
   189→      return { x: view.x1 - dx, y: view.y0 + dy };
   190→    case 270:
   191→      return { x: view.x1 - dy, y: view.y1 - dx };
   192→    default:
   193→      return { x: dx + view.x0, y: view.y1 - dy };
   194→  }
   195→}
   196→
   197→/** PDF rect (y from bottom) → display rect (y from top), /Rotate-aware. */
   198→export function pdfRectToDisplay(rect: ObjRect, view: PageView): ObjRect {
   199→  const corners = [
   200→    pdfToDisplayPoint(rect.x, rect.y, view),
   201→    pdfToDisplayPoint(rect.x + rect.w, rect.y + rect.h, view),
   202→  ];
   203→  const xs = corners.map((c) => c.x);
   204→  const ys = corners.map((c) => c.y);
   205→  const x = Math.min(...xs);
   206→  const y = Math.min(...ys);
   207→  return { x, y, w: Math.max(...xs) - x, h: Math.max(...ys) - y };
   208→}
   209→
   210→/** Display rect (y from top) → PDF rect (y from bottom), /Rotate-aware. */
   211→export function displayRectToPdf(rect: ObjRect, view: PageView): ObjRect {
   212→  const corners = [
   213→    displayToPdfPoint(rect.x, rect.y, view),
   214→    displayToPdfPoint(rect.x + rect.w, rect.y + rect.h, view),
   215→  ];
   216→  const xs = corners.map((c) => c.x);
   217→  const ys = corners.map((c) => c.y);
   218→  const x = Math.min(...xs);
   219→  const y = Math.min(...ys);
   220→  return { x, y, w: Math.max(...xs) - x, h: Math.max(...ys) - y };
   221→}

   254→/** #rgb / #rrggbb → pdf-lib rgb() components. White for garbage. */
   255→export function hexToRgb(hex: string): {
   256→  r: number;
   257→  g: number;
   258→  b: number;
   259→} {
   260→  const m = /^#?([0-9a-f]{6}|[0-9a-f]{3})$/i.exec(hex.trim());
   261→  if (!m) return { r: 1, g: 1, b: 1 };
   262→  let h = m[1];
   263→  if (h.length === 3) h = h[0] + h[0] + h[1] + h[1] + h[2] + h[2];
   264→  const num = Number.parseInt(h, 16);
   265→  return {
   266→    r: ((num >> 16) & 255) / 255,
   267→    g: ((num >> 8) & 255) / 255,
   268→    b: (num & 255) / 255,
   269→  };
   270→}

   401→/** Suggests a unique field name: "Field", "Field 2", … against the
   402→ * names (and radio group names) already on the page/document. */
   403→export function suggestFieldName(taken: ReadonlySet<string>): string {
   404→  const base = "Field";
   405→  if (!taken.has(base)) return base;
   406→  for (let n = 2; n < 500; n += 1) {
   407→    const candidate = `${base} ${n}`;
   408→    if (!taken.has(candidate)) return candidate;
   409→  }
   410→  return `${base} ${Date.now()}`;
   411→}

   471→/**
   472→ * Converts raw display-space sample points (pt) into a bbox + fractional
   473→ * point list. Points outside the page are NOT clamped (the caller clamps
   474→ * the gesture to the overlay).
   475→ */
   476→export function normaliseDraw(
   477→  pts: { x: number; y: number }[]
   478→): { rect: ObjRect; fracs: number[][] } | null {
   479→  if (pts.length < MIN_DRAW_POINTS) return null;
   480→  let minX = Infinity;
   481→  let minY = Infinity;
   482→  let maxX = -Infinity;
   483→  let maxY = -Infinity;
   484→  for (const p of pts) {
   485→    minX = Math.min(minX, p.x);
   486→    minY = Math.min(minY, p.y);
   487→    maxX = Math.max(maxX, p.x);
   488→    maxY = Math.max(maxY, p.y);
   489→  }
   490→  const w = maxX - minX;
   491→  const h = maxY - minY;
   492→  // Degenerate (a dot / a straight horizontal or vertical line) — keep a
   493→  // minimal bbox so the stroke still renders with its width.
   494→  const bw = Math.max(w, 2);
   495→  const bh = Math.max(h, 2);
   496→  const fracs = pts.map((p) => [
   497→    Number(((p.x - minX) / bw).toFixed(4)),
   498→    Number(((p.y - minY) / bh).toFixed(4)),
   499→  ]);
   500→  return {
   501→    rect: { x: minX, y: minY, w: bw, h: bh },
   502→    fracs,
   503→  };
   504→}

   522→export function hitRect(disp: ObjRect, px: number, py: number): boolean {
   523→  return (
   524→    px >= disp.x &&

   526→    py >= disp.y &&

   528→  );

   531→/** Corner handle the point is over (within `tol` pt), if any. */
   532→export function hitHandle(
   533→  disp: ObjRect,
   534→  px: number,
   535→  py: number,
   536→  tol: number
   537→): HitEdge | null {
   538→  const corners: { edge: HitEdge; cx: number; cy: number }[] = [
   539→    { edge: "nw", cx: disp.x, cy: disp.y },
   540→    { edge: "ne", cx: disp.x + disp.w, cy: disp.y },
   541→    { edge: "sw", cx: disp.x, cy: disp.y + disp.h },
   542→    { edge: "se", cx: disp.x + disp.w, cy: disp.y + disp.h },
   543→  ];
   544→  for (const c of corners) {
   545→    if (Math.abs(px - c.cx) <= tol && Math.abs(py - c.cy) <= tol) return c.edge;
   546→  }
   547→  return null;
   548→}

   550→/** Topmost (last in array) object at the display point. */
   551→export function pickObject(
   552→  objects: EditorObject[],
   553→  view: PageView,
   554→  px: number,
   555→  py: number
   556→): EditorObject | null {
   557→  for (let i = objects.length - 1; i >= 0; i -= 1) {
   558→    const obj = objects[i];
   559→    if (obj.hidden) continue;
   560→    if (!hitRect(pdfRectToDisplay(obj, view), px, py)) continue;
   561→    return obj;
   562→  }
   563→  return null;
   564→}

   566→/** Resize a rect from one corner, keeping w/h ≥ min. Returns display rect. */
   567→export function resizeRect(
   568→  orig: ObjRect,
   569→  edge: HitEdge,
   570→  x2: number,
   571→  y2: number,
   572→  min: number,
   573→  keepAspect: boolean,
   574→  aspect: number
   575→): ObjRect {
   576→  let { x, y, w, h } = orig;
   577→  if (edge === "se") {
   578→    w = Math.max(min, x2 - x);
   579→    h = Math.max(min, y2 - y);
   580→    if (keepAspect) {
   581→      w = Math.max(min, Math.min(w, h * aspect));
   582→      h = w / aspect;
   583→    }
   584→  } else if (edge === "ne") {
   585→    const ny = Math.min(y + h - min, y2);
   586→    w = Math.max(min, x2 - x);
   587→    h = Math.max(min, y + h - ny);
   588→    y = ny;
   589→    if (keepAspect) {
   590→      w = Math.max(min, Math.min(w, h * aspect));
   591→      h = w / aspect;
   592→      y = orig.y + orig.h - h;
   593→    }
   594→  } else if (edge === "sw") {
   595→    const nx = Math.min(x + w - min, x2);
   596→    w = Math.max(min, x + w - nx);
   597→    x = nx;
   598→    h = Math.max(min, y2 - y);
   599→    if (keepAspect) {
   600→      h = Math.max(min, Math.min(h, w / aspect));
   601→      w = h * aspect;
   602→      x = orig.x + orig.w - w;
   603→    }
   604→  } else {
   605→    const nx = Math.min(x + w - min, x2);
   606→    const ny = Math.min(y + h - min, y2);
   607→    w = Math.max(min, x + w - nx);
   608→    h = Math.max(min, y + h - ny);
   609→    x = nx;
   610→    y = ny;
   611→    if (keepAspect) {
   612→      w = Math.max(min, Math.min(w, h * aspect));
   613→      h = w / aspect;
   614→      x = orig.x + orig.w - w;
   615→      y = orig.y + orig.h - h;
   616→    }
   617→  }
   618→  return { x, y, w, h };
   619→}

   622→// Z-order operations (array order == z order, last = front)

   625→export function bringToFront(
   626→  objects: EditorObject[],
   627→  id: string
   628→): EditorObject[] {
   629→  const obj = objects.find((o) => o.id === id);
   630→  if (!obj) return objects;
   631→  return [...objects.filter((o) => o.id !== id), obj];
   632→}

   634→export function sendToBack(
   635→  objects: EditorObject[],
   636→  id: string
   637→): EditorObject[] {
   638→  const obj = objects.find((o) => o.id === id);
   639→  if (!obj) return objects;
   640→  return [obj, ...objects.filter((o) => o.id !== id)];
   641→}

   643→export function raiseOne(objects: EditorObject[], id: string): EditorObject[] {
   644→  const idx = objects.findIndex((o) => o.id === id);
   645→  if (idx < 0 || idx === objects.length - 1) return objects;
   646→  const next = [...objects];
   647→  const tmp = next[idx];
   648→  next[idx] = next[idx + 1];
   649→  next[idx + 1] = tmp;
   650→  return next;
   651→}
   652→
   653→export function lowerOne(objects: EditorObject[], id: string): EditorObject[] {
   654→  const idx = objects.findIndex((o) => o.id === id);
   655→  if (idx <= 0) return objects;
   656→  const next = [...objects];
   657→  const tmp = next[idx];
   658→  next[idx] = next[idx - 1];
   659→  next[idx - 1] = tmp;
   660→  return next;
   661→}

   663→export function duplicateObject(
   664→  objects: EditorObject[],
   665→  id: string
   666→): { objects: EditorObject[]; copy: EditorObject | null } {
   667→  const idx = objects.findIndex((o) => o.id === id);
   668→  if (idx < 0) return { objects, copy: null };
   669→  const src = objects[idx];
   670→  const copy: EditorObject = {
   671→    ...src,
   672→    id: nextObjId("c"),
   673→    x: roundPt(src.x + 12),
   674→    y: roundPt(Math.max(0, src.y - 12)),
   675→  };
   676→  // Duplicated fields need their own name — same-name fields share one
   677→  // AcroForm value in every reader (radio groups excepted, where the
   678→  // caller duplicates on purpose).
   679→  if (copy.kind === "field" && copy.fieldType !== "radio") {
   680→    copy.name = suggestFieldName(takenFieldNames(objects));
   681→  }
   682→  const next = [...objects];
   683→  next.splice(idx + 1, 0, copy);
   684→  return { objects: next, copy };
   685→}

   691→export const HISTORY_LIMIT = 120;

   693→export function pushSnapshot(
   694→  past: string[],
   695→  snapshot: string,
   696→  future: string[]
   697→): { past: string[]; future: string[] } {
   698→  const nextPast = [...past, snapshot];
   699→  if (nextPast.length > HISTORY_LIMIT) nextPast.shift();
   700→  return { past: nextPast, future: [] };
   701→}

   703→/** Undo: pops past into future, returns the restored snapshot. */
   704→export function undoStep(
   705→  past: string[],
   706→  future: string[],
   707→  current: string
   708→): { past: string[]; future: string[]; snapshot: string | null } {
   709→  if (past.length === 0) return { past, future, snapshot: null };
   710→  const nextPast = [...past];
   711→  const snapshot = nextPast.pop() as string;
   712→  return { past: nextPast, future: [current, ...future], snapshot };
   713→}
   714→
   715→/** Redo: pops future into past, returns the restored snapshot. */
   716→export function redoStep(
   717→  past: string[],
   718→  future: string[],
   719→  current: string
   720→): { past: string[]; future: string[]; snapshot: string | null } {
   721→  if (future.length === 0) return { past, future, snapshot: null };
   722→  const [snapshot, ...rest] = future;
   723→  return { past: [...past, current], future: rest, snapshot };
   724→}

   774→/** Parses + clamps one raw object from the wire blob. null = drop it. */
   775→export function sanitizeObject(
   776→  raw: unknown,
   777→  pageCount: number,
   778→  knownAssets: ReadonlySet<string>
   779→): EditorObject | null {
   780→  if (typeof raw !== "object" || raw === null) return null;
   781→  const o = raw as Record<string, unknown>;
   782→  const kind = KINDS.includes(o.kind as ObjKind) ? (o.kind as ObjKind) : null;
   783→  if (kind === null) return null;
   784→  const page = Math.min(
   785→    Math.max(0, Math.trunc(num(o.page, 0))),
   786→    Math.max(0, pageCount - 1)
   787→  );
   788→  const base = {
   789→    id: str(o.id, nextObjId("x")),
   790→    page,
   791→    x: num(o.x, 72),
   792→    y: num(o.y, 72),
   793→    w: Math.max(MIN_OBJ_PT, num(o.w, 120)),
   794→    h: Math.max(MIN_OBJ_PT, num(o.h, 24)),
   795→    opacity: clampN(num(o.opacity, 1), 0.05, 1),
   796→    locked: bool(o.locked, false),
   797→    hidden: bool(o.hidden, false),
   798→  };
   799→  switch (kind) {
   800→    case "text":
   801→      return {
   802→        ...base,
   803→        kind,
   804→        text: str(o.text, ""),
   805→        font: FONTS.includes(o.font as EditorFontId)
   806→          ? (o.font as EditorFontId)
   807→          : "helvetica",
   808→        size: clampN(num(o.size, 12), 4, 200),
   809→        color: color(o.color, "#111827"),
   810→        bold: bool(o.bold, false),
   811→        italic: bool(o.italic, false),
   812→        underline: bool(o.underline, false),
   813→        align: o.align === "center" || o.align === "right" ? o.align : "left",
   814→        bg: optColor(o.bg),
   815→      };
   816→    case "image": {
   817→      const key = str(o.assetKey, "");
   818→      if (!knownAssets.has(key)) return null;
   819→      return { ...base, kind, assetKey: key };
   820→    }
   821→    case "rect":
   822→    case "ellipse":
   823→    case "line":
   824→    case "arrow":
   825→      return {
   826→        ...base,
   827→        kind,
   828→        fill: optColor(o.fill),
   829→        stroke: optColor(o.stroke),
   830→        strokeW: clampN(num(o.strokeW, 2), 0.25, 40),
   831→        dash: bool(o.dash, false),
   832→        flip: bool(o.flip, false),
   833→      };
   834→    case "draw": {
   835→      const rawPts = Array.isArray(o.pts) ? o.pts : [];
   836→      const pts = rawPts
   837→        .filter(
   838→          (p): p is number[] =>
   839→            Array.isArray(p) &&
   840→            p.length === 2 &&
   841→            typeof p[0] === "number" &&
   842→            typeof p[1] === "number"
   843→        )
   844→        .map((p) => [clampN(p[0], 0, 1), clampN(p[1], 0, 1)]);
   845→      if (pts.length < 2) return null;
   846→      return {
   847→        ...base,
   848→        kind,
   849→        pts,
   850→        stroke: color(o.stroke, "#dc2626"),
   851→        strokeW: clampN(num(o.strokeW, 2.5), 0.25, 40),
   852→      };
   853→    }
   854→    case "whiteout":
   855→      return { ...base, kind, fill: color(o.fill, "#ffffff") };
   856→    case "field": {
   857→      const fieldType = FIELD_TYPES.includes(o.fieldType as FieldType)
   858→        ? (o.fieldType as FieldType)
   859→        : "text";
   860→      const rawOptions = Array.isArray(o.options) ? o.options : [];
   861→      const options = [
   862→        ...new Set(
   863→          rawOptions
   864→            .filter((v): v is string => typeof v === "string")
   865→            .map((v) => v.trim().slice(0, 80))
   866→            .filter((v) => v.length > 0)
   867→        ),
   868→      ].slice(0, 50);
   869→      const value = str(o.value, "").slice(0, 200);
   870→      const name =
   871→        str(o.name, "").replace(/\s+/g, " ").trim().slice(0, 60) ||
   872→        "Field";
   873→      let h = base.h;
   874→      if (fieldType === "checkbox" || fieldType === "radio") {
   875→        // Square widgets (the worker squares them at export too — this
   876→        // keeps the editor preview honest about the final shape).
   877→        const size = Math.max(MIN_OBJ_PT, Math.min(base.w, base.h));
   878→        base.w = size;
   879→        h = size;
   880→      }
   881→      return {
   882→        ...base,
   883→        w: base.w,
   884→        h,
   885→        kind,
   886→        fieldType,
   887→        name,
   888→        value: fieldType === "dropdown" && !options.includes(value) ? "" : value,
   889→        required: bool(o.required, false),
   890→        options: fieldType === "dropdown" ? options : [],
   891→      };
   892→    }
   893→  }
   894→}

   896→/** Parses the full objects array from the wire blob. */
   897→export function sanitizeObjects(
   898→  raw: unknown,
   899→  pageCount: number,
   900→  knownAssets: ReadonlySet<string>
   901→): EditorObject[] {
   902→  if (!Array.isArray(raw)) return [];

  1035→function normaliseFontName(raw: string): string {
  1036→  // Strip a subset prefix ("ABCDEF+ArialMT") and punctuation, lowercase.
  1037→  const plus = raw.indexOf("+");
  1038→  const base = plus >= 0 ? raw.slice(plus + 1) : raw;
  1039→  return base.replace(/[\s_-]+/g, "").toLowerCase();
  1040→}

  1042→/**
  1043→ * Maps an original PDF font onto the editor's embedded fonts (PRD §8.2:
  1044→ * "font match — standard fonts, embedded font agar subset allow kare, warna
  1045→ * closest font suggest"). Exact only for the standard-14 families (and the
  1046→ * Devanagari font we embed verbatim); everything else maps to the closest
  1047→ * family by name table, class flags or CSS fallback family.
  1048→ */
  1049→export function matchFont(
  1050→  rawName: string,
  1051→  hints: FontMatchHints
  1052→): FontMatch {
  1053→  const norm = normaliseFontName(rawName);
  1054→  const bold =
  1055→    hints.bold === true ||
  1056→    (hints.bold !== false && /bold|black|heavy|semibold|demibold/i.test(rawName));
  1057→  const italic =
  1058→    hints.italic === true ||
  1059→    (hints.italic !== false && /italic|oblique/i.test(rawName));
  1060→
  1061→  // Devanagari text needs the Devanagari pipeline regardless of the name.
  1062→  if (hints.hasDeva === true) {
  1063→    return {
  1064→      font: "devanagari",
  1065→      bold,
  1066→      italic,
  1067→      exact: norm === "notosansdevanagari",
  1068→      original: rawName,
  1069→    };
  1070→  }
  1071→
  1072→  let font: EditorFontId | null = null;
  1073→  if (norm !== "" && FAMILY_BY_NAME[norm] !== undefined) {
  1074→    font = FAMILY_BY_NAME[norm];
  1075→  } else if (hints.mono === true || hints.cssFamily === "monospace") {
  1076→    font = "courier";
  1077→  } else if (hints.serif === true || hints.cssFamily === "serif") {
  1078→    font = "times";
  1079→  } else if (norm.includes("devanagari")) {
  1080→    font = "devanagari";
  1081→  } else if (
  1082→    norm.includes("mono") ||
  1083→    norm.includes("courier") ||
  1084→    norm.includes("consolas") ||
  1085→    norm.includes("menlo")
  1086→  ) {
  1087→    font = "courier";
  1088→  } else if (
  1089→    norm.includes("serif") ||
  1090→    norm.includes("roman") ||
  1091→    norm.includes("times") ||
  1092→    norm.includes("georgia") ||
  1093→    norm.includes("garamond")
  1094→  ) {
  1095→    font = "times";
  1096→  } else if (
  1097→    norm.includes("sans") ||
  1098→    norm.includes("helvetica") ||
  1099→    norm.includes("arial") ||
  1100→    norm.includes("verdana") ||
  1101→    norm.includes("tahoma") ||
  1102→    norm.includes("segoe")
  1103→  ) {
  1104→    font = "helvetica";
  1105→  } else if (font === null) {
  1106→    font = "helvetica";
  1107→  }
  1108→
  1109→  return {
  1110→    font,
  1111→    bold,
  1112→    italic,
  1113→    exact: EXACT_NAMES.has(norm),
  1114→    original: rawName,
  1115→  };
  1116→}

  1118→/** Vertical metrics used to expand a piece into a comfortable cover rect. */
  1119→export const PIECE_ASCENT = 0.85;
  1120→export const PIECE_DESCENT = 0.25;
  1121→export const PIECE_PAD = 1.5;
  1122→
  1123→/**
  1124→ * The DISPLAY rect a piece's cover/text objects occupy: the glyph band
  1125→ * (ascent above the baseline, descent below) plus a small pad, widened by
  1126→ * 1pt on each side so ascenders/descenders never peek out.
  1127→ */
  1128→export function pieceRect(piece: {
  1129→  x: number;
  1130→  xEnd: number;
  1131→  baseline: number;
  1132→  h: number;
  1133→}): ObjRect {
  1134→  const top = piece.baseline - PIECE_ASCENT * piece.h - PIECE_PAD;
  1135→  const bottom = piece.baseline + PIECE_DESCENT * piece.h + PIECE_PAD;
  1136→  return {
  1137→    x: piece.x - 1,
  1138→    y: top,
  1139→    w: Math.max(8, piece.xEnd - piece.x + 2),
  1140→    h: Math.max(10, bottom - top),
  1141→  };
  1142→}

  1144→/**
  1145→ * Builds the whiteout + retype pair for one harvested piece (PRD §8.2
  1146→ * "original text area whiteout + naya text same position par"). Both
  1147→ * objects share the piece rect; the cover is inserted BEFORE the text so
  1148→ * array z-order draws it first. One history group per call is the caller's
  1149→ * responsibility.
  1150→ */
  1151→export function buildReplacePieceObjects(
  1152→  page: number,
  1153→  view: PageView,
  1154→  piece: HarvestPiece,
  1155→  bgHex: string,
  1156→  inkHex: string
  1157→): { cover: WhiteoutObj; text: TextObj } {
  1158→  const disp = pieceRect(piece);
  1159→  const pdf = displayRectToPdf(disp, view);
  1160→  const cover = makeWhiteout(page, pdf);
  1161→  cover.fill = bgHex;
  1162→  const text = makeText(page, pdf, {
  1163→    text: piece.str,
  1164→    font: piece.font.font,
  1165→    size: clampN(piece.h, 4, 200),
  1166→    color: inkHex,
  1167→    bold: piece.font.bold,
  1168→    italic: piece.font.italic,
  1169→    align: "left",
  1170→  });
  1171→  return { cover, text };
  1172→}

  1190→/**
  1191→ * Groups a page's pieces into visual lines: baseline bucket (Δ ≤ 0.45 ×
  1192→ * the larger em) + horizontal gap ≤ 1.2 × em joins as a space, pieces sorted
  1193→ * by x. The same-baseline rule keeps columns separate (P4.3 insight).
  1194→ */
  1195→export function groupPiecesIntoLines(
  1196→  page: number,
  1197→  pieces: HarvestPiece[]
  1198→): PieceLine[] {
  1199→  const sorted = [...pieces].sort((a, b) => a.x - b.x);
  1200→  const lines: PieceLine[] = [];
  1201→  for (const p of sorted) {
  1202→    const line = lines.find((l) => {
  1203→      if (Math.abs(l.baseline - p.baseline) > 0.45 * Math.max(l.h, p.h)) {
  1204→        return false;
  1205→      }
  1206→      // Gap between the line's right edge and this piece (x-sorted so ≥ 0).
  1207→      return p.x - l.xEnd <= 1.2 * Math.max(l.h, p.h);
  1208→    });
  1209→    if (line) {
  1210→      line.pieces.push(p.index);
  1211→      line.xEnd = Math.max(line.xEnd, p.xEnd);
  1212→      line.text = `${line.text} ${p.str}`;
  1213→      line.h = Math.max(line.h, p.h);
  1214→    } else {
  1215→      lines.push({
  1216→        page,
  1217→        pieces: [p.index],
  1218→        text: p.str,
  1219→        x: p.x,
  1220→        xEnd: p.xEnd,
  1221→        baseline: p.baseline,
  1222→        h: p.h,
  1223→      });
  1224→    }
  1225→  }
  1226→  return lines;
  1227→}

  1238→/** Literal (non-regex) occurrences of the query across all lines. */
  1239→export function planLineReplacements(
  1240→  lines: PieceLine[],
  1241→  query: string,
  1242→  matchCase: boolean
  1243→): LineMatch[] {
  1244→  const out: LineMatch[] = [];
  1245→  const needle = matchCase ? query : query.toLowerCase();
  1246→  if (needle.length === 0) return out;
  1247→  for (let i = 0; i < lines.length; i += 1) {
  1248→    const line = lines[i];
  1249→    const hay = matchCase ? line.text : line.text.toLowerCase();
  1250→    let from = 0;
  1251→    for (;;) {
  1252→      const at = hay.indexOf(needle, from);
  1253→      if (at < 0) break;
  1254→      out.push({ page: line.page, line: i, start: at, end: at + needle.length });
  1255→      from = at + needle.length;
  1256→    }
  1257→  }
  1258→  return out;

  1261→/**
  1262→ * Builds the whiteout + retype pair for one VISUAL LINE (find & replace):
  1263→ * the cover spans the union of the line's piece rects, the retype uses the
  1264→ * dominant piece's font match and the substituted text. One history group
  1265→ * per call is the caller's responsibility.
  1266→ */
  1267→export function buildReplaceLineObjects(
  1268→  page: number,
  1269→  view: PageView,
  1270→  line: PieceLine,
  1271→  pagePieces: HarvestPiece[],
  1272→  newText: string,
  1273→  bgHex: string,
  1274→  inkHex: string
  1275→): { cover: WhiteoutObj; text: TextObj } {
  1276→  let x0 = Infinity;
  1277→  let y0 = Infinity;
  1278→  let x1 = -Infinity;
  1279→  let y1 = -Infinity;
  1280→  for (const idx of line.pieces) {
  1281→    const p = pagePieces[idx];
  1282→    if (!p) continue;
  1283→    const r = pieceRect(p);
  1284→    x0 = Math.min(x0, r.x);
  1285→    y0 = Math.min(y0, r.y);
  1286→    x1 = Math.max(x1, r.x + r.w);
  1287→    y1 = Math.max(y1, r.y + r.h);
  1288→  }
  1289→  if (!Number.isFinite(x0) || !Number.isFinite(y0)) {
  1290→    x0 = line.x;
  1291→    y0 = line.baseline - line.h;
  1292→    x1 = line.xEnd;
  1293→    y1 = line.baseline;
  1294→  }
  1295→  const disp = { x: x0, y: y0, w: Math.max(8, x1 - x0), h: Math.max(10, y1 - y0) };
  1296→  const pdf = displayRectToPdf(disp, view);
  1297→  const cover = makeWhiteout(page, pdf);
  1298→  cover.fill = bgHex;
  1299→  // Dominant piece = largest em (ties: first in x order).
  1300→  let dom = pagePieces[line.pieces[0]] ?? null;
  1301→  for (const idx of line.pieces) {
  1302→    const p = pagePieces[idx];
  1303→    if (p && dom !== null && p.h > dom.h) dom = p;
  1304→    else if (p && dom === null) dom = p;
  1305→  }
  1306→  const text = makeText(page, pdf, {
  1307→    text: newText,
  1308→    font: dom !== null ? dom.font.font : "helvetica",
  1309→    size: clampN(dom !== null ? dom.h : 12, 4, 200),
  1310→    color: inkHex,
  1311→    bold: dom !== null ? dom.font.bold : false,
  1312→    italic: dom !== null ? dom.font.italic : false,
  1313→    align: "left",
  1314→  });
  1315→  return { cover, text };
  1316→}

  1318→/**
  1319→ * Line text with the given occurrence ranges substituted (literal
  1320→ * semantics — split/join so regex metacharacters in the query are safe).
  1321→ */
  1322→export function substituteOccurrences(
  1323→  text: string,
  1324→  ranges: { start: number; end: number }[],
  1325→  replacement: string
  1326→): string {
  1327→  if (ranges.length === 0) return text;
  1328→  const sorted = [...ranges].sort((a, b) => a.start - b.start);
  1329→  let out = "";
  1330→  let cursor = 0;
  1331→  for (const r of sorted) {
  1332→    if (r.start < cursor) continue; // overlap guard
  1333→    out += text.slice(cursor, r.start) + replacement;
  1334→    cursor = r.end;
  1335→  }
  1336→  return out + text.slice(cursor);
  1337→}

  1349→export function luminance(r: number, g: number, b: number): number {
  1350→  return (0.2126 * r + 0.7152 * g + 0.0722 * b) / 255;
  1351→}

  1353→/** Median of an RGBA pixel run (ignores alpha — pages render opaque). */
  1354→export function medianRgb(pixels: number[]): { r: number; g: number; b: number } {
  1355→  if (pixels.length === 0) return { r: 255, g: 255, b: 255 };
  1356→  const by = (ch: 0 | 1 | 2): number => {
  1357→    const vals = pixels.filter((_, i) => i % 4 === ch).sort((a, b) => a - b);
  1358→    return vals[Math.floor(vals.length / 2)] ?? 255;
  1359→  };
  1360→  return { r: by(0), g: by(1), b: by(2) };
  1361→}

  1363→/**
  1364→ * Picks the text ink colour from the pixels under a piece: the darkest
  1365→ * cluster on a light background, the brightest on a dark one. Returns null
  1366→ * when the contrast is too weak to trust (caller falls back to a default).
  1367→ */
  1368→export function pickInkFromPixels(
  1369→  pixels: number[],
  1370→  bgLum: number
  1371→): { r: number; g: number; b: number } | null {
  1372→  if (pixels.length < 16) return null;
  1373→  const lums = pixels
  1374→    .filter((_, i) => i % 4 === 3 && pixels[i] > 40)
  1375→    .length;
  1376→  if (lums < 4) return null;
  1377→  const entries: { lum: number; r: number; g: number; b: number }[] = [];
  1378→  for (let i = 0; i + 3 < pixels.length; i += 4) {
  1379→    const { r, g, b } = { r: pixels[i], g: pixels[i + 1], b: pixels[i + 2] };
  1380→    entries.push({ lum: luminance(r, g, b), r, g, b });
  1381→  }
  1382→  entries.sort((a, b) => a.lum - b.lum);
  1383→  const dark = entries[Math.floor(entries.length * 0.12)];
  1384→  const bright = entries[Math.floor(entries.length * 0.88)];
  1385→  const pick = bgLum > 0.5 ? dark : bright;
  1386→  const contrast = Math.abs(pick.lum - bgLum);
  1387→  if (contrast < 0.16) return null;
  1388→  return { r: pick.r, g: pick.g, b: pick.b };
  1389→}

  1391→/** Greedy word-wrap that hard-breaks overlong words (surface CSS + worker). */
  1392→export function wrapText(
  1393→  text: string,
  1394→  maxWidth: number,
  1395→  measure: (s: string) => number
  1396→): string[] {
  1397→  const lines: string[] = [];
  1398→  for (const para of text.split("\n")) {
  1399→    if (para.length === 0) {
  1400→      lines.push("");
  1401→      continue;
  1402→    }
  1403→    let current = "";
  1404→    for (const word of para.split(/ +/)) {
  1405→      const candidate = current.length === 0 ? word : `${current} ${word}`;
  1406→      if (measure(candidate) <= maxWidth) {
  1407→        current = candidate;
  1408→        continue;
  1409→      }
  1410→      if (current.length > 0) {
  1411→        lines.push(current);
  1412→        current = "";
  1413→      }
  1414→      if (measure(word) <= maxWidth) {
  1415→        current = word;
  1416→      } else {
  1417→        // Hard-break the overlong word by characters (a single character
  1418→        // wider than maxWidth is still emitted — best effort).
  1419→        let chunk = "";
  1420→        for (const ch of word) {
  1421→          if (measure(chunk + ch) > maxWidth && chunk.length > 0) {
  1422→            lines.push(chunk);
  1423→            chunk = ch;
  1424→          } else {
  1425→            chunk += ch;
  1426→          }
  1427→        }
  1428→        current = chunk;
  1429→      }
  1430→    }
  1431→    lines.push(current);
  1432→  }
  1433→  return lines;
  1434→}

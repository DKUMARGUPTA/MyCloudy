    77→
    78→/**
    79→ * PageGrid surface (PRD §8.1 "Rotate / Rearrange / Remove / Extract
    80→ * (Page-grid family)") — ONE shared interactive page grid, preset-tuned:
    81→ *
    82→ * - common actions: select (click / shift-click / Ctrl+A / range input),
    83→ *   drag reorder, rotate 90 L/R per page or scope, delete, duplicate,
    84→ *   insert blank, undo/redo, zoom slider
    85→ * - rotate preset adds odd/even scope buttons + live thumbnail rotation
    86→ * - remove preset adds "remove blank pages" auto-detect
    87→ * - extract preset keeps the grid as the picker (selection = pages out)
    88→ * - large docs (>VIRTUALIZE_ABOVE pages) render virtualized rows with lazy
    89→ *   thumbnails; drag swaps for the always-available move buttons
    90→ *
    91→ * The edited page list is synced to the tool options ("grid" / "selected")
    92→ * after every action, so the shared worker always sees the exact state.
    93→ */
    94→
    95→const ZOOM_MIN = 88;
    96→const ZOOM_MAX = 168;
    97→const ZOOM_STEP = 16;
    98→const ZOOM_DEFAULT = 120;
    99→const CELL_GAP = 12;
   100→const CELL_FOOTER = 0; // cells size themselves via aspect-ratio
   101→const OVERSCAN_ROWS = 2;
   102→const DEFAULT_ASPECT = 0.707; // A4 portrait while the real aspect loads
   103→
   104→/** Toolbar buttons stay 44px touch targets (a11y hard rule). */
   105→const TOOL_BTN =
   106→  "h-11 min-h-11 gap-1.5 rounded-btn px-2.5 text-xs font-medium outline-none transition-colors focus-visible:ring-[3px] focus-visible:ring-ring/50";
   107→
   108→type ScopeId = "all" | "selected" | "odd" | "even";
   109→
   110→export function GridSurface({
   111→  preset,
   112→  files,
   113→  onChange,
   114→  setFileOrder,
   115→  setRunBlocked,
   116→}: ToolSurfaceProps & { preset: GridPreset }) {
   117→  const { t } = useI18n();
   118→  const { toast } = useToast();
   119→  const file = files.length > 0 ? files[0].file : null;
   120→
   121→  const {
   122→    pages,
   123→    canUndo,
   124→    canRedo,
   125→    commit,
   126→    reset,
   127→    undo,
   128→    redo,
   129→  } = useGridHistory([]);
   130→
   131→  const [selection, setSelection] = useState<Set<number>>(new Set());
   132→  const [focusIndex, setFocusIndex] = useState(0);
   133→  const [anchorIndex, setAnchorIndex] = useState<number | null>(null);
   134→  const [scope, setScope] = useState<ScopeId>("all");
   135→  const [zoom, setZoom] = useState(ZOOM_DEFAULT);
   136→  const [rangeText, setRangeText] = useState("");
   137→  const [blankScan, setBlankScan] = useState<{
   138→    done: number;
   139→    total: number;
   140→  } | null>(null);
   141→  const [docInfo, setDocInfo] = useState<{
   142→    file: File;
   143→    count: number;
   144→    aspect: number;
   145→  } | null>(null);
   146→
   147→  const rootRef = useRef<HTMLDivElement>(null);
   148→  const scrollRef = useRef<HTMLDivElement>(null);
   149→  const [scrollTop, setScrollTop] = useState(0);
   150→  const [viewportH, setViewportH] = useState(480);
   151→  const [containerW, setContainerW] = useState(0);
   152→
   153→  const virtualized = pages.length > VIRTUALIZE_ABOVE;
   154→  const thumbDim = gridThumbMaxDim();
   155→  const aspect = docInfo?.file === file ? docInfo.aspect : DEFAULT_ASPECT;
   156→
   157→  /* ---------------- CTA gating (event-context sync, no effect-setState) */
   158→
   159→  const syncBlocked = useCallback(
   160→    (nextPages: GridPage[], nextSelection: ReadonlySet<number>) => {
   161→      if (preset === "extract") {
   162→        setRunBlocked(nextSelection.size === 0 ? "grid.ctaSelectPages" : null);
   163→        return;
   164→      }
   165→      setRunBlocked(nextPages.length === 0 ? "grid.ctaAllRemoved" : null);
   166→    },
   167→    [preset, setRunBlocked]
   168→  );
   169→
   170→  /* ---------------- Document detection (page count + aspect) ---------------- */
   171→
   172→  useEffect(() => {
   173→    if (!file) return;
   174→    let cancelled = false;
   175→    void Promise.all([getPdfPageCount(file), getPdfPageAspect(file)])
   176→      .then(([count, fileAspect]) => {
   177→        if (cancelled) return;
   178→        if (count < 1) {
   179→          setDocInfo({ file, count: 0, aspect: DEFAULT_ASPECT });
   180→          syncBlocked([], new Set());
   181→          return;
   182→        }
   183→        const initial = createInitialGrid(count);
   184→        reset(initial);
   185→        setDocInfo({ file, count, aspect: fileAspect });
   186→        setSelection(new Set());
   187→        setFocusIndex(0);
   188→        setAnchorIndex(null);
   189→        onChange("grid", serializeGrid(initial));
   190→        onChange("selected", "[]");
   191→        onChange("originalCount", count);
   192→        syncBlocked(initial, new Set());
   193→      })
   194→      .catch(() => {
   195→        if (cancelled) return;
   196→        setDocInfo({ file, count: 0, aspect: DEFAULT_ASPECT });
   197→        syncBlocked([], new Set());
   198→      });
   199→    return () => {
   200→      cancelled = true;
   201→    };
   202→  }, [file, onChange, reset, syncBlocked]);
   203→
   204→  /* ---------------- Viewport measurement (virtualization) ---------------- */
   205→
   206→  useEffect(() => {
   207→    const el = scrollRef.current;
   208→    if (!el) return;
   209→    const measure = () => {
   210→      setViewportH(el.clientHeight);
   211→      setContainerW(el.clientWidth);
   212→    };
   213→    measure();
   214→    const observer = new ResizeObserver(measure);
   215→    observer.observe(el);
   216→    return () => {
   217→      observer.disconnect();
   218→    };
   219→  }, []);
   220→
   221→  const handleScroll = useCallback(() => {
   222→    const el = scrollRef.current;
   223→    if (el) setScrollTop(el.scrollTop);
   224→  }, []);
   225→
   226→  /* ---------------- Mutations (all flow through apply/commit) ---------------- */
   227→
   228→  const serializeSelection = (next: ReadonlySet<number>): string =>
   229→    JSON.stringify([...next].sort((a, b) => a - b));
   230→
   231→  const apply = (next: GridPage[], nextSelection?: Set<number>) => {
   232→    const clamped =
   233→      nextSelection ??
   234→      new Set([...selection].filter((index) => index < next.length));
   235→    commit(next);
   236→    setSelection(clamped);
   237→    setFocusIndex((prev) => Math.min(prev, Math.max(0, next.length - 1)));
   238→    onChange("grid", serializeGrid(next));
   239→    onChange("selected", serializeSelection(clamped));
   240→    syncBlocked(next, clamped);
   241→  };
   242→
   243→  const changeSelection = (next: Set<number>) => {
   244→    setSelection(next);
   245→    onChange("selected", serializeSelection(next));
   246→    syncBlocked(pages, next);
   247→  };
   248→
   249→  const selectedIndices = useMemo(
   250→    () => [...selection].sort((a, b) => a - b),
   251→    [selection]
   252→  );
   253→
   254→  const scopeIndices = (): number[] => {
   255→    if (scope === "odd") return oddIndices(pages);
   256→    if (scope === "even") return evenIndices(pages);
   257→    if (scope === "selected") return selectedIndices;
   258→    return pages.map((_, index) => index);
   259→  };
   260→
   261→  const rotate = (delta: number) => {
   262→    const targets = scopeIndices();
   263→    if (targets.length === 0) return;
   264→    apply(rotateIndices(pages, targets, delta));
   265→  };
   266→
   267→  const deleteSelected = () => {
   268→    if (selection.size === 0) return;
   269→    apply(deleteIndices(pages, selectedIndices), new Set());
   270→  };
   271→
   272→  const duplicateSelected = () => {
   273→    if (selection.size === 0) return;
   274→    apply(duplicateIndices(pages, selectedIndices));
   275→  };
   276→
   277→  const insertBlank = () => {
   278→    apply(insertBlankAfter(pages, selectedIndices));
   279→  };
   280→
   281→  const moveSelected = (direction: -1 | 1) => {
   282→    if (selection.size === 0) return;
   283→    apply(moveSelectionByOne(pages, selectedIndices, direction));
   284→  };
   285→
   286→  const handleUndo = () => {
   287→    const restored = undo();
   288→    if (restored === null) {
   289→      toast({ title: t("grid.toastNoUndo") });
   290→      return;
   291→    }
   292→    const clamped = new Set(
   293→      [...selection].filter((index) => index < restored.length)
   294→    );
   295→    setSelection(clamped);
   296→    onChange("grid", serializeGrid(restored));
   297→    onChange("selected", serializeSelection(clamped));
   298→    syncBlocked(restored, clamped);
   299→  };
   300→
   301→  const handleRedo = () => {
   302→    const redone = redo();
   303→    if (redone === null) {
   304→      toast({ title: t("grid.toastNoRedo") });
   305→      return;
   306→    }
   307→    const clamped = new Set(
   308→      [...selection].filter((index) => index < redone.length)
   309→    );
   310→    setSelection(clamped);
   311→    onChange("grid", serializeGrid(redone));
   312→    onChange("selected", serializeSelection(clamped));
   313→    syncBlocked(redone, clamped);
   314→  };
   315→
   316→  /* ---------------- Selection interactions ---------------- */
   317→
   318→  const toggleCell = (
   319→    index: number,
   320→    event: ReactMouseEvent<HTMLButtonElement>
   321→  ) => {
   322→    const additive = event.ctrlKey || event.metaKey;
   323→    if (event.shiftKey && anchorIndex !== null) {
   324→      const from = Math.min(anchorIndex, index);
   325→      const to = Math.max(anchorIndex, index);
   326→      const next = new Set<number>();
   327→      for (let i = from; i <= to; i += 1) next.add(i);
   328→      changeSelection(next);
   329→      return;
   330→    }
   331→    setAnchorIndex(index);
   332→    if (additive) {
   333→      const next = new Set(selection);
   334→      if (next.has(index)) next.delete(index);
   335→      else next.add(index);
   336→      changeSelection(next);
   337→      return;
   338→    }
   339→    if (selection.size === 1 && selection.has(index)) {
   340→      changeSelection(new Set());
   341→      return;
   342→    }
   343→    changeSelection(new Set([index]));
   344→  };
   345→
   346→  const selectAll = () => {
   347→    changeSelection(new Set(pages.map((_, index) => index)));
   348→  };
   349→
   350→  const clearSelection = () => {
   351→    changeSelection(new Set());
   352→    setAnchorIndex(null);
   353→  };
   354→
   355→  const applyRange = () => {
   356→    const parsed = parseGridRange(rangeText, pages.length);
   357→    if (parsed === null) {
   358→      toast({ title: t("grid.rangeInvalid"), variant: "destructive" });
   359→      return;
   360→    }
   361→    changeSelection(new Set(parsed));
   362→    setAnchorIndex(parsed.length > 0 ? parsed[0] : null);
   363→  };
   364→
   365→  /* ---------------- Remove-blanks auto-detect (remove preset) ---------------- */
   366→
   367→  const removeBlanks = () => {
   368→    if (!file || blankScan !== null) return;
   369→    setBlankScan({ done: 0, total: docInfo?.count ?? 0 });
   370→    void detectBlankPages(file, (progress) => {
   371→      setBlankScan({ done: progress.done, total: progress.total });
   372→    })
   373→      .then((blanks) => {
   374→        setBlankScan(null);
   375→        if (blanks.length === 0) {
   376→          toast({ title: t("grid.removeBlanksNone") });
   377→          return;
   378→        }
   379→        const blankSources = new Set(blanks.map((page) => page - 1));
   380→        const targets = pages
   381→          .map((entry, index) =>
   382→            entry.source >= 0 && blankSources.has(entry.source) ? index : -1
   383→          )
   384→          .filter((index) => index >= 0);
   385→        apply(deleteIndices(pages, targets), new Set());
   386→        toast({ title: t("grid.removeBlanksFound", { count: blanks.length }) });
   387→      })
   388→      .catch(() => {
   389→        setBlankScan(null);
   390→        toast({
   391→          title: t("tool.errorTitle"),
   392→          description: t("tool.errorCorrupt"),
   393→          variant: "destructive",
   394→        });
   395→      });
   396→  };
   397→
   398→  /* ---------------- Keyboard (scoped to the surface) ---------------- */
   399→
   400→  const onRootKeyDown = (event: ReactKeyboardEvent<HTMLDivElement>) => {
   401→    const target = event.target as HTMLElement | null;
   402→    const tag = target?.tagName?.toLowerCase();
   403→    const inField = tag === "input" || tag === "textarea" || tag === "select";
   404→    if (event.key === "Escape") {
   405→      if (selection.size > 0) {
   406→        clearSelection();
   407→        event.preventDefault();
   408→      }
   409→      return;
   410→    }
   411→    if (inField) return;
   412→    const mod = event.ctrlKey || event.metaKey;
   413→    if (mod && event.key.toLowerCase() === "a") {
   414→      event.preventDefault();
   415→      selectAll();
   416→      return;
   417→    }
   418→    if (mod && event.key.toLowerCase() === "z") {
   419→      event.preventDefault();
   420→      if (event.shiftKey) handleRedo();
   421→      else handleUndo();
   422→      return;
   423→    }
   424→    if (mod && event.key.toLowerCase() === "y") {
   425→      event.preventDefault();
   426→      handleRedo();
   427→      return;
   428→    }
   429→    if (
   430→      (event.key === "Delete" || event.key === "Backspace") &&
   431→      selection.size > 0
   432→    ) {
   433→      event.preventDefault();
   434→      deleteSelected();
   435→    }
   436→  };
   437→
   438→  /* ---------------- Grid geometry (virtualization math) ---------------- */
   439→
   440→  const cellW = zoom;
   441→  const cellH = Math.round(cellW / aspect) + CELL_FOOTER;
   442→  const cols = Math.max(
   443→    1,
   444→    Math.floor((containerW + CELL_GAP) / (cellW + CELL_GAP))
   445→  );
   446→  const rowCount = Math.ceil(pages.length / cols);
   447→  const rowH = cellH + CELL_GAP;
   448→  const firstRow = Math.max(0, Math.floor(scrollTop / rowH) - OVERSCAN_ROWS);
   449→  const lastRow = Math.min(
   450→    rowCount - 1,
   451→    Math.ceil((scrollTop + viewportH) / rowH) + OVERSCAN_ROWS
   452→  );
   453→  const windowedPages = virtualized
   454→    ? pages.slice(firstRow * cols, (lastRow + 1) * cols)
   455→    : pages;
   456→  const windowOffset = virtualized ? firstRow * cols : 0;
   457→
   458→  const scopeOptions: { id: ScopeId; key: string }[] = [
   459→    { id: "all", key: "grid.scopeAll" },
   460→    { id: "selected", key: "grid.scopeSelected" },
   461→  ];
   462→  if (preset === "rotate") {
   463→    scopeOptions.push(
   464→      { id: "odd", key: "grid.scopeOdd" },
   465→      { id: "even", key: "grid.scopeEven" }
   466→    );
   467→  }
   468→
   469→  /* ---------------- Drag reorder (only when fully mounted) ---------------- */
   470→
   471→  const sensors = useSensors(
   472→    useSensor(PointerSensor, { activationConstraint: { distance: 6 } }),
   473→    useSensor(TouchSensor, { activationConstraint: { delay: 200, tolerance: 8 } }),
   474→    useSensor(KeyboardSensor, {
   475→      coordinateGetter: sortableKeyboardCoordinates,
   476→    })
   477→  );
   478→
   479→  const handleDragEnd = (event: DragEndEvent) => {
   480→    const { active, over } = event;
   481→    if (!over || active.id === over.id) return;
   482→    const from = pages.findIndex((page) => page.uid === active.id);
   483→    const to = pages.findIndex((page) => page.uid === over.id);
   484→    if (from < 0 || to < 0) return;
   485→    apply(moveEntry(pages, from, to));
   486→  };
   487→
   488→  /* ---------------- Cells ---------------- */
   489→
   490→  const cells = windowedPages.map((entry, offset) => {
   491→    const index = windowOffset + offset;
   492→    return (
   493→      <GridCell
   494→        key={entry.uid}
   495→        entry={entry}
   496→        index={index}
   497→        file={file}
   498→        aspect={aspect}
   499→        selected={selection.has(index)}
   500→        focusable={index === focusIndex}
   501→        sortable={!virtualized}
   502→        thumbDim={thumbDim}
   503→        onToggle={toggleCell}
   504→        onFocusCell={setFocusIndex}
   505→        onRotate={(cellIndex, delta) =>
   506→          apply(rotateIndices(pages, [cellIndex], delta))
   507→        }
   508→        onDelete={(cellIndex) =>
   509→          apply(deleteIndices(pages, [cellIndex]), new Set())
   510→        }
   511→      />
   512→    );
   513→  });
   514→
   515→  /* ---------------- Render ---------------- */
   516→
   517→  const selectionLabel =
   518→    selection.size === 0
   519→      ? t("grid.selectionNone")
   520→      : t("grid.selectionCount", {
   521→          count: selection.size,
   522→          total: pages.length,
   523→        });
   524→
   525→  return (
   526→    <section
   527→      ref={rootRef}
   528→      aria-label={t("grid.surfaceLabel")}
   529→      onKeyDown={onRootKeyDown}
   530→      className="flex flex-col gap-3"
   531→    >
   532→      {/* File header — the surface replaces the FileList for grid tools */}
   533→      <div className="flex items-center gap-3 rounded-card border border-border bg-card p-3">
   534→        <span
   535→          className="flex size-10 shrink-0 items-center justify-center rounded-btn bg-cat-organize/10 text-cat-organize"
   536→          aria-hidden="true"
   537→        >
   538→          <FileText className="size-5" />
   539→        </span>
   540→        <div className="min-w-0 flex-1">
   541→          <p
   542→            className="truncate text-sm font-semibold text-foreground"
   543→            title={file?.name}
   544→          >
   545→            {file?.name}
   546→          </p>
   547→          <p className="text-[11px] text-muted-foreground">
   548→            {file
   549→              ? `${formatBytes(file.size)}${
   550→                  docInfo?.file === file && docInfo.count > 0
   551→                    ? ` · ${t("tool.pageCount", { count: docInfo.count })}`
   552→                    : ""
   553→                }`
   554→              : ""}
   555→          </p>
   556→        </div>
   557→        {file ? (
   558→          <Button
   559→            type="button"
   560→            variant="outline"
   561→            className="h-11 min-h-11 rounded-btn px-3 text-xs"
   562→            onClick={() => setFileOrder([])}
   563→          >
   564→            {t("tool.removeFile")}
   565→          </Button>
   566→        ) : null}
   567→      </div>
   568→
   569→      {/* Toolbar — selection row */}
   570→      <div className="flex flex-col gap-3 rounded-card border border-border bg-card p-3">
   571→        <div className="flex flex-wrap items-center gap-2">
   572→          <span
   573→            className={cn(
   574→              "rounded-chip px-2.5 py-1 text-[11px] font-semibold",
   575→              selection.size > 0
   576→                ? "bg-primary/10 text-primary"
   577→                : "bg-muted text-muted-foreground"
   578→            )}
   579→          >
   580→            {selectionLabel}
   581→          </span>
   582→          <div className="flex min-w-44 flex-1 items-center gap-1.5">
   583→            <Input
   584→              value={rangeText}
   585→              onChange={(event) => setRangeText(event.target.value)}
   586→              onKeyDown={(event) => {
   587→                if (event.key === "Enter") {
   588→                  event.preventDefault();
   589→                  applyRange();
   590→                }
   591→              }}
   592→              placeholder={t("grid.rangePlaceholder")}
   593→              aria-label={t("grid.rangeLabel")}
   594→              spellCheck={false}
   595→              autoComplete="off"
   596→              className="h-11 rounded-btn text-xs font-medium"
   597→            />
   598→            <Button
   599→              type="button"
   600→              variant="outline"
   601→              className={TOOL_BTN}
   602→              onClick={applyRange}
   603→              aria-label={t("grid.rangeLabel")}
   604→            >
   605→              <MousePointerClick className="size-4" aria-hidden="true" />
   606→              <span className="hidden sm:inline">{t("grid.rangeApply")}</span>
   607→            </Button>
   608→          </div>
   609→          <div className="flex items-center gap-1.5">
   610→            <Button

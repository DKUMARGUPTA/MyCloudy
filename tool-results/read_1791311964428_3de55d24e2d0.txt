    79→/**
    80→ * ToolShell (PRD §6.3): the universal 4-state tool template.
    81→ * Landing (upload + SEO copy) → Workspace (preview + options) →
    82→ * Processing (real worker progress) → Result (download/share/chain).
    83→ * Every enabled tool renders from its registry entry + config — this
    84→ * component is the ONLY tool page in the app.
    85→ */
    86→
    87→type ShellPhase = "landing" | "workspace" | "processing" | "result";
    88→
    89→let fileSeq = 0;
    90→function nextFileId(): string {
    91→  fileSeq += 1;
    92→  return `wf-${Date.now().toString(36)}-${fileSeq}`;
    93→}
    94→
    95→export function ToolShell({ tool, config }: { tool: ToolDef; config: ToolConfig }) {
    96→  const { t, locale } = useI18n();
    97→  const { toast } = useToast();
    98→  const category = getCategory(tool.category);
    99→
   100→  // One-time chain handoff peek — set by a previous tool's result view or
   101→  // the home Smart Drop Zone right before navigating here. Consumed via
   102→  // the lazy initializer (pure read); cleared in the effect below.
   103→  const [chainIn] = useState<{
   104→    fromToolId: string;
   105→    items: WorkspaceFile[];
   106→  } | null>(() => {
   107→    const handoff = peekChainHandoff();
   108→    if (!handoff || handoff.items.length === 0) return null;
   109→    return {
   110→      fromToolId: handoff.fromToolId,
   111→      items: handoff.items.map((item) => ({
   112→        id: item.id,
   113→        file: item.file,
   114→      })),
   115→    };
   116→  });
   117→
   118→  const [phase, setPhase] = useState<ShellPhase>(() => (chainIn ? "workspace" : "landing"));
   119→  const [files, setFiles] = useState<WorkspaceFile[]>(() => chainIn?.items ?? []);
   120→  const [options, setOptions] = useState<OptionValues>(() =>
   121→    defaultOptionValues(config.options)
   122→  );
   123→  const [progress, setProgress] = useState<RunProgress>({
   124→    pct: 0,
   125→    stage: "reading",
   126→  });
   127→  const [outputs, setOutputs] = useState<ToolOutput[]>([]);
   128→  const [inputBytes, setInputBytes] = useState(0);
   129→  const [runSummary, setRunSummary] = useState<RunDetail | null>(null);
   130→  const [optionsOpen, setOptionsOpen] = useState(false);
   131→  const [helpOpen, setHelpOpen] = useState(false);
   132→
   133→  // Chain handoff consumed once at mount (lazy initializer = pure read);
   134→  // clearing the module store is an external-system sync in this effect.
   135→  const chainFrom: ToolDef | null = chainIn
   136→    ? (getTool(chainIn.fromToolId) ?? null)
   137→    : null;
   138→  useEffect(() => {
   139→    if (chainIn) clearChainHandoff();
   140→  }, [chainIn]);
   141→
   142→  // Input kind drives the DropZone sniffing + accept filter + labels.
   143→  const acceptKind =
   144→    tool.accepts === "image" ? "image" : tool.accepts === "pdf" ? "pdf" : "any";
   145→  const selectLabelKey =
   146→    tool.accepts === "image" ? "tool.selectImageFiles" : "tool.selectPdfFiles";
   147→  const addMoreLabelKey =
   148→    tool.accepts === "image" ? "tool.addMoreImages" : "tool.addMoreFiles";
   149→  const recentKind = tool.accepts === "image" ? "image" : "pdf";
   150→
   151→  // Chain/SmartDrop files get the same treatment as dropped files (PRD
   152→  // §6.1): OPFS working copy + a metadata-only recent entry with thumbnail.
   153→  // External-system sync only — no React state is touched here.
   154→  useEffect(() => {
   155→    if (!chainIn) return;
   156→    for (const item of chainIn.items) {
   157→      void writeWorkingFile(item.id, item.file).catch(() => undefined);
   158→      void makeFileThumbnail(item.file, 160)
   159→        .then((blob) =>
   160→          addRecentFile({
   161→            id: item.id,
   162→            kind: recentKind as "pdf" | "image",
   163→            label: item.file.name,
   164→            size: item.file.size,
   165→            toolId: tool.id,
   166→            thumbnail: blob ?? undefined,
   167→          })
   168→        )
   169→        .catch(() => undefined);
   170→    }
   171→  }, [chainIn, recentKind, tool.id]);
   172→
   173→  const runHandleRef = useRef<RunHandle | null>(null);
   174→
   175→  // Optional custom options panel (PRD §2.6 Panel.tsx pattern). Direct
   176→  // map lookup — a stable module-level reference, never created in render.
   177→  const CustomPanel = TOOL_PANELS[tool.id];
   178→
   179→  // Optional workspace surface (PageGrid family): rendered in the main
   180→  // column instead of the FileList; can gate the CTA via runBlocked.
   181→  const WorkspaceSurface = TOOL_SURFACES[tool.id];
   182→
   183→  // Surface-driven CTA gating (PRD §8.1 safety: "sab pages delete ho to
   184→  // Apply disabled" / extract needs a selection). Event-context only —
   185→  // surfaces call this from their action handlers, never during render.
   186→  const [runBlockedKey, setRunBlockedKey] = useState<string | null>(null);
   187→  const handleRunBlocked = useCallback((reasonKey: string | null) => {
   188→    setRunBlockedKey(reasonKey);
   189→  }, []);
   190→  const runDisabled =
   191→    files.length < config.minFiles || runBlockedKey !== null;
   192→
   193→  const handleOptionChange = useCallback(
   194→    (id: string, value: string | number | boolean) => {
   195→      setOptions((prev) => ({ ...prev, [id]: value }));
   196→    },
   197→    []
   198→  );
   199→
   200→  const relatedTools = useMemo(
   201→    () =>
   202→      toolsByCategory(tool.category)
   203→        .filter((item) => item.id !== tool.id)
   204→        .slice(0, 6),
   205→    [tool.category, tool.id]
   206→  );
   207→
   208→  const chainTargets = useMemo(() => {
   209→    // PRD §6.4 "Ab aage kya?" — natural next steps. Image outputs chain
   210→    // to image-accepting tools (the PDF default set would be a type lie);
   211→    // configs can override with their own curated list.
   212→    const suggestionIds =
   213→      config.chainToolIds ?? [
   214→        "compress-pdf",
   215→        "protect-pdf",
   216→        "sign-pdf",
   217→        "split-pdf",
   218→        "edit-pdf",
   219→      ];
   220→    const resolved = suggestionIds
   221→      .map((id) => getTool(id))
   222→      .filter((item): item is ToolDef => item !== undefined && item.id !== tool.id);
   223→    return resolved.slice(0, 5);
   224→  }, [config.chainToolIds, tool.id]);
   225→
   226→  // Working copies are cleared when the tool session ends (PRD §2.4).
   227→  useEffect(
   228→    () => () => {
   229→      void clearWorkingFiles().catch(() => undefined);
   230→    },
   231→    []
   232→  );
   233→
   234→  const addFiles = useCallback(
   235→    (incoming: File[]) => {
   236→      if (incoming.length === 0) return;
   237→      const room = config.maxFiles - files.length;
   238→      if (room <= 0) {
   239→        toast({
   240→          title: t("dropzone.errTooManyTitle"),
   241→          description: t("dropzone.errTooManyDesc", {
   242→            count: config.maxFiles,
   243→          }),
   244→          variant: "destructive",
   245→        });
   246→        return;
   247→      }
   248→      const accepted = incoming.slice(0, room);
   249→      const items = accepted.map((file) => ({ id: nextFileId(), file }));
   250→
   251→      // Side effects run ONCE here — never inside the setFiles updater
   252→      // (React StrictMode double-invokes updaters; they must stay pure).
   253→      for (const item of items) {
   254→        void writeWorkingFile(item.id, item.file).catch(() => undefined);
   255→        // Recent entry: metadata + thumbnail only, never the bytes.
   256→        void makeFileThumbnail(item.file, 160)
   257→          .then(async (blob) => {
   258→            await addRecentFile({
   259→              id: item.id,
   260→              kind: recentKind as "pdf" | "image",
   261→              label: item.file.name,
   262→              size: item.file.size,
   263→              toolId: tool.id,
   264→              thumbnail: blob ?? undefined,
   265→            });
   266→          })
   267→          .catch(() => undefined);
   268→      }
   269→
   270→      setFiles((prev) => [...prev, ...items]);
   271→      setPhase("workspace");
   272→    },
   273→    [config.maxFiles, files.length, recentKind, t, toast, tool.id]
   274→  );
   275→
   276→  const removeFile = useCallback((id: string) => {
   277→    setFiles((prev) => prev.filter((item) => item.id !== id));
   278→  }, []);
   279→
   280→  const moveFile = useCallback((from: number, to: number) => {
   281→    setFiles((prev) => {
   282→      if (to < 0 || to >= prev.length) return prev;
   283→      const next = [...prev];
   284→      const [moved] = next.splice(from, 1);
   285→      next.splice(to, 0, moved);
   286→      return next;
   287→    });
   288→  }, []);
   289→
   290→  const clearAll = useCallback(() => {
   291→    setFiles([]);
   292→    void clearWorkingFiles().catch(() => undefined);
   293→    setPhase("landing");
   294→  }, []);
   295→
   296→  // Stable function ref from the config (module constant) — keeps the
   297→  // React Compiler's dependency inference happy.
   298→  const collectAssets = config.collectAssets;
   299→
   300→  const startRun = useCallback(() => {
   301→    if (files.length < config.minFiles || runBlockedKey !== null) return;
   302→    trackToolUse(tool.id);
   303→    setInputBytes(files.reduce((sum, item) => sum + item.file.size, 0));
   304→    setProgress({ pct: 0, stage: "reading" });
   305→    setPhase("processing");
   306→    // Extra in-memory assets (watermark logo etc.) ride the worker message.
   307→    const assets = collectAssets?.() ?? undefined;
   308→    runHandleRef.current = toolRunner.run(
   309→      tool.id,
   310→      options,
   311→      files.map((item) => item.file),
   312→      setProgress,
   313→      files.map((item) => item.id),
   314→      assets
   315→    );
   316→    runHandleRef.current.promise
   317→      .then((result) => {
   318→        runHandleRef.current = null;
   319→        // Working copies of the INPUTS die with the run (PRD §2.4);
   320→        // results live in memory + can chain into the next tool.
   321→        setOutputs(result.outputs);
   322→        setRunSummary(result.detail ?? null);
   323→        setPhase("result");
   324→        void clearWorkingFiles().catch(() => undefined);
   325→      })
   326→      .catch((error: unknown) => {
   327→        runHandleRef.current = null;
   328→        if (error instanceof RunCancelledError) {
   329→          setPhase("workspace");
   330→          toast({ title: t("tool.cancelledToast") });
   331→          return;
   332→        }
   333→        setPhase("workspace");
   334→        const code =
   335→          error instanceof ToolRunError ? error.code : "tool.errorGeneric";
   336→        toast({
   337→          title: t("tool.errorTitle"),
   338→          description: t(code, {
   339→            detail: error instanceof ToolRunError ? error.detail ?? "" : "",
   340→          }),
   341→          variant: "destructive",
   342→        });
   343→      });
   344→  }, [collectAssets, config.minFiles, files, options, runBlockedKey, t, toast, tool.id]);
   345→
   346→  const cancelRun = useCallback(() => {
   347→    runHandleRef.current?.cancel();
   348→  }, []);
   349→
   350→  const handleChain = useCallback(
   351→    (target: ToolDef, resultFiles: File[]) => {
   352→      if (!target.enabled) {
   353→        toast({
   354→          title: t("tools.comingToastTitle", { tool: target.name[locale] }),
   355→          description: t("tools.comingToastDesc", {
   356→            phase: target.phase,
   357→          }),
   358→        });
   359→        return;
   360→      }
   361→      setChainHandoff({
   362→        fromToolId: tool.id,
   363→        items: resultFiles.map((file) => ({ id: nextFileId(), file })),
   364→      });
   365→      navigateToHash(ROUTES.tool(target.id));
   366→    },
   367→    [locale, t, toast, tool.id]
   368→  );
   369→
   370→  const removeFromDevice = useCallback(() => {
   371→    void clearWorkingFiles().catch(() => undefined);
   372→    // Remove this run's recent entries by their file metadata (label+size).
   373→    void removeRecentFilesByMeta(
   374→      files.map((item) => ({ label: item.file.name, size: item.file.size }))
   375→    ).catch(() => undefined);
   376→    setOutputs([]);
   377→    setFiles([]);
   378→    setPhase("landing");
   379→    toast({ title: t("result.removedToast") });
   380→  }, [files, t, toast]);
   381→
   382→  const totalSize = files.reduce((sum, item) => sum + item.file.size, 0);
   383→  const ctaLabel = `${tool.name[locale]} ${t("tool.ctaArrow")}`;
   384→  const Icon = tool.icon;
   385→
   386→  return (
   387→    <div className="mx-auto w-full max-w-6xl px-4 pb-28 sm:pb-24">
   388→      {/* Breadcrumb (PRD §6.3): Home > Category > Tool */}
   389→      <nav aria-label={t("a11y.breadcrumb")} className="pt-4">
   390→        <ol className="flex flex-wrap items-center gap-1.5 text-xs text-muted-foreground">
   391→          <li>
   392→            <a
   393→              href={ROUTES.home}
   394→              className="inline-flex items-center gap-1 rounded-btn outline-none hover:text-foreground focus-visible:ring-[3px] focus-visible:ring-ring/50"
   395→            >
   396→              <Home className="size-3" aria-hidden="true" />
   397→              {t("nav.home")}
   398→            </a>
   399→          </li>
   400→          <li aria-hidden="true">/</li>
   401→          <li>
   402→            <a
   403→              href={ROUTES.category(tool.category)}
   404→              className="rounded-btn outline-none hover:text-foreground focus-visible:ring-[3px] focus-visible:ring-ring/50"
   405→            >
   406→              {category.name[locale]}
   407→            </a>
   408→          </li>
   409→          <li aria-hidden="true">/</li>
   410→          <li aria-current="page" className="font-medium text-foreground">
   411→            {tool.name[locale]}
   412→          </li>
   413→        </ol>
   414→      </nav>
   415→
   416→      {phase === "landing" ? (
   417→        <section className="flex flex-col gap-8 pt-6">
   418→          <header className="flex flex-col gap-3">
   419→            <div className="flex items-center gap-3">
   420→              <span
   421→                className={cn(
   422→                  "flex size-11 items-center justify-center rounded-btn",
   423→                  "bg-cat-organize/10 text-cat-organize"
   424→                )}
   425→                aria-hidden="true"
   426→              >
   427→                <Icon className="size-5" />
   428→              </span>
   429→              <h1 className="text-2xl font-bold tracking-tight text-foreground sm:text-3xl">
   430→                {tool.name[locale]}
   431→              </h1>
   432→            </div>
   433→            <p className="max-w-2xl text-sm text-muted-foreground sm:text-base">
   434→              {tool.description[locale]}
   435→            </p>
   436→            <div className="flex flex-wrap gap-2">
   437→              <Badge variant={tool.mode === "D" ? "private" : "server"} className="rounded-chip">
   438→                <Lock className="size-3" aria-hidden="true" />
   439→                {tool.mode === "D"
   440→                  ? t("tools.privateBadge")
   441→                  : t("tools.serverBadge")}
   442→              </Badge>
   443→              <Badge variant="outline" className="rounded-chip">
   444→                {t("tool.freeBadge")}
   445→              </Badge>
   446→            </div>
   447→          </header>
   448→
   449→          <DropZone
   450→            labelKey={selectLabelKey}
   451→            multiple={config.maxFiles > 1}
   452→            maxFiles={config.maxFiles}
   453→            acceptKind={acceptKind}
   454→            onAccept={addFiles}
   455→            currentCount={files.length}
   456→          />
   457→
   458→          <div className="grid gap-8 lg:grid-cols-[2fr_1fr]">
   459→            <div className="flex flex-col gap-8">
   460→              {/* How it works */}
   461→              <section aria-labelledby="how-heading">
   462→                <h2 id="how-heading" className="flex items-center gap-2 text-sm font-semibold text-foreground">
   463→                  <ListChecks className="size-4 text-primary" aria-hidden="true" />
   464→                  {t("tool.howTitle")}
   465→                </h2>
   466→                <ol className="mt-3 grid gap-3 sm:grid-cols-3">
   467→                  {["tool.how1", "tool.how2", "tool.how3"].map((key, index) => (
   468→                    <li
   469→                      key={key}
   470→                      className="rounded-card border border-border bg-card p-4"
   471→                    >
   472→                      <span className="flex size-7 items-center justify-center rounded-full bg-primary/10 text-xs font-bold text-primary">
   473→                        {index + 1}
   474→                      </span>
   475→                      <p className="mt-2 text-xs leading-relaxed text-muted-foreground">
   476→                        {t(key)}
   477→                      </p>
   478→                    </li>
   479→                  ))}
   480→                </ol>
   481→              </section>
   482→
   483→              {/* FAQ */}
   484→              <section aria-labelledby="faq-heading">
   485→                <h2 id="faq-heading" className="text-sm font-semibold text-foreground">
   486→                  {t("tool.faqTitle")}
   487→                </h2>
   488→                <Accordion type="single" collapsible className="mt-3">
   489→                  {config.faq.map((item, index) => (
   490→                    <AccordionItem key={index} value={`faq-${index}`}>
   491→                      <AccordionTrigger className="text-left text-sm">
   492→                        {item.q[locale]}
   493→                      </AccordionTrigger>
   494→                      <AccordionContent className="text-xs leading-relaxed text-muted-foreground">
   495→                        {item.a[locale]}
   496→                      </AccordionContent>
   497→                    </AccordionItem>
   498→                  ))}
   499→                </Accordion>
   500→              </section>
   501→            </div>
   502→
   503→            <aside className="flex flex-col gap-8">
   504→              {/* Features */}
   505→              <section aria-labelledby="features-heading">
   506→                <h2 id="features-heading" className="flex items-center gap-2 text-sm font-semibold text-foreground">
   507→                  <Sparkles className="size-4 text-primary" aria-hidden="true" />
   508→                  {t("tool.featuresTitle")}
   509→                </h2>
   510→                <ul className="mt-3 flex flex-col gap-2.5">
   511→                  {config.features.map((feature, index) => (
   512→                    <li
   513→                      key={index}
   514→                      className="flex gap-2 rounded-card border border-border bg-card p-3 text-xs leading-relaxed text-muted-foreground"
   515→                    >
   516→                      <span
   517→                        aria-hidden="true"
   518→                        className="mt-0.5 size-1.5 shrink-0 rounded-full bg-primary"
   519→                      />
   520→                      {feature[locale]}
   521→                    </li>
   522→                  ))}
   523→                </ul>
   524→              </section>
   525→
   526→              {/* Related */}
   527→              <section aria-labelledby="related-heading">
   528→                <h2 id="related-heading" className="text-sm font-semibold text-foreground">
   529→                  {t("tool.relatedTitle")}
   530→                </h2>
   531→                <ul className="mt-3 flex flex-col gap-1.5">
   532→                  {relatedTools.map((item) => {
   533→                    const RelatedIcon = item.icon;
   534→                    return (
   535→                      <li key={item.id}>
   536→                        <a
   537→                          href={item.enabled ? ROUTES.tool(item.id) : ROUTES.category(item.category)}
   538→                          className="flex min-h-11 items-center gap-2.5 rounded-btn px-2 text-sm outline-none transition-colors hover:bg-accent focus-visible:ring-[3px] focus-visible:ring-ring/50"
   539→                        >
   540→                          <RelatedIcon className="size-4 text-muted-foreground" aria-hidden="true" />
   541→                          <span className="truncate text-foreground">
   542→                            {item.name[locale]}
   543→                          </span>
   544→                          {!item.enabled ? (
   545→                            <Badge variant="outline" className="ml-auto rounded-chip text-[10px] text-muted-foreground">
   546→                              {t("tools.soonBadge")}
   547→                            </Badge>
   548→                          ) : null}
   549→                        </a>
   550→                      </li>
   551→                    );
   552→                  })}
   553→                </ul>
   554→              </section>
   555→            </aside>
   556→          </div>
   557→        </section>
   558→      ) : null}
   559→
   560→      {phase === "workspace" ? (
   561→        <section className="pt-6">
   562→          {/* Workspace toolbar (PRD §6.3 state 2) */}
   563→          <div className="flex flex-wrap items-center gap-2">
   564→            <Button
   565→              type="button"
   566→              variant="ghost"
   567→              size="sm"
   568→              onClick={() => setPhase("landing")}
   569→              className="rounded-btn"
   570→            >
   571→              <ArrowLeft className="size-4" aria-hidden="true" />
   572→              {t("tool.back")}
   573→            </Button>
   574→            <span className="ml-auto flex items-center gap-2">
   575→              <Button
   576→                type="button"
   577→                variant="outline"
   578→                size="sm"
   579→                className="rounded-btn"
   580→                onClick={() => setHelpOpen(true)}
   581→                aria-label={t("tool.helpLabel")}
   582→              >
   583→                <CircleHelp className="size-4" aria-hidden="true" />
   584→                <span className="hidden sm:inline">{t("tool.helpLabel")}</span>
   585→              </Button>
   586→              <Button
   587→                type="button"
   588→                variant="outline"
   589→                size="sm"
   590→                className="rounded-btn"
   591→                onClick={clearAll}
   592→              >
   593→                <Trash2 className="size-4" aria-hidden="true" />
   594→                {t("tool.clearAll")}
   595→              </Button>
   596→            </span>
   597→          </div>
   598→
   599→          <div className="mt-4 grid grid-cols-1 gap-6 lg:grid-cols-[minmax(0,1fr)_320px]">
   600→            {/* Main column: the surface owns file presentation when one is
   601→                registered (PageGrid family); otherwise the classic
   602→                FileList + add-more DropZone (PRD §6.3). */}
   603→            <div className="flex flex-col gap-3">
   604→              {WorkspaceSurface && files.length > 0 ? (
   605→                <WorkspaceSurface
   606→                  files={files}
   607→                  values={options}
   608→                  onChange={handleOptionChange}
   609→                  setFileOrder={setFiles}
   610→                  setRunBlocked={handleRunBlocked}
   611→                  addFiles={addFiles}
   612→                />
   613→              ) : (
   614→                <>
   615→                  <FileList
   616→                    items={files}
   617→                    onMove={moveFile}
   618→                    onRemove={removeFile}
   619→                  />
   620→                  <DropZone
   621→                    labelKey={addMoreLabelKey}
   622→                    multiple={config.maxFiles > 1}
   623→                    maxFiles={config.maxFiles}
   624→                    acceptKind={acceptKind}
   625→                    onAccept={addFiles}
   626→                    currentCount={files.length}
   627→                    className="border-dashed py-4"
   628→                  />
   629→                </>
   630→              )}
   631→            </div>
   632→
   633→            {/* Options: desktop side panel / mobile bottom sheet.
   634→                Height budget: sticky header (top-20) + sticky CTA bar at the
   635→                bottom — so the panel never slides under the CTA bar. */}
   636→            <aside className="hidden lg:block">
   637→              <div className="sticky top-20 mb-20 flex max-h-[calc(100vh-11.5rem)] flex-col gap-5 overflow-y-auto scrollbar-thin rounded-card border border-border bg-card p-5">
   638→                <div>
   639→                  <h2 className="text-sm font-semibold text-foreground">
   640→                    {t("tool.optionsTitle")}
   641→                  </h2>
   642→                  {CustomPanel ? (
   643→                    <div className="mt-4">
   644→                      <CustomPanel
   645→                        files={files}
   646→                        values={options}
   647→                        onChange={handleOptionChange}
   648→                        setFileOrder={setFiles}
   649→                      />
   650→                    </div>
   651→                  ) : null}
   652→                  <OptionsPanel
   653→                    fields={config.options}
   654→                    values={options}
   655→                    onChange={handleOptionChange}
   656→                    className="mt-4"
   657→                  />
   658→                </div>
   659→                <p className="border-t border-border pt-3 text-[11px] text-muted-foreground">
   660→                  {files.length > 0 && files.length < config.minFiles
   661→                    ? t("tool.summaryMinFiles", { count: config.minFiles })
   662→                    : t("tool.summary", {
   663→                        count: files.length,
   664→                        size: formatBytes(totalSize),
   665→                      })}
   666→                </p>
   667→              </div>
   668→            </aside>
   669→          </div>
   670→
   671→          {/* Sticky CTA (PRD §6.3): count + estimated size + action */}
   672→          <div className="fixed inset-x-0 bottom-0 z-40 border-t border-border bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/85">
   673→            <div className="mx-auto flex max-w-6xl items-center gap-3 px-4 py-3">
   674→              <div className="min-w-0 flex-1">
   675→                <p
   676→                  className={cn(
   677→                    "flex items-center gap-1.5 truncate text-xs",
   678→                    runBlockedKey !== null
   679→                      ? "font-medium text-amber-600 dark:text-amber-400"
   680→                      : files.length > 0 && files.length < config.minFiles
   681→                        ? "font-medium text-primary"
   682→                        : "text-muted-foreground"
   683→                  )}
   684→                >
   685→                  {runBlockedKey !== null ? (
   686→                    <TriangleAlert
   687→                      className="size-3.5 shrink-0"
   688→                      aria-hidden="true"
   689→                    />
   690→                  ) : null}
   691→                  <span className="truncate">
   692→                    {runBlockedKey !== null
   693→                      ? t(runBlockedKey)
   694→                      : files.length > 0 && files.length < config.minFiles
   695→                        ? t("tool.summaryMinFiles", { count: config.minFiles })
   696→                        : t("tool.summary", {
   697→                            count: files.length,
   698→                            size: formatBytes(totalSize),
   699→                          })}
   700→                  </span>
   701→                </p>
   702→                {chainFrom ? (
   703→                  <p className="truncate text-[11px] text-primary">
   704→                    {t("tool.chainFrom", { tool: chainFrom.name[locale] })}
   705→                  </p>
   706→                ) : null}
   707→              </div>
   708→              <Button
   709→                type="button"
   710→                variant="outline"
   711→                size="touch"
   712→                className="rounded-btn lg:hidden"
   713→                onClick={() => setOptionsOpen(true)}
   714→              >
   715→                {t("tool.optionsButton")}
   716→              </Button>
   717→              <Button
   718→                type="button"
   719→                size="touch"
   720→                className="rounded-btn font-semibold"
   721→                onClick={startRun}
   722→                disabled={runDisabled}
   723→              >
   724→                {ctaLabel}
   725→                <ArrowRight className="size-4" aria-hidden="true" />
   726→              </Button>
   727→            </div>
   728→          </div>
   729→
   730→          <BottomSheet
   731→            open={optionsOpen}
   732→            onOpenChange={setOptionsOpen}
   733→            title={t("tool.optionsTitle")}
   734→            description={tool.description[locale]}
   735→            footer={
   736→              <Button
   737→                type="button"
   738→                size="touch"
   739→                className="w-full rounded-btn font-semibold"
   740→                onClick={() => {
   741→                  setOptionsOpen(false);
   742→                  startRun();
   743→                }}
   744→                disabled={runDisabled}
   745→              >
   746→                {ctaLabel}
   747→                <ArrowRight className="size-4" aria-hidden="true" />
   748→              </Button>
   749→            }
   750→          >
   751→            {CustomPanel ? (
   752→              <div className="mb-4">
   753→                <CustomPanel
   754→                  files={files}
   755→                  values={options}
   756→                  onChange={handleOptionChange}
   757→                  setFileOrder={setFiles}
   758→                />
   759→              </div>
   760→            ) : null}
   761→            <OptionsPanel
   762→              fields={config.options}
   763→              values={options}
   764→              onChange={handleOptionChange}
   765→            />
   766→          </BottomSheet>
   767→        </section>
   768→      ) : null}
   769→
   770→      {phase === "processing" ? (
   771→        <ProcessingView
   772→          toolName={tool.name[locale]}
   773→          progress={progress}
   774→          onCancel={cancelRun}
   775→        />
   776→      ) : null}
   777→
   778→      {phase === "result" ? (
   779→        <ResultView
   780→          tool={tool}
   781→          outputs={outputs}
   782→          inputBytes={inputBytes}
   783→          runSummary={runSummary}
   784→          chainFromName={chainFrom ? chainFrom.name[locale] : undefined}
   785→          chainTargets={chainTargets}
   786→          onChain={handleChain}
   787→          onRemoveFromDevice={removeFromDevice}
   788→        />
   789→      ) : null}
   790→
   791→      {/* Help dialog */}
   792→      <Modal
   793→        open={helpOpen}
   794→        onOpenChange={setHelpOpen}
   795→        title={t("tool.helpTitle")}
   796→        footer={
   797→          <Button
   798→            type="button"
   799→            variant="default"
   800→            size="touch"
   801→            className="w-full rounded-btn sm:w-auto"
   802→            onClick={() => setHelpOpen(false)}
   803→          >
   804→            {t("pro.action")}
   805→          </Button>
   806→        }
   807→      >
   808→        <div className="flex flex-col gap-3 text-sm text-muted-foreground">
   809→          <p className="flex gap-2">
   810→            <FilePlus2 className="mt-0.5 size-4 shrink-0 text-primary" aria-hidden="true" />
   811→            {t("tool.how2")}
   812→          </p>
   813→          <p>{t("tool.helpPrivacy")}</p>
   814→        </div>
   815→      </Modal>
   816→    </div>
   817→  );
   818→}

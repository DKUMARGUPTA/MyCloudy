     1→"use client";
     2→
     3→import {
     4→  Fragment,
     5→  useEffect,
     6→  useMemo,
     7→  useRef,
     8→  useState,
     9→  type KeyboardEvent,
    10→} from "react";
    11→import {
    12→  Bookmark,
    13→  Files,
    14→  Loader2,
    15→  Scissors,
    16→  TriangleAlert,
    17→} from "lucide-react";
    18→
    19→import { Input } from "@/components/ui/input";
    20→import { Label } from "@/components/ui/label";
    21→import { Switch } from "@/components/ui/switch";
    22→import { useI18n } from "@/components/providers/locale-provider";
    23→import type { ToolPanelProps } from "@/lib/tools/panel-types";
    24→import {
    25→  detectBlankPages,
    26→  getPdfPageCount,
    27→  getPdfTopLevelOutline,
    28→  renderPdfThumbnail,
    29→  type PdfOutlineEntry,
    30→} from "@/lib/pdf/thumbnails";
    31→import {
    32→  computeSplitPlan,
    33→  fileKey,
    34→  parseNumberListOption,
    35→  parseRangeList,
    36→  rangesOverlap,
    37→  type SplitMode,
    38→  type SplitPart,
    39→} from "./plan";
    40→import { cn } from "@/lib/utils";
    41→
    42→/**
    43→ * Split PDF custom panel (PRD §8.1): mode tabs (7 modes), per-mode inputs
    44→ * and the LIVE preview strip — cut markers, color-coded parts and the
    45→ * "N files banengi" count — all computed with the same pure plan module the
    46→ * worker executes. Bookmark + blank-page detection run on the main thread
    47→ * with the self-hosted pdf.js and reach the worker as plain options.
    48→ */
    49→
    50→const MODE_IDS: { id: SplitMode; labelKey: string }[] = [
    51→  { id: "range", labelKey: "split.modeRange" },
    52→  { id: "everyN", labelKey: "split.modeEveryN" },
    53→  { id: "bookmarks", labelKey: "split.modeBookmarks" },
    54→  { id: "size", labelKey: "split.modeSize" },
    55→  { id: "visual", labelKey: "split.modeVisual" },
    56→  { id: "oddEven", labelKey: "split.modeOddEven" },
    57→  { id: "blanks", labelKey: "split.modeBlanks" },
    58→];
    59→
    60→const PART_STYLES = [
    61→  "border-primary/30 bg-primary/10 text-primary dark:text-primary",
    62→  "border-purple-500/30 bg-purple-500/10 text-purple-600 dark:text-purple-400",
    63→  "border-emerald-500/30 bg-emerald-500/10 text-emerald-600 dark:text-emerald-400",
    64→  "border-amber-500/30 bg-amber-500/10 text-amber-600 dark:text-amber-400",
    65→];
    66→
    67→function partStyle(index: number): string {
    68→  return PART_STYLES[index % PART_STYLES.length];
    69→}
    70→
    71→export function SplitPanel({
    72→  files,
    73→  values,
    74→  onChange,
    75→}: ToolPanelProps) {
    76→  const { t } = useI18n();
    77→  const file = files.length > 0 ? files[0].file : null;
    78→  const tabRefs = useRef<(HTMLButtonElement | null)[]>([]);
    79→
    80→  const mode: SplitMode = useMemo(() => {
    81→    const raw = values.mode;
    82→    return typeof raw === "string" &&
    83→      MODE_IDS.some((entry) => entry.id === raw)
    84→      ? (raw as SplitMode)
    85→      : "range";
    86→  }, [values.mode]);
    87→
    88→  const ranges = typeof values.ranges === "string" ? values.ranges : "";
    89→  const oneFilePerRange = values.oneFilePerRange !== false;
    90→  const everyN = typeof values.everyN === "number" ? values.everyN : 5;
    91→  const maxSizeMb =
    92→    typeof values.maxSizeMb === "number" ? values.maxSizeMb : 5;
    93→  const cutAfter = useMemo(
    94→    () => parseNumberListOption(values.cutAfter),
    95→    [values.cutAfter]
    96→  );
    97→
    98→  /* ---------------- Page count + outline detection ---------------- */
    99→  // Async results are keyed by their File and only applied to state inside
   100→  // promise callbacks (never synchronously in the effect body).
   101→
   102→  const [pageCountState, setPageCountState] = useState<{
   103→    file: File;
   104→    count: number;
   105→  } | null>(null);
   106→  const pageCount =
   107→    pageCountState !== null && pageCountState.file === file
   108→      ? pageCountState.count
   109→      : null;
   110→
   111→  const [outlineState, setOutlineState] = useState<{
   112→    file: File;
   113→    entries: PdfOutlineEntry[];
   114→  } | null>(null);
   115→  const outline =
   116→    outlineState !== null && outlineState.file === file
   117→      ? outlineState.entries
   118→      : null;
   119→
   120→  useEffect(() => {
   121→    if (!file) return;
   122→    let cancelled = false;
   123→    void getPdfPageCount(file)
   124→      .then((count) => {
   125→        if (!cancelled) setPageCountState({ file, count });
   126→      })
   127→      .catch(() => {
   128→        if (!cancelled) setPageCountState({ file, count: 0 });
   129→      });
   130→    return () => {
   131→      cancelled = true;
   132→    };
   133→  }, [file]);
   134→
   135→  // Outline runs for every file (it decides whether the bookmarks tab is
   136→  // usable) and mirrors its result into options for the worker.
   137→  useEffect(() => {
   138→    if (!file) return;
   139→    let cancelled = false;
   140→    const key = fileKey(file.name, file.size);
   141→    void getPdfTopLevelOutline(file)
   142→      .then((entries) => {
   143→        if (cancelled) return;
   144→        setOutlineState({ file, entries });
   145→        onChange("bookmarkStatus", "done");
   146→        onChange("bookmarkFileKey", key);
   147→        onChange(
   148→          "bookmarkCuts",
   149→          JSON.stringify([...new Set(entries.map((entry) => entry.page))].sort((a, b) => a - b))
   150→        );
   151→      })
   152→      .catch(() => {
   153→        if (cancelled) return;
   154→        setOutlineState({ file, entries: [] });
   155→        onChange("bookmarkStatus", "done");
   156→        onChange("bookmarkFileKey", key);
   157→        onChange("bookmarkCuts", "[]");
   158→      });
   159→    return () => {
   160→      cancelled = true;
   161→    };
   162→  }, [file, onChange]);
   163→
   164→  /* ---------------- Blank-page detection (mode 7) ---------------- */
   165→
   166→  const [blankProgress, setBlankProgress] = useState<{
   167→    file: File;
   168→    done: number;
   169→    total: number;
   170→  } | null>(null);
   171→
   172→  const blankDetectionDone =
   173→    file !== null &&
   174→    values.blankStatus === "done" &&
   175→    values.blankFileKey === fileKey(file.name, file.size);
   176→  const blankPages = blankDetectionDone
   177→    ? parseNumberListOption(values.blankPages)
   178→    : [];
   179→  const blanksDetecting =
   180→    file !== null && mode === "blanks" && !blankDetectionDone;
   181→
   182→  useEffect(() => {
   183→    if (!file || mode !== "blanks") return;
   184→    if (
   185→      values.blankStatus === "done" &&
   186→      values.blankFileKey === fileKey(file.name, file.size)
   187→    ) {
   188→      return; // already detected for this exact file
   189→    }
   190→    let cancelled = false;
   191→    void detectBlankPages(file, (progress) => {
   192→      if (!cancelled) {
   193→        setBlankProgress({ file, done: progress.done, total: progress.total });
   194→      }
   195→    })
   196→      .then((pages) => {
   197→        if (cancelled) return;
   198→        setBlankProgress(null);
   199→        onChange("blankStatus", "done");
   200→        onChange("blankFileKey", fileKey(file.name, file.size));
   201→        onChange("blankPages", JSON.stringify(pages));
   202→      })
   203→      .catch(() => {
   204→        if (cancelled) return;
   205→        setBlankProgress(null);
   206→        onChange("blankStatus", "done");
   207→        onChange("blankFileKey", fileKey(file.name, file.size));
   208→        onChange("blankPages", "[]");
   209→      });
   210→    return () => {
   211→      cancelled = true;
   212→    };
   213→  }, [file, mode, onChange, values.blankStatus, values.blankFileKey]);
   214→
   215→  /* ---------------- Live plan preview ---------------- */
   216→
   217→  const rangeItems =
   218→    mode === "range" && pageCount !== null
   219→      ? parseRangeList(ranges, pageCount)
   220→      : null;
   221→  const rangesValid = mode !== "range" || rangeItems !== null;
   222→  const overlapping =
   223→    rangeItems !== null && rangesOverlap(rangeItems) ? true : false;
   224→
   225→  const plan = useMemo(() => {
   226→    if (!file || pageCount === null || pageCount < 1) return null;
   227→    return computeSplitPlan({
   228→      mode,
   229→      pageCount,
   230→      fileSize: file.size,
   231→      ranges,
   232→      oneFilePerRange,
   233→      everyN,
   234→      maxSizeMb,
   235→      cutAfter,
   236→      bookmarkCuts: outline ? [...new Set(outline.map((e) => e.page))] : [],
   237→      blankPages,
   238→    });
   239→  }, [
   240→    file,
   241→    pageCount,
   242→    mode,
   243→    ranges,
   244→    oneFilePerRange,
   245→    everyN,
   246→    maxSizeMb,
   247→    cutAfter,
   248→    outline,
   249→    blankPages,
   250→  ]);
   251→
   252→  const bookmarkTabDisabled = outline !== null && outline.length === 0;
   253→  const canShowCount =
   254→    plan !== null && !(mode === "blanks" && blankDetectionDone && blankPages.length === 0);
   255→
   256→  const toggleCut = (page: number) => {
   257→    const next = cutAfter.includes(page)
   258→      ? cutAfter.filter((item) => item !== page)
   259→      : [...cutAfter, page].sort((a, b) => a - b);
   260→    onChange("cutAfter", JSON.stringify(next));
   261→  };
   262→
   263→  const selectMode = (next: SplitMode) => {
   264→    if (next === "bookmarks" && bookmarkTabDisabled) return;
   265→    onChange("mode", next);
   266→  };
   267→
   268→  const onTablistKeyDown = (event: KeyboardEvent<HTMLDivElement>) => {
   269→    if (event.key !== "ArrowRight" && event.key !== "ArrowLeft") return;
   270→    event.preventDefault();
   271→    const direction = event.key === "ArrowRight" ? 1 : -1;
   272→    const enabledIndexes = MODE_IDS.map((entry, index) => ({
   273→      entry,
   274→      index,
   275→    }))
   276→      .filter(({ entry }) => !(entry.id === "bookmarks" && bookmarkTabDisabled))
   277→      .map(({ index }) => index);
   278→    if (enabledIndexes.length === 0) return;
   279→    const currentIndex = MODE_IDS.findIndex((entry) => entry.id === mode);
   280→    const position = enabledIndexes.indexOf(currentIndex);
   281→    const nextPosition =
   282→      (position === -1 ? 0 : position + direction + enabledIndexes.length) %
   283→      enabledIndexes.length;
   284→    const nextIndex = enabledIndexes[nextPosition];
   285→    const nextEntry = MODE_IDS[nextIndex];
   286→    tabRefs.current[nextIndex]?.focus();
   287→    selectMode(nextEntry.id);
   288→  };
   289→
   290→  /* ---------------- Render ---------------- */
   291→
   292→  return (
   293→    <section
   294→      aria-label={t("split.panelTitle")}
   295→      className="flex flex-col gap-3"
   296→    >
   297→      <div className="flex items-center justify-between gap-2">
   298→        <h3 className="flex items-center gap-1.5 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   299→          <Scissors className="size-3.5" aria-hidden="true" />
   300→          {t("split.panelTitle")}
   301→        </h3>
   302→        {file ? (
   303→          <span className="text-[11px] text-muted-foreground">
   304→            {pageCount !== null
   305→              ? t("split.pagesInFile", { count: pageCount })
   306→              : t("split.detecting")}
   307→          </span>
   308→        ) : null}
   309→      </div>
   310→
   311→      {!file ? (
   312→        <p className="rounded-btn border border-dashed border-border bg-muted/30 px-3 py-2 text-[11px] text-muted-foreground">
   313→          {t("split.emptyHint")}
   314→        </p>
   315→      ) : (
   316→        <>
   317→          {/* Mode tabs (PRD §8.1 "Modes (tabs)") */}
   318→          <div
   319→            role="tablist"
   320→            aria-label={t("split.modeLabel")}
   321→            className="flex flex-wrap gap-1.5"
   322→            onKeyDown={onTablistKeyDown}
   323→          >
   324→            {MODE_IDS.map((entry, index) => {
   325→              const disabled = entry.id === "bookmarks" && bookmarkTabDisabled;
   326→              const selected = mode === entry.id;
   327→              return (
   328→                <button
   329→                  key={entry.id}
   330→                  ref={(node) => {
   331→                    tabRefs.current[index] = node;
   332→                  }}
   333→                  type="button"
   334→                  role="tab"
   335→                  aria-selected={selected}
   336→                  aria-disabled={disabled}
   337→                  disabled={disabled}
   338→                  title={
   339→                    disabled ? t("split.bookmarksDisabled") : undefined
   340→                  }
   341→                  onClick={() => selectMode(entry.id)}
   342→                  className={cn(
   343→                    "min-h-9 rounded-btn border px-2.5 text-xs font-medium outline-none transition-colors",
   344→                    selected
   345→                      ? "border-primary bg-primary text-primary-foreground"
   346→                      : "border-border bg-background text-foreground hover:border-primary/40 hover:bg-accent",
   347→                    disabled && "cursor-not-allowed opacity-50",
   348→                    "focus-visible:ring-[3px] focus-visible:ring-ring/50"
   349→                  )}
   350→                >
   351→                  {t(entry.labelKey)}
   352→                </button>
   353→              );
   354→            })}
   355→          </div>
   356→
   357→          {/* Per-mode controls */}
   358→          {mode === "range" ? (
   359→            <div className="flex flex-col gap-2">
   360→              <div className="flex flex-col gap-1.5">
   361→                <Label htmlFor="split-ranges" className="text-xs font-medium">
   362→                  {t("split.rangeLabel")}
   363→                </Label>
   364→                <Input
   365→                  id="split-ranges"
   366→                  type="text"
   367→                  spellCheck={false}
   368→                  autoComplete="off"
   369→                  value={ranges}
   370→                  placeholder={t("split.rangePlaceholder")}
   371→                  onChange={(event) => onChange("ranges", event.target.value)}
   372→                  aria-invalid={!rangesValid && ranges.trim() !== ""}
   373→                  className={cn(
   374→                    "h-10 rounded-btn text-xs font-medium",
   375→                    !rangesValid &&
   376→                      ranges.trim() !== "" &&
   377→                      "border-destructive focus-visible:ring-destructive/30"
   378→                  )}
   379→                />
   380→                {ranges.trim() === "" ? (
   381→                  <p className="text-[11px] leading-relaxed text-muted-foreground">
   382→                    {t("split.rangeHint")}
   383→                  </p>
   384→                ) : rangesValid ? (
   385→                  overlapping ? (
   386→                    <p className="flex items-start gap-1.5 rounded-btn border border-amber-500/40 bg-amber-500/10 px-2 py-1.5 text-[11px] font-medium leading-relaxed text-amber-600 dark:text-amber-400">
   387→                      <TriangleAlert className="mt-0.5 size-3.5 shrink-0" aria-hidden="true" />
   388→                      {t("split.overlapWarning")}
   389→                    </p>
   390→                  ) : (
   391→                    <p className="text-[11px] leading-relaxed text-muted-foreground">
   392→                      {t("split.rangeHint")}
   393→                    </p>
   394→                  )
   395→                ) : (
   396→                  <p className="text-[11px] font-medium leading-relaxed text-destructive">
   397→                    {t("split.invalidRanges")}
   398→                  </p>
   399→                )}
   400→              </div>
   401→              <div className="flex items-center justify-between gap-3 rounded-btn border border-border bg-background px-2.5 py-2">
   402→                <Label
   403→                  htmlFor="split-one-per-range"
   404→                  className="text-xs font-medium leading-snug"
   405→                >
   406→                  {t("split.oneFilePerRangeLabel")}
   407→                  <span className="block text-[11px] font-normal text-muted-foreground">
   408→                    {t("split.oneFilePerRangeHint")}
   409→                  </span>
   410→                </Label>
   411→                <Switch
   412→                  id="split-one-per-range"
   413→                  checked={oneFilePerRange}
   414→                  onCheckedChange={(next) => onChange("oneFilePerRange", next)}
   415→                  aria-label={t("split.oneFilePerRangeLabel")}
   416→                />
   417→              </div>
   418→            </div>
   419→          ) : null}
   420→
   421→          {mode === "everyN" ? (
   422→            <div className="flex flex-col gap-1.5">
   423→              <Label htmlFor="split-every-n" className="text-xs font-medium">
   424→                {t("split.everyNLabel")}
   425→              </Label>
   426→              <div className="flex items-center gap-2">
   427→                <Input
   428→                  id="split-every-n"
   429→                  type="number"
   430→                  inputMode="numeric"
   431→                  min={1}
   432→                  max={pageCount ?? 9999}
   433→                  step={1}
   434→                  value={everyN}
   435→                  onChange={(event) =>
   436→                    onChange("everyN", event.target.valueAsNumber || 1)
   437→                  }
   438→                  className="h-10 w-24 rounded-btn text-xs font-medium"
   439→                />
   440→                <span className="text-xs text-muted-foreground">
   441→                  {t("split.pagesSuffix")}
   442→                </span>
   443→              </div>
   444→              <p className="text-[11px] leading-relaxed text-muted-foreground">
   445→                {t("split.everyNHint")}
   446→              </p>
   447→            </div>
   448→          ) : null}
   449→
   450→          {mode === "bookmarks" ? (
   451→            outline === null ? (
   452→              <p className="flex items-center gap-2 rounded-btn border border-dashed border-border bg-muted/30 px-3 py-2 text-[11px] text-muted-foreground">
   453→                <Loader2
   454→                  className="size-3.5 animate-spin"
   455→                  aria-hidden="true"
   456→                />
   457→                {t("split.detecting")}
   458→              </p>
   459→            ) : outline.length === 0 ? (
   460→              <p className="flex items-start gap-1.5 rounded-btn border border-dashed border-border bg-muted/30 px-3 py-2 text-[11px] text-muted-foreground">
   461→                <Bookmark className="mt-0.5 size-3.5 shrink-0" aria-hidden="true" />
   462→                {t("split.bookmarksNone")}
   463→              </p>
   464→            ) : (
   465→              <div className="flex flex-col gap-1.5">
   466→                <p className="text-[11px] text-muted-foreground">
   467→                  {t("split.bookmarksCount", { count: outline.length })} —{" "}
   468→                  {t("split.bookmarksList")}
   469→                </p>
   470→                <ul className="scrollbar-thin flex max-h-36 flex-col gap-1 overflow-y-auto pr-1">
   471→                  {outline.map((entry, index) => (
   472→                    <li
   473→                      key={`${entry.page}-${index}`}
   474→                      className="flex items-center gap-2 rounded-btn border border-border bg-background px-2 py-1.5"
   475→                    >
   476→                      <span
   477→                        className={cn(
   478→                          "flex size-5 shrink-0 items-center justify-center rounded-full text-[10px] font-bold",
   479→                          partStyle(index)
   480→                        )}
   481→                        aria-hidden="true"
   482→                      >
   483→                        {index + 1}
   484→                      </span>
   485→                      <span className="min-w-0 flex-1 truncate text-xs text-foreground" title={entry.title}>
   486→                        {entry.title}
   487→                      </span>
   488→                      <span className="shrink-0 rounded-chip bg-muted px-1.5 py-0.5 text-[10px] font-semibold text-muted-foreground">
   489→                        {t("split.pageChipLabel", { page: entry.page })}
   490→                      </span>
   491→                    </li>
   492→                  ))}
   493→                </ul>
   494→              </div>
   495→            )
   496→          ) : null}
   497→
   498→          {mode === "size" ? (
   499→            <div className="flex flex-col gap-1.5">
   500→              <Label htmlFor="split-max-size" className="text-xs font-medium">
   501→                {t("split.sizeLabel")}
   502→              </Label>
   503→              <div className="flex items-center gap-2">
   504→                <Input
   505→                  id="split-max-size"
   506→                  type="number"
   507→                  inputMode="decimal"
   508→                  min={0.1}
   509→                  max={2000}
   510→                  step={0.5}
   511→                  value={maxSizeMb}
   512→                  onChange={(event) =>
   513→                    onChange(
   514→                      "maxSizeMb",
   515→                      event.target.valueAsNumber > 0
   516→                        ? event.target.valueAsNumber
   517→                        : 0.1
   518→                    )
   519→                  }
   520→                  className="h-10 w-24 rounded-btn text-xs font-medium"
   521→                />
   522→                <span className="text-xs text-muted-foreground">MB</span>
   523→              </div>
   524→              <p className="text-[11px] leading-relaxed text-muted-foreground">
   525→                {t("split.sizeHint")}
   526→              </p>
   527→            </div>
   528→          ) : null}
   529→
   530→          {mode === "visual" ? (
   531→            <div className="flex flex-col gap-2">
   532→              <p className="text-[11px] leading-relaxed text-muted-foreground">
   533→                {t("split.visualHint")}
   534→              </p>
   535→              {cutAfter.length > 0 ? (
   536→                <button
   537→                  type="button"
   538→                  onClick={() => onChange("cutAfter", "[]")}
   539→                  className="flex min-h-9 w-fit items-center gap-1.5 rounded-btn border border-border bg-background px-2.5 text-xs font-medium text-muted-foreground outline-none transition-colors hover:border-destructive/40 hover:text-destructive focus-visible:ring-[3px] focus-visible:ring-ring/50"
   540→                >
   541→                  <Scissors className="size-3.5" aria-hidden="true" />
   542→                  {t("split.visualClear", { count: cutAfter.length })}
   543→                </button>
   544→              ) : null}
   545→            </div>
   546→          ) : null}
   547→
   548→          {mode === "oddEven" ? (
   549→            <p className="text-[11px] leading-relaxed text-muted-foreground">
   550→              {t("split.oddEvenHint")}
   551→            </p>
   552→          ) : null}
   553→
   554→          {mode === "blanks" ? (
   555→            blanksDetecting ? (
   556→              <p className="flex items-center gap-2 rounded-btn border border-dashed border-border bg-muted/30 px-3 py-2 text-[11px] text-muted-foreground">
   557→                <Loader2 className="size-3.5 animate-spin" aria-hidden="true" />
   558→                {blankProgress !== null && blankProgress.file === file
   559→                  ? t("split.blanksDetectingPct", {
   560→                      percent: Math.round(
   561→                        (blankProgress.done / Math.max(1, blankProgress.total)) * 100
   562→                      ),
   563→                    })
   564→                  : t("split.blanksDetecting")}
   565→              </p>
   566→            ) : blankPages.length === 0 ? (
   567→              <p className="flex items-start gap-1.5 rounded-btn border border-dashed border-border bg-muted/30 px-3 py-2 text-[11px] text-muted-foreground">
   568→                <TriangleAlert className="mt-0.5 size-3.5 shrink-0" aria-hidden="true" />
   569→                {t("split.blanksNone")}
   570→              </p>
   571→            ) : (
   572→              <div className="flex flex-col gap-2">
   573→                <p className="text-[11px] font-medium leading-relaxed text-foreground">
   574→                  {t("split.blanksFound", { count: blankPages.length })}
   575→                </p>
   576→                <div className="flex flex-wrap gap-1" aria-label={t("split.blanksChipLabel")}>
   577→                  {blankPages.map((page) => (
   578→                    <span
   579→                      key={page}
   580→                      className="rounded-md border border-border bg-muted px-1.5 py-0.5 text-[11px] font-semibold text-muted-foreground line-through"
   581→                    >
   582→                      {page}
   583→                    </span>
   584→                  ))}
   585→                </div>
   586→              </div>
   587→            )
   588→          ) : null}
   589→
   590→          {/* Live preview (PRD §8.1: cut markers + color-coded parts +
   591→              live count) */}
   592→          <div className="flex flex-col gap-2.5 rounded-card border border-border bg-background p-3">
   593→            <div className="flex items-center justify-between gap-2">
   594→              <h4 className="flex items-center gap-1.5 text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">
   595→                <Files className="size-3.5" aria-hidden="true" />
   596→                {t("split.previewTitle")}
   597→              </h4>
   598→              <div className="flex items-center gap-1.5">
   599→                {plan?.estimated ? (
   600→                  <span
   601→                    className="rounded-chip border border-amber-500/40 bg-amber-500/10 px-1.5 py-0.5 text-[10px] font-semibold text-amber-600 dark:text-amber-400"
   602→                    title={t("split.sizeHint")}
   603→                  >
   604→                    ≈ {t("split.estimated")}
   605→                  </span>
   606→                ) : null}
   607→                {canShowCount && plan !== null ? (
   608→                  <span className="rounded-chip bg-primary/10 px-2 py-0.5 text-[11px] font-semibold text-primary">
   609→                    {plan.parts.length === 1
   610→                      ? t("split.willCreateOne")
   611→                      : t("split.willCreate", { count: plan.parts.length })}
   612→                  </span>
   613→                ) : null}
   614→              </div>
   615→            </div>
   616→
   617→            {mode === "visual" && pageCount !== null ? (
   618→              <VisualStrip
   619→                file={file}
   620→                pageCount={pageCount}
   621→                cutAfter={cutAfter}
   622→                parts={plan?.parts ?? []}
   623→                onToggleCut={toggleCut}
   624→              />
   625→            ) : pageCount === null ? (
   626→              <p className="flex items-center gap-2 px-1 py-2 text-[11px] text-muted-foreground">
   627→                <Loader2 className="size-3.5 animate-spin" aria-hidden="true" />
   628→                {t("split.detecting")}
   629→              </p>
   630→            ) : plan === null ? (
   631→              ranges.trim() === "" ? (
   632→                <p className="px-1 py-2 text-[11px] text-muted-foreground">
   633→                  {t("split.rangePrompt")}
   634→                </p>
   635→              ) : (
   636→                <p className="px-1 py-2 text-[11px] font-medium text-destructive">
   637→                  {t("split.invalidRanges")}
   638→                </p>
   639→              )
   640→            ) : plan.parts.length === 0 ? (
   641→              <p className="px-1 py-2 text-[11px] text-muted-foreground">
   642→                {t("split.noParts")}
   643→              </p>
   644→            ) : (
   645→              <CompactStrip parts={plan.parts} />
   646→            )}
   647→          </div>
   648→        </>
   649→      )}
   650→    </section>
   651→  );
   652→}

   713→/** Thumbnail picker: real page thumbs + scissors buttons between pages. */
   714→function VisualStrip({
   715→  file,
   716→  pageCount,
   717→  cutAfter,
   718→  parts,
   719→  onToggleCut,
   720→}: {
   721→  file: File;
   722→  pageCount: number;
   723→  cutAfter: number[];
   724→  parts: SplitPart[];
   725→  onToggleCut: (page: number) => void;
   726→}) {
   727→  const { t } = useI18n();
   728→  const partIndexOf = new Map<number, number>();
   729→  parts.forEach((part, partIndex) => {
   730→    for (const pageIndex of part.pages) partIndexOf.set(pageIndex, partIndex);
   731→  });
   732→
   733→  return (
   734→    <div
   735→      role="group"
   736→      aria-label={t("split.stripLabel")}
   737→      className="scrollbar-thin flex max-h-72 flex-wrap items-center gap-1 overflow-y-auto pr-1"
   738→    >
   739→      {Array.from({ length: pageCount }, (_, index) => index + 1).map(
   740→        (pageNumber) => {
   741→          const partIndex = partIndexOf.get(pageNumber - 1) ?? 0;
   742→          const cut = cutAfter.includes(pageNumber);
   743→          return (
   744→            <Fragment key={pageNumber}>
   745→              <div className="flex w-14 flex-col items-center gap-1">
   746→                <VisualThumb file={file} pageIndex={pageNumber - 1} />
   747→                <span
   748→                  className={cn(
   749→                    "rounded-md border px-1 text-[10px] font-bold",
   750→                    partStyle(partIndex)
   751→                  )}
   752→                >
   753→                  {pageNumber}
   754→                </span>
   755→              </div>
   756→              {pageNumber < pageCount ? (
   757→                <button
   758→                  type="button"
   759→                  aria-pressed={cut}
   760→                  aria-label={t("split.cutAfter", { page: pageNumber })}
   761→                  title={t("split.cutAfter", { page: pageNumber })}
   762→                  onClick={() => onToggleCut(pageNumber)}
   763→                  className={cn(
   764→                    "flex size-11 shrink-0 items-center justify-center rounded-full border outline-none transition-colors focus-visible:ring-[3px] focus-visible:ring-ring/50",
   765→                    cut
   766→                      ? "border-primary bg-primary text-primary-foreground"
   767→                      : "border-dashed border-border bg-background text-muted-foreground hover:border-primary/50 hover:text-primary"
   768→                  )}
   769→                >
   770→                  <Scissors className="size-4" aria-hidden="true" />
   771→                </button>
   772→              ) : null}
   773→            </Fragment>
   774→          );
   775→        }
   776→      )}
   777→    </div>
   778→  );
   779→}

   781→/** One queued thumbnail (render queue caps concurrency at 2). */
   782→function VisualThumb({
   783→  file,
   784→  pageIndex,
   785→}: {
   786→  file: File;
   787→  pageIndex: number;
   788→}) {
   789→  const [url, setUrl] = useState<string | null>(null);
   790→
   791→  useEffect(() => {
   792→    let cancelled = false;
   793→    let objectUrl: string | null = null;
   794→    void renderPdfThumbnail(file, { pageIndex, maxDim: 112 })
   795→      .then((result) => {
   796→        if (cancelled) return;
   797→        objectUrl = URL.createObjectURL(result.blob);
   798→        setUrl(objectUrl);
   799→      })
   800→      .catch(() => undefined);
   801→    return () => {
   802→      cancelled = true;
   803→      if (objectUrl) URL.revokeObjectURL(objectUrl);
   804→    };
   805→  }, [file, pageIndex]);
   806→
   807→  return (
   808→    <div className="relative flex aspect-[3/4] w-full items-center justify-center overflow-hidden rounded-md border border-border bg-muted/40">
   809→      {url ? (
   810→        <img
   811→          src={url}
   812→          alt=""
   813→          loading="lazy"
   814→          decoding="async"
   815→          className="size-full object-contain"
   816→        />
   817→      ) : (
   818→        <div
   819→          aria-hidden="true"
   820→          className="size-full animate-pulse bg-gradient-to-br from-muted to-muted/40"
   821→        />
   822→      )}
   823→    </div>
   824→  );
   825→}

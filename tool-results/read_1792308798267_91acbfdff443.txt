     1→     1→/**
     2→     2→ * Card maker shared model (PRD §"Card" — the 63rd tool) — PURE data +
     3→     3→ * math used by the surface (live preview), the worker (PDF/JPG/PNG
     4→     4→ * export) and the smoke script, so what you preview is exactly what
     5→     5→ * prints. No platform API is touched here; every text-layout routine
     6→     6→ * takes an injected `measure` function so the battery can verify the
     7→     7→ * math with a fake measurer.
     8→     8→ *
     9→     9→ * ── FOLD MODEL (fold-a4) ────────────────────────────────────────────
    10→    10→ * One A4 LANDSCAPE sheet (297×210 mm) per printed side, folded once at
    11→    11→ * the vertical centre → the folded card is A5-portrait-sized. The PDF
    12→    12→ * has TWO pages per card:
    13→    13→ *   Page 1 OUTSIDE — LEFT half = back cover, RIGHT half = front cover.
    14→    14→ *   Page 2 INSIDE  — LEFT half = inside-left (date/venue panel),
    15→    15→ *                    RIGHT half = inside-right (the main message panel,
    16→    16→ *                    i.e. the back of the front cover).
    17→    17→ * Printing advice given in the FAQ (honest): print page 1, re-insert
    18→    18→ * the sheet to print page 2 on the blank side in the same orientation,
    19→    19→ * then fold at the middle line — and do a TEST PRINT first. Which
    20→    20→ * duplex flip mode a printer needs (long-edge vs short-edge) varies by
    21→    21→ * model, so this tool deliberately does NOT assume one.
    22→    22→ * The fold line is exactly W/2 and both halves cover the full sheet
    23→    23→ * (smoke asserts this).
    24→    24→ *
    25→    25→ * ── TEXT-RESET RULES (standard card-maker behaviour) ─────────────────
    26→    26→ * Switching TEMPLATE resets every text field to the new template's
    27→    27→ * defaults for the CURRENT text locale (tracked when the texts were
    28→    28→ * last set — Canva-like). Switching the UI LANGUAGE never touches
    29→    29→ * typed texts; it only re-serializes the option (the output file name
    30→    30→ * is localized). Both rules live in the surface; documented here so
    31→    31→ * the model and the UI agree.
    32→    32→ */
    33→    33→
    34→    34→import type { L10n } from "@/lib/tools/registry";
    35→    35→
    36→    36→/* --------------------------- exact geometry --------------------------- */
    37→    37→
    38→    38→/** mm → PDF points, exact factor. */
    39→    39→export const PT_PER_MM = 72 / 25.4;
    40→    40→
    41→    41→/** Worker/export canvases render at 2 px per PDF point (collage policy). */
    42→    42→export const PX_PER_PT = 2;
    43→    43→
    44→    44→export type CardFormat = "flat-a6" | "flat-a5" | "flat-4x6" | "fold-a4";
    45→    45→export type CardOrientation = "portrait" | "landscape";
    46→    46→export type ExportFormat = "pdf" | "jpg" | "png";
    47→    47→
    48→    48→/** Trim sizes in mm. fold-a4 is the A4 LANDSCAPE sheet, folded once. */
    49→    49→export const FORMAT_MM: Record<CardFormat, [w: number, h: number]> = {
    50→    50→  "flat-a6": [105, 148],
    51→    51→  "flat-a5": [148, 210],
    52→    52→  "flat-4x6": [101.6, 152.4],
    53→    53→  "fold-a4": [297, 210],
    54→    54→};
    55→    55→
    56→    56→export const CARD_FORMATS = ["flat-a6", "flat-a5", "flat-4x6", "fold-a4"] as const;
    57→    57→export const CARD_ORIENTATIONS = ["portrait", "landscape"] as const;
    58→    58→export const EXPORT_FORMATS = ["pdf", "jpg", "png"] as const;
    59→    59→
    60→    60→export interface SizePt {
    61→    61→  w: number;
    62→    62→  h: number;
    63→    63→}
    64→    64→
    65→    65→/**
    66→    66→ * Page size in PDF points. fold-a4 is ALWAYS the landscape sheet
    67→    67→ * (297×210 mm) — orientation only matters for the flat formats.
    68→    68→ */
    69→    69→export function pagePtSize(
    70→    70→  format: CardFormat,
    71→    71→  orientation: CardOrientation
    72→    72→): SizePt {
    73→    73→  const [mmW, mmH] = FORMAT_MM[format];
    74→    74→  const ptW = mmW * PT_PER_MM;
    75→    75→  const ptH = mmH * PT_PER_MM;
    76→    76→  const swap = format !== "fold-a4" && orientation === "landscape";
    77→    77→  return swap ? { w: ptH, h: ptW } : { w: ptW, h: ptH };
    78→    78→}
    79→    79→
    80→    80→/** Worker/export canvas size in pixels (pt × PX_PER_PT, rounded). */
    81→    81→export function pagePixelSize(
    82→    82→  format: CardFormat,
    83→    83→  orientation: CardOrientation
    84→    84→): { w: number; h: number } {
    85→    85→  const pt = pagePtSize(format, orientation);
    86→    86→  return {
    87→    87→    w: Math.round(pt.w * PX_PER_PT),
    88→    88→    h: Math.round(pt.h * PX_PER_PT),
    89→    89→  };
    90→    90→}

    92→    92→/* ------------------------------ sheets ------------------------------ */
    93→    93→
    94→    94→export type SheetKind = "flat" | "fold-outside" | "fold-inside";
    95→    95→
    96→    96→export interface CardSheet {
    97→    97→  kind: SheetKind;
    98→    98→  /** 0-based mail-merge row this sheet belongs to. */
    99→    99→  rowIndex: number;
   100→   100→}
   101→   101→
   102→   102→/**
   103→   103→ * Sheets of one export, in print order. Flat formats make one sheet
   104→   104→ * per merge row; fold-a4 makes TWO per row (outside then inside).
   105→   105→ * N rows → flat: N sheets / fold: 2N sheets.
   106→   106→ */
   107→   107→export function computeSheets(
   108→   108→  format: CardFormat,
   109→   109→  mergeRows: number
   110→   110→): CardSheet[] {
   111→   111→  const rows = Math.max(1, Math.floor(mergeRows));
   112→   112→  const sheets: CardSheet[] = [];
   113→   113→  for (let row = 0; row < rows; row += 1) {
   114→   114→    if (format === "fold-a4") {
   115→   115→      sheets.push({ kind: "fold-outside", rowIndex: row });
   116→   116→      sheets.push({ kind: "fold-inside", rowIndex: row });
   117→   117→    } else {
   118→   118→      sheets.push({ kind: "flat", rowIndex: row });
   119→   119→    }
   120→   120→  }
   121→   121→  return sheets;
   122→   122→}
   123→   123→
   124→   124→export interface RectPt {
   125→   125→  x: number;
   126→   126→  y: number;
   127→   127→  w: number;
   128→   128→  h: number;
   129→   129→}
   130→   130→
   131→   131→/**
   132→   132→ * Panels of one printed sheet in PDF points. Folded sheets are EXACT
   133→   133→ * halves: the fold line sits at W/2 and both halves cover the full
   134→   134→ * sheet height. Panel order matches the fold model above:
   135→   135→ *   fold-outside → [back (left), front (right)]
   136→   136→ *   fold-inside  → [inside-left, inside-right]
   137→   137→ *   flat         → [whole page]
   138→   138→ */
   139→   139→export function sheetPanelsPt(
   140→   140→  format: CardFormat,
   141→   141→  orientation: CardOrientation,
   142→   142→  kind: SheetKind
   143→   143→): RectPt[] {
   144→   144→  const { w, h } = pagePtSize(format, orientation);
   145→   145→  if (kind === "flat") return [{ x: 0, y: 0, w, h }];
   146→   146→  const half = w / 2;
   147→   147→  return [
   148→   148→    { x: 0, y: 0, w: half, h },
   149→   149→    { x: half, y: 0, w: half, h },
   150→   150→  ];
   151→   151→}

   492→   492→/* --------------------------- text wrapping --------------------------- */
   493→   493→
   494→   494→/**
   495→   495→ * Word-wraps `text` to `maxWidth` using an injected measure function
   496→   496→ * (PURE — the smoke verifies it with a fake measurer).
   497→   497→ *
   498→   498→ * Rules:
   499→   499→ *  - Splits on "\n" first: an explicit newline is always a hard break.
   500→   500→ *  - Each paragraph wraps by spaces (runs of spaces collapse to one).
   501→   501→ *  - A word wider than the line breaks CHARACTER-WISE (Devanagari text
   502→   502→ *    often contains long unbroken strings); only a single character
   503→   503→ *    wider than the whole line is ever emitted over width.
   504→   504→ *  - Trailing whitespace is trimmed per emitted line.
   505→   505→ *  - Whitespace-only text → zero lines. Blank paragraphs at the very
   506→   506→ *    start/end are dropped; blank paragraphs BETWEEN text are kept as
   507→   507→ *    one empty line (a deliberate blank line).
   508→   508→ */
   509→   509→export function wrapText(
   510→   510→  text: string,
   511→   511→  maxWidth: number,
   512→   512→  measure: (t: string) => number
   513→   513→): string[] {
   514→   514→  if (typeof text !== "string" || text.trim() === "") return [];
   515→   515→  if (!(maxWidth > 0)) return [];
   516→   516→  const paragraphs = text.split("\n");
   517→   517→  let start = 0;
   518→   518→  let end = paragraphs.length - 1;
   519→   519→  while (start <= end && paragraphs[start].trim() === "") start += 1;
   520→   520→  while (end >= start && paragraphs[end].trim() === "") end -= 1;
   521→   521→  const lines: string[] = [];
   522→   522→  for (let i = start; i <= end; i += 1) {
   523→   523→    const paragraph = paragraphs[i];
   524→   524→    if (paragraph.trim() === "") {
   525→   525→      lines.push("");
   526→   526→      continue;
   527→   527→    }
   528→   528→    lines.push(...wrapParagraph(paragraph, maxWidth, measure));
   529→   529→  }
   530→   530→  return lines;
   531→   531→}
   532→   532→
   533→   533→function wrapParagraph(
   534→   534→  paragraph: string,
   535→   535→  maxWidth: number,
   536→   536→  measure: (t: string) => number
   537→   537→): string[] {
   538→   538→  const words = paragraph.split(" ").filter((word) => word !== "");
   539→   539→  const lines: string[] = [];
   540→   540→  let line = "";
   541→   541→  for (const word of words) {
   542→   542→    const candidate = line === "" ? word : `${line} ${word}`;
   543→   543→    if (measure(candidate) <= maxWidth) {
   544→   544→      line = candidate;
   545→   545→      continue;
   546→   546→    }
   547→   547→    if (line !== "") {
   548→   548→      lines.push(line);
   549→   549→      line = "";
   550→   550→    }
   551→   551→    if (measure(word) <= maxWidth) {
   552→   552→      line = word;
   553→   553→      continue;
   554→   554→    }
   555→   555→    // Character-break the over-long word; every chunk except the last
   556→   556→    // becomes its own (full) line, the last stays open for more words.
   557→   557→    const chunks = breakWord(word, maxWidth, measure);
   558→   558→    for (let c = 0; c < chunks.length - 1; c += 1) lines.push(chunks[c]);
   559→   559→    line = chunks[chunks.length - 1] ?? "";
   560→   560→  }
   561→   561→  if (line !== "") lines.push(line);
   562→   562→  return lines;
   563→   563→}
   564→   564→
   565→   565→function breakWord(
   566→   566→  word: string,
   567→   567→  maxWidth: number,
   568→   568→  measure: (t: string) => number
   569→   569→): string[] {
   570→   570→  const chunks: string[] = [];
   571→   571→  let chunk = "";
   572→   572→  for (const char of [...word]) {
   573→   573→    const next = chunk + char;
   574→   574→    if (chunk !== "" && measure(next) > maxWidth) {
   575→   575→      chunks.push(chunk);
   576→   576→      chunk = char;
   577→   577→    } else {
   578→   578→      chunk = next;
   579→   579→    }
   580→   580→  }
   581→   581→  if (chunk !== "") chunks.push(chunk);
   582→   582→  return chunks;
   583→   583→}
   584→   584→
   585→   585→/**
   586→   586→ * Longest prefix of `text` + "…" that fits `maxWidth` (binary search).
   587→   587→ * "" → ""; nothing fits → the bare ellipsis.
   588→   588→ */
   589→   589→export function truncateWithEllipsis(
   590→   590→  text: string,
   591→   591→  maxWidth: number,
   592→   592→  measure: (t: string) => number
   593→   593→): string {
   594→   594→  if (text === "") return "";
   595→   595→  if (measure(`${text}\u2026`) <= maxWidth) return `${text}\u2026`;
   596→   596→  const chars = [...text];
   597→   597→  let lo = 0;
   598→   598→  let hi = chars.length;
   599→   599→  while (lo < hi) {
   600→   600→    const mid = Math.ceil((lo + hi) / 2);
   601→   601→    if (measure(`${chars.slice(0, mid).join("")}\u2026`) <= maxWidth) lo = mid;
   602→   602→    else hi = mid - 1;
   603→   603→  }
   604→   604→  return `${chars.slice(0, lo).join("")}\u2026`;
   605→   605→}

   609→   609→export interface HeadlineFit {
   610→   610→  /** The chosen font size in pt. */
   611→   611→  sizePt: number;
   612→   612→  /** Wrapped lines at that size (ALL of them — never truncated). */
   613→   613→  lines: string[];
   614→   614→  /** True when even the minimum size does not fit. */
   615→   615→  overflowed: boolean;
   616→   616→}
   617→   617→
   618→   618→export interface HeadlineOptions {
   619→   619→  baseSizePt: number;
   620→   620→  minSizePt: number;
   621→   621→  /** Deterministic step-down (default 2). */
   622→   622→  stepPt: number;
   623→   623→  maxWidthPt: number;
   624→   624→  maxLines: number;
   625→   625→  lineHeightRatio: number;
   626→   626→  maxHeightPt: number;
   627→   627→}
   628→   628→
   629→   629→/**
   630→   630→ * Headline shrink-to-fit: start at the template's base size, step down
   631→   631→ * by `stepPt` (−2pt) until the wrapped headline fits the panel within
   632→   632→ * the allowed line count and height budget. Returns the FIRST size
   633→   633→ * that fits. If nothing fits down to the minimum, keeps the minimum
   634→   634→ * size with its full wrap (never truncates silently) and flags
   635→   635→ * `overflowed`. The loop is bounded by an explicit iteration cap, so
   636→   636→ * it always terminates (smoke verifies with a fake measurer).
   637→   637→ */
   638→   638→export function fitHeadline(
   639→   639→  text: string,
   640→   640→  opts: HeadlineOptions,
   641→   641→  measureAt: (t: string, sizePt: number) => number
   642→   642→): HeadlineFit {
   643→   643→  const step = opts.stepPt > 0 ? opts.stepPt : 2;
   644→   644→  const min = Math.max(1, opts.minSizePt);
   645→   645→  const base = Math.max(min, opts.baseSizePt);
   646→   646→  if (typeof text !== "string" || text.trim() === "") {
   647→   647→    return { sizePt: base, lines: [], overflowed: false };
   648→   648→  }
   649→   649→  const ITERATION_CAP = 512; // (base−min)/step + 1 in practice — hard cap.
   650→   650→  let size = base;
   651→   651→  for (let i = 0; i < ITERATION_CAP; i += 1) {
   652→   652→    const lines = wrapText(text, opts.maxWidthPt, (t) => measureAt(t, size));
   653→   653→    const fits =
   654→   654→      lines.length <= opts.maxLines &&
   655→   655→      lines.length * size * opts.lineHeightRatio <= opts.maxHeightPt;
   656→   656→    if (fits) return { sizePt: size, lines, overflowed: false };
   657→   657→    if (size <= min) return { sizePt: min, lines, overflowed: true };
   658→   658→    size = Math.max(min, size - step);
   659→   659→  }
   660→   660→  const lines = wrapText(text, opts.maxWidthPt, (t) => measureAt(t, min));
   661→   661→  return { sizePt: min, lines, overflowed: true };
   662→   662→}

   666→   666→export interface MessageLayout {
   667→   667→  lines: string[];
   668→   668→  overflowed: boolean;
   669→   669→}
   670→   670→
   671→   671→export interface MessageOptions {
   672→   672→  fontPt: number;
   673→   673→  lineHeightRatio: number;
   674→   674→  maxWidthPt: number;
   675→   675→  maxHeightPt: number;
   676→   676→}
   677→   677→
   678→   678→/**
   679→   679→ * Message body at a FIXED size. Wraps freely; when the measured block
   680→   680→ * overflows the panel bottom, the last fitting line gets an ellipsis
   681→   681→ * "…" and `overflowed` flags it (the surface shows an honest badge
   682→   682→ * from the SAME engine's metrics).
   683→   683→ */
   684→   684→export function layoutMessage(
   685→   685→  text: string,
   686→   686→  opts: MessageOptions,
   687→   687→  measureAt: (t: string, sizePt: number) => number
   688→   688→): MessageLayout {
   689→   689→  if (typeof text !== "string" || text.trim() === "") {
   690→   690→    return { lines: [], overflowed: false };
   691→   691→  }
   692→   692→  const measure = (t: string): number => measureAt(t, opts.fontPt);
   693→   693→  const lines = wrapText(text, opts.maxWidthPt, measure);
   694→   694→  const lineH = opts.fontPt * opts.lineHeightRatio;
   695→   695→  const maxLines = Math.max(1, Math.floor(opts.maxHeightPt / lineH));
   696→   696→  if (lines.length <= maxLines) return { lines, overflowed: false };
   697→   697→  const kept = lines.slice(0, maxLines);
   698→   698→  kept[maxLines - 1] = truncateWithEllipsis(
   699→   699→    kept[maxLines - 1] ?? "",
   700→   700→    opts.maxWidthPt,
   701→   701→    measure
   702→   702→  );
   703→   703→  return { lines: kept, overflowed: true };
   704→   704→}

   706→   706→/* ------------------------------ CSV merge ------------------------------ */
   707→   707→
   708→   708→export const MAX_CSV_ROWS = 200;
   709→   709→/**
   710→   710→ * 2 MB CSV file cap, measured in characters (a UTF-8 file never has
   711→   711→ * more chars than bytes, so chars > cap ⇔ file > 2 MB for valid UTF-8).
   712→   712→ */
   713→   713→export const CSV_MAX_CHARS = 2 * 1024 * 1024;
   714→   714→
   715→   715→export class CsvError extends Error {
   716→   716→  /** i18n error key: card.errorCsvQuote | card.errorCsvRows | card.errorCsvSize. */
   717→   717→  code: string;
   718→   718→  constructor(code: string) {
   719→   719→    super(code);
   720→   720→    this.name = "CsvError";
   721→   721→    this.code = code;
   722→   722→  }
   723→   723→}
   724→   724→
   725→   725→export interface ParsedCsv {
   726→   726→  /** First row, EXACTLY as written — case-sensitive, NOT trimmed. */
   727→   727→  columns: string[];
   728→   728→  rows: string[][];
   729→   729→}
   730→   730→
   731→   731→/**
   732→   732→ * RFC-4180 CSV parser (PURE):
   733→   733→ *  - quoted fields with embedded commas and newlines; `""` escapes a
   734→   734→ *    literal quote;
   735→   735→ *  - CRLF, LF and lone CR line endings all work;
   736→   736→ *  - a leading UTF-8 BOM is stripped;
   737→   737→ *  - the first record is the header (column names preserved exactly);
   738→   738→ *  - a SINGLE trailing empty line is skipped (more are kept);
   739→   739→ *  - unterminated quote → CsvError("card.errorCsvQuote");
   740→   740→ *  - more than MAX_CSV_ROWS data rows → CsvError("card.errorCsvRows");
   741→   741→ *  - text larger than the 2 MB cap → CsvError("card.errorCsvSize").
   742→   742→ * Header-only / empty input → zero rows = NO merge (single card).
   743→   743→ */
   744→   744→export function parseCsv(text: string): ParsedCsv {
   745→   745→  if (typeof text !== "string") text = "";
   746→   746→  if (text.startsWith("\uFEFF")) text = text.slice(1);
   747→   747→  if (text.length > CSV_MAX_CHARS) {
   748→   748→    throw new CsvError("card.errorCsvSize");
   749→   749→  }
   750→   750→  const records: string[][] = [];
   751→   751→  let field = "";
   752→   752→  let record: string[] = [];
   753→   753→  let inQuotes = false;
   754→   754→  for (let i = 0; i < text.length; i += 1) {
   755→   755→    const char = text[i];
   756→   756→    if (inQuotes) {
   757→   757→      if (char === '"') {
   758→   758→        if (text[i + 1] === '"') {
   759→   759→          field += '"';
   760→   760→          i += 1;
   761→   761→        } else {
   762→   762→          inQuotes = false;
   763→   763→        }
   764→   764→      } else {
   765→   765→        field += char;
   766→   766→      }
   767→   767→      continue;
   768→   768→    }
   769→   769→    if (char === '"' && field === "") {
   770→   770→      inQuotes = true;
   771→   771→      continue;
   772→   772→    }
   773→   773→    if (char === ",") {
   774→   774→      record.push(field);
   775→   775→      field = "";
   776→   776→      continue;
   777→   777→    }
   778→   778→    if (char === "\n" || char === "\r") {
   779→   779→      if (char === "\r" && text[i + 1] === "\n") i += 1;
   780→   780→      record.push(field);
   781→   781→      field = "";
   782→   782→      records.push(record);
   783→   783→      record = [];
   784→   784→      continue;
   785→   785→    }
   786→   786→    field += char;
   787→   787→  }
   788→   788→  if (inQuotes) throw new CsvError("card.errorCsvQuote");
   789→   789→  if (field !== "" || record.length > 0) {
   790→   790→    record.push(field);
   791→   791→    records.push(record);
   792→   792→  }
   793→   793→  // Skip a single trailing empty line (an empty record with one field).
   794→   794→  if (
   795→   795→    records.length > 0 &&
   796→   796→    records[records.length - 1].length === 1 &&
   797→   797→    records[records.length - 1][0] === ""
   798→   798→  ) {
   799→   799→    records.pop();
   800→   800→  }
   801→   801→  const columns = records[0] ?? [];
   802→   802→  const rows = records.slice(1);
   803→   803→  if (rows.length > MAX_CSV_ROWS) {
   804→   804→    throw new CsvError("card.errorCsvRows");
   805→   805→  }
   806→   806→  return { columns, rows };
   807→   807→}
   808→   808→
   809→   809→/**
   810→   810→ * Replaces `{{column}}` placeholders with row values. Exact column
   811→   811→ * name match (case-sensitive). Unknown placeholders and non-matching
   812→   812→ * forms (`{{}}`, `{{ name}}`) are left untouched — they render
   813→   813→ * literally and never crash. Missing row fields become "".
   814→   814→ */
   815→   815→export function mergeText(text: string, row: string[], columns: string[]): string {
   816→   816→  return text.replace(/\{\{([^{}\s]+)\}\}/g, (whole, name: string) => {
   817→   817→    const index = columns.indexOf(name);
   818→   818→    if (index === -1) return whole;
   819→   819→    return row[index] ?? "";
   820→   820→  });
   821→   821→}

   823→   823→/** The effective texts of one merge row (or the base texts). */
   824→   824→export function mergedTextsFor(
   825→   825→  texts: CardTexts,
   826→   826→  row: string[] | null,
   827→   827→  columns: string[]
   828→   828→): CardTexts {
   829→   829→  if (row === null) return texts;
   830→   830→  const merged = {} as CardTexts;
   831→   831→  for (const field of TEXT_FIELD_ORDER) {
   832→   832→    merged[field] = mergeText(texts[field], row, columns);
   833→   833→  }
   834→   834→  return merged;
   835→   835→}

   837→   837→/**
   838→   838→ * Collects ALL unknown `{{column}}` names across every text field, in
   839→   839→ * first-appearance order (field order is TEXT_FIELD_ORDER — stable).
   840→   840→ * Empty result = everything merges cleanly.
   841→   841→ */
   842→   842→export function validatePlaceholders(
   843→   843→  texts: CardTexts,
   844→   844→  columns: string[]
   845→   845→): string[] {
   846→   846→  const unknown: string[] = [];
   847→   847→  const seen = new Set<string>();
   848→   848→  const re = /\{\{([^{}\s]+)\}\}/g;
   849→   849→  for (const field of TEXT_FIELD_ORDER) {
   850→   850→    const value = texts[field];
   851→   851→    if (typeof value !== "string") continue;
   852→   852→    for (const match of value.matchAll(re)) {
   853→   853→      const name = match[1];
   854→   854→      if (!columns.includes(name) && !seen.has(name)) {
   855→   855→        seen.add(name);
   856→   856→        unknown.push(name);
   857→   857→      }
   858→   858→    }
   859→   859→  }
   860→   860→  return unknown;
   861→   861→}

   896→   896→  if (typeof value !== "string") return "";

   898→   898→    .replace(/[\u0000-\u0009\u000b-\u001f]/g, "")
   899→   899→    .slice(0, max)

   919→   919→function pickString<T extends string>(
   920→   920→  value: unknown,
   921→   921→  allowed: readonly T[],
   922→   922→  fallback: T
   923→   923→): T {
   924→   924→  return typeof value === "string" && (allowed as readonly string[]).includes(value)
   925→   925→    ? (value as T)
   926→   926→    : fallback;

   929→   929→/**
   930→   930→ * Parses the "card" option. Garbage JSON → safe defaults; every number
   931→   931→ * is clamped (photoIndex into the file range), every enum falls back,
   932→   932→ * the locale falls back to "en". Structurally invalid data never kills
   933→   933→ * a run — the card always draws.
   934→   934→ */
   935→   935→export function parseCardOption(value: unknown, fileCount: number): CardOption {
   936→   936→  let parsed: unknown = null;
   937→   937→  if (typeof value === "string" && value !== "") {
   938→   938→    try {
   939→   939→      parsed = JSON.parse(value);
   940→   940→    } catch {
   941→   941→      parsed = null;
   942→   942→    }
   943→   943→  }
   944→   944→  const raw = (typeof parsed === "object" && parsed !== null ? parsed : {}) as Record<
   945→   945→    string,
   946→   946→    unknown
   947→   947→  >;
   948→   948→  let photoIndex: number | null = null;
   949→   949→  if (
   950→   950→    typeof raw.photoIndex === "number" &&
   951→   951→    Number.isInteger(raw.photoIndex) &&
   952→   952→    fileCount > 0
   953→   953→  ) {
   954→   954→    // Clamp into [0, fileCount): out-of-range indexes land on the
   955→   955→    // nearest photo, never crash the render.
   956→   956→    photoIndex = Math.min(Math.max(raw.photoIndex, 0), fileCount - 1);
   957→   957→  }
   958→   958→  return {
   959→   959→    template: isTemplateId(String(raw.template))
   960→   960→      ? String(raw.template)
   961→   961→      : DEFAULT_TEMPLATE_ID,
   962→   962→    format: pickString(raw.format, CARD_FORMATS, "flat-a6"),
   963→   963→    orientation: pickString(raw.orientation, CARD_ORIENTATIONS, "portrait"),
   964→   964→    texts: sanitizeTexts(raw.texts),
   965→   965→    photoIndex,
   966→   966→    background: sanitizeHexColor(raw.background, "#ffffff"),
   967→   967→    csvText: typeof raw.csvText === "string" ? raw.csvText.slice(0, CSV_MAX_CHARS) : "",
   968→   968→    csvName:
   969→   969→      typeof raw.csvName === "string"
   970→   970→        ? raw.csvName.replace(/[\u0000-\u001f]/g, "").slice(0, 120)
   971→   971→        : "",
   972→   972→    exportFormat: pickString(raw.exportFormat, EXPORT_FORMATS, "pdf"),
   973→   973→    locale: pickString(raw.locale, ["en", "hi"] as const, "en"),
   974→   974→  };
   975→   975→}

   979→   979→export interface CropRect {
   980→   980→  sx: number;
   981→   981→  sy: number;
   982→   982→  sw: number;
   983→   983→  sh: number;
   984→   984→}

   986→   986→/**
   987→   987→ * Cover-crop: the largest centred source rect whose aspect matches the
   988→   988→ * destination (collage's exact math — reused so backgrounds and
   989→   989→ * photos crop identically).
   990→   990→ */
   991→   991→export function coverCrop(sw: number, sh: number, dw: number, dh: number): CropRect {
   992→   992→  if (sw <= 0 || sh <= 0 || dw <= 0 || dh <= 0) return { sx: 0, sy: 0, sw: 0, sh: 0 };
   993→   993→  const scale = Math.max(dw / sw, dh / sh);
   994→   994→  const cropW = dw / scale;
   995→   995→  const cropH = dh / scale;
   996→   996→  return {
   997→   997→    sx: (sw - cropW) / 2,
   998→   998→    sy: (sh - cropH) / 2,
   999→   999→    sw: cropW,
  1000→  1000→    sh: cropH,
  1001→  1001→  };
  1002→  1002→}

     1→/**
     2→ * OpenDocument to PDF renderer (RB2-b — the 72nd tool) — the vector
     3→ * page painter.
     4→ *
     5→ * Extracted so the worker and the smoke script drive the EXACT drawing
     6→ * code headless in bun (create-resume precedent): same geometry, same
     7→ * wrap math, same pagination.
     8→ *
     9→ *  - Text kind (word-to-pdf lineage):
    10→ *      A4 595.28×841.89 (or US Letter 612×792), 56 pt margins (28 pt
    11→ *      narrow). Body 10.5 pt / 15 pt leading. Headings 1-6 = 16/13/11.5/
    12→ *      10.5 pt — bold when a bold face exists (base-14 HelveticaBold),
    13→ *      size-only hierarchy with the Noto Devanagari face (Regular only).
    14→ *      List items: bullet glyphs "•" / "–" / "·" (all WinAnsi AND present
    15→ *      in the Noto face — verified) with hanging indents (14 pt/level).
    16→ *      Tables: measured column widths, hairline 0.5 pt borders, wrapped
    17→ *      top-aligned cells, row-wise pagination.
    18→ *  - Sheets kind (excel-to-pdf lineage): each sheet starts a fresh
    19→ *      page with its name as a 12 pt header + hairline rule; column
    20→ *      widths from a content heuristic, fit-scaled to the printable
    21→ *      width; wrapped cells with honest "…" clipping when one row would
    22→ *      outgrow a page; hairline grid, row-wise pagination.
    23→ *
    24→ * Fonts arrive already embedded (worker): `font` is either the
    25→ * non-embedded base-14 Helvetica (pure-WinAnsi docs) or the Noto
    26→ * Devanagari subset; `boldFont` may be null — bold then falls back to
    27→ * the regular face. Every glyph is drawText — REAL selectable text;
    28→ * images/drawings from the source document are NEVER drawn.
    29→ */

   100→class Painter {
   101→  readonly pdfDoc: PDFDocument;
   102→  readonly font: PDFFont;
   103→  readonly boldFont: PDFFont | null;
   104→  readonly pages: PDFPage[] = [];
   105→  readonly pageW: number;
   106→  readonly pageH: number;
   107→  readonly margin: number;
   108→  readonly contentW: number;
   109→  page: PDFPage;
   110→  y!: number;
   111→
   112→  constructor(
   113→    pdfDoc: PDFDocument,
   114→    font: PDFFont,
   115→    boldFont: PDFFont | null,
   116→    pageSize: "a4" | "letter",
   117→    marginKind: "normal" | "narrow"
   118→  ) {
   119→    this.pdfDoc = pdfDoc;
   120→    this.font = font;
   121→    this.boldFont = boldFont;
   122→    const [w, h] = PAGE_SIZES[pageSize];
   123→    this.pageW = w;
   124→    this.pageH = h;
   125→    this.margin = MARGIN_PT[marginKind];
   126→    this.contentW = w - this.margin * 2;
   127→    this.page = this.newPage();
   128→  }
   129→
   130→  newPage(): PDFPage {
   131→    this.page = this.pdfDoc.addPage([this.pageW, this.pageH]);
   132→    this.pages.push(this.page);
   133→    this.y = this.pageH - this.margin;
   134→    return this.page;
   135→  }
   136→
   137→  /** Page-break unless `need` points fit above the bottom margin. */
   138→  ensure(need: number): void {
   139→    if (this.y - need < this.margin) this.newPage();
   140→  }
   141→
   142→  fontFor(bold: boolean): PDFFont {
   143→    return bold && this.boldFont ? this.boldFont : this.font;
   144→  }
   145→
   146→  width(text: string, size: number, bold: boolean): number {
   147→    if (!text) return 0;
   148→    return this.fontFor(bold).widthOfTextAtSize(text, size);
   149→  }
   150→
   151→  /** Baseline-aligned draw at (x, yTop) — the house text() convention. */
   152→  text(
   153→    s: string,
   154→    x: number,
   155→    yTop: number,
   156→    size: number,
   157→    bold: boolean,
   158→    baseline: number
   159→  ): void {
   160→    if (!s) return;
   161→    this.page.drawText(s, {
   162→      x,
   163→      y: yTop - baseline,
   164→      size,
   165→      font: this.fontFor(bold),
   166→      color: COL_TEXT,
   167→    });
   168→  }
   169→}

   175→interface Seg {
   176→  text: string;
   177→  bold: boolean;
   178→  size: number;
   179→}
   180→
   181→interface Line {
   182→  segs: Seg[];
   183→  width: number;
   184→  lh: number;
   185→  /** baseline offset = the largest seg size in the line. */
   186→  maxSeg: number;
   187→}
   188→
   189→/**
   190→ * Word items derived from the runs' MERGED text — whitespace is the
   191→ * only true word boundary; each word carries its per-segment styles;
   192→ * "\n" (from text:line-break) is a hard break item.
   193→ */
   194→type Item = { kind: "word"; segs: Seg[] } | { kind: "break" };
   195→
   196→function paraItems(runs: OdfRun[], baseSize: number): Item[] {
   197→  let merged = "";
   198→  const bolds: boolean[] = [];
   199→  const sizes: number[] = [];
   200→  for (const r of runs) {
   201→    const size = r.sizePt ?? baseSize;
   202→    for (const ch of r.text) {
   203→      merged += ch;
   204→      bolds.push(r.bold);
   205→      sizes.push(size);
   206→    }
   207→  }
   208→  const items: Item[] = [];
   209→  const n = merged.length;
   210→  let i = 0;
   211→  while (i < n) {
   212→    const ch = merged[i];
   213→    if (ch === "\n") {
   214→      items.push({ kind: "break" });
   215→      i += 1;
   216→      continue;
   217→    }
   218→    if (ch === " " || ch === "\t") {
   219→      i += 1;
   220→      continue;
   221→    }
   222→    const start = i;
   223→    while (i < n) {
   224→      const c = merged[i];
   225→      if (c === " " || c === "\t" || c === "\n") break;
   226→      i += 1;
   227→    }
   228→    const segs: Seg[] = [];
   229→    let s = start;
   230→    while (s < i) {
   231→      let e = s + 1;
   232→      while (e < i && bolds[e] === bolds[s] && sizes[e] === sizes[s]) e += 1;
   233→      segs.push({ text: merged.slice(s, e), bold: bolds[s], size: sizes[s] });
   234→      s = e;
   235→    }
   236→    items.push({ kind: "word", segs });
   237→  }
   238→  return items;
   239→}
   240→
   241→const segW = (p: Painter, seg: Seg): number =>
   242→  p.width(seg.text, seg.size, seg.bold);
   243→const wordW = (p: Painter, segs: Seg[]): number =>
   244→  segs.reduce((sum, seg) => sum + segW(p, seg), 0);
   245→
   246→/** Over-wide word → char-level pieces that each fit (styles preserved). */
   247→function hardBreak(p: Painter, segs: Seg[], width: number): Seg[][] {
   248→  const pieces: Seg[][] = [];
   249→  let cur: Seg[] = [];
   250→  let curW = 0;
   251→  for (const seg of segs) {
   252→    const text = seg.text;
   253→    let off = 0;
   254→    while (off < text.length) {
   255→      let take = text.length - off;
   256→      while (
   257→        take > 1 &&
   258→        curW + p.width(text.slice(off, off + take), seg.size, seg.bold) > width
   259→      ) {
   260→        take -= 1;
   261→      }
   262→      const piece = text.slice(off, off + take);
   263→      const pieceW = p.width(piece, seg.size, seg.bold);
   264→      if (curW > 0 && curW + pieceW > width) {
   265→        pieces.push(cur);
   266→        cur = [];
   267→        curW = 0;
   268→        continue; // retry the same piece on an empty line
   269→      }
   270→      const last = cur[cur.length - 1];
   271→      if (last && last.bold === seg.bold && last.size === seg.size) {
   272→        last.text += piece;
   273→      } else {
   274→        cur.push({ text: piece, bold: seg.bold, size: seg.size });
   275→      }
   276→      curW += pieceW;
   277→      off += take;
   278→    }
   279→  }
   280→  if (cur.length > 0) pieces.push(cur);
   281→  return pieces;
   282→}

   284→/** Greedy fill of styled words; always returns at least one line. */
   285→function layoutLines(
   286→  p: Painter,
   287→  runs: OdfRun[],
   288→  width: number,
   289→  baseSize: number,
   290→  baseLH: number
   291→): Line[] {
   292→  const items = paraItems(runs, baseSize);
   293→  const lines: Line[] = [];
   294→  let segs: Seg[] = [];
   295→  let lineW = 0;
   296→  let maxSeg = 0;
   297→
   298→  const flush = (): void => {
   299→    lines.push({
   300→      segs,
   301→      width: lineW,
   302→      lh: Math.max(baseLH, maxSeg * 1.3),
   303→      maxSeg: Math.max(maxSeg, 1),
   304→    });
   305→    segs = [];
   306→    lineW = 0;
   307→    maxSeg = 0;
   308→  };
   309→
   310→  const addWord = (segsIn: Seg[], spaceBefore: boolean): void => {
   311→    const w = wordW(p, segsIn);
   312→    const spaceW = spaceBefore ? p.width(" ", segsIn[0].size, segsIn[0].bold) : 0;
   313→    if (w + spaceW > width && width > 0) {
   314→      // pathological single word — hard-break it first (pieces join
   315→      // without spaces; only the first can carry the preceding space)
   316→      const pieces = hardBreak(p, segsIn, width);
   317→      pieces.forEach((piece, idx) => {
   318→        addWord(piece, idx === 0 ? spaceBefore && lineW > 0 : false);
   319→      });
   320→      return;
   321→    }
   322→    if (lineW > 0 && lineW + spaceW + w > width) flush();
   323→    const needSpace = lineW > 0 && spaceBefore;
   324→    let first = true;
   325→    for (const seg of segsIn) {
   326→      if (first && needSpace) {
   327→        const last = segs[segs.length - 1];
   328→        if (last && last.bold === seg.bold && last.size === seg.size) {
   329→          last.text += " " + seg.text;
   330→        } else {
   331→          segs.push({ text: " " + seg.text, bold: seg.bold, size: seg.size });
   332→        }
   333→      } else {
   334→        const last = segs[segs.length - 1];
   335→        if (!first && last && last.bold === seg.bold && last.size === seg.size) {
   336→          last.text += seg.text;
   337→        } else {
   338→          segs.push({ text: seg.text, bold: seg.bold, size: seg.size });
   339→        }
   340→      }
   341→      first = false;
   342→    }
   343→    lineW += spaceW + w;
   344→    for (const seg of segsIn) maxSeg = Math.max(maxSeg, seg.size);
   345→  };
   346→
   347→  let haveWord = false;
   348→  for (const item of items) {
   349→    if (item.kind === "break") {
   350→      flush();
   351→      continue;
   352→    }
   353→    addWord(item.segs, haveWord);
   354→    haveWord = true;
   355→  }
   356→  if (segs.length > 0 || lines.length === 0) flush();
   357→  return lines;
   358→}

   360→/** Plain (unstyled) wrap for table cells. */
   361→function layoutPlain(p: Painter, text: string, width: number): string[] {
   362→  const out: string[] = [];
   363→  for (const para of text.split("\n")) {
   364→    const words = para.split(/\s+/).filter(Boolean);
   365→    if (words.length === 0) {
   366→      out.push("");
   367→      continue;
   368→    }
   369→    let line = "";
   370→    let lineW = 0;
   371→    for (const word of words) {
   372→      let piece = word;
   373→      // hard-break pathological single words
   374→      while (p.width(piece, TABLE_CELL_SIZE, false) > width && piece.length > 1) {
   375→        let cut = piece.length;
   376→        while (
   377→          cut > 1 &&
   378→          p.width(piece.slice(0, cut - 1), TABLE_CELL_SIZE, false) > width
   379→        ) {
   380→          cut -= 1;
   381→        }
   382→        out.push((line ? line + " " : "") + piece.slice(0, Math.max(1, cut - 1)));
   383→        line = "";
   384→        lineW = 0;
   385→        piece = piece.slice(Math.max(1, cut - 1));
   386→      }
   387→      const w = p.width(piece, TABLE_CELL_SIZE, false);
   388→      const space = lineW > 0 ? p.width(" ", TABLE_CELL_SIZE, false) : 0;
   389→      if (lineW > 0 && lineW + space + w > width) {
   390→        out.push(line);
   391→        line = piece;
   392→        lineW = w;
   393→      } else {
   394→        line = line ? line + " " + piece : piece;
   395→        lineW += space + w;
   396→      }
   397→    }
   398→    if (line) out.push(line);
   399→  }
   400→  return out;
   401→}

   407→function drawHeadingOrPara(
   408→  p: Painter,
   409→  blk: OdfHeadingBlock | OdfParaBlock | OdfListBlock
   410→): void {
   411→  const isHeading = blk.kind === "heading";
   412→  const h = isHeading ? HEADINGS[blk.level] : undefined;
   413→  const baseSize = h ? h.size : BODY_SIZE;
   414→  const baseLH = h ? h.lh : BODY_LH;
   415→  const isList = blk.kind === "list-item";
   416→  const level = Math.min(3, blk.kind === "list-item" ? blk.level : 0);
   417→
   418→  if (blk.runs.length === 0 && !isList) {
   419→    // ODT's blank spacing paragraph — one honest empty line.
   420→    p.y -= baseLH;
   421→    return;
   422→  }
   423→
   424→  const bulletX = p.margin + 4 + BULLET_STEP * level;
   425→  const textX = isList ? bulletX + BULLET_GAP : p.margin;
   426→  const avail = Math.max(MIN_LINE, p.margin + p.contentW - textX);
   427→
   428→  if (h) {
   429→    // keep the heading with at least its first content line
   430→    p.ensure(h.lh + h.after + baseLH);
   431→    p.y -= h.before;
   432→  }
   433→
   434→  const lines = layoutLines(p, blk.runs, avail, baseSize, baseLH);
   435→  let first = true;
   436→  for (const ln of lines) {
   437→    p.ensure(ln.lh);
   438→    if (isList && first) {
   439→      p.text(BULLETS[level] + " ", bulletX, p.y, baseSize, false, ln.maxSeg);
   440→    }
   441→    let cx = textX;
   442→    for (const seg of ln.segs) {
   443→      if (seg.text) {
   444→        p.text(seg.text, cx, p.y, seg.size, seg.bold || isHeading, ln.maxSeg);
   445→      }
   446→      cx += p.width(seg.text, seg.size, seg.bold);
   447→    }
   448→    p.y -= ln.lh;
   449→    first = false;
   450→  }
   451→  p.y -= h ? h.after : isList ? BULLET_AFTER : PARA_AFTER;
   452→}

   458→function columnWidths(p: Painter, rows: string[][], ncols: number): number[] {
   459→  const natural = new Array<number>(ncols).fill(0);
   460→  const minWord = new Array<number>(ncols).fill(TABLE_MIN_COL);
   461→  for (const row of rows) {
   462→    for (let c = 0; c < ncols; c++) {
   463→      const cell = row[c] ?? "";
   464→      const full = p.width(cell, TABLE_CELL_SIZE, false);
   465→      if (full > natural[c]) natural[c] = Math.min(full, p.contentW);
   466→      for (const word of cell.split(/\s+/)) {
   467→        const w = p.width(word, TABLE_CELL_SIZE, false) + TABLE_PAD * 2;
   468→        if (w > minWord[c]) minWord[c] = Math.min(w, p.contentW);
   469→      }
   470→    }
   471→  }
   472→  const ideal = natural.map((n, c) =>
   473→    Math.max(Math.min(n, p.contentW), minWord[c], TABLE_MIN_COL)
   474→  );
   475→  const sum = ideal.reduce((a, b) => a + b, 0);
   476→  if (sum > p.contentW) {
   477→    const scale = p.contentW / sum;
   478→    return ideal.map((w) => Math.max(w * scale, 24));
   479→  }
   480→  if (sum < p.contentW) {
   481→    const extra = (p.contentW - sum) / ncols;
   482→    return ideal.map((w) => w + extra);
   483→  }
   484→  return ideal;
   485→}

   487→function drawTextTable(p: Painter, blk: OdfTableBlock): void {
   488→  const rows = blk.rows;
   489→  if (rows.length === 0) return;
   490→  const ncols = Math.max(1, ...rows.map((r) => r.length));
   491→  const widths = columnWidths(p, rows, ncols);
   492→
   493→  // keep the first row + one more line together on a page
   494→  let firstRowLines = 1;
   495→  rows[0].forEach((cell, c) => {
   496→    firstRowLines = Math.max(
   497→      firstRowLines,
   498→      layoutPlain(p, cell, widths[c] - TABLE_PAD * 2).length
   499→    );
   500→  });
   501→  p.ensure(firstRowLines * TABLE_CELL_LH + TABLE_PAD * 2 + TABLE_CELL_LH);
   502→
   503→  for (const row of rows) {
   504→    const cellLines = row.map(
   505→      (cell, c) => layoutPlain(p, cell, widths[c] - TABLE_PAD * 2).slice(0, 40)
   506→    );
   507→    let rowLines = 1;
   508→    for (const lines of cellLines) rowLines = Math.max(rowLines, lines.length);
   509→    const rowH = rowLines * TABLE_CELL_LH + TABLE_PAD * 2;
   510→    p.ensure(rowH);
   511→    let x = p.margin;
   512→    for (let c = 0; c < ncols; c++) {
   513→      const w = widths[c];
   514→      p.page.drawRectangle({
   515→        x,
   516→        y: p.y - rowH,
   517→        width: w,
   518→        height: rowH,
   519→        borderColor: COL_BORDER,
   520→        borderWidth: 0.5,
   521→      });
   522→      const lines = cellLines[c] ?? [];
   523→      let ty = p.y - TABLE_PAD;
   524→      for (const line of lines) {
   525→        if (line) {
   526→          p.text(line, x + TABLE_PAD, ty, TABLE_CELL_SIZE, false, TABLE_CELL_SIZE);
   527→        }
   528→        ty -= TABLE_CELL_LH;
   529→      }
   530→      x += w;
   531→    }
   532→    p.y -= rowH;
   533→  }
   534→  p.y -= TABLE_AFTER;
   535→}

   550→/** Natural column widths in pt from content length (excel heuristic). */
   551→function sheetColumnWidths(grid: string[][]): number[] {
   552→  const ncols = grid.reduce((m, row) => Math.max(m, row.length), 0);
   553→  const widths = new Array<number>(ncols).fill(SHEET_MIN_COL_PT);
   554→  for (const row of grid) {
   555→    for (let c = 0; c < row.length && c < ncols; c++) {
   556→      const len = row[c]?.length ?? 0;
   557→      // only grow for the first 48 chars per column — long cells wrap
   558→      const eff = Math.min(len, 48);
   559→      const w = 10 + eff * 5.6;
   560→      if (w > widths[c]) widths[c] = w;
   561→    }
   562→  }
   563→  for (let c = 0; c < ncols; c++) {
   564→    widths[c] = Math.min(SHEET_MAX_COL_PT, Math.max(SHEET_MIN_COL_PT, widths[c]));
   565→  }
   566→  return widths;
   567→}
   568→
   569→type Measure = (text: string, size: number) => number;
   570→
   571→/** Word-wrap with char-breaking for over-long words. */
   572→function wrapCellText(
   573→  text: string,
   574→  width: number,
   575→  size: number,
   576→  measure: Measure
   577→): string[] {
   578→  const lines: string[] = [];
   579→  for (const para of text.split("\n")) {
   580→    if (para === "") {
   581→      lines.push("");
   582→      continue;
   583→    }
   584→    let line = "";
   585→    for (const word of para.split(" ")) {
   586→      if (word === "") continue;
   587→      const cand = line === "" ? word : `${line} ${word}`;
   588→      if (measure(cand, size) <= width) {
   589→        line = cand;
   590→        continue;
   591→      }
   592→      if (line !== "") {
   593→        lines.push(line);
   594→        line = "";
   595→      }
   596→      if (measure(word, size) <= width) {
   597→        line = word;
   598→        continue;
   599→      }
   600→      // char-break the over-long word (progress guaranteed)
   601→      let piece = "";
   602→      for (const ch of word) {
   603→        if (measure(piece + ch, size) <= width) {
   604→          piece += ch;
   605→        } else {
   606→          lines.push(piece);
   607→          piece = ch;
   608→        }
   609→      }
   610→      line = piece;
   611→    }
   612→    if (line !== "") lines.push(line);
   613→  }
   614→  return lines.length > 0 ? lines : [""];
   615→}
   616→
   617→/** Clips wrapped lines to `cap`, honestly marking the cut with "…". */
   618→function clipLines(
   619→  lines: string[],
   620→  cap: number,
   621→  width: number,
   622→  size: number,
   623→  measure: Measure
   624→): string[] {
   625→  if (cap < 1 || lines.length <= cap) return lines;
   626→  const out = lines.slice(0, cap);
   627→  let last = out[cap - 1];
   628→  while (last.length > 1 && measure(`${last}\u2026`, size) > width) {
   629→    last = last.slice(0, -1);
   630→  }
   631→  out[cap - 1] = `${last}\u2026`;
   632→  return out;
   633→}

   635→function drawSheets(
   636→  pdfDoc: PDFDocument,
   637→  font: PDFFont,
   638→  boldFont: PDFFont | null,
   639→  doc: Extract<ParsedOdf, { kind: "sheets" }>,
   640→  opts: OdfRenderOpts
   641→): OdfRenderReport {
   642→  const [pageW, pageH] = PAGE_SIZES[opts.pageSize === "letter" ? "letter" : "a4"];
   643→  const margin = MARGIN_PT[opts.margin === "narrow" ? "narrow" : "normal"];
   644→  const x0 = margin;
   645→  const contentWidth = pageW - 2 * margin;
   646→  const tableTopDrop = 54;
   647→  const sheetNameDrop = 34;
   648→  const measure: Measure = (text, size) => font.widthOfTextAtSize(text, size);
   649→  const tableTop = pageH - tableTopDrop;
   650→  const bottom = margin;
   651→  let pagesDrawn = 0;
   652→
   653→  const drawCellText = (
   654→    page: PDFPage,
   655→    text: string,
   656→    x: number,
   657→    y: number,
   658→    bold: boolean
   659→  ): void => {
   660→    if (text === "") return;
   661→    const face = bold && boldFont ? boldFont : font;
   662→    page.drawText(text, { x, y, size: SHEET_BODY_SIZE, font: face, color: COL_TEXT });
   663→    if (bold && !boldFont) {
   664→      // Noto ships Regular only — the invoice double-draw faux-bold
   665→      page.drawText(text, { x: x + 0.3, y, size: SHEET_BODY_SIZE, font: face, color: COL_TEXT });
   666→    }
   667→  };
   668→
   669→  for (const sheet of doc.sheets) {
   670→    const grid = sheet.grid;
   671→    const natural = sheetColumnWidths(grid);
   672→    const naturalW = natural.reduce((a, b) => a + b, 0);
   673→    const scale = naturalW <= 0 ? 1 : Math.min(1, contentWidth / naturalW);
   674→    const widths = natural.map((w) => w * scale);
   675→    const xEnds: number[] = [];
   676→    let acc = 0;
   677→    for (const w of widths) {
   678→      acc += w;
   679→      xEnds.push(acc);
   680→    }
   681→    const tableWidth = Math.min(acc, contentWidth);
   682→
   683→    // ---- first page of the sheet: name header + rule ----
   684→    const page = pdfDoc.addPage([pageW, pageH]);
   685→    pagesDrawn += 1;
   686→    const nameFace = boldFont ?? font;
   687→    page.drawText(sheet.name, {
   688→      x: x0,
   689→      y: pageH - sheetNameDrop,
   690→      size: SHEET_NAME_SIZE,
   691→      font: nameFace,
   692→      color: COL_TEXT,
   693→    });
   694→    page.drawLine({
   695→      start: { x: x0, y: pageH - sheetNameDrop - 8 },
   696→      end: { x: x0 + contentWidth, y: pageH - sheetNameDrop - 8 },
   697→      thickness: 0.75,
   698→      color: COL_BORDER,
   699→    });
   700→
   701→    // header row = grid row 0 (bold, neutral fill, repeated on breaks)
   702→    const headerRow = grid[0] ?? null;
   703→    const contentHeight = pageH - tableTopDrop - bottom;
   704→    const maxLinesCap = Math.max(
   705→      1,
   706→      Math.floor((contentHeight * 0.6) / SHEET_LINE_H)
   707→    );
   708→    const headerCells =
   709→      headerRow !== null
   710→        ? widths.map((w, c) =>
   711→            clipLines(
   712→              wrapCellText(headerRow[c] ?? "", w - 6, SHEET_HEADER_SIZE, measure),
   713→              maxLinesCap,
   714→              w - 6,
   715→              SHEET_HEADER_SIZE,
   716→              measure
   717→            )
   718→          )
   719→        : null;
   720→    const headerH =
   721→      headerCells !== null
   722→        ? Math.max(...headerCells.map((l) => l.length), 1) * SHEET_LINE_H + SHEET_ROW_PAD
   723→        : 0;
   724→
   725→    let y = tableTop;
   726→    let pageTopY = tableTop;
   727→    let pageIdx = 0;
   728→    const pages: PDFPage[] = [page];
   729→    let justBroke = false;
   730→
   731→    const finishPage = (): void => {
   732→      const p = pages[pageIdx];
   733→      p.drawLine({
   734→        start: { x: x0, y: pageTopY },
   735→        end: { x: x0 + tableWidth, y: pageTopY },
   736→        thickness: 0.75,
   737→        color: COL_GRID,
   738→      });
   739→      for (let c = 0; c <= widths.length; c++) {
   740→        const vx = x0 + (c === 0 ? 0 : xEnds[c - 1]);
   741→        p.drawLine({
   742→          start: { x: vx, y: pageTopY },
   743→          end: { x: vx, y },
   744→          thickness: 0.5,
   745→          color: COL_GRID,
   746→        });
   747→      }
   748→    };
   749→
   750→    const drawHeaderRow = (target: PDFPage, top: number): number => {
   751→      if (headerCells === null) return top;
   752→      const h = headerH;
   753→      target.drawRectangle({
   754→        x: x0,
   755→        y: top - h,
   756→        width: tableWidth,
   757→        height: h,
   758→        color: COL_HEADER_FILL,
   759→      });
   760→      for (let c = 0; c < widths.length; c++) {
   761→        const lines = headerCells[c];
   762→        const cx = x0 + (c === 0 ? 0 : xEnds[c - 1]) + 3;
   763→        for (let i = 0; i < lines.length; i++) {
   764→          drawCellText(
   765→            target,
   766→            lines[i],
   767→            cx,
   768→            top - SHEET_FIRST_BASELINE - i * SHEET_LINE_H,
   769→            true
   770→          );
   771→        }
   772→      }
   773→      target.drawLine({
   774→        start: { x: x0, y: top - h },
   775→        end: { x: x0 + tableWidth, y: top - h },
   776→        thickness: 1,
   777→        color: COL_BORDER,
   778→      });
   779→      return top - h;
   780→    };
   781→
   782→    y = drawHeaderRow(page, y);
   783→
   784→    for (let r = 1; r < grid.length; r++) {
   785→      const row = grid[r];
   786→      const cells = widths.map((w, c) =>
   787→        wrapCellText(row[c] ?? "", w - 6, SHEET_BODY_SIZE, measure)
   788→      );
   789→      let lineCount = 1;
   790→      for (const l of cells) lineCount = Math.max(lineCount, l.length);
   791→      const rowH = lineCount * SHEET_LINE_H + SHEET_ROW_PAD;
   792→
   793→      if (y - rowH < bottom && y < tableTop - 0.5 && !justBroke) {
   794→        finishPage();
   795→        const next = pdfDoc.addPage([pageW, pageH]);
   796→        pagesDrawn += 1;
   797→        pages.push(next);
   798→        pageIdx += 1;
   799→        pageTopY = tableTop;
   800→        y = tableTop;
   801→        y = drawHeaderRow(next, y);
   802→        justBroke = true;
   803→      } else {
   804→        justBroke = false;
   805→      }
   806→
   807→      const target = pages[pageIdx];
   808→      for (let c = 0; c < widths.length; c++) {
   809→        const lines = cells[c];
   810→        const cx = x0 + (c === 0 ? 0 : xEnds[c - 1]) + 3;
   811→        for (let i = 0; i < lines.length; i++) {
   812→          drawCellText(
   813→            target,
   814→            lines[i],
   815→            cx,
   816→            y - SHEET_FIRST_BASELINE - i * SHEET_LINE_H,
   817→            false
   818→          );
   819→        }
   820→      }
   821→      target.drawLine({
   822→        start: { x: x0, y: y - rowH },
   823→        end: { x: x0 + tableWidth, y: y - rowH },
   824→        thickness: 0.5,
   825→        color: COL_GRID,
   826→      });
   827→      y -= rowH;
   828→    }
   829→    finishPage();
   830→  }
   831→
   832→  if (pagesDrawn === 0) {
   833→    // defensive: a workbook with no renderable sheets still yields one page
   834→    pdfDoc.addPage([pageW, pageH]);
   835→    pagesDrawn = 1;
   836→  }
   837→  return { pages: pagesDrawn };
   838→}

   853→/**
   854→ * Paint the whole document into `pdfDoc` (pages added as needed).
   855→ * `font` must already be embedded (Helvetica standard OR the Noto
   856→ * subset); `boldFont` may be null — bold text then falls back to the
   857→ * regular face (Noto ships Regular only).
   858→ */
   859→export function drawOdfPdf(
   860→  pdfDoc: PDFDocument,
   861→  font: PDFFont,
   862→  boldFont: PDFFont | null,
   863→  doc: ParsedOdf,
   864→  opts: OdfRenderOpts
   865→): OdfRenderReport {
   866→  if (doc.kind === "sheets") {
   867→    return drawSheets(pdfDoc, font, boldFont, doc, opts);
   868→  }
   869→  const p = new Painter(
   870→    pdfDoc,
   871→    font,
   872→    boldFont,
   873→    opts.pageSize === "letter" ? "letter" : "a4",
   874→    opts.margin === "narrow" ? "narrow" : "normal"
   875→  );
   876→  for (const blk of doc.blocks) {
   877→    if (blk.kind === "table") {
   878→      drawTextTable(p, blk);
   879→    } else {
   880→      drawHeadingOrPara(p, blk);
   881→    }
   882→  }
   883→  return { pages: p.pages.length };
   884→}

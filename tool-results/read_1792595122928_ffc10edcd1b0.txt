    94→/** Strips control chars (keeps \n and \t), then caps the length. */
    95→export function cleanText(v: unknown, max: number): string {
    96→  if (typeof v !== "string") return "";
    97→  return v
    98→    .replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F\u007F]/g, "")
    99→    .slice(0, max);
   100→}
   101→
   102→/** Single-line variant: also collapses every run of whitespace. */
   103→export function cleanLine(v: unknown, max: number): string {
   104→  return cleanText(v, max).replace(/\s+/g, " ").trim();
   105→}

   109→
   110→function img(v: unknown): string {
   111→  return typeof v === "string" &&
   112→    v.length <= MAX_IMG_DATAURL &&

   118→/** New unique-enough id for sections/items (main thread + worker safe). */
   119→let idSeq = 0;
   120→export function newId(prefix: string): string {
   121→  idSeq = (idSeq + 1) % 1_000_000;
   122→  return `${prefix}-${Date.now().toString(36)}-${idSeq.toString(36)}`;
   123→}

   125→function dedupeId(id: unknown, taken: Set<string>, prefix: string): string {
   126→  const base = cleanLine(id, 40) || newId(prefix);
   127→  let out = base;
   128→  let n = 2;
   129→  while (taken.has(out)) out = `${base}-${n++}`;
   130→  taken.add(out);
   131→  return out;
   132→}

   167→/**
   168→ * One month value → display string.
   169→ * "2021-06-15" / "2021-06" → "Jun 2021" / "जून 2021"; "2021" → "2021";
   170→ * any other non-empty text is used verbatim (honest pass-through).
   171→ */
   172→export function formatMonth(value: string, locale: ResumeLocale): string {
   173→  const v = cleanLine(value, LIMITS.date);
   174→  if (!v) return "";
   175→  const m = /^(\d{4})-(\d{1,2})(?:-\d{1,2})?$/.exec(v);
   176→  if (m) {
   177→    const year = m[1];
   178→    const idx = Number.parseInt(m[2], 10);
   179→    if (idx >= 1 && idx <= 12) {
   180→      return locale === "hi"
   181→        ? `${HI_MONTHS[idx - 1]} ${year}`
   182→        : `${EN_MONTHS[idx - 1]} ${year}`;
   183→    }
   184→    return year;
   185→  }
   186→  if (/^\d{4}$/.test(v)) return v;
   187→  return v;
   188→}
   189→
   190→/**
   191→ * "Jan 2021 – Mar 2023" / "जन 2021 – मार्च 2023"; end ignored + `present`
   192→ * shown when `current`. One-sided ranges degrade to the single month.
   193→ */
   194→export function formatDateRange(
   195→  start: string,
   196→  end: string,
   197→  current: boolean,
   198→  locale: ResumeLocale,
   199→  present: string
   200→): string {
   201→  const s = formatMonth(start, locale);
   202→  if (!s) return current ? present : "";
   203→  const e = current ? present : formatMonth(end, locale);
   204→  return e ? `${s} – ${e}` : s;
   205→}

   216→export function atsContactLine(
   217→  h: ResumeDoc["header"]

   219→  return [

   223→    cleanLine(h.linkedin, LIMITS.contact),

   225→  ]
   226→    .filter(Boolean)
   227→    .join(" \u00b7 ");
   228→}

   252→/** true when every char is encodable in WinAnsi (CP1252) minus controls. */
   253→export function isWinAnsiText(s: string): boolean {
   254→  for (let i = 0; i < s.length; i++) {
   255→    const c = s.charCodeAt(i);
   256→    if (c >= 0x20 && c <= 0x7e) continue;
   257→    if (c >= 0xa0 && c <= 0xff) continue;
   258→    if (WINANSI_EXTRA.has(c)) continue;
   259→    return false;
   260→  }
   261→  return true;
   262→}

   264→/** Every string the renderer can draw for this run (content + labels + dates). */
   265→export function resumeAllText(run: ResumeRun): string[] {
   266→  const out: string[] = [];
   267→  const d = run.d;
   268→  const push = (s: string): void => {
   269→    if (s) out.push(s);
   270→  };
   271→  push(d.header.fullName);
   272→  push(d.header.headline);
   273→  push(atsContactLine(d.header));
   274→  for (const s of d.sections) {
   275→    push(s.title);
   276→    switch (s.kind) {
   277→      case "summary":
   278→        push(s.text);
   279→        break;
   280→      case "experience":
   281→        for (const it of s.items) {
   282→          push(it.role);
   283→          push(it.company);
   284→          push(it.location);
   285→          push(
   286→            formatDateRange(
   287→              it.start,
   288→              it.end,
   289→              it.current,
   290→              run.locale,
   291→              run.labels.present ?? "Present"
   292→            )
   293→          );
   294→          for (const b of it.bullets) push(b);
   295→        }
   296→        break;
   297→      case "education":
   298→        for (const it of s.items) {
   299→          push(it.degree);
   300→          push(it.school);
   301→          push(it.location);
   302→          push(it.grade);
   303→          push(formatDateRange(it.start, it.end, false, run.locale, ""));
   304→        }
   305→        break;
   306→      case "skills":
   307→        for (const g of s.groups) {
   308→          push(g.name);
   309→          for (const sk of g.items) push(sk);
   310→        }
   311→        break;
   312→      case "projects":
   313→        for (const it of s.items) {
   314→          push(it.name);
   315→          push(it.link);
   316→          push(it.description);
   317→        }
   318→        break;
   319→      case "certifications":
   320→        for (const it of s.items) push(it.name + " " + it.issuer + " " + it.year);
   321→        break;
   322→      case "languages":
   323→        for (const it of s.items) push(it.name + " " + it.level);
   324→        break;
   325→      case "custom":
   326→        for (const it of s.items) {
   327→          push(it.title);
   328→          push(it.text);
   329→        }
   330→        break;
   331→    }
   332→  }
   333→  for (const v of Object.values(run.labels)) push(v);
   334→  return out;
   335→}

   337→/**
   338→ * true → the renderer must embed the self-hosted Noto Sans Devanagari
   339→ * (fontkit subset). false → pure-WinAnsi content, and the PDF uses the
   340→ * non-embedded base-14 Helvetica (smallest possible, 100% ATS).
   341→ */
   342→export function resumeNeedsNoto(run: ResumeRun): boolean {
   343→  return resumeAllText(run).some((s) => !isWinAnsiText(s));

   453→export type MeasureFn = (text: string, size: number, bold?: boolean) => number;
   454→
   455→/**
   456→ * Greedy word-wrap honouring explicit \n. A word wider than the whole
   457→ * width is hard-broken at glyph level (long URLs never overflow).
   458→ * Never returns [].
   459→ */
   460→export function wrapText(
   461→  text: string,
   462→  width: number,
   463→  size: number,
   464→  measure: MeasureFn,
   465→  bold = false
   466→): string[] {
   467→  const out: string[] = [];
   468→  const paras = String(text ?? "")
   469→    .split("\n")
   470→    .map((p) => p.replace(/[\t\r]+/g, " ").trim());
   471→  for (const para of paras) {
   472→    if (para === "") {
   473→      out.push("");
   474→      continue;
   475→    }
   476→    let line = "";
   477→    for (let word of para.split(" ").filter(Boolean)) {
   478→      // hard-break pathological single words
   479→      while (measure(word, size, bold) > width && word.length > 1) {
   480→        let cut = word.length;
   481→        while (
   482→          cut > 1 &&
   483→          measure(word.slice(0, cut - 1) + "…", size, bold) > width
   484→        ) {
   485→          cut--;
   486→        }
   487→        const head = word.slice(0, Math.max(1, cut - 1));
   488→        const tail = word.slice(Math.max(1, cut - 1));
   489→        const piece = line ? line + " " + head : head;
   490→        out.push(piece);
   491→        line = "";
   492→        word = tail;
   493→      }
   494→      const next = line ? line + " " + word : word;
   495→      if (measure(next, size, bold) <= width || !line) line = next;
   496→      else {
   497→        out.push(line);
   498→        line = word;
   499→      }
   500→    }
   501→    out.push(line);
   502→  }
   503→  return out.length ? out : [""];
   504→}
   505→
   506→/**
   507→ * Helvetica-flavoured approximation (per-char class widths, thousandths
   508→ * of an em) — good enough for the honest "≈N pages" badge without any
   509→ * DOM or pdf-lib dependency. Non-ASCII (Devanagari etc.) is treated
   510→ * slightly wider than the Latin average.
   511→ */
   512→const NARROW = new Set(
   513→  "iljt.,:;'!|()[]{}fIrI/\\-\" ".split("")
   514→);
   515→const WIDE = new Set("mwMW@%&".split(""));
   516→
   517→export function approxMeasure(text: string, size: number): number {
   518→  let units = 0;
   519→  for (let i = 0; i < text.length; i++) {
   520→    const c = text.charAt(i);
   521→    const code = text.charCodeAt(i);
   522→    if (code < 0x80) {
   523→      if (NARROW.has(c)) units += 0.29;
   524→      else if (WIDE.has(c)) units += 0.82;
   525→      else if (c >= "A" && c <= "Z") units += 0.68;
   526→      else if (c >= "0" && c <= "9") units += 0.56;
   527→      else units += 0.52;
   528→    } else {
   529→      units += 0.62; // Devanagari and other scripts — honest average
   530→    }
   531→  }
   532→  return units * size;
   533→}

   565→          for (const b of it.bullets) push(b);

   572→        }
   573→        break;
   574→      case "skills":
   575→        for (const g of s.groups) {

   614→
   615→/**
   616→ * Simulates the renderer's flow with the given measure (the real font
   617→ * inside the worker, approxMeasure on the surface): every block height
   618→ * uses the SAME wrapText + ResumeGeometry the drawing walks, breaking
   619→ * pages at the same bottom limit. Returns the page count (≥1) — the

   621→ */
   622→export function estimatePageBreaks(
   623→  doc: ResumeDoc,
   624→  measure: MeasureFn,
   625→  locale: ResumeLocale = doc.locale,
   626→  present = "Present"
   627→): number {
   628→  const geo = geometryFor(doc.template, doc.pageSize);
   629→  const measureWrapped: MeasureFn = (t, s, b) =>
   630→    measure(t, s, b ?? false);
   631→  let pages = 1;
   632→  let y = geo.pageH - geo.margin;
   633→  const block = (h: number): void => {
   634→    if (y - h < geo.bottom) {
   635→      pages++;
   636→      y = geo.pageH - geo.margin;
   637→    }
   638→    y -= h;
   639→  };
   640→  const lines = (text: string, width: number, size: number, bold = false) => {
   641→    for (const ln of wrapText(text, width, size, measureWrapped, bold)) {
   642→      block(geo.lh);
   643→    }
   644→  };
   645→
   646→  /* main column flow */
   647→  const hasPhoto = doc.header.photo !== "";
   648→  if (doc.template !== "modern") {
   649→    const w = hasPhoto ? geo.w - geo.photoMain - 16 : geo.w;
   650→    block(geo.name * 1.25);
   651→    if (doc.header.headline) lines(doc.header.headline, w, geo.headline);
   652→    lines(atsContactLine(doc.header), w, geo.contact);
   653→    block(geo.gapHeader);
   654→  } else {
   655→    block(geo.name * 1.25);
   656→    if (doc.header.headline) lines(doc.header.headline, geo.w, geo.headline);
   657→    block(geo.gapHeader);
   658→  }
   659→
   660→  for (const s of doc.sections) {
   661→    if (doc.template === "modern" && s.kind === "skills") continue; // sidebar
   662→    block(geo.title * 1.35 + 6 + geo.gapSection * 0.35);
   663→    const w = geo.w - geo.bulletIndent;
   664→    switch (s.kind) {
   665→      case "summary":
   666→        lines(s.text, geo.w, geo.body);
   667→        break;
   668→      case "experience":
   669→        for (const it of s.items) {
   670→          block(geo.lh * 2 + 2);
   671→          for (const b of it.bullets)
   672→            lines(atsBulletLine(b), w, geo.body);
   673→          block(geo.gapItem);
   674→        }
   675→        break;
   676→      case "education":
   677→        for (const it of s.items) {
   678→          block(geo.lh * 2);
   679→          block(geo.gapItem);
   680→        }
   681→        break;
   682→      case "skills":
   683→        for (const g of s.groups) {
   684→          lines(joinMeta([g.name]) + " " + g.items.join(" · "), geo.w, geo.body);
   685→        }
   686→        break;
   687→      case "projects":
   688→        for (const it of s.items) {
   689→          block(geo.lh);
   690→          lines(it.description, w, geo.body);
   691→          block(geo.gapItem);
   692→        }
   693→        break;
   694→      case "certifications":
   695→      case "languages":
   696→        for (const it of s.items) lines(Object.values(it).join(" — "), geo.w, geo.body);
   697→        break;
   698→      case "custom":
   699→        for (const it of s.items) {
   700→          if (it.title) block(geo.lh);
   701→          lines(it.text, geo.w, geo.body);
   702→          block(geo.gapItem);
   703→        }
   704→        break;
   705→    }
   706→    block(geo.gapSection * 0.65);
   707→  }
   708→
   709→  /* sidebar flow (modern): photo + contact + skills */
   710→  if (doc.template === "modern") {
   711→    let sidePages = 1;
   712→    let sy = geo.pageH - geo.margin;
   713→    const sBlock = (h: number): void => {
   714→      if (sy - h < geo.margin) {
   715→        sidePages++;
   716→        sy = geo.pageH - geo.margin;
   717→      }
   718→      sy -= h;
   719→    };
   720→    const sLines = (text: string, size: number): void => {
   721→      for (const _ln of wrapText(text, geo.sideW, size, measureWrapped)) {
   722→        sBlock(geo.lh);
   723→      }
   724→    };
   725→    if (doc.header.photo) sBlock(geo.photoSide + 14);
   726→    sLines(doc.header.fullName, 12);
   727→    sLines(atsContactLine(doc.header), geo.small);
   728→    for (const s of doc.sections) {
   729→      if (s.kind !== "skills") continue;
   730→      sBlock(geo.title * 1.3 + 4);
   731→      for (const g of s.groups) {
   732→        sLines(g.name, geo.body);
   733→        sLines(g.items.join(" · "), geo.small);
   734→        sBlock(4);
   735→      }
   736→    }
   737→    pages = Math.max(pages, sidePages);
   738→  }
   739→
   740→  return pages;
   741→}

  1165→function parseExperience(v: unknown): ExperienceItem {
  1166→  const o = (typeof v === "object" && v !== null ? v : {}) as Record<
  1167→    string,
  1168→    unknown
  1169→  >;
  1170→  return {
  1171→    id: String(o.id ?? ""),
  1172→    role: line(o.role, LIMITS.role),
  1173→    company: line(o.company, LIMITS.company),
  1174→    location: line(o.location, LIMITS.company),
  1175→    start: line(o.start, LIMITS.date),
  1176→    end: line(o.end, LIMITS.date),
  1177→    current: o.current === true,
  1178→    bullets: (Array.isArray(o.bullets) ? o.bullets : [])
  1179→      .slice(0, MAX_BULLETS_PER_ITEM)
  1180→      .map((b) => line(b, LIMITS.bullet))
  1181→      .filter((b) => b !== ""),
  1182→  };

  1185→function parseEducation(v: unknown): EducationItem {
  1186→  const o = (typeof v === "object" && v !== null ? v : {}) as Record<
  1187→    string,
  1188→    unknown
  1189→  >;
  1190→  return {

  1364→function coerceDoc(v: unknown): ResumeDoc {
  1365→  const o = (typeof v === "object" && v !== null ? v : {}) as Record<
  1366→    string,
  1367→    unknown
  1368→  >;
  1369→  const h = (typeof o.header === "object" && o.header !== null ? o.header : {}) as Record<
  1370→    string,
  1371→    unknown
  1372→  >;
  1373→  const rawSections = Array.isArray(o.sections)
  1374→    ? o.sections.slice(0, MAX_SECTIONS)
  1375→    : [];
  1376→  const taken = new Set<string>();
  1377→  const sections: ResumeSection[] = [];
  1378→  for (const sv of rawSections) {
  1379→    const so = (typeof sv === "object" && sv !== null ? sv : {}) as Record<
  1380→      string,
  1381→      unknown
  1382→    >;
  1383→    const kind = so.kind;
  1384→    const common = {
  1385→      id: dedupeId(so.id, taken, String(kind ?? "sec")),
  1386→      title: line(so.title, LIMITS.sectionTitle),
  1387→    };
  1388→    if (kind === "summary") {
  1389→      sections.push({
  1390→        ...common,
  1391→        kind: "summary",
  1392→        text: str(so.text, LIMITS.summary),
  1393→      });
  1394→    } else if (kind === "experience") {
  1395→      const p = {
  1396→        id: "",
  1397→        kind: "experience" as const,
  1398→        title: common.title,
  1399→        items: (Array.isArray(so.items) ? so.items : [])
  1400→          .slice(0, MAX_ITEMS_PER_SECTION)
  1401→          .map(parseExperience),
  1402→      };
  1403→      sections.push({ ...p, id: common.id });
  1404→    } else if (kind === "education") {
  1405→      sections.push({
  1406→        id: common.id,
  1407→        kind: "education",
  1408→        title: common.title,
  1409→        items: (Array.isArray(so.items) ? so.items : [])
  1410→          .slice(0, MAX_ITEMS_PER_SECTION)
  1411→          .map(parseEducation),
  1412→      });
  1413→    } else if (kind === "skills") {
  1414→      const p = parseSkills(so);
  1415→      sections.push({ ...p, id: common.id, title: common.title });
  1416→    } else if (kind === "projects") {
  1417→      const p = parseProjects(so);
  1418→      sections.push({ ...p, id: common.id, title: common.title });
  1419→    } else if (kind === "certifications") {
  1420→      const p = parseCerts(so);
  1421→      sections.push({ ...p, id: common.id, title: common.title });
  1422→    } else if (kind === "languages") {
  1423→      const p = parseLanguages(so);
  1424→      sections.push({ ...p, id: common.id, title: common.title });
  1425→    } else if (kind === "custom") {
  1426→      const p = parseCustom(so);
  1427→      sections.push({ ...p, id: common.id, title: common.title });
  1428→    }
  1429→    // unknown kinds are dropped (worker-boundary contract)
  1430→  }
  1431→
  1432→  const template = TEMPLATES.includes(o.template as ResumeTemplate)
  1433→    ? (o.template as ResumeTemplate)
  1434→    : "classic";
  1435→  const pageSize = PAGE_SIZES.includes(o.pageSize as ResumePageSize)
  1436→    ? (o.pageSize as ResumePageSize)
  1437→    : "a4";
  1438→  const locale: ResumeLocale = o.locale === "hi" ? "hi" : "en";
  1439→  const accent =
  1440→    typeof o.accentColor === "string" && AC_RE.test(o.accentColor)
  1441→      ? o.accentColor
  1442→      : DEFAULT_ACCENT;
  1443→
  1444→  return {
  1445→    header: {
  1446→      fullName: line(h.fullName, LIMITS.fullName),
  1447→      headline: line(h.headline, LIMITS.headline),
  1448→      email: line(h.email, LIMITS.contact),
  1449→      phone: line(h.phone, LIMITS.contact),
  1450→      location: line(h.location, LIMITS.contact),
  1451→      linkedin: line(h.linkedin, LIMITS.contact),
  1452→      website: line(h.website, LIMITS.contact),
  1453→      photo: img(h.photo),
  1454→    },
  1455→    sections,
  1456→    accentColor: accent,
  1457→    template,
  1458→    pageSize,
  1459→    locale,
  1460→  };
  1461→}

  1463→/**
  1464→ * Parse the serialized `resume` option / run payload. Everything
  1465→ * untrusted is clamped/validated — a tampered or stale payload can
  1466→ * never reach the renderer as a wrong type, an unknown kind or an
  1467→ * oversized image. Unreadable JSON falls back to a blank resume.
  1468→ */
  1469→export function parseResumeRun(raw: string): ResumeRun {
  1470→  let parsed: unknown;
  1471→  try {
  1472→    parsed = JSON.parse(raw);
  1473→  } catch {
  1474→    parsed = {};
  1475→  }
  1476→  const o = (typeof parsed === "object" && parsed !== null ? parsed : {}) as Record<
  1477→    string,
  1478→    unknown
  1479→  >;
  1480→  const labels: Record<string, string> = {};
  1481→  if (typeof o.labels === "object" && o.labels !== null) {
  1482→    for (const [k, v] of Object.entries(o.labels as Record<string, unknown>)) {
  1483→      if (typeof v === "string" && k.length <= 16) labels[k] = v.slice(0, 300);
  1484→    }
  1485→  }
  1486→  return {
  1487→    d: coerceDoc(o.d),
  1488→    labels,
  1489→    locale: o.locale === "hi" ? "hi" : "en",
  1490→  };
  1491→}

  1494→/* Output file name (unicode-safe, no regex property escapes)           */
  1495→/* ------------------------------------------------------------------ */

  1497→function safeNameChar(c: string): boolean {
  1498→  const code = c.charCodeAt(0);
  1499→  if (code >= 0x80) return true; // Devanagari and every other script
  1500→  return /[A-Za-z0-9._-]/.test(c);
  1501→}
  1502→
  1503→/**
  1504→ * "Priya Sharma" + "Resume" → "Priya-Sharma-Resume". Falls back to the
  1505→ * file word alone, then "resume". Unicode letters survive (Hindi names
  1506→ * make Hindi file names), path separators and symbols do not.
  1507→ */
  1508→export function outputBaseName(doc: ResumeDoc, fileWord: string): string {
  1509→  const parts = [cleanLine(doc.header.fullName, 120), cleanLine(fileWord, 60)]
  1510→    .filter(Boolean)
  1511→    .map((p) =>
  1512→      p
  1513→        .split(" ")
  1514→        .map((word) =>
  1515→          [...word].filter((c) => c !== " " && safeNameChar(c)).join("")
  1516→        )
  1517→        .filter((word) => word.length > 0)
  1518→        .join("-")
  1519→    )
  1520→    .filter((p) => p.length > 0);
  1521→  const name = parts.join("-").slice(0, 80);
  1522→  return name || "resume";
  1523→}

     6→/**
     7→ * Repair PDF model (RB3-c — the 75th tool) — a raw-byte PDF salvager that
     8→ * runs 100% on-device in a Web Worker (or headless in the smoke; no DOM).
     9→ *
    10→ * HONEST ARCHITECTURE (why two stages):
    11→ *
    12→ * pdf-lib's parser is deliberately lenient — it scans objects linearly and
    13→ * can "load" a PDF whose xref/startxref/trailer are entirely missing. A
    14→ * repair tool that trusts that load alone would call stripped files
    15→ * "valid". The structure check is therefore two-part:
    16→ *
    17→ *  Stage 1 (structure check):
    18→ *   a. A raw damage scan (this file): %PDF- header at offset 0, startxref
    19→ *      present and pointing at a real xref whose sampled entry offsets
    20→ *      actually hit "N G obj" headers, %%EOF near the tail, plus the object
    21→ *      scan's findings (duplicate object numbers, wrong/missing stream
    22→ *      /Length entries, an unterminated final object).
    23→ *   b. pdf-lib load with throwOnInvalidObject: true.
    24→ *   Only when BOTH are clean is the file "valid": it is still rebuilt
    25→ *   (fresh header + xref via pdf-lib save, object streams off for maximum
    26→ *   viewer compatibility) and reported honestly as such — or, with the
    27→ *   rebuild switch off, returned byte-for-byte unchanged.
    28→ *
    29→ *  Stage 2 (raw salvage — the real work), whenever the structure check
    30→ *   finds ANY damage:
    31→ *   a. Regex-scan the raw bytes for every "N G obj" header (%PDF- is
    32→ *      located anywhere — email wrappers and garbage prefixes included;
    33→ *      a missing %PDF- is reported via headerFixed).
    34→ *   b. Extract each object's bytes up to its endobj / the next header /
    35→ *      EOF — truncated files still yield the objects that survived.
    36→ *   c. Streams: a DIRECT /Length is trusted only when "endstream" really
    37→ *      follows at that offset (and inflates, for FlateDecode); otherwise
    38→ *      the endstream-scan extent is used. Missing/indirect/wrong /Length
    39→ *      values are rewritten as the true direct number and counted.
    40→ *      Non-inflating extents are kept anyway — viewers are lenient.
    41→ *   d. Duplicate object numbers (concatenated files): the LAST complete
    42→ *      occurrence wins, matching what viewers do when the xref points
    43→ *      late. The duplicate count is reported.
    44→ *   e. Rebuild: fresh %PDF-1.7 header + objects in ascending object-number
    45→ *      order + a correct xref table + a minimal trailer whose /Root is
    46→ *      found by scanning the objects for /Type /Catalog (the catalog with
    47→ *      the most reachable /Pages leaves wins); /Size from max objnum+1;
    48→ *      a surviving trailer /Info is carried over; %%EOF is always written.
    49→ *   f. No catalog / no reachable page → honest failure (errNoCatalog /
    50→ *      errNoPages) — pages are never fabricated.
    51→ *
    52→ * Verification (always, both stages): pdf-lib strict load of the OUTPUT
    53→ * (throwOnInvalidObject: true) — the reported page count is what that
    54→ * verification sees, never a declared /Count.
    55→ *
    56→ * Encrypted PDFs are rejected up front (repair ≠ crack — use unlock-pdf):
    57→ * the raw bytes are checked for the trailer "/Encrypt N G R" reference and
    58→ * the "/Filter /Standard" encryption-dict signature, and a pdf-lib
    59→ * encryption error is mapped the same way.
    60→ */
    61→

   182→/** Lossless byte→string (one char per byte, codepoints 0x00–0xFF). */
   183→function toLatin(bytes: Uint8Array): string {
   184→  let out = "";
   185→  const CHUNK = 8192;
   186→  for (let i = 0; i < bytes.length; i += CHUNK) {
   187→    out += String.fromCharCode.apply(
   188→      null,
   189→      Array.prototype.slice.call(bytes.subarray(i, i + CHUNK)) as number[]
   190→    );
   191→  }
   192→  return out;
   193→}
   194→
   195→/** Inverse of toLatin for chars ≤ 0xFF (exactly what toLatin produces). */
   196→function latin1Encode(text: string): Uint8Array {
   197→  const out = new Uint8Array(text.length);
   198→  for (let i = 0; i < text.length; i += 1) {
   199→    out[i] = text.charCodeAt(i) & 0xff;
   200→  }
   201→  return out;
   202→}

   204→function skipWs(s: string, pos: number, limit: number): number {
   205→  let i = pos;
   206→  while (i < limit && WS_CHARS.indexOf(s[i]) >= 0) i += 1;
   207→  return i;
   208→}
   209→
   210→/** Strips trailing PDF whitespace (incl. NUL). */
   211→function trimPdfWs(s: string): string {
   212→  let end = s.length;
   213→  while (end > 0 && WS_CHARS.indexOf(s[end - 1]) >= 0) end -= 1;
   214→  return s.slice(0, end);
   215→}

   221→/**
   222→ * Index just past the ">>" matching the "<<" at `start`, or -1 when the
   223→ * dict is unbalanced before `limit` (an incomplete object).
   224→ */
   225→function findDictEnd(s: string, start: number, limit: number): number {
   226→  let depth = 0;
   227→  let state: "n" | "s" | "h" | "c" = "n";
   228→  let i = start;
   229→  while (i < limit) {
   230→    const ch = s[i];
   231→    if (state === "s") {
   232→      if (ch === "\\") i += 1;
   233→      else if (ch === ")") state = "n";
   234→    } else if (state === "h") {
   235→      if (ch === ">") state = "n";
   236→    } else if (state === "c") {
   237→      if (ch === "\n" || ch === "\r") state = "n";
   238→    } else {
   239→      if (ch === "%") state = "c";
   240→      else if (ch === "(") state = "s";
   241→      else if (ch === "<" && s[i + 1] === "<") {
   242→        depth += 1;
   243→        i += 1;
   244→      } else if (ch === ">" && s[i + 1] === ">") {
   245→        depth -= 1;
   246→        i += 1;
   247→        if (depth === 0) return i + 1;
   248→      } else if (ch === "<") {
   249→        state = "h";
   250→      }
   251→    }
   252→    i += 1;
   253→  }
   254→  return -1;
   255→}

   257→interface LengthSpan {
   258→  /** Index of the first value char (a digit). */
   259→  valueStart: number;
   260→  /** Index just past the value (digits, or "N G R" for an indirect ref). */
   261→  valueEnd: number;
   262→  /** Parsed direct number (null for indirect refs). */
   263→  direct: number | null;
   264→}
   265→
   266→/** Locates the top-level /Length value inside a dict string. */
   267→function findLengthSpan(dict: string): LengthSpan | null {
   268→  let depth = 0;
   269→  let state: "n" | "s" | "h" | "c" = "n";
   270→  let i = 0;
   271→  while (i < dict.length) {
   272→    const ch = dict[i];
   273→    if (state === "s") {
   274→      if (ch === "\\") i += 1;
   275→      else if (ch === ")") state = "n";
   276→    } else if (state === "h") {
   277→      if (ch === ">") state = "n";
   278→    } else if (state === "c") {
   279→      if (ch === "\n" || ch === "\r") state = "n";
   280→    } else {
   281→      if (ch === "%") state = "c";
   282→      else if (ch === "(") state = "s";
   283→      else if (ch === "<" && dict[i + 1] === "<") {
   284→        depth += 1;
   285→        i += 1;
   286→      } else if (ch === ">" && dict[i + 1] === ">") {
   287→        depth -= 1;
   288→        i += 1;
   289→      } else if (ch === "<") {
   290→        state = "h";
   291→      } else if (ch === "/" && dict.startsWith("/Length", i) && depth === 1) {
   292→        const afterName = i + 7;
   293→        const next = dict[afterName];
   294→        if (next === undefined || WS_CHARS.indexOf(next) >= 0 || next === "/") {
   295→          const valueStart = skipWs(dict, afterName, dict.length);
   296→          const digits = /^[0-9]+/.exec(dict.slice(valueStart, valueStart + 12));
   297→          if (digits) {
   298→            let valueEnd = valueStart + digits[0].length;
   299→            const rest = new RegExp(
   300→              `${WS_CLASS_PLUS}([0-9]{1,9})${WS_CLASS_PLUS}R`
   301→            ).exec(dict.slice(valueEnd, valueEnd + 32));
   302→            if (rest) {
   303→              // Indirect reference value "N G R".
   304→              valueEnd += rest[0].length;
   305→              return { valueStart, valueEnd, direct: null };
   306→            }
   307→            return {
   308→              valueStart,
   309→              valueEnd,
   310→              direct: Number.parseInt(digits[0], 10),
   311→            };
   312→          }
   313→        }
   314→        i = afterName - 1;
   315→      }
   316→    }
   317→    i += 1;
   318→  }
   319→  return null;
   320→}

   322→/** Rewrites the dict's /Length to the true direct number. */
   323→function spliceLength(
   324→  dict: string,
   325→  actualLen: number
   326→): { dict: string; fixed: boolean } {
   327→  const span = findLengthSpan(dict);
   328→  if (span === null) {
   329→    // No /Length at all — insert one before the dict's final ">>".
   330→    const close = dict.lastIndexOf(">>");
   331→    if (close < 0) return { dict, fixed: false };
   332→    return {
   333→      dict: `${dict.slice(0, close)} /Length ${actualLen} ${dict.slice(close)}`,
   334→      fixed: true,
   335→    };
   336→  }
   337→  if (span.direct !== null && span.direct === actualLen) {
   338→    return { dict, fixed: false };
   339→  }
   340→  return {
   341→    dict:
   342→      dict.slice(0, span.valueStart) +
   343→      String(actualLen) +
   344→      dict.slice(span.valueEnd),
   345→    fixed: true,
   346→  };
   347→}

   393→/** Verifies a classic xref table: sampled in-use entries must hit real
   394→ *  "N G obj" headers at their recorded offsets. */
   395→function verifyClassicXref(s: string, at: number): boolean {
   396→  let pos = skipWs(s, at + 4, s.length);
   397→  const head = /^([0-9]{1,9})[ \t]+([0-9]{1,9})/.exec(s.slice(pos, pos + 32));
   398→  if (!head) return false;
   399→  const first = Number.parseInt(head[1], 10);
   400→  const count = Number.parseInt(head[2], 10);
   401→  if (!Number.isFinite(first) || !Number.isFinite(count) || count <= 0) {
   402→    return false;
   403→  }
   404→  pos += head[0].length;
   405→
   406→  // Deterministic sample: first, middle, last entry.
   407→  const samples = new Set<number>([0, Math.floor(count / 2), count - 1]);
   408→  let parsed = 0;
   409→  let checked = 0;
   410→  let sawInUse = false;
   411→  for (let k = 0; k < count && k < 100_000; k += 1) {
   412→    pos = skipWs(s, pos, s.length);
   413→    const entry = /^([0-9]{1,10})[ \t]+([0-9]{1,5})[ \t]*([nf])(?![0-9a-zA-Z])/.exec(
   414→      s.slice(pos, pos + 24)
   415→    );
   416→    if (!entry) break;
   417→    parsed += 1;
   418→    if (entry[3] === "n") {
   419→      sawInUse = true;
   420→      if (samples.has(k)) {
   421→        const off = Number.parseInt(entry[1], 10);
   422→        const objnum = first + k;
   423→        const atObj = /^([0-9]{1,9})[ \t\r\n\f\x00]+([0-9]{1,3})[ \t\r\n\f\x00]+obj/.exec(
   424→          s.slice(off, off + 40)
   425→        );
   426→        if (!atObj || Number.parseInt(atObj[1], 10) !== objnum) return false;
   427→        checked += 1;
   428→      }
   429→    }
   430→    pos += entry[0].length;
   431→  }
   432→  return parsed > 0 && (!sawInUse || checked > 0);
   433→}

   435→function scanStructure(s: string, objects: ScannedObject[]): StructureInfo {
   436→  const headerIdx = s.indexOf("%PDF-");
   437→  const eofIdx = s.lastIndexOf("%%EOF");
   438→
   439→  let xrefOk = false;
   440→  const sxIdx = s.lastIndexOf("startxref");
   441→  if (sxIdx >= 0) {
   442→    const numAt = skipWs(s, sxIdx + 9, s.length);
   443→    const num = /^[0-9]{1,10}/.exec(s.slice(numAt, numAt + 12));
   444→    if (num) {
   445→      const off = Number.parseInt(num[0], 10);
   446→      if (s.startsWith("xref", off)) {
   447→        xrefOk = verifyClassicXref(s, off);
   448→      } else if (
   449→        new RegExp(`^([0-9]{1,9})${WS_CLASS_PLUS}([0-9]{1,3})${WS_CLASS_PLUS}obj`).test(
   450→          s.slice(off, off + 40)
   451→        )
   452→      ) {
   453→        // xref stream — presence accepted (entries are compressed).
   454→        xrefOk = true;
   455→      }
   456→    }
   457→  }
   458→
   459→  // /Info from the LAST original trailer, when its object survived.
   460→  let infoRef: { num: number; gen: number } | null = null;
   461→  const trIdx = s.lastIndexOf("trailer");
   462→  if (trIdx >= 0) {
   463→    const region = s.slice(trIdx, trIdx + 512);
   464→    const m = new RegExp(
   465→      `\\/Info${WS_CLASS_PLUS}([0-9]{1,9})${WS_CLASS_PLUS}([0-9]{1,3})${WS_CLASS_PLUS}R`
   466→    ).exec(region);
   467→    if (m) {
   468→      const num = Number.parseInt(m[1], 10);
   469→      const gen = Number.parseInt(m[2], 10);
   470→      if (objects.some((o) => o.num === num)) {
   471→        infoRef = { num, gen };
   472→      }
   473→    }
   474→  }
   475→
   476→  return {
   477→    headerAtStart: headerIdx === 0,
   478→    headerFound: headerIdx >= 0,
   479→    eofOk: eofIdx >= 0 && eofIdx >= s.length - 64,
   480→    xrefOk,
   481→    infoRef,
   482→  };
   483→}

   485→/** One raw "N G obj" match with its body-start position. */
   486→interface HeaderMatch {
   487→  num: number;
   488→  gen: number;
   489→  matchStart: number;
   490→  bodyStart: number;
   491→}
   492→
   493→function collectHeaders(s: string): HeaderMatch[] {
   494→  const matches: HeaderMatch[] = [];
   495→  OBJ_HEADER_RE.lastIndex = 0;
   496→  let m: RegExpExecArray | null;
   497→  while ((m = OBJ_HEADER_RE.exec(s)) !== null) {
   498→    // The char before a real header is never a digit (a longer number's
   499→    // tail would otherwise read as an object number).
   500→    const prev = m.index > 0 ? s[m.index - 1] : "";
   501→    if (prev >= "0" && prev <= "9") continue;
   502→    matches.push({
   503→      num: Number.parseInt(m[1], 10),
   504→      gen: Number.parseInt(m[2], 10),
   505→      matchStart: m.index,
   506→      bodyStart: skipWs(s, m.index + m[0].length, s.length),
   507→    });
   508→  }
   509→  return matches;
   510→}
   511→
   512→/**
   513→ * Extracts every complete object between a header and the next header /
   514→ * endobj / EOF. Stream extents honour a DIRECT /Length only when
   515→ * "endstream" really follows it (and inflates); otherwise the
   516→ * endstream-scan extent is used. An incomplete FINAL object sets the
   517→ * truncation flag (mid-file incomplete objects are just dropped).
   518→ */
   519→function extractObjects(s: string, headers: HeaderMatch[]): {
   520→  objects: ScannedObject[];
   521→  streamLengthsFixed: number;
   522→  truncationDetected: boolean;
   523→} {
   524→  const objects: ScannedObject[] = [];
   525→  let streamLengthsFixed = 0;
   526→  let truncationDetected = false;
   527→
   528→  for (let i = 0; i < headers.length; i += 1) {
   529→    const h = headers[i];
   530→    const isLast = i === headers.length - 1;
   531→    const extentEnd =
   532→      i + 1 < headers.length ? headers[i + 1].matchStart : s.length;
   533→    if (h.num === 0 || h.num > MAX_OBJNUM || h.gen > 65535) continue;
   534→
   535→    let obj: ScannedObject | null = null;
   536→    let incomplete = false;
   537→
   538→    if (s[h.bodyStart] === "<" && s[h.bodyStart + 1] === "<") {
   539→      const dictEnd = findDictEnd(s, h.bodyStart, extentEnd);
   540→      if (dictEnd < 0) {
   541→        incomplete = true; // dict never closes — the object was cut short
   542→      } else {
   543→        const afterDict = skipWs(s, dictEnd, extentEnd);
   544→        const isStream =
   545→          s.startsWith("stream", afterDict) &&
   546→          (s[afterDict + 6] === "\r" || s[afterDict + 6] === "\n");
   547→        if (isStream) {
   548→          // ---- stream object: dict + stream + data + endstream ----
   549→          let dataStart = afterDict + 6;
   550→          if (s[dataStart] === "\r" && s[dataStart + 1] === "\n") dataStart += 2;
   551→          else if (s[dataStart] === "\r" || s[dataStart] === "\n") dataStart += 1;
   552→
   553→          const dictStr = s.slice(h.bodyStart, dictEnd);
   554→          const span = findLengthSpan(dictStr);
   555→          const isFlate = FLATE_RE.test(dictStr);
   556→
   557→          let dataEnd = -1;
   558→          if (span !== null && span.direct !== null) {
   559→            const cand = Math.min(dataStart + span.direct, extentEnd);
   560→            if (cand === extentEnd || s.startsWith("endstream", skipWs(s, cand, extentEnd))) {
   561→              dataEnd = cand;
   562→            }
   563→          }
   564→          if (dataEnd >= 0 && isFlate && span !== null && span.direct !== null) {
   565→            // The declared extent must actually inflate for FlateDecode;
   566→            // when it doesn't but the FIRST endstream extent does, the
   567→            // wrong length had landed on a later "endstream" — prefer the
   568→            // inflating extent (the task's inflate verification).
   569→            const scanEnd = s.indexOf("endstream", dataStart);
   570→            if (
   571→              scanEnd >= 0 &&
   572→              scanEnd < dataEnd &&
   573→              inflates(s.slice(dataStart, scanEnd)) &&
   574→              !inflates(s.slice(dataStart, dataEnd))
   575→            ) {
   576→              dataEnd = scanEnd;
   577→            }
   578→          } else if (dataEnd < 0) {
   579→            // Missing/indirect/wrong length — take the endstream extent.
   580→            const scanEnd = s.indexOf("endstream", dataStart);
   581→            if (scanEnd >= 0 && scanEnd < extentEnd) {
   582→              dataEnd = scanEnd;
   583→            }
   584→          }
   585→          if (dataEnd < 0) {
   586→            // Stream runs to the extent end (a truncated tail).
   587→            dataEnd = extentEnd;
   588→            incomplete = true;
   589→          }
   590→
   591→          const data = s.slice(dataStart, dataEnd);
   592→          const spliced = spliceLength(dictStr, dataEnd - dataStart);
   593→          if (spliced.fixed) streamLengthsFixed += 1;
   594→          obj = {
   595→            num: h.num,
   596→            gen: h.gen,
   597→            isStream: true,
   598→            dict: spliced.dict,
   599→            data,
   600→            body: "",
   601→          };
   602→        } else {
   603→          // ---- plain dict object ----
   604→          const endobj = s.indexOf("endobj", dictEnd);
   605→          const bodyEnd = endobj >= 0 && endobj < extentEnd ? endobj : extentEnd;
   606→          obj = {
   607→            num: h.num,
   608→            gen: h.gen,
   609→            isStream: false,
   610→            dict: "",
   611→            data: "",
   612→            body: trimPdfWs(s.slice(h.bodyStart, bodyEnd)),
   613→          };
   614→        }
   615→      }
   616→    } else if (h.bodyStart < extentEnd) {
   617→      // ---- non-dict body (numbers, refs, arrays, strings) ----
   618→      const endobj = s.indexOf("endobj", h.bodyStart);
   619→      const bodyEnd = endobj >= 0 && endobj < extentEnd ? endobj : extentEnd;
   620→      const body = trimPdfWs(s.slice(h.bodyStart, bodyEnd));
   621→      const lastChar = body.length > 0 ? body[body.length - 1] : "";
   622→      if (body.length > 0 && PLAIN_BODY_END_CHARS.has(lastChar)) {
   623→        obj = {
   624→          num: h.num,
   625→          gen: h.gen,
   626→          isStream: false,
   627→          dict: "",
   628→          data: "",
   629→          body,
   630→        };
   631→      } else {
   632→        incomplete = true; // never closed — garbage or cut short
   633→      }
   634→    }
   635→
   636→    if (obj !== null) objects.push(obj);
   637→    if (isLast && incomplete) truncationDetected = true;
   638→  }
   639→
   640→  return { objects, streamLengthsFixed, truncationDetected };
   641→}
   642→
   643→function scanRaw(s: string): ScanResult {
   644→  const headers = collectHeaders(s);
   645→
   646→  // Duplicate object numbers — count of nums seen more than once.
   647→  const seen = new Set<number>();
   648→  const dup = new Set<number>();
   649→  for (const h of headers) {
   650→    if (seen.has(h.num)) dup.add(h.num);
   651→    else seen.add(h.num);
   652→  }
   653→
   654→  const extracted = extractObjects(s, headers);
   655→  const structure = scanStructure(s, extracted.objects);
   656→
   657→  return {
   658→    objects: extracted.objects,
   659→    dupObjNums: dup.size,
   660→    streamLengthsFixed: extracted.streamLengthsFixed,
   661→    truncationDetected: extracted.truncationDetected,
   662→    structure,
   663→  };
   664→}

   666→/* ------------------------------------------------------------------ */
   667→/* Catalog + page-tree walk                                            */
   668→/* ------------------------------------------------------------------ */
   669→
   670→function extractRefFrom(
   671→  body: string,
   672→  key: string
   673→): { num: number; gen: number } | null {
   674→  const m = new RegExp(
   675→    `\\/${key}${WS_CLASS_PLUS}([0-9]{1,9})${WS_CLASS_PLUS}([0-9]{1,3})${WS_CLASS_PLUS}R`
   676→  ).exec(body);
   677→  if (!m) return null;
   678→  return {
   679→    num: Number.parseInt(m[1], 10),
   680→    gen: Number.parseInt(m[2], 10),
   681→  };
   682→}
   683→
   684→function extractKidsRefs(body: string): Array<{ num: number; gen: number }> {
   685→  const kidsAt = new RegExp(`\\/Kids${WS_CLASS_PLUS}`).exec(body);
   686→  if (!kidsAt) return [];
   687→  const open = body.indexOf("[", kidsAt.index + kidsAt[0].length);
   688→  if (open < 0) return [];
   689→  const close = body.indexOf("]", open);
   690→  if (close < 0) return [];
   691→  const refs: Array<{ num: number; gen: number }> = [];
   692→  REF_RE.lastIndex = 0;
   693→  let m: RegExpExecArray | null;
   694→  const region = body.slice(open, close);
   695→  while ((m = REF_RE.exec(region)) !== null) {
   696→    refs.push({
   697→      num: Number.parseInt(m[1], 10),
   698→      gen: Number.parseInt(m[2], 10),
   699→    });
   700→  }
   701→  return refs;
   702→}
   703→
   704→/** Reachable /Type /Page leaves under a /Pages ref (missing kids skip). */
   705→function countLeaves(
   706→  byNum: ReadonlyMap<number, ScannedObject>,
   707→  root: { num: number; gen: number }
   708→): number {
   709→  const visited = new Set<number>();
   710→  const walk = (ref: { num: number; gen: number }, depth: number): number => {
   711→    if (depth > MAX_TREE_DEPTH || visited.has(ref.num)) return 0;
   712→    visited.add(ref.num);
   713→    const node = byNum.get(ref.num);
   714→    if (node === undefined || node.isStream) return 0;
   715→    if (PAGE_LEAF_RE.test(node.body)) return 1;
   716→    if (!PAGE_TREE_RE.test(node.body)) return 0;
   717→    let leaves = 0;
   718→    for (const kid of extractKidsRefs(node.body)) {
   719→      leaves += walk(kid, depth + 1);
   720→      if (leaves >= MAX_TREE_LEAVES) return MAX_TREE_LEAVES;
   721→    }
   722→    return leaves;
   723→  };
   724→  return walk(root, 0);
   725→}
   726→
   727→/**
   728→ * Finds the catalog to anchor the rebuilt trailer: every recovered object
   729→ * with /Type /Catalog is a candidate; the one with the most reachable
   730→ * /Pages leaves wins (lowest object number breaks ties). No catalogs at
   731→ * all → errNoCatalog; catalogs but zero reachable pages → errNoPages.
   732→ */
   733→function pickCatalog(
   734→  byNum: ReadonlyMap<number, ScannedObject>
   735→): { num: number; gen: number; leaves: number } {
   736→  const catalogs = [...byNum.values()].filter(
   737→    (o) => !o.isStream && CATALOG_RE.test(o.body)
   738→  );
   739→  if (catalogs.length === 0) {
   740→    throw new RepairError("errNoCatalog");
   741→  }
   742→  let best: { num: number; gen: number; leaves: number } | null = null;
   743→  for (const cat of catalogs) {
   744→    const pagesRef = extractRefFrom(cat.body, "Pages");
   745→    const leaves = pagesRef === null ? 0 : countLeaves(byNum, pagesRef);
   746→    if (
   747→      best === null ||
   748→      leaves > best.leaves ||
   749→      (leaves === best.leaves && cat.num < best.num)
   750→    ) {
   751→      best = { num: cat.num, gen: cat.gen, leaves };
   752→    }
   753→  }
   754→  if (best === null || best.leaves <= 0) {
   755→    throw new RepairError("errNoPages");
   756→  }
   757→  return best;
   758→}

   760→/* ------------------------------------------------------------------ */
   761→/* Rebuild                                                            */
   762→/* ------------------------------------------------------------------ */
   763→
   764→/** Assembles header + objects + xref + trailer into a fresh PDF. */
   765→function buildRebuilt(
   766→  byNum: ReadonlyMap<number, ScannedObject>,
   767→  root: { num: number; gen: number },
   768→  info: { num: number; gen: number } | null
   769→): Uint8Array {
   770→  const nums = [...byNum.keys()].sort((a, b) => a - b);
   771→  const maxNum = nums[nums.length - 1];
   772→  const parts: Array<string | Uint8Array> = [];
   773→  let length = 0;
   774→  const push = (part: string | Uint8Array): void => {
   775→    parts.push(part);
   776→    length += part.length;
   777→  };
   778→
   779→  push("%PDF-1.7\n");
   780→  const offsets = new Map<number, number>();
   781→  for (const num of nums) {
   782→    const o = byNum.get(num);
   783→    if (o === undefined) continue;
   784→    offsets.set(num, length);
   785→    if (o.isStream) {
   786→      push(`${num} ${o.gen} obj\n`);
   787→      push(o.dict);
   788→      push("\nstream\n");
   789→      push(latin1Encode(o.data));
   790→      push("\nendstream\nendobj\n");
   791→    } else {
   792→      push(`${num} ${o.gen} obj\n`);
   793→      push(o.body);
   794→      push("\nendobj\n");
   795→    }
   796→  }
   797→
   798→  const xrefOffset = length;
   799→  const size = maxNum + 1;
   800→  let table = `xref\n0 ${size}\n0000000000 65535 f \n`;
   801→  for (let n = 1; n <= maxNum; n += 1) {
   802→    const off = offsets.get(n);
   803→    const o = off === undefined ? undefined : byNum.get(n);
   804→    if (off === undefined || o === undefined) {
   805→      table += "0000000000 00000 f \n";
   806→    } else {
   807→      table += `${String(off).padStart(10, "0")} ${String(o.gen).padStart(5, "0")} n \n`;
   808→    }
   809→  }
   810→  push(table);
   811→  push(
   812→    `trailer\n<< /Size ${size} /Root ${root.num} ${root.gen} R` +
   813→      (info !== null ? ` /Info ${info.num} ${info.gen} R` : "") +
   814→      ` >>\nstartxref\n${xrefOffset}\n%%EOF\n`
   815→  );
   816→
   817→  const out = new Uint8Array(length);
   818→  let pos = 0;
   819→  for (const part of parts) {
   820→    if (typeof part === "string") {
   821→      for (let i = 0; i < part.length; i += 1) {
   822→        out[pos] = part.charCodeAt(i) & 0xff;
   823→        pos += 1;
   824→      }
   825→    } else {
   826→      out.set(part, pos);
   827→      pos += part.length;
   828→    }
   829→  }
   830→  return out;
   831→}

   834→/* Verification                                                        */
   835→/* ------------------------------------------------------------------ */

   837→async function strictVerify(bytes: Uint8Array): Promise<number> {
   838→  try {
   839→    const doc = await PDFDocument.load(bytes, {
   840→      throwOnInvalidObject: true,
   841→      updateMetadata: false,
   842→    });
   843→    return doc.getPageCount();
   844→  } catch (error) {
   845→    throw new RepairError(
   846→      "errVerify",
   847→      error instanceof Error ? error.message : String(error)
   848→    );
   849→  }
   850→}

   852→/* ------------------------------------------------------------------ */
   853→/* The pipeline                                                        */
   854→/* ------------------------------------------------------------------ */
   855→
   856→/**
   857→ * repairPdf — the whole two-stage pipeline. Never touches the network,
   858→ * is deterministic for a given input, and throws RepairError for the
   859→ * honest failures (encrypted, no objects, no catalog, no pages, failed
   860→ * verification).
   861→ */
   862→export async function repairPdf(
   863→  bytes: Uint8Array,
   864→  options: RepairOptions
   865→): Promise<RepairOutcome> {
   866→  const s = toLatin(bytes);
   867→
   868→  // ---- Encryption first: repair is not crack. Both raw signatures are
   869→  // checked so an encrypted file whose trailer was also damaged is still
   870→  // rejected honestly instead of being "salvaged" into ciphertext soup.
   871→  if (ENCRYPT_TRAILER_RE.test(s) || ENCRYPT_DICT_RE.test(s)) {
   872→    throw new RepairError("errEncrypted");
   873→  }
   874→
   875→  // ---- Raw damage scan (ALWAYS — pdf-lib's lenient load alone cannot
   876→  // tell a stripped file from a valid one).
   877→  const scan = scanRaw(s);
   878→  const damaged =
   879→    !scan.structure.headerAtStart ||
   880→    !scan.structure.xrefOk ||
   881→    !scan.structure.eofOk ||
   882→    scan.truncationDetected ||
   883→    scan.streamLengthsFixed > 0 ||
   884→    scan.dupObjNums > 0;
   885→
   886→  // ---- pdf-lib structure check.
   887→  let loaded: PDFDocument | null = null;
   888→  let loadFailed = false;
   889→  try {
   890→    loaded = await PDFDocument.load(bytes, {
   891→      throwOnInvalidObject: true,
   892→      updateMetadata: false,
   893→    });
   894→  } catch (error) {
   895→    loadFailed = true;
   896→    const message = error instanceof Error ? error.message : String(error);
   897→    if (/encrypt/i.test(message)) {
   898→      throw new RepairError("errEncrypted", message);
   899→    }
   900→  }
   901→
   902→  /* ---------------- Stage 1: clean file ---------------------------- */
   903→  if (loaded !== null && !damaged) {
   904→    const objectCount = loaded.context.enumerateIndirectObjects().length;
   905→    if (options.alsoRebuildValid) {
   906→      // Honest rebuild of an already-valid file: fresh header + xref via
   907→      // pdf-lib save; object streams OFF for maximum viewer compatibility.
   908→      const out = await loaded.save({ useObjectStreams: false });
   909→      const pages = await strictVerify(out);
   910→      return {
   911→        bytes: out,
   912→        report: {
   913→          structureWasValid: true,
   914→          unchanged: false,
   915→          loadFailed,
   916→          xrefRebuilt: false,
   917→          headerFixed: false,
   918→          eofMissing: false,
   919→          truncationDetected: false,
   920→          objectsRecovered: objectCount,
   921→          duplicatesResolved: 0,
   922→          streamLengthsFixed: 0,
   923→          pages,
   924→          bytesIn: bytes.length,
   925→          bytesOut: out.length,
   926→        },
   927→      };
   928→    }
   929→    const pages = loaded.getPageCount();
   930→    return {
   931→      bytes,
   932→      report: {
   933→        structureWasValid: true,
   934→        unchanged: true,
   935→        loadFailed,
   936→        xrefRebuilt: false,
   937→        headerFixed: false,
   938→        eofMissing: false,
   939→        truncationDetected: false,
   940→        objectsRecovered: objectCount,
   941→        duplicatesResolved: 0,
   942→        streamLengthsFixed: 0,
   943→        pages,
   944→        bytesIn: bytes.length,
   945→        bytesOut: bytes.length,
   946→      },
   947→    };
   948→  }
   949→
   950→  /* ---------------- Stage 2: raw salvage ---------------------------- */
   951→  if (scan.objects.length === 0) {
   952→    throw new RepairError("errNoObjects");
   953→  }
   954→  const salvage = scan.objects.slice(0, MAX_OBJECTS);
   955→
   956→  // Last complete occurrence wins for duplicate object numbers (an
   957→  // earlier complete copy survives when the later one was cut short).
   958→  const byNum = new Map<number, ScannedObject>();
   959→  for (const o of salvage) {
   960→    byNum.set(o.num, o);
   961→  }
   962→
   963→  const root = pickCatalog(byNum);
   964→  const out = buildRebuilt(byNum, root, scan.structure.infoRef);
   965→  const pages = await strictVerify(out);
   966→
   967→  return {
   968→    bytes: out,
   969→    report: {
   970→      structureWasValid: false,
   971→      unchanged: false,
   972→      loadFailed,
   973→      xrefRebuilt: !scan.structure.xrefOk,
   974→      headerFixed: !scan.structure.headerAtStart,
   975→      eofMissing: !scan.structure.eofOk,
   976→      truncationDetected: scan.truncationDetected,
   977→      objectsRecovered: byNum.size,
   978→      duplicatesResolved: scan.dupObjNums,
   979→      streamLengthsFixed: scan.streamLengthsFixed,
   980→      pages,
   981→      bytesIn: bytes.length,
   982→      bytesOut: out.length,
   983→    },
   984→  };
   985→}

   991→function sanitizeFileWord(raw: string): string {
   992→  let out = "";
   993→  for (const ch of raw) {
   994→    const code = ch.codePointAt(0) ?? 0;
   995→    if (code >= 32 && code !== 127) out += ch;
   996→  }
   997→  return out.replace(/\s+/g, " ").trim().slice(0, 60);
   998→}

  1000→/**
  1001→ * parseRun — the options payload the surface sends as options.run.
  1002→ * Garbage JSON degrades to the defaults; only an explicit false disables
  1003→ * the rebuild switch; the locale must be exactly "hi".
  1004→ */
  1005→export function parseRun(raw: string): RepairOptions {
  1006→  let parsed: unknown;
  1007→  try {
  1008→    parsed = JSON.parse(raw);
  1009→  } catch {
  1010→    parsed = {};
  1011→  }
  1012→  const o =
  1013→    typeof parsed === "object" && parsed !== null
  1014→      ? (parsed as Record<string, unknown>)
  1015→      : {};
  1016→  return {
  1017→    alsoRebuildValid: o.alsoRebuildValid !== false,
  1018→    locale: o.locale === "hi" ? "hi" : "en",
  1019→    fileWord: sanitizeFileWord(typeof o.fileWord === "string" ? o.fileWord : ""),
  1020→  };
  1021→}

  1023→/** Download base name: strips .pdf, sanitizes, localized fallback. */
  1024→export function outputBaseName(
  1025→  name: string,
  1026→  locale: "en" | "hi",
  1027→  fileWord = ""
  1028→): string {
  1029→  const base = name
  1030→    .replace(/\.pdf$/i, "")
  1031→    .replace(/[\\/:*?"<>|\u0000-\u001f]/g, "_")
  1032→    .trim();
  1033→  if (base.length > 0) return base.slice(0, 120);
  1034→  const word = fileWord.trim() || (locale === "hi" ? "दस्तावेज़" : "document");
  1035→  return word.slice(0, 60);
  1036→}

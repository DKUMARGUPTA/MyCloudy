     5→/**
     6→ * PDF → PowerPoint model (RB3-d). PURE TypeScript: no pdf.js, no DOM,
     7→ * no pdf-lib — the smoke drives every export headless in bun, and the
     8→ * worker feeds it real pdf.js text items + page viewports.
     9→ *
    10→ * Pipeline (all in this file):
    11→ *  1. collectPieces — coerce + position-tag the raw pdf.js items (the
    12→ *     pdf-to-word recipe: whitespace-only items become exactly ONE
    13→ *     space piece, per-item splits keep their edge spaces).
    14→ *  2. buildLines — group items into visual lines (same baseline y
    15→ *     within half a font height); within a line the pieces are ordered
    16→ *     by x and joined into runs, with a space inserted wherever the
    17→ *     horizontal gap exceeds a quarter of the font size (pdf.js emits
    18→ *     one item per Tj without carrying spaces).
    19→ *  3. buildBlocks — lines merge into text blocks on y-proximity
    20→ *     (baseline gap ≤ 1.7 × font size) AND font-size similarity
    21→ *     (≤ 25% change) — a title above body never glues to its body.
    22→ *  4. Title detection — the largest font size among blocks whose first
    23→ *     line sits in the TOP THIRD of the page; ties go to the topmost.
    24→ *     The winning block becomes its own bold title shape.
    25→ *  5. Layout mapping — page pt → slide EMU. Slide sizes: "16:9"
    26→ *     (12192000×6858000), "4:3" (9144000×6858000) or "match page"
    27→ *     (page pt × 12700, orientation-aware). Every page is CONTAIN-FIT
    28→ *     into the slide (scale = min(slideW/pageW, slideH/pageH)) and
    29→ *     centered when the aspects differ, so no content ever lands off
    30→ *     the slide. Font size × scale, clamped 8–96 pt (a:p sz is
    31→ *     hundredths). One text box (p:sp + p:txBody) per block, one a:p
    32→ *     per line, body width = the remaining column width (block x →
    33→ *     page content right edge).
    34→ *  6. Font naming — PPTX does NOT embed fonts: each run references a
    35→ *     typeface by name. Devanagari text → "Noto Sans Devanagari"
    36→ *     (a:latin + a:ea + a:cs), everything else → "Calibri". This is
    37→ *     the honest advantage over PDF export: हिंदी needs no font bytes.
    38→ *  7. buildPptx — a REAL minimal Office Open XML package via fflate
    39→ *     zipSync (the pdf-to-excel writer recipe): [Content_Types].xml,
    40→ *     _rels/.rels, ppt/presentation.xml (+ rels), slideMaster + rels +
    41→ *     theme, slideLayout + rels, one slideN.xml (+ rels) per page,
    42→ *     optional ppt/media/imageN.png for page-images mode, docProps
    43→ *     core.xml + app.xml (creator "MyCloudy"). NO external pptx
    44→ *     library — and the parts are shaped exactly for the mirror tool
    45→ *     powerpoint-to-pdf's parsePptx to read back (the smoke proves it).
    46→ *
    47→ * Honest limits (mirrored in config.ts + the surface):
    48→ *  - text mode carries TEXT ONLY: vector art, fills, images and
    49→ *    tables come out as plain text boxes at approximated positions;
    50→ *  - page-images mode renders each page as one full-bleed PNG —
    51→ *    pixel-exact look, nothing editable;
    52→ *  - scanned pages in text mode produce empty slides (never a fake
    53→ *    failure) — counted honestly in the run detail.
    54→ */

   169→const WINANSI_EXTRA = new Set([
   170→  0x20ac, 0x201a, 0x0192, 0x201e, 0x2026, 0x2020, 0x2021, 0x02c6, 0x2030,
   171→  0x0160, 0x2039, 0x0152, 0x017d, 0x2018, 0x2019, 0x201c, 0x201d, 0x2022,
   172→  0x2013, 0x2014, 0x02dc, 0x2122, 0x0161, 0x203a, 0x0153, 0x017e, 0x0178,
   173→]);
   174→
   175→const DEVA_RE = /[\u0900-\u097F]/;
   176→
   177→/**
   178→ * true when every char is encodable in WinAnsi (CP1252) minus controls.
   179→ * Not a font-encoding question here — the PPTX is UTF-8 XML; it only
   180→ * decides WHICH typeface name a run should reference.
   181→ */
   182→export function isWinAnsiText(s: string): boolean {
   183→  for (let i = 0; i < s.length; i++) {
   184→    const c = s.charCodeAt(i);
   185→    if (c === 0x0a || c === 0x09) continue;
   186→    if (c >= 0x20 && c <= 0x7e) continue;
   187→    if (c >= 0xa0 && c <= 0xff) continue;
   188→    if (WINANSI_EXTRA.has(c)) continue;
   189→    return false;
   190→  }
   191→  return true;
   192→}

   194→/** The typeface a run references: Devanagari → the Noto face,
   195→ *  everything else → Calibri (no embedding, just the name). */
   196→export function typefaceFor(text: string): string {
   197→  return !isWinAnsiText(text) && DEVA_RE.test(text)
   198→    ? "Noto Sans Devanagari"
   199→    : "Calibri";
   200→}

   205→/* ------------------------------------------------------------------ */
   206→/* Devanagari visual→logical repair (pdf-to-excel copy)                 */
   207→/* ------------------------------------------------------------------ */
   208→
   209→/* pdf.js concatenates per-glyph ToUnicode in stream order, and shaping
   210→ * puts the three PRE-base matras (ि े ै) BEFORE their base cluster —
   211→ * so text drawn from shaped glyph streams (pdf-lib/Noto, many cheap
   212→ * generators) extracts with "िप्रया" instead of "प्रिया". The inverse
   213→ * move is only applied where it is PROVABLY visual: the matra starts
   214→ * the run, or follows anything that can never end a logical syllable
   215→ * (space, vowel, another sign, virama, Latin…). After a plain
   216→ * CONSONANT the reading is ambiguous and is left untouched — OCR-style
   217→ * logical text is never damaged. */
   218→
   219→const DEV_PRE_MATRA = new Set(["\u093F", "\u0947", "\u0948"]); // ि े ै
   220→const DEV_VIRAMA = "\u094D";
   221→const DEV_NUKTA = "\u093C";
   222→
   223→function isDevConsonant(ch: string | undefined): boolean {
   224→  if (ch === undefined) return false;
   225→  const c = ch.charCodeAt(0);
   226→  return (
   227→    (c >= 0x0915 && c <= 0x0939) || (c >= 0x0958 && c <= 0x095f) ||
   228→    c === 0x0979 || c === 0x097a
   229→  );
   230→}
   231→
   232→/** Greedy match of a base/conjunct cluster: C (nukta?) (virama C (nukta?))*. */
   233→function matchDevCluster(text: string, i: number): string | null {
   234→  let j = i;
   235→  if (!isDevConsonant(text[j])) return null;
   236→  j += 1;
   237→  if (text[j] === DEV_NUKTA) j += 1;
   238→  for (;;) {
   239→    if (text[j] !== DEV_VIRAMA || !isDevConsonant(text[j + 1])) break;
   240→    j += 2;
   241→    if (text[j] === DEV_NUKTA) j += 1;
   242→  }
   243→  return text.slice(i, j);
   244→}
   245→
   246→/** Move provably-visual pre-base matras behind their cluster. */
   247→export function fixDevanagariVisual(text: string): string {
   248→  let needsFix = false;
   249→  for (const ch of text) {
   250→    if (DEV_PRE_MATRA.has(ch)) {
   251→      needsFix = true;
   252→      break;
   253→    }
   254→  }
   255→  if (!needsFix) return text;
   256→  let out = "";
   257→  let prev = "";
   258→  let i = 0;
   259→  while (i < text.length) {
   260→    const ch = text[i] as string;
   261→    if (DEV_PRE_MATRA.has(ch)) {
   262→      const cluster = matchDevCluster(text, i + 1);
   263→      const ambiguous =
   264→        isDevConsonant(prev) || prev === DEV_NUKTA; // logical reading exists
   265→      if (cluster !== null && !ambiguous) {
   266→        out += cluster + ch;
   267→        prev = cluster[cluster.length - 1] as string;
   268→        i += 1 + cluster.length;
   269→        continue;
   270→      }
   271→    }
   272→    out += ch;
   273→    prev = ch;
   274→    i += 1;
   275→  }
   276→  return out;
   277→}

   296→/** Strips C0/C1 controls, collapses whitespace to single spaces. */
   297→export function sanitizeRunText(text: string): string {
   298→  let out = "";
   299→  for (const ch of text) {
   300→    const code = ch.codePointAt(0) ?? 0;
   301→    if (code >= 28 && code <= 32) out += " ";
   302→    else if (code < 32 || (code >= 127 && code <= 159)) continue;
   303→    else out += ch;
   304→  }
   305→  return out.replace(/\s+/g, " ").trim();
   306→}

   364→interface Piece {
   365→  text: string;
   366→  x: number;
   367→  y: number;
   368→  w: number;
   369→  size: number;
   370→  bold: boolean;
   371→}
   372→
   373→/** Coerces one page of raw pdf.js items into position-tagged pieces. */
   374→function collectPieces(items: readonly PdfTextItemInput[]): Piece[] {
   375→  const pieces: Piece[] = [];
   376→  for (const item of items) {
   377→    if (typeof item !== "object" || item === null) continue;
   378→    const str = typeof item.str === "string" ? item.str : "";
   379→    if (str.length === 0) continue;
   380→    const transform = item.transform;
   381→    if (!Array.isArray(transform) || transform.length < 6) continue;
   382→    const x = Number(transform[4]);
   383→    const y = Number(transform[5]);
   384→    if (!Number.isFinite(x) || !Number.isFinite(y)) continue;
   385→    const size = Math.hypot(
   386→      Number(transform[1]) || 0,
   387→      Number(transform[3]) || 0
   388→    );
   389→    const w = Number(item.width) || 0;
   390→    const fontName = typeof item.fontName === "string" ? item.fontName : "";
   391→    const bold = fontName.length > 0 && BOLD_NAME_RE.test(fontName);
   392→    // sanitizeRunText trims the ends — but pdf.js drops its own split
   393→    // items exactly there (whitespace-only items carry the visual
   394→    // spaces). The raw item's edge whitespace is restored as a single
   395→    // space; a whitespace-only item becomes exactly ONE space piece
   396→    // (never two glued ends — the RB1 lesson).
   397→    const text = sanitizeRunText(str);
   398→    const glued =
   399→      text.length > 0
   400→        ? (/^\s/.test(str) ? " " : "") +
   401→          text +
   402→          (/\s$/.test(str) ? " " : "")
   403→        : " ";
   404→    if (glued.length === 0) continue;
   405→    pieces.push({
   406→      text: glued,
   407→      x,
   408→      y,
   409→      w,
   410→      size: size > 0 ? size : 10,
   411→      bold,
   412→    });
   413→  }
   414→  return pieces;
   415→}

   436→/** Rounded font size (0.5 pt buckets) for stable comparisons. */
   437→function roundSize(size: number): number {
   438→  return Math.round(size * 2) / 2;
   439→}

   441→/** Groups pieces into visual lines with styled runs. */
   442→function buildLines(pieces: Piece[]): Line[] {
   443→  if (pieces.length === 0) return [];
   444→  const sorted = [...pieces].sort((a, b) => b.y - a.y || a.x - b.x);
   445→
   446→  const groups: Piece[][] = [];
   447→  let current: Piece[] = [sorted[0] as Piece];
   448→  let groupY = (sorted[0] as Piece).y;
   449→  let groupTol = Math.max(1.5, (sorted[0] as Piece).size * LINE_Y_TOL);
   450→
   451→  for (let i = 1; i < sorted.length; i += 1) {
   452→    const piece = sorted[i] as Piece;
   453→    if (Math.abs(piece.y - groupY) <= groupTol) {
   454→      current.push(piece);
   455→    } else {
   456→      groups.push(current);
   457→      current = [piece];
   458→      groupY = piece.y;
   459→      groupTol = Math.max(1.5, piece.size * LINE_Y_TOL);
   460→    }
   461→  }
   462→  groups.push(current);
   463→
   464→  return groups.map((group) => {
   465→    const ordered = [...group].sort((a, b) => a.x - b.x);
   466→    let x = ordered[0]?.x ?? 0;
   467→    let endX = x;
   468→    let size = 0;
   469→    const runs: LineRun[] = [];
   470→    let lastEnd: number | null = null;
   471→    for (const piece of ordered) {
   472→      x = Math.min(x, piece.x);
   473→      endX = Math.max(endX, piece.x + piece.w);
   474→      size = Math.max(size, piece.size);
   475→      const typeface = typefaceFor(piece.text);
   476→      const last = runs[runs.length - 1];
   477→      if (
   478→        lastEnd !== null &&
   479→        piece.x - lastEnd > piece.size * SPACE_GAP &&
   480→        !last?.text.endsWith(" ")
   481→      ) {
   482→        // pdf.js items carry no spaces — a real gap becomes one.
   483→        if (last) last.text += " ";
   484→        else
   485→          runs.push({
   486→            text: " ",
   487→            bold: piece.bold,
   488→            size: roundSize(piece.size),
   489→            typeface,
   490→          });
   491→      }
   492→      if (
   493→        last &&
   494→        last.bold === piece.bold &&
   495→        last.size === roundSize(piece.size) &&
   496→        last.typeface === typeface
   497→      ) {
   498→        last.text += piece.text;
   499→      } else {
   500→        runs.push({
   501→          text: piece.text,
   502→          bold: piece.bold,
   503→          size: roundSize(piece.size),
   504→          typeface,
   505→        });
   506→      }
   507→      lastEnd = Math.max(lastEnd ?? 0, piece.x + piece.w);
   508→    }
   509→    // Merge identical-style neighbours created by the space glue, then
   510→    // repair visual-ordered Devanagari matras per merged run (the fix
   511→    // needs contiguous text — cluster + matra may straddle pieces).
   512→    const merged: LineRun[] = [];
   513→    for (const run of runs) {
   514→      const prev = merged[merged.length - 1];
   515→      if (
   516→        prev &&
   517→        prev.bold === run.bold &&
   518→        prev.size === run.size &&
   519→        prev.typeface === run.typeface
   520→      ) {
   521→        prev.text += run.text;
   522→      } else {
   523→        merged.push({ ...run, text: fixDevanagariVisual(run.text) });
   524→      }
   525→    }
   526→    return {
   527→      runs: merged.filter((run) => run.text.trim().length > 0),
   528→      y: group[0]?.y ?? 0,
   529→      x,
   530→      endX,
   531→      size: roundSize(size),
   532→    };
   533→  });
   534→}

   542→  x: number;
   543→  endX: number;
   544→  /** dominant (char-weighted, 0.5 pt buckets) size. */
   545→  size: number;
   546→}
   547→
   548→/** Char-weighted dominant size of a block. */
   549→function dominantSize(lines: readonly Line[]): number {
   550→  let weighted = 0;
   551→  let total = 0;
   552→  for (const line of lines) {
   553→    weighted += line.size * line.runs.reduce((n, r) => n + r.text.length, 0);
   554→    total += line.runs.reduce((n, r) => n + r.text.length, 0);
   555→  }
   556→  if (total === 0) return 10;
   557→  return Math.round((weighted / total) * 2) / 2;
   558→}
   559→
   560→/** Merges lines into blocks: y-proximity + font-size similarity. */
   561→function buildBlocks(lines: readonly Line[]): Block[] {
   562→  const blocks: Block[] = [];
   563→  let current: Line[] = [];
   564→  let prevY: number | null = null;
   565→  let prevSize = 10;
   566→
   567→  const flush = (): void => {
   568→    if (current.length === 0) return;
   569→    blocks.push({
   570→      lines: current,
   571→      x: Math.min(...current.map((line) => line.x)),
   572→      endX: Math.max(...current.map((line) => line.endX)),
   573→      size: dominantSize(current),
   574→    });
   575→    current = [];
   576→  };
   577→
   578→  // lines arrive top-of-page first (y descends).
   579→  for (const line of lines) {
   580→    if (line.runs.length === 0) continue;
   581→    if (current.length === 0) {
   582→      current = [line];
   583→    } else if (prevY !== null && prevY - line.y > BLOCK_GAP * Math.max(prevSize, line.size)) {
   584→      // clear vertical gap → new block
   585→      flush();
   586→      current = [line];
   587→    } else if (
   588→      Math.abs(line.size - prevSize) >
   589→      Math.max(1.5, prevSize * BLOCK_SIZE_TOL)
   590→    ) {
   591→      // font-size change (title above body) → new block
   592→      flush();
   593→      current = [line];
   594→    } else {
   595→      current.push(line);
   596→    }
   597→    prevY = line.y;
   598→    prevSize = line.size;
   599→  }
   600→  flush();
   601→  return blocks;
   602→}
   603→
   604→/** Title = the largest dominant font among blocks whose first line sits
   605→ *  in the top third of the page (ties → the topmost); −1 when none. */
   606→export function detectTitleIndex(
   607→  blocks: readonly { lines: Line[]; size: number }[],
   608→  pageH: number
   609→): number {
   610→  const topThird = pageH * (2 / 3);
   611→  let best = -1;
   612→  let bestSize = -1;
   613→  let bestY = -Infinity;
   614→  for (let i = 0; i < blocks.length; i += 1) {
   615→    const block = blocks[i];
   616→    if (!block) continue;
   617→    const first = block.lines[0];
   618→    const chars = block.lines.reduce(
   619→      (n, line) => n + line.runs.reduce((m, r) => m + r.text.trim().length, 0),
   620→      0
   621→    );
   622→    if (!first || chars < 3) continue;
   623→    if (first.y < topThird) continue; // not in the top third
   624→    if (
   625→      block.size > bestSize ||
   626→      (block.size === bestSize && first.y > bestY)
   627→    ) {
   628→      best = i;
   629→      bestSize = block.size;
   630→      bestY = first.y;
   631→    }
   632→  }
   633→  return best;
   634→}

   640→/** Contain-fit scale + centering offsets (pt space). */
   641→export function fitTransform(
   642→  pageW: number,
   643→  pageH: number,
   644→  slide: SlideSize
   645→): { scale: number; offX: number; offY: number } {
   646→  const slideW = slide.cx / EMU_PER_PT;
   647→  const slideH = slide.cy / EMU_PER_PT;
   648→  const w = pageW > 0 ? pageW : 1;
   649→  const h = pageH > 0 ? pageH : 1;
   650→  const scale = Math.min(slideW / w, slideH / h);
   651→  return {
   652→    scale,
   653→    offX: (slideW - w * scale) / 2,
   654→    offY: (slideH - h * scale) / 2,
   655→  };
   656→}
   657→
   658→/** Scaled + clamped font size in pt (8–96, the pptx family clamp). */
   659→export function scaledFontPt(sizePt: number, scale: number): number {
   660→  const v = sizePt * (Number.isFinite(scale) && scale > 0 ? scale : 1);
   661→  return Math.min(MAX_FONT_PT, Math.max(MIN_FONT_PT, Math.round(v * 100) / 100));
   662→}

   664→/**
   665→ * Converts one page of raw pdf.js items into a slide of positioned
   666→ * text boxes. Pure: the same function feeds the smoke and the worker.
   667→ */
   668→export function extractTextSlide(
   669→  items: readonly PdfTextItemInput[],
   670→  pageW: number,
   671→  pageH: number,
   672→  slide: SlideSize
   673→): SlidePage {
   674→  const pieces = collectPieces(items);
   675→  if (pieces.length === 0) return { boxes: [], empty: true };
   676→  const lines = buildLines(pieces).filter((line) => line.runs.length > 0);
   677→  if (lines.length === 0) return { boxes: [], empty: true };
   678→
   679→  const blocks = buildBlocks(lines);
   680→  const titleIndex = detectTitleIndex(blocks, pageH);
   681→  const { scale, offX, offY } = fitTransform(pageW, pageH, slide);
   682→
   683→  // body width = the remaining column width: block x → the page's
   684→  // content right edge (never narrower than the block's own text).
   685→  let contentRight = Math.min(pageW - 18, pageW * 0.9);
   686→  for (const block of blocks) {
   687→    if (block.endX > contentRight) contentRight = block.endX;
   688→  }
   689→
   690→  const boxes: SlideBox[] = [];
   691→  for (let bi = 0; bi < blocks.length && boxes.length < MAX_BOXES_PER_SLIDE; bi += 1) {
   692→    const block = blocks[bi];
   693→    if (!block) continue;
   694→    const first = block.lines[0] as Line;
   695→    const isTitle = bi === titleIndex;
   696→
   697→    const boxX = emuOff((offX + block.x * scale) * EMU_PER_PT);
   698→    // box top = the first line's ascender top (PDF y is the baseline,
   699→    // growing up; EMU grows down from the slide origin).
   700→    const boxY = emuOff(
   701→      (offY + (pageH - (first.y + first.size * ASCENT)) * scale) * EMU_PER_PT
   702→    );
   703→    const widthPt = Math.max(block.endX - block.x, contentRight - block.x, 12);
   704→    const boxCx = Math.max(MIN_EXT_EMU, emu(widthPt * scale * EMU_PER_PT));
   705→    const heightPt = block.lines.reduce(
   706→      (sum, line) => sum + line.size * LINE_LEADING,
   707→      0
   708→    );
   709→    const boxCy = Math.max(
   710→      MIN_EXT_EMU,
   711→      emu(Math.max(heightPt * scale, first.size * 1.1) * EMU_PER_PT)
   712→    );
   713→
   714→    const paras: SlidePara[] = [];
   715→    for (const line of block.lines) {
   716→      if (paras.length >= MAX_LINES_PER_BOX) break;
   717→      const runs: SlideRun[] = [];
   718→      for (const run of line.runs) {
   719→        const text = run.text.slice(0, MAX_RUN_TEXT);
   720→        if (text.trim().length === 0) continue;
   721→        runs.push({
   722→          text,
   723→          sizePt: scaledFontPt(run.size, scale),
   724→          bold: run.bold || isTitle,
   725→          typeface: run.typeface,
   726→        });
   727→      }
   728→      if (runs.length > 0) paras.push({ runs });
   729→    }
   730→    if (paras.length === 0) continue;
   731→    boxes.push({ x: boxX, y: boxY, cx: boxCx, cy: boxCy, paras, isTitle });
   732→  }
   733→
   734→  return { boxes, empty: false };
   735→}

   956→/**
   957→ * Builds a valid minimal .pptx (Uint8Array) with fflate zipSync — the
   958→ * OOXML writer recipe from pdf-to-excel: [Content_Types].xml with
   959→ * per-part overrides (+ a png Default in images mode), _rels/.rels,
   960→ * presentation + master + layout + theme chain, one slideN.xml (+rels)
   961→ * per page and the docProps pair PowerPoint's strict loader expects.
   962→ * Everything clamped to the caps above.
   963→ */
   964→export function buildPptx(opts: PptxBuildOptions): Uint8Array {
   965→  let mode = opts.mode;
   966→  let count =
   967→    mode === "images"
   968→      ? Math.min(MAX_SLIDES, opts.images?.length ?? 0)
   969→      : Math.min(MAX_SLIDES, opts.pages?.length ?? 0);
   970→  if (count === 0) {
   971→    // an honest empty presentation (one blank slide) — never invalid
   972→    mode = "text";
   973→    count = 1;
   974→  }
   975→  const n = count;
   976→
   977→  /* [Content_Types].xml */
   978→  const ctParts = [
   979→    XML_DECL,
   980→    '<Types xmlns="http://schemas.openxmlformats.org/package/2006/content-types">',
   981→    '<Default Extension="rels" ContentType="application/vnd.openxmlformats-package.relationships+xml"/>',
   982→    '<Default Extension="xml" ContentType="application/xml"/>',
   983→  ];
   984→  if (mode === "images") {
   985→    ctParts.push('<Default Extension="png" ContentType="image/png"/>');
   986→  }
   987→  ctParts.push(
   988→    `<Override PartName="/ppt/presentation.xml" ContentType="${CT_PRESENTATION}"/>`,
   989→    `<Override PartName="/ppt/slideMasters/slideMaster1.xml" ContentType="${CT_SLIDE_MASTER}"/>`,
   990→    `<Override PartName="/ppt/slideLayouts/slideLayout1.xml" ContentType="${CT_SLIDE_LAYOUT}"/>`,
   991→    `<Override PartName="/ppt/theme/theme1.xml" ContentType="${CT_THEME}"/>`
   992→  );
   993→  for (let i = 1; i <= n; i += 1) {
   994→    ctParts.push(
   995→      `<Override PartName="/ppt/slides/slide${i}.xml" ContentType="${CT_SLIDE}"/>`
   996→    );
   997→  }
   998→  ctParts.push(

  1093→/**
  1094→ * coerceRunOptions — the options record the surface/shell sends.
  1095→ * Garbage values degrade to the defaults; unknown enum values fall
  1096→ * back honestly; the page spec is sanitized to a string; the file word
  1097→ * is control-char-stripped + clamped.
  1098→ */
  1099→export function coerceRunOptions(
  1100→  raw: Record<string, unknown>
  1101→): P2PptRunOptions {
  1102→  const str = (v: unknown, max: number): string =>
  1103→    typeof v === "string"
  1104→      ? v.replace(/[\u0000-\u001f]/g, "").trim().slice(0, max)
  1105→      : "";
  1106→  return {
  1107→    mode: raw.mode === "images" ? "images" : "text",
  1108→    slideSize:
  1109→      raw.slideSize === "4-3" || raw.slideSize === "match"
  1110→        ? raw.slideSize
  1111→        : "16-9",
  1112→    pages: str(raw.pages, 200) || "all",
  1113→    locale: raw.locale === "hi" ? "hi" : "en",
  1114→    fileWord: str(raw.fileWord, 40),
  1115→  };
  1116→}

  1122→/**
  1123→ * pdf.js failure message → honest i18n error code. Encrypted PDFs ask

  1126→export function pdfErrorMessageToCode(message: string): string {

  1131→/**
  1132→ * "My Deck.pdf" → "My-Deck" (spaces collapse to hyphens, path-unsafe
  1133→ * chars stripped, Devanagari survives); a nameless file falls back to
  1134→ * the localized word. The caller appends ".pptx".
  1135→ */
  1136→export function outputBaseName(fileName: string, lang: string): string {
  1137→  const base = fileName
  1138→    .replace(/\.pdf$/i, "")
  1139→    .replace(/[\\/:*?"<>|\u0000-\u001f]/g, "")
  1140→    .replace(/\s+/g, "-")
  1141→    .replace(/-{2,}/g, "-")
  1142→    .replace(/^[-.]+|[-.]+$/g, "")
  1143→    .slice(0, 120);
  1144→  return base || (lang === "hi" ? "प्रेज़ेंटेशन" : "Presentation");
  1145→}

    81→    81→/** Coerces raw panel values into typed options. */
    82→    82→export function resolveMdOptions(values: OptionValues): MdOptions {
    83→    83→  const margin = Number(values.margin);
    84→    84→  return {
    85→    85→    pageSize: values.pageSize === "letter" ? "letter" : "a4",
    86→    86→    margin: Number.isFinite(margin)
    87→    87→      ? Math.min(40, Math.max(10, margin))
    88→    88→      : 20,
    89→    89→    bodySize: pickNumber(values.bodySize, [10, 11, 12], 11),
    90→    90→    monoSize: pickNumber(values.monoSize, [8, 9, 10], 9),
    91→    91→    theme: values.theme === "warm" ? "warm" : "clean",
    92→    92→  };
    93→    93→}

   136→   136→function baseRun(text: string): MdInlineRun {
   137→   137→  return {
   138→   138→    text,
   139→   139→    bold: false,
   140→   140→    italic: false,
   141→   141→    code: false,
   142→   142→    link: false,

   147→   147→function sameStyle(a: MdInlineRun, b: MdInlineRun): boolean {
   148→   148→  return (
   149→   149→    a.bold === b.bold &&
   150→   150→    a.italic === b.italic &&
   151→   151→    a.code === b.code &&
   152→   152→    a.link === b.link &&
   153→   153→    a.sup === b.sup
   154→   154→  );
   155→   155→}

   157→   157→/** Merges adjacent runs with identical style (fewer draw calls). */
   158→   158→function mergeRuns(runs: MdInlineRun[]): MdInlineRun[] {
   159→   159→  const out: MdInlineRun[] = [];
   160→   160→  for (const run of runs) {
   161→   161→    if (run.text.length === 0) continue;
   162→   162→    const last = out[out.length - 1];
   163→   163→    if (
   164→   164→      last &&
   165→   165→      sameStyle(last, run) &&
   166→   166→      last.text !== "\n" &&
   167→   167→      run.text !== "\n"
   168→   168→    ) {
   169→   169→      last.text += run.text;
   170→   170→    } else {
   171→   171→      out.push({ ...run });
   172→   172→    }
   173→   173→  }
   174→   174→  return out;
   175→   175→}

   177→   177→interface InlineStyle {
   178→   178→  bold: boolean;
   179→   179→  italic: boolean;
   180→   180→  link: boolean;
   181→   181→}
   182→   182→
   183→   183→function walkPhrasing(
   184→   184→  nodes: readonly PhrasingContent[],
   185→   185→  style: InlineStyle,
   186→   186→  ctx: ParseCtx,
   187→   187→  out: MdInlineRun[]
   188→   188→): void {
   189→   189→  for (const node of nodes) {
   190→   190→    switch (node.type) {
   191→   191→      case "text":
   192→   192→        out.push({
   193→   193→          ...baseRun(node.value),
   194→   194→          bold: style.bold,
   195→   195→          italic: style.italic,
   196→   196→          link: style.link,
   197→   197→        });
   198→   198→        break;
   199→   199→      case "emphasis":
   200→   200→        walkPhrasing(
   201→   201→          node.children,
   202→   202→          { ...style, italic: true },
   203→   203→          ctx,
   204→   204→          out
   205→   205→        );
   206→   206→        break;
   207→   207→      case "strong":
   208→   208→        walkPhrasing(node.children, { ...style, bold: true }, ctx, out);
   209→   209→        break;
   210→   210→      case "inlineCode":
   211→   211→        out.push({ ...baseRun(node.value), code: true });
   212→   212→        break;
   213→   213→      case "break":
   214→   214→        out.push(baseRun("\n"));
   215→   215→        break;
   216→   216→      case "link": {
   217→   217→        walkPhrasing(node.children, { ...style, link: true }, ctx, out);
   218→   218→        const n = ctx.links.push(node.url);
   219→   219→        out.push({
   220→   220→          ...baseRun(` [${n}]`),
   221→   221→          link: true,
   222→   222→          sup: true,
   223→   223→        });
   224→   224→        break;
   225→   225→      }
   226→   226→      case "linkReference": {
   227→   227→        const url = ctx.defs.get(node.identifier);
   228→   228→        if (url !== undefined) {
   229→   229→          walkPhrasing(node.children, { ...style, link: true }, ctx, out);
   230→   230→          const n = ctx.links.push(url);
   231→   231→          out.push({ ...baseRun(` [${n}]`), link: true, sup: true });
   232→   232→        } else {
   233→   233→          out.push(baseRun(`[${node.label ?? node.identifier}]`));
   234→   234→        }
   235→   235→        break;
   236→   236→      }
   237→   237→      case "image":
   238→   238→        out.push({ ...baseRun(`[${node.alt ?? ""}]`), italic: true });
   239→   239→        break;
   240→   240→      case "imageReference": {
   241→   241→        const url = ctx.defs.get(node.identifier);
   242→   242→        const alt = `[${node.alt ?? ""}]`;
   243→   243→        if (url !== undefined) {
   244→   244→          const n = ctx.links.push(url);
   245→   245→          out.push({ ...baseRun(`${alt} [${n}]`), link: true, sup: true });
   246→   246→        } else {
   247→   247→          out.push({ ...baseRun(alt), italic: true });
   248→   248→        }
   249→   249→        break;
   250→   250→      }
   251→   251→      case "html":
   252→   252→        // inline raw HTML — skipped (honest omission, see FAQ)
   253→   253→        break;
   254→   254→      default:
   255→   255→        break;
   256→   256→    }
   257→   257→  }
   258→   258→}
   259→   259→
   260→   260→function inlineRuns(
   261→   261→  nodes: readonly PhrasingContent[],
   262→   262→  ctx: ParseCtx
   263→   263→): MdInlineRun[] {
   264→   264→  const out: MdInlineRun[] = [];
   265→   265→  walkPhrasing(nodes, { bold: false, italic: false, link: false }, ctx, out);
   266→   266→  return mergeRuns(out);
   267→   267→}

   269→   269→/** Collects reference definitions anywhere in the document (label → url). */
   270→   270→function collectDefinitions(
   271→   271→  nodes: readonly RootContent[],
   272→   272→  into: Map<string, string>
   273→   273→): void {
   274→   274→  for (const node of nodes) {
   275→   275→    if (node.type === "definition") {
   276→   276→      into.set(node.identifier, node.url);
   277→   277→    } else if ("children" in node && Array.isArray(node.children)) {
   278→   278→      collectDefinitions(
   279→   279→        node.children as RootContent[],
   280→   280→        into
   281→   281→      );
   282→   282→    }
   283→   283→  }
   284→   284→}
   285→   285→
   286→   286→function codeLines(value: string): string[] {
   287→   287→  const cleaned = value.replace(/\r\n?/g, "\n").replace(/\t/g, "    ");
   288→   288→  const lines = cleaned.split("\n");
   289→   289→  // fenced blocks implicitly end with a newline — drop that phantom line
   290→   290→  if (lines.length > 1 && lines[lines.length - 1] === "") lines.pop();
   291→   291→  return lines;
   292→   292→}
   293→   293→
   294→   294→function walkBlocks(
   295→   295→  nodes: readonly BlockContent[] | readonly RootContent[],
   296→   296→  ctx: ParseCtx,
   297→   297→  out: MdBlock[],
   298→   298→  depth: number
   299→   299→): void {
   300→   300→  for (const node of nodes) {
   301→   301→    switch (node.type) {
   302→   302→      case "heading":
   303→   303→        out.push({
   304→   304→          kind: "heading",
   305→   305→          level: node.depth,
   306→   306→          runs: inlineRuns(node.children, ctx),
   307→   307→        });
   308→   308→        break;
   309→   309→      case "paragraph":
   310→   310→        out.push({
   311→   311→          kind: "paragraph",
   312→   312→          runs: inlineRuns(node.children, ctx),
   313→   313→        });
   314→   314→        break;
   315→   315→      case "code":
   316→   316→        out.push({ kind: "code", lines: codeLines(node.value) });
   317→   317→        break;
   318→   318→      case "thematicBreak":
   319→   319→        out.push({ kind: "rule" });
   320→   320→        break;
   321→   321→      case "blockquote":
   322→   322→        out.push({
   323→   323→          kind: "quote",
   324→   324→          children: walkBlocksInto(node.children, ctx),
   325→   325→        });
   326→   326→        break;
   327→   327→      case "list": {
   328→   328→        const list = node as List;
   329→   329→        const start = typeof list.start === "number" ? list.start : 1;
   330→   330→        const hoisted: MdBlock[] = [];
   331→   331→        list.children.forEach((item, index) => {
   332→   332→          hoisted.push(...listItemBlocks(item, ctx, depth, list.ordered === true, start + index));
   333→   333→        });
   334→   334→        out.push(...hoisted);
   335→   335→        break;
   336→   336→      }
   337→   337→      case "html":
   338→   338→      case "definition":
   339→   339→        // block raw HTML / reference definitions — skipped (see FAQ)
   340→   340→        break;
   341→   341→      default:
   342→   342→        break;
   343→   343→    }
   344→   344→  }
   345→   345→}
   346→   346→
   347→   347→/**
   348→   348→ * One mdast listItem → its own block (phrasing content merged with hard
   349→   349→ * breaks) + nested lists flattened as deeper item blocks. Block-level
   350→   350→ * code/quote/heading inside an item is hoisted as a top-level block right
   351→   351→ * after the list (a documented v1 simplification — rare in the wild).
   352→   352→ */
   353→   353→function listItemBlocks(
   354→   354→  item: ListItem,
   355→   355→  ctx: ParseCtx,
   356→   356→  depth: number,
   357→   357→  ordered: boolean,
   358→   358→  number: number
   359→   359→): MdBlock[] {
   360→   360→  const out: MdBlock[] = [];
   361→   361→  const runs: MdInlineRun[] = [];
   362→   362→  const after: MdBlock[] = [];
   363→   363→  for (const child of item.children) {
   364→   364→    if (child.type === "paragraph") {
   365→   365→      if (runs.length > 0) runs.push(baseRun("\n"));
   366→   366→      runs.push(...inlineRuns(child.children, ctx));
   367→   367→    } else if (child.type === "list") {
   368→   368→      const list = child as List;
   369→   369→      const start = typeof list.start === "number" ? list.start : 1;
   370→   370→      list.children.forEach((nested, index) => {
   371→   371→        after.push(
   372→   372→          ...listItemBlocks(
   373→   373→            nested,
   374→   374→            ctx,
   375→   375→            depth + 1,
   376→   376→            list.ordered === true,
   377→   377→            start + index
   378→   378→          )
   379→   379→        );
   380→   380→      });
   381→   381→    } else if (child.type === "blockquote") {
   382→   382→      after.push({
   383→   383→        kind: "quote",
   384→   384→        children: walkBlocksInto(child.children, ctx),
   385→   385→      });
   386→   386→    } else if (child.type === "code") {
   387→   387→      after.push({ kind: "code", lines: codeLines(child.value) });
   388→   388→    } else if (child.type === "heading") {
   389→   389→      after.push({
   390→   390→        kind: "heading",
   391→   391→        level: child.depth,
   392→   392→        runs: inlineRuns(child.children, ctx),
   393→   393→      });
   394→   394→    } else if (child.type === "thematicBreak") {
   395→   395→      after.push({ kind: "rule" });
   396→   396→    }
   397→   397→    // html/definition inside items: skipped like everywhere else
   398→   398→  }
   399→   399→  if (runs.length > 0) {
   400→   400→    out.push({ kind: "listItem", depth, ordered, number, runs: mergeRuns(runs) });
   401→   401→  }
   402→   402→  out.push(...after);
   403→   403→  return out;
   404→   404→}
   405→   405→
   406→   406→function walkBlocksInto(
   407→   407→  nodes: readonly BlockContent[],
   408→   408→  ctx: ParseCtx
   409→   409→): MdBlock[] {
   410→   410→  const out: MdBlock[] = [];
   411→   411→  walkBlocks(nodes, ctx, out, 0);
   412→   412→  return out;
   413→   413→}

   415→   415→/** Parses markdown source into the document model + numbered links. */
   416→   416→export function parseMarkdown(source: string): ParsedMarkdown {
   417→   417→  const root = fromMarkdown(source);
   418→   418→  const ctx: ParseCtx = { links: [], defs: new Map() };
   419→   419→  collectDefinitions(root.children, ctx.defs);
   420→   420→  const blocks: MdBlock[] = [];
   421→   421→  walkBlocks(root.children, ctx, blocks, 0);
   422→   422→  return { blocks, links: ctx.links };
   423→   423→}

   499→   499→        HEADING_LINE_HEIGHT

   504→   504→    case "paragraph":
   505→   505→    case "listItem":
   506→   506→    case "rule":
   507→   507→      return bodyAdvance;
   508→   508→  }

   511→   511→export function layoutMd(
   512→   512→  parsed: ParsedMarkdown,
   513→   513→  options: MdOptions,

   518→   518→  const marginPt = options.margin * MM_TO_PT;
   519→   519→  const geometry: PageGeometry = {
   520→   520→    pageW: size.width,
   521→   521→    pageH: size.height,
   522→   522→    marginLeft: marginPt,
   523→   523→    marginRight: marginPt,
   524→   524→    marginTop: marginPt,
   525→   525→    marginBottom: marginPt,
   526→   526→  };
   527→   527→  const flow = new PageFlow(geometry);
   528→   528→  const pages: MdPageDraw[] = [
   529→   529→    { runs: [], rects: [], warm: options.theme === "warm" },
   530→   530→  ];
   531→   531→  const ensurePage = (index: number): MdPageDraw => {
   532→   532→    while (pages.length <= index) {
   533→   533→      pages.push({ runs: [], rects: [], warm: options.theme === "warm" });
   534→   534→    }
   535→   535→    return pages[index];
   536→   536→  };
   537→   537→
   538→   538→  const bodyAdvance = options.bodySize * BODY_LINE_HEIGHT;
   539→   539→  const monoAdvance = options.monoSize * MONO_LINE_HEIGHT;
   540→   540→  const monoSpec: FontSpec = { family: "mono", bold: false, italic: false };
   541→   541→  const sansSpec: FontSpec = { family: "sans", bold: false, italic: false };
   542→   542→
   543→   543→  /** Open quote extents, innermost last: per-page line extents. */
   544→   544→  const quoteStack: {
   545→   545→    extents: { page: number; top: number; bottom: number }[];
   546→   546→  }[] = [];
   547→   547→
   548→   548→  let indent = 0;
   549→   549→  let muted = false;
   550→   550→  let gapScale = 1;
   551→   551→  let placedAny = false;
   552→   552→
   553→   553→  const contentLeft = (): number => geometry.marginLeft + indent;
   554→   554→
   555→   555→  const toWrapRuns = (runs: readonly MdInlineRun[]): WrapRun[] => {
   556→   556→    const out: WrapRun[] = [];
   557→   557→    for (const run of runs) {
   558→   558→      const color: RunColor = run.link
   559→   559→        ? "link"
   560→   560→        : muted
   561→   561→          ? "muted"
   562→   562→          : "text";
   563→   563→      out.push({
   564→   564→        text: run.text,
   565→   565→        spec: fontSpecFor(run),
   566→   566→        color,
   567→   567→        sup: run.sup,
   568→   568→      });
   569→   569→    }
   570→   570→    return out;
   571→   571→  };
   572→   572→
   573→   573→  /**
   574→   574→   * Wraps one run group (hard-break segments flow without extra gaps) and
   575→   575→   * places it with orphan/widow protection. Registers extents with every
   576→   576→   * open quote so their bars size correctly across page breaks.
   577→   577→   */
   578→   578→  const placeRuns = (
   579→   579→    runs: readonly MdInlineRun[],
   580→   580→    x: number,
   581→   581→    maxWidth: number,
   582→   582→    fontSize: number,
   583→   583→    advance: number
   584→   584→  ): { page: number; top: number; bottom: number; baseline: number }[] => {
   585→   585→    const script = blockScript(runsText(runs));
   586→   586→    const segments = splitRunsAtBreaks(toWrapRuns(runs));
   587→   587→    const allLines: WrapRun[][] = [];
   588→   588→    for (const segment of segments) {
   589→   589→      if (segment.length === 0) {
   590→   590→        allLines.push([]);
   591→   591→        continue;
   592→   592→      }
   593→   593→      allLines.push(
   594→   594→        ...wrapRuns(segment, { maxWidth, size: fontSize, script, measure })
   595→   595→      );
   596→   596→    }
   597→   597→    const placements = planLineGroup(flow, allLines.length, advance, fontSize);
   598→   598→    const extents: {
   599→   599→      page: number;
   600→   600→      top: number;
   601→   601→      bottom: number;
   602→   602→      baseline: number;
   603→   603→    }[] = [];
   604→   604→    for (let i = 0; i < placements.length; i += 1) {
   605→   605→      const place = placements[i];
   606→   606→      const page = ensurePage(place.pageIndex);
   607→   607→      page.runs.push(
   608→   608→        ...positionRuns(allLines[i], x, place.baseline, fontSize, script, measure)
   609→   609→      );
   610→   610→      const extent = {
   611→   611→        page: place.pageIndex,
   612→   612→        top: place.top,
   613→   613→        bottom: place.top + advance,
   614→   614→        baseline: place.baseline,
   615→   615→      };
   616→   616→      extents.push(extent);
   617→   617→      for (const open of quoteStack) open.extents.push(extent);
   618→   618→    }
   619→   619→    return extents;
   620→   620→  };
   621→   621→
   622→   622→  const gapBefore = (gap: number): void => {
   623→   623→    if (!placedAny) return;
   624→   624→    flow.consumeGap(gap * gapScale);
   625→   625→  };
   626→   626→
   627→   627→  const placeCodeBlock = (lines: string[]): void => {
   628→   628→    if (lines.length === 0) return;
   629→   629→    const blockX = contentLeft();
   630→   630→    const height = lines.length * monoAdvance + 2 * CODE_PAD;
   631→   631→    // ≤3-line blocks never split across pages (when a full page could hold
   632→   632→    // them); longer blocks split without 1-line fragments (strictSingle)
   633→   633→    if (
   634→   634→      lines.length <= 3 &&
   635→   635→      flow.pageHasContent &&
   636→   636→      !flow.fits(height) &&
   637→   637→      height <= flow.contentHeight
   638→   638→    ) {
   639→   639→      flow.breakPage();
   640→   640→    }
   641→   641→    gapBefore(bodyAdvance * BLOCK_GAP);
   642→   642→
   643→   643→    // Wrapped + positioned per source line (code wraps only overlong
   644→   644→    // lines; indentation is preserved as an x offset per line)
   645→   645→    const indentWidths = lines.map((line) => {
   646→   646→      const indentMatch = /^[ ]*/.exec(line);
   647→   647→      const indentText = indentMatch ? indentMatch[0] : "";
   648→   648→      return {
   649→   649→        text: indentText,
   650→   650→        width: measure(indentText, monoSpec, options.monoSize, "std"),
   651→   651→        body: line.slice(indentText.length).replace(/[ ]+$/, ""),
   652→   652→      };
   653→   653→    });
   654→   654→    const maxIndent = indentWidths.reduce(
   655→   655→      (max, entry) => Math.max(max, entry.width),
   656→   656→      0
   657→   657→    );
   658→   658→    const wrapWidth = Math.max(
   659→   659→      24,
   660→   660→      flow.contentWidth - indent - 2 * CODE_PAD - maxIndent
   661→   661→    );
   662→   662→    const wrappedLines: { runs: WrapRun[]; script: ScriptFont; indent: number }[] =
   663→   663→      indentWidths.map((entry) => {
   664→   664→        const script = blockScript(entry.text + entry.body || " ");
   665→   665→        const wrapped = wrapRuns([{ text: entry.body, spec: monoSpec }], {
   666→   666→          maxWidth: wrapWidth,
   667→   667→          size: options.monoSize,
   668→   668→          script,
   669→   669→          measure,
   670→   670→        });
   671→   671→        return {
   672→   672→          runs: wrapped[0] ?? [],
   673→   673→          script,
   674→   674→          indent: entry.width,
   675→   675→        };
   676→   676→      });
   677→   677→
   678→   678→    const placements = planLineGroup(
   679→   679→      flow,
   680→   680→      wrappedLines.length,
   681→   681→      monoAdvance,
   682→   682→      options.monoSize,
   683→   683→      true,
   684→   684→      CODE_PAD
   685→   685→    );
   686→   686→    const fragmentByPage = new Map<number, { top: number; count: number }>();
   687→   687→    for (let i = 0; i < placements.length; i += 1) {
   688→   688→      const place = placements[i];
   689→   689→      const fragment = fragmentByPage.get(place.pageIndex);
   690→   690→      if (fragment) {
   691→   691→        fragment.count += 1;
   692→   692→      } else {
   693→   693→        fragmentByPage.set(place.pageIndex, { top: place.top, count: 1 });
   694→   694→      }
   695→   695→      const line = wrappedLines[i];
   696→   696→      const page = ensurePage(place.pageIndex);
   697→   697→      page.runs.push(
   698→   698→        ...positionRuns(
   699→   699→          line.runs,
   700→   700→          blockX + CODE_PAD + line.indent,
   701→   701→          place.baseline,
   702→   702→          options.monoSize,
   703→   703→          line.script,
   704→   704→          measure
   705→   705→        )
   706→   706→      );
   707→   707→      const extent = {
   708→   708→        page: place.pageIndex,
   709→   709→        top: place.top,
   710→   710→        bottom: place.top + monoAdvance,
   711→   711→      };
   712→   712→      for (const open of quoteStack) open.extents.push(extent);
   713→   713→    }
   714→   714→    for (const [pageIndex, fragment] of fragmentByPage) {
   715→   715→      const page = ensurePage(pageIndex);
   716→   716→      // no top padding when the fragment starts at the content top — the
   717→   717→      // page margin already provides the gap there
   718→   718→      const topPad = fragment.top > 0 ? CODE_PAD : 0;
   719→   719→      const topY =
   720→   720→        geometry.pageH - geometry.marginTop - fragment.top + topPad;
   721→   721→      page.rects.push({
   722→   722→        x: blockX,
   723→   723→        y: topY - fragment.count * monoAdvance - topPad - CODE_PAD,
   724→   724→        w: flow.contentWidth - indent,
   725→   725→        h: fragment.count * monoAdvance + topPad + CODE_PAD,
   726→   726→        kind: "codeBg",
   727→   727→      });
   728→   728→    }
   729→   729→  };
   730→   730→
   731→   731→  const placeBlock = (
   732→   732→    block: MdBlock,
   733→   733→    next: MdBlock | null
   734→   734→  ): void => {
   735→   735→    const bodyGap = bodyAdvance * BLOCK_GAP;
   736→   736→    switch (block.kind) {
   737→   737→      case "heading": {
   738→   738→        const level = Math.min(6, Math.max(1, block.level));
   739→   739→        const fontSize = HEADING_SIZES[level - 1];
   740→   740→        const advance = fontSize * HEADING_LINE_HEIGHT;
   741→   741→        // keep-with-next: heading + ~2 lines of the following block, else
   742→   742→        // the heading never dangles at the very bottom of a page
   743→   743→        const nextAdvance = next
   744→   744→          ? firstLineAdvance(next, bodyAdvance, monoAdvance)
   745→   745→          : bodyAdvance;
   746→   746→        const needed = advance + 2 * nextAdvance;
   747→   747→        if (flow.pageHasContent && !flow.fits(needed)) flow.breakPage();
   748→   748→        gapBefore(HEADING_SPACING[level - 1][0]);
   749→   749→        const headingRuns: MdInlineRun[] = block.runs.map((run) => ({
   750→   750→          ...run,
   751→   751→          bold: true,
   752→   752→          italic: level === 6,
   753→   753→          code: false,
   754→   754→        }));
   755→   755→        placeRuns(
   756→   756→          headingRuns,
   757→   757→          contentLeft(),
   758→   758→          flow.contentWidth - indent,
   759→   759→          fontSize,
   760→   760→          advance
   761→   761→        );
   762→   762→        gapBefore(HEADING_SPACING[level - 1][1]);
   763→   763→        break;
   764→   764→      }
   765→   765→      case "paragraph": {
   766→   766→        if (block.runs.length === 0) return;
   767→   767→        gapBefore(bodyGap);
   768→   768→        placeRuns(
   769→   769→          block.runs,
   770→   770→          contentLeft(),
   771→   771→          flow.contentWidth - indent,
   772→   772→          options.bodySize,
   773→   773→          bodyAdvance
   774→   774→        );
   775→   775→        break;
   776→   776→      }
   777→   777→      case "code": {
   778→   778→        placeCodeBlock(block.lines);
   779→   779→        break;
   780→   780→      }
   781→   781→      case "rule": {
   782→   782→        gapBefore(bodyGap);
   783→   783→        if (flow.pageHasContent && !flow.fits(RULE_THICKNESS * 2)) {
   784→   784→          flow.breakPage();
   785→   785→        }
   786→   786→        const page = ensurePage(flow.pageIndex);
   787→   787→        const y = flow.contentTopY() - RULE_THICKNESS;
   788→   788→        page.rects.push({
   789→   789→          x: contentLeft(),
   790→   790→          y,
   791→   791→          w: flow.contentWidth - indent,
   792→   792→          h: RULE_THICKNESS * 2,
   793→   793→          kind: "rule",
   794→   794→        });
   795→   795→        flow.consume(RULE_THICKNESS * 2);
   796→   796→        gapBefore(bodyGap * 0.5);
   797→   797→        break;
   798→   798→      }
   799→   799→      case "quote": {
   800→   800→        gapBefore(bodyGap);
   801→   801→        indent += QUOTE_INDENT;
   802→   802→        muted = true;
   803→   803→        gapScale *= 0.7;
   804→   804→        quoteStack.push({ extents: [] });
   805→   805→        for (let i = 0; i < block.children.length; i += 1) {
   806→   806→          placeBlock(
   807→   807→            block.children[i],
   808→   808→            block.children[i + 1] ?? null
   809→   809→          );
   810→   810→        }
   811→   811→        const open = quoteStack.pop();
   812→   812→        indent -= QUOTE_INDENT;
   813→   813→        muted = false;
   814→   814→        gapScale /= 0.7;
   815→   815→        if (open && open.extents.length > 0) {
   816→   816→          const byPage = new Map<number, { top: number; bottom: number }>();
   817→   817→          for (const extent of open.extents) {
   818→   818→            const span = byPage.get(extent.page);
   819→   819→            if (span) {
   820→   820→              span.top = Math.min(span.top, extent.top);
   821→   821→              span.bottom = Math.max(span.bottom, extent.bottom);
   822→   822→            } else {
   823→   823→              byPage.set(extent.page, {
   824→   824→                top: extent.top,
   825→   825→                bottom: extent.bottom,
   826→   826→              });
   827→   827→            }
   828→   828→          }
   829→   829→          for (const [pageIndex, span] of byPage) {
   830→   830→            const page = ensurePage(pageIndex);
   831→   831→            const topY = geometry.pageH - geometry.marginTop - span.top;
   832→   832→            page.rects.push({
   833→   833→              x: contentLeft(),
   834→   834→              y: topY - span.bottom - QUOTE_BAR_OVERHANG,
   835→   835→              w: QUOTE_BAR_WIDTH,
   836→   836→              h: span.bottom - span.top + 2 * QUOTE_BAR_OVERHANG,
   837→   837→              kind: "quoteBar",
   838→   838→            });
   839→   839→          }
   840→   840→        }
   841→   841→        break;
   842→   842→      }
   843→   843→      case "listItem": {
   844→   844→        gapBefore(bodyAdvance * ITEM_GAP);
   845→   845→        const markerX = contentLeft() + block.depth * LIST_INDENT;
   846→   846→        const marker = block.ordered
   847→   847→          ? `${block.number}.`
   848→   848→          : MARKERS[Math.min(MARKERS.length - 1, block.depth)];
   849→   849→        const markerWidth = measure(
   850→   850→          marker,
   851→   851→          sansSpec,
   852→   852→          options.bodySize,
   853→   853→          "std"
   854→   854→        );
   855→   855→        const textX = markerX + markerWidth + MARKER_GAP;
   856→   856→        const maxWidth = Math.max(
   857→   857→          24,
   858→   858→          geometry.marginLeft + flow.contentWidth - textX
   859→   859→        );
   860→   860→        const extents = placeRuns(
   861→   861→          block.runs,
   862→   862→          textX,
   863→   863→          maxWidth,
   864→   864→          options.bodySize,
   865→   865→          bodyAdvance
   866→   866→        );
   867→   867→        if (extents.length > 0) {
   868→   868→          const page = ensurePage(extents[0].page);
   869→   869→          page.runs.push({
   870→   870→            text: marker,
   871→   871→            x: markerX,
   872→   872→            baseline: extents[0].baseline,
   873→   873→            size: options.bodySize,
   874→   874→            spec: sansSpec,
   875→   875→            script: "std",
   876→   876→            color: muted ? "muted" : "text",
   877→   877→          });
   878→   878→        }
   879→   879→        break;
   880→   880→      }
   881→   881→    }
   882→   882→    if (block.kind !== "paragraph" || block.runs.length > 0) {
   883→   883→      placedAny = true;
   884→   884→    }
   885→   885→  };
   886→   886→
   887→   887→  for (let i = 0; i < blocks.length; i += 1) {
   888→   888→    placeBlock(blocks[i], blocks[i + 1] ?? null);
   889→   889→  }
   890→   890→
   891→   891→  // Links section: bold "Links" heading + one numbered line per URL
   892→   892→  if (links.length > 0) {
   893→   893→    gapBefore(bodyAdvance * BLOCK_GAP);
   894→   894→    placeRuns(
   895→   895→      [{ ...baseRun("Links"), bold: true }],
   896→   896→      contentLeft(),
   897→   897→      flow.contentWidth - indent,
   898→   898→      12,
   899→   899→      12 * HEADING_LINE_HEIGHT
   900→   900→    );
   901→   901→    for (let n = 1; n <= links.length; n += 1) {
   902→   902→      placeRuns(
   903→   903→        [baseRun(`${n}. `), { ...baseRun(links[n - 1]), link: true }],
   904→   904→        contentLeft(),
   905→   905→        flow.contentWidth - indent,
   906→   906→        options.bodySize,
   907→   907→        bodyAdvance
   908→   908→      );
   909→   909→    }
   910→   910→  }
   911→   911→
   912→   912→  return { pages, linkCount: links.length };
   913→   913→}

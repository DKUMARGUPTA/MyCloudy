     3→/**
     4→ * Publisher to PDF surface (RB4-e). .pub → PDF text salvage, 100%
     5→ * on-device, EN + हिंदी.
     6→ *
     7→ * Landing (landingSurface — word-to-pdf pattern): a .pub drop zone
     8→ * that validates the OLE2 compound-file magic itself (the shared
     9→ * DropZone only knows pdf/image/zip), so a renamed ZIP, a text file or
    10→ * a stray binary can never enter the pipeline.
    11→ *
    12→ * Workspace: the file card, a LIVE salvage report from the exact same
    13→ * parsePub the worker runs (streams detected with their text verdicts,
    14→ * paragraphs / characters / ≈pages — the numbers can never disagree
    15→ * with the output), A4/Letter + normal/narrow + font-size chips bound
    16→ * to the shared option values (the aside/bottom-sheet selects and
    17→ * these chips write through one source of truth), the honest
    18→ * text-focused-conversion explainer, and the Convert button that
    19→ * spawns the worker (create-invoice pattern) → download + toast.
    20→ *
    21→ * A .pub that fails parsePub (not OLE2, corrupt container, no text
    22→ * found) shows its honest localized reason inline instead of a
    23→ * generic error.
    24→ *
    25→ * Mobile-safe: min-w-0 on every grid/text wrapper (the pinned
    26→ * grid-blowout lesson). Everything is real DOM text — no canvas.
    27→ */
    28→

    39→  FileDown,

    68→
    69→function isMobileDevice(): boolean {
    70→  if (typeof window === "undefined" || !window.matchMedia) return false;
    71→  return window.matchMedia("(pointer: coarse)").matches;
    72→}
    73→
    74→function sizeLimitBytes(): number {
    75→  return isMobileDevice() ? 50 * 1024 * 1024 : 200 * 1024 * 1024;
    76→}
    77→
    78→function sizeLimitLabel(): string {
    79→  return isMobileDevice() ? "50 MB" : "200 MB";
    80→}
    81→
    82→/** .pub files are OLE2 compound files: D0 CF 11 E0 A1 B1 1A E1. */
    83→async function sniffIsOle2(file: File): Promise<boolean> {
    84→  try {
    85→    const head = new Uint8Array(await file.slice(0, 8).arrayBuffer());
    86→    return isOle2File(head);
    87→  } catch {
    88→    return false;
    89→  }
    90→}

    92→/** The live salvage report, keyed to the file identity it belongs to. */
    93→interface SummaryState {
    94→  key: string;
    95→  doc: ParsedPub | null;
    96→  /** PubError code when the file was rejected by the model. */
    97→  errorCode: string | null;
    98→}

   101→/* Surface                                                              */
   102→/* ------------------------------------------------------------------ */
   103→
   104→export function PublisherToPdfSurface({
   105→  files,
   106→  values,
   107→  onChange,
   108→  setFileOrder,
   109→  setRunBlocked,
   110→  addFiles,
   111→}: ToolSurfaceProps): React.JSX.Element {
   112→  const { t, locale } = useI18n();
   113→  const { toast } = useToast();
   114→
   115→  const file = files.length > 0 ? (files[files.length - 1]?.file ?? null) : null;
   116→  const fileKey = file ? `${file.name}:${file.size}:${file.lastModified}` : "";
   117→
   118→  const [dragOver, setDragOver] = useState(false);
   119→  const dragCounter = useRef(0);
   120→  const inputRef = useRef<HTMLInputElement>(null);
   121→  const inputId = useId();
   122→
   123→  const [summary, setSummary] = useState<SummaryState | null>(null);
   124→  const [pct, setPct] = useState<number | null>(null);
   125→  const workerRef = useRef<Worker | null>(null);
   126→
   127→  /* ---------- CTA gate: no file → blocked (scan-pdf pattern) ---------- */
   128→  useEffect(() => {
   129→    let cancelled = false;
   130→    void Promise.resolve().then(() => {
   131→      if (cancelled) return;
   132→      setRunBlocked(files.length === 0 ? "pub.ctaNoFile" : null);
   133→    });
   134→    return () => {
   135→      cancelled = true;
   136→    };
   137→  }, [files.length, setRunBlocked]);
   138→
   139→  /* ---------- localized fallback word for the shell-CTA run ---------- */
   140→  useEffect(() => {
   141→    let cancelled = false;
   142→    void Promise.resolve().then(() => {
   143→      if (cancelled) return;
   144→      onChange("fileWord", t("pub.fileWord"));
   145→    });
   146→    return () => {
   147→      cancelled = true;
   148→    };
   149→  }, [onChange, t]);
   150→
   151→  /* ---------- live salvage report: the SAME parser the worker uses ---------- */
   152→  useEffect(() => {
   153→    let cancelled = false;
   154→    if (!file) {
   155→      void Promise.resolve().then(() => {
   156→        if (!cancelled) setSummary(null);
   157→      });
   158→      return () => {
   159→        cancelled = true;
   160→      };
   161→    }
   162→    void (async () => {
   163→      try {
   164→        const bytes = new Uint8Array(await file.arrayBuffer());
   165→        if (cancelled) return;
   166→        try {
   167→          const doc = parsePub(bytes);
   168→          if (!cancelled) setSummary({ key: fileKey, doc, errorCode: null });
   169→        } catch (error) {
   170→          const code =
   171→            error instanceof PubError ? error.code : "errCorrupt";
   172→          if (!cancelled) setSummary({ key: fileKey, doc: null, errorCode: code });
   173→        }
   174→      } catch {
   175→        if (!cancelled) {
   176→          setSummary({ key: fileKey, doc: null, errorCode: "errCorrupt" });
   177→        }
   178→      }
   179→    })();
   180→    return () => {
   181→      cancelled = true;
   182→    };
   183→  }, [file, fileKey]);
   184→
   185→  const parsedDoc =
   186→    summary && summary.key === fileKey && file ? summary.doc : null;
   187→  const docOk = parsedDoc !== null && parsedDoc.paras.length > 0;
   188→  const errorCode =
   189→    summary !== null && summary.key === fileKey && !docOk
   190→      ? (summary.errorCode ?? "errNoText")
   191→      : null;
   192→
   193→  /* ---------- input pipeline (magic bytes + size limit) ---------- */
   194→
   195→  const acceptFile = useCallback(
   196→    (incoming: File[]): void => {
   197→      if (incoming.length === 0 || files.length > 0) return;
   198→      const f = incoming[0];
   199→      if (!f) return;
   200→      if (f.size > sizeLimitBytes()) {
   201→        toast({
   202→          title: t("pub.errTooLargeTitle"),
   203→          description: t("pub.errTooLargeDesc", { size: sizeLimitLabel() }),
   204→          variant: "destructive",
   205→        });
   206→        return;
   207→      }
   208→      void sniffIsOle2(f).then((isOle2) => {
   209→        if (!isOle2) {
   210→          toast({
   211→            title: t("pub.errNotPubTitle"),
   212→            description: t("pub.errNotPubDesc"),
   213→            variant: "destructive",
   214→          });
   215→          return;
   216→        }
   217→        addFiles([f]);
   218→      });
   219→    },
   220→    [addFiles, files.length, t, toast]
   221→  );
   222→
   223→  const handleInputChange = useCallback(
   224→    (event: ChangeEvent<HTMLInputElement>): void => {
   225→      acceptFile([...(event.target.files ?? [])]);
   226→      event.target.value = ""; // allow re-selecting the same file
   227→    },
   228→    [acceptFile]
   229→  );
   230→
   231→  const handleDrop = useCallback(
   232→    (event: DragEvent<HTMLDivElement>): void => {
   233→      event.preventDefault();
   234→      dragCounter.current = 0;
   235→      setDragOver(false);
   236→      acceptFile([...event.dataTransfer.files]);
   237→    },
   238→    [acceptFile]
   239→  );
   240→
   241→  // Ctrl+V of a file from the desktop (house DropZone behaviour).
   242→  useEffect(() => {
   243→    const onPaste = (event: globalThis.ClipboardEvent): void => {
   244→      const pasted = [...(event.clipboardData?.files ?? [])];
   245→      if (pasted.length === 0) return;
   246→      event.preventDefault();
   247→      acceptFile(pasted);
   248→    };
   249→    document.addEventListener("paste", onPaste);
   250→    return () => document.removeEventListener("paste", onPaste);
   251→  }, [acceptFile]);
   252→
   253→  /* ---------- Convert → worker spawn (create-invoice pattern) ---------- */
   254→
   255→  const pageSize = String(values.pageSize ?? "a4") === "letter" ? "letter" : "a4";
   256→  const margin = String(values.margin ?? "normal") === "narrow" ? "narrow" : "normal";
   257→  const fontSizeRaw = Number(values.fontSize);
   258→  const fontSize = Number.isFinite(fontSizeRaw)
   259→    ? Math.min(18, Math.max(8, fontSizeRaw))
   260→    : 10.5;
   261→
   262→  const convert = useCallback((): void => {
   263→    if (!file || workerRef.current || !docOk) return;
   264→    setPct(0);
   265→    const worker = new Worker(new URL("./worker.ts", import.meta.url), {
   266→      type: "module",
   267→    });
   268→    workerRef.current = worker;
   269→    const runId = Date.now();
   270→    const cleanup = (): void => {
   271→      worker.terminate();
   272→      if (workerRef.current === worker) workerRef.current = null;
   273→    };
   274→    worker.addEventListener(
   275→      "message",
   276→      (event: MessageEvent<WorkerResponse>) => {
   277→        const data = event.data;
   278→        if (!data || data.runId !== runId) return;
   279→        if (data.type === "progress") {
   280→          setPct(data.pct);
   281→          return;
   282→        }
   283→        if (data.type === "result") {
   284→          setPct(null);
   285→          const output = data.outputs[0];
   286→          if (output) {
   287→            downloadBlob(output.blob, output.name);
   288→            toast({
   289→              title: t("pub.doneTitle"),
   290→              description: t("pub.doneDesc", {
   291→                name: output.name,
   292→                pages: data.detail?.pages ?? 1,
   293→              }),
   294→            });
   295→          }
   296→          cleanup();
   297→          return;
   298→        }
   299→        if (data.type === "error") {
   300→          setPct(null);
   301→          toast({
   302→            title: t("tool.errorTitle"),
   303→            description: data.code.startsWith("pub.")
   304→              ? t(data.code)
   305→              : t("tool.errorGeneric"),
   306→            variant: "destructive",
   307→          });
   308→          cleanup();
   309→        }
   310→      }
   311→    );
   312→    worker.addEventListener("error", () => {
   313→      setPct(null);
   314→      toast({
   315→        title: t("tool.errorTitle"),
   316→        description: t("tool.errorGeneric"),
   317→        variant: "destructive",
   318→      });
   319→      cleanup();
   320→    });
   321→    worker.postMessage({
   322→      type: "run",
   323→      runId,
   324→      toolId: "publisher-to-pdf",
   325→      options: {
   326→        pageSize,
   327→        margin,
   328→        fontSize,
   329→        fileWord: t("pub.fileWord"),
   330→      },
   331→      files: [file],
   332→    });
   333→  }, [docOk, file, fontSize, margin, pageSize, t, toast]);
   334→
   335→  // Kill a stuck worker if the surface unmounts mid-run.
   336→  useEffect(() => {
   337→    return () => {
   338→      workerRef.current?.terminate();
   339→    };
   340→  }, []);
   341→
   342→  const removeFile = useCallback((): void => {
   343→    setFileOrder([]);
   344→  }, [setFileOrder]);
   345→
   346→  const countFmt = useCallback(
   347→    (n: number): string => n.toLocaleString(locale === "hi" ? "hi-IN" : "en-IN"),
   348→    [locale]
   349→  );
   350→
   351→  const estPages = parsedDoc
   352→    ? estimatePages(parsedDoc, {
   353→        pageSize,
   354→        margin,
   355→        baseSize: fontSize,
   356→      })
   357→    : 0;
   358→
   359→  /* ---------------------------------------------------------------- */
   360→
   361→  return (
   362→    <div className="flex flex-col gap-4 sm:gap-6">
   363→      {/* ---------------- file input / file card ---------------- */}
   364→      {!file ? (
   365→        <section aria-label={t("pub.selectFile")}>
   366→          <div
   367→            onDragEnter={(event) => {
   368→              event.preventDefault();
   369→              dragCounter.current += 1;
   370→              setDragOver(true);
   371→            }}
   372→            onDragOver={(event) => event.preventDefault()}
   373→            onDragLeave={(event) => {
   374→              event.preventDefault();
   375→              dragCounter.current = Math.max(0, dragCounter.current - 1);
   376→              if (dragCounter.current === 0) setDragOver(false);
   377→            }}
   378→            onDrop={handleDrop}
   379→            className={cn(
   380→              "relative rounded-card border-2 border-dashed transition-all duration-200",
   381→              dragOver
   382→                ? "scale-[1.01] border-primary bg-primary/5 ring-4 ring-primary/10"
   383→                : "border-border bg-card hover:border-primary/40"
   384→            )}
   385→          >
   386→            <label
   387→              htmlFor={inputId}
   388→              className="flex min-h-44 cursor-pointer flex-col items-center justify-center gap-3 px-6 py-8 text-center"
   389→            >
   390→              <span
   391→                aria-hidden="true"
   392→                className={cn(
   393→                  "flex size-14 items-center justify-center rounded-full border transition-all duration-200",
   394→                  dragOver
   395→                    ? "scale-110 border-primary bg-primary/10 text-primary"
   396→                    : "border-border bg-muted/50 text-muted-foreground"
   397→                )}
   398→              >
   399→                {dragOver ? (
   400→                  <FileUp className="size-6 animate-in fade-in-50 zoom-in-75" />
   401→                ) : (
   402→                  <FolderOpen className="size-6" />
   403→                )}
   404→              </span>
   405→              <span className="text-sm font-semibold text-foreground">
   406→                {t("pub.selectFile")}
   407→              </span>
   408→              <span className="flex flex-wrap items-center justify-center gap-x-2 gap-y-1 text-xs text-muted-foreground">
   409→                <span>{t("pub.dropHint")}</span>
   410→                <span aria-hidden="true">·</span>
   411→                <span className="font-medium">
   412→                  <kbd className="rounded border border-border bg-muted px-1 py-0.5 font-sans text-[10px]">
   413→                    Ctrl
   414→                  </kbd>
   415→                  <span> + </span>
   416→                  <kbd className="rounded border border-border bg-muted px-1 py-0.5 font-sans text-[10px]">
   417→                    V
   418→                  </kbd>
   419→                  <span className="sr-only"> </span>
   420→                  <span> {t("pub.pasteHint")}</span>
   421→                </span>
   422→              </span>
   423→              <span className="text-[11px] text-muted-foreground">
   424→                {t("pub.limits", { size: sizeLimitLabel() })}
   425→              </span>
   426→            </label>
   427→            <input
   428→              ref={inputRef}
   429→              id={inputId}
   430→              type="file"
   431→              accept=".pub,application/x-mspublisher,application/vnd.ms-publisher"
   432→              className="sr-only"
   433→              onChange={handleInputChange}
   434→              aria-label={t("pub.selectFile")}
   435→            />
   436→          </div>
   437→        </section>
   438→      ) : (
   439→        <section aria-label={file.name}>
   440→          <div className="rounded-card border border-border bg-card p-4 sm:p-5">
   441→            <div className="flex flex-wrap items-center gap-3">
   442→              <span
   443→                className="flex size-11 shrink-0 items-center justify-center rounded-btn bg-primary/10 text-primary"
   444→                aria-hidden="true"
   445→              >
   446→                <FileText className="size-5" />
   447→              </span>
   448→              <div className="min-w-0 flex-1">
   449→                <p
   450→                  className="truncate text-sm font-semibold text-foreground"
   451→                  title={file.name}
   452→                >
   453→                  {file.name}
   454→                </p>
   455→                <p className="text-xs text-muted-foreground">
   456→                  {formatBytes(file.size)}
   457→                </p>
   458→              </div>
   459→              <span className="shrink-0 rounded-chip bg-muted px-2.5 py-1 text-xs font-medium text-foreground">
   460→                {t("pub.pubChip")}
   461→              </span>
   462→              <Button
   463→                type="button"
   464→                variant="outline"
   465→                size="sm"
   466→                className="rounded-btn"
   467→                onClick={removeFile}
   468→                aria-label={t("pub.removeFile")}
   469→              >
   470→                <X className="size-4" aria-hidden="true" />
   471→                <span className="hidden sm:inline">{t("pub.removeFile")}</span>
   472→              </Button>
   473→            </div>
   474→          </div>
   475→        </section>
   476→      )}
   477→
   478→      {/* ---------------- live salvage report ---------------- */}
   479→      {file ? (
   480→        <section aria-labelledby="pub-summary-heading">
   481→          <h2
   482→            id="pub-summary-heading"
   483→            className="text-sm font-semibold text-foreground"
   484→          >
   485→            {t("pub.summaryTitle")}
   486→          </h2>
   487→          {parsedDoc === null && errorCode === null ? (
   488→            <div
   489→              className="mt-3 flex items-center gap-2 rounded-card border border-border bg-card px-4 py-3 text-sm text-muted-foreground"
   490→              aria-live="polite"
   491→            >
   492→              <Loader2
   493→                className="size-4 shrink-0 animate-spin text-primary"
   494→                aria-hidden="true"
   495→              />
   496→              {t("conv.reading")}
   497→            </div>
   498→          ) : null}
   499→          {errorCode !== null ? (
   500→            <div
   501→              role="status"
   502→              className="mt-3 flex items-start gap-2 rounded-card border border-amber-500/40 bg-amber-500/10 px-4 py-3 text-xs leading-relaxed text-amber-900 dark:text-amber-200"
   503→            >
   504→              <TriangleAlert
   505→                className="mt-0.5 size-4 shrink-0"
   506→                aria-hidden="true"
   507→              />
   508→              <span>{t(`pub.${errorCode}`)}</span>
   509→            </div>
   510→          ) : null}
   511→          {parsedDoc !== null ? (
   512→            <div className="mt-3">
   513→              <dl className="grid grid-cols-3 gap-3">
   514→                {(
   515→                  [
   516→                    ["pub.statParas", parsedDoc.paraCount],
   517→                    ["pub.statChars", parsedDoc.charCount],
   518→                    ["pub.statPages", estPages],
   519→                  ] as const
   520→                ).map(([key, n]) => (
   521→                  <div
   522→                    key={key}
   523→                    className="min-w-0 rounded-card border border-border bg-card p-3 text-center"
   524→                  >
   525→                    <dt className="order-2 text-[11px] font-medium uppercase tracking-wide text-muted-foreground">
   526→                      {t(key)}
   527→                    </dt>
   528→                    <dd className="order-1 text-lg font-bold tabular-nums text-foreground">
   529→                      {countFmt(n)}
   530→                    </dd>
   531→                  </div>
   532→                ))}
   533→              </dl>
   534→
   535→              {/* detected streams — the honest container preview */}
   536→              {parsedDoc.stories.length > 0 ? (
   537→                <div className="mt-3 min-w-0 rounded-card border border-border bg-card p-4">
   538→                  <p className="text-xs font-medium text-muted-foreground">
   539→                    {t("pub.streamsTitle", {
   540→                      used: parsedDoc.streamsUsed,
   541→                      scanned: parsedDoc.streamsScanned,
   542→                    })}
   543→                  </p>
   544→                  <ul
   545→                    className="mt-2 max-h-96 overflow-y-auto pr-1"
   546→                    aria-label={t("pub.streamsTitle", {
   547→                      used: parsedDoc.streamsUsed,
   548→                      scanned: parsedDoc.streamsScanned,
   549→                    })}
   550→                  >
   551→                    {parsedDoc.stories.map((s) => (
   552→                      <li
   553→                        key={`${s.id}-${s.name}`}
   554→                        className="flex min-w-0 items-center gap-2 border-b border-border/60 py-1.5 text-xs last:border-b-0"
   555→                      >
   556→                        <span className="min-w-0 flex-1 truncate font-medium text-foreground">
   557→                          {s.name}
   558→                        </span>
   559→                        <span className="shrink-0 rounded-chip bg-muted px-1.5 py-0.5 text-[10px] font-medium uppercase text-muted-foreground">
   560→                          {s.encoding === "utf16" ? "UTF-16" : "ASCII"}
   561→                        </span>
   562→                        <span className="shrink-0 rounded-chip bg-muted px-1.5 py-0.5 text-[10px] font-medium uppercase text-muted-foreground">
   563→                          {s.path === "mini"
   564→                            ? t("pub.pathMini")
   565→                            : t("pub.pathNormal")}
   566→                        </span>
   567→                        <span className="shrink-0 tabular-nums text-muted-foreground">
   568→                          {countFmt(s.chars)}
   569→                        </span>
   570→                      </li>
   571→                    ))}
   572→                  </ul>
   573→                </div>
   574→              ) : null}
   575→
   576→              {parsedDoc.streamsFailed > 0 ? (
   577→                <div
   578→                  role="status"
   579→                  className="mt-3 flex items-start gap-2 rounded-btn border border-amber-500/40 bg-amber-500/10 px-3 py-2.5 text-xs leading-relaxed text-amber-900 dark:text-amber-200"
   580→                >
   581→                  <TriangleAlert
   582→                    className="mt-0.5 size-3.5 shrink-0"
   583→                    aria-hidden="true"
   584→                  />
   585→                  {t("pub.streamsFailedNote", {
   586→                    count: parsedDoc.streamsFailed,
   587→                  })}
   588→                </div>
   589→              ) : null}
   590→              {parsedDoc.truncated ? (
   591→                <div
   592→                  role="status"
   593→                  className="mt-3 flex items-start gap-2 rounded-btn border border-amber-500/40 bg-amber-500/10 px-3 py-2.5 text-xs leading-relaxed text-amber-900 dark:text-amber-200"
   594→                >
   595→                  <TriangleAlert

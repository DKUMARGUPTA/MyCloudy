   128→   128→/** Collects per-page crop-box + rotation views (pdf.js, main thread). */
   129→   129→async function collectPageViews(
   130→   130→  file: File,
   131→   131→  maxPages: number
   132→   132→): Promise<PageView[]> {
   133→   133→  const pdfjs = await loadPdfjs();
   134→   134→  const data = new Uint8Array(await file.arrayBuffer());
   135→   135→  const doc: PDFDocumentProxy = await pdfjs.getDocument({
   136→   136→    data,
   137→   137→    cMapUrl: "/pdfjs/cmaps/",
   138→   138→    cMapPacked: true,
   139→   139→    standardFontDataUrl: "/pdfjs/standard_fonts/",
   140→   140→  }).promise;
   141→   141→  const count = Math.min(doc.numPages, maxPages);
   142→   142→  const views: PageView[] = [];
   143→   143→  for (let p = 1; p <= count; p += 1) {
   144→   144→    const page = await doc.getPage(p);
   145→   145→    const raw = (page as unknown as { view: number[] }).view ?? [0, 0, 612, 792];
   146→   146→    const rotateRaw = (page as unknown as { rotate: number }).rotate ?? 0;
   147→   147→    const r = ((rotateRaw % 360) + 360) % 360;
   148→   148→    views.push({
   149→   149→      x0: raw[0],
   150→   150→      y0: raw[1],
   151→   151→      x1: raw[2],
   152→   152→      y1: raw[3],
   153→   153→      rotate: r === 90 || r === 180 || r === 270 ? r : 0,
   154→   154→    });
   155→   155→  }
   156→   156→  return views;
   157→   157→}

   159→   159→function parseBlob(raw: unknown): EditorBlob {
   160→   160→  try {
   161→   161→    const parsed = JSON.parse(String(raw ?? "{}")) as Partial<EditorBlob>;
   162→   162→    return {
   163→   163→      objects: Array.isArray(parsed.objects)
   164→   164→        ? (parsed.objects as EditorObject[])
   165→   165→        : [],
   166→   166→      pageViews: Array.isArray(parsed.pageViews)
   167→   167→        ? (parsed.pageViews as PageView[])
   168→   168→        : [],
   169→   169→      assetOrder: Array.isArray(parsed.assetOrder)
   170→   170→        ? parsed.assetOrder.filter((k): k is string => typeof k === "string")
   171→   171→        : [],
   172→   172→    };
   173→   173→  } catch {
   174→   174→    return { objects: [], pageViews: [], assetOrder: [] };
   175→   175→  }
   176→   176→}
   177→   177→
   178→   178→/** Asset keys referenced by visible objects, first-seen order. */
   179→   179→function orderFromObjects(objects: EditorObject[]): string[] {
   180→   180→  const seen: string[] = [];
   181→   181→  for (const o of objects) {
   182→   182→    if (o.kind === "image" && !seen.includes(o.assetKey)) seen.push(o.assetKey);
   183→   183→  }
   184→   184→  return seen;
   185→   185→}

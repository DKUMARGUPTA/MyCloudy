    78→/**
    79→ * DocoPDF tool registry — THE source of truth for every tool (PRD §7).
    80→ *
    81→ * One config object per tool: id (== URL slug), category, mode D/S, roadmap
    82→ * phase, icon, localized name/description (en + hi), search tags (en + hi),
    83→ * accepted input type, curated popularity, and the `enabled` flag that flips
    84→ * to true when the tool ships (Phases 2+). The shared ToolShell renders every
    85→ * enabled tool from this registry — no per-tool pages are hand-written.
    86→ *
    87→ * NOTE on i18n: tool catalog strings live HERE as typed locale maps (not in
    88→ * en.json/hi.json). This keeps the catalog a single source of truth and makes
    89→ * a missing Hindi string a type error. UI chrome strings stay in the JSON
    90→ * catalogs. Deploying with path locales (/hi) reads the same maps.
    91→ */
    92→
    93→export type ToolMode = "D" | "S";
    94→export type RoadmapPhase = "P1" | "P2" | "P3";

     3→     3→/**
     4→     4→ * PDF → Markdown model (RB4-c) — 100% pure TypeScript: no DOM, no pdf.js,
     5→     5→ * no pdf-lib. The extraction takes PLAIN text-item arrays (the shape pdf.js
     6→     6→ * getTextContent emits) plus optional link-annotation rects, so the smoke
     7→     7→ * can drive it headless, and the emitter produces clean Markdown that the
     8→     8→ * house's own markdown-to-pdf parser understands (the mirror rule: write
     9→     9→ * what it reads — headings #..######, **bold**, *italic*, - bullets,
    10→    10→ * 1. ordered items, > quotes, [text](url) links, --- thematic breaks).
    11→    11→ *
    12→    12→ * Pipeline (adapted from pdf-to-word, the extraction layer of record):
    13→    13→ *  1. collectPieces — coerce + position-tag the raw pdf.js items; bold /
    14→    14→ *     italic resolved from the item, the worker's font-table enrichment
    15→    15→ *     (pdf.js 4.x reports internal ids like "g_d0_f3") or the font name;
    16→    16→ *     pieces covered by a link-annotation rect carry that URL.
    17→    17→ *  2. buildLines — group by y-proximity (half a font height), order by x,
    18→    18→ *     insert spaces at horizontal gaps (pdf.js emits one item per Tj
    19→    19→ *     operator without carrying spaces — the pinned pdf-to-word lesson:
    20→    20→ *     a whitespace-only item is EXACTLY ONE space, never two glued).
    21→    21→ *  3. Lines → blocks — a line starting with a bullet glyph (•, -, …) or
    22→    22→ *     a "1." / "2)" number becomes a LIST ITEM (never merged with the
    23→    23→ *     previous block); tight non-marker lines continue an item or soft-
    24→    24→ *     wrap-join a paragraph (a baseline gap noticeably larger than one
    25→    25→ *     line height, or a font-size change, starts a new block; hyphenated
    26→    26→ *     line endings merge back before a lowercase letter).
    27→    27→ *  4. Headings — font-size clusters: the document's body size is the
    28→    28→ *     char-weighted mode; sizes clearly above it form tiers, largest
    29→    29→ *     first; tier i becomes heading level i+1, capped at ######. Heading
    30→    30→ *     text is emitted PLAIN (no bold or italic markers inside —
    31→    31→ *     anchor-friendly).
    32→    32→ *  5. Block quotes — only when visually distinct: the block starts
    33→    33→ *     ≥ QUOTE_INDENT_PT right of the document's modal left edge AND is
    34→    34→ *     italic-dominant. Honest: no indent/italics → plain paragraph.
    35→    35→ *  6. Emission — blocks joined with \n\n; consecutive same-kind list
    36→    36→ *     items share a single \n; optional "---" between PDF pages; optional
    37→    37→ *     "<!-- converted by MyCloudy -->" comment; optional YAML front
    38→    38→ *     matter. Literal line-start markdown (* # _ [ + > - and ordered
    39→    39→ *     markers) is backslash-escaped; HTML entities are NEVER used (raw
    40→    40→ *     UTF-8 — Hindi passes through, with the visual→logical Devanagari
    41→    41→ *     matra repair from pdf-to-powerpoint).
    42→    42→ *
    43→    43→ * HONEST limits (mirrored in the FAQ copy): text + structure only.
    44→    44→ * Tables are NOT converted (point at pdf-to-excel), images are skipped,
    45→    45→ * two-column pages come out as a line-by-line text flow (rows of both
    46→    46→ * columns may join), scanned pages need the OCR tool first (the worker
    47→    47→ * fails with tool.errorNoText).
    48→    48→ */

   171→   171→/**
   172→   172→ * Strips C0/C1 control characters (whitespace controls normalise to a
   173→   173→ * plain space) — every run text passes through here before the model.
   174→   174→ */
   175→   175→export function sanitizeRunText(text: string): string {
   176→   176→  let out = "";
   177→   177→  for (const ch of text) {
   178→   178→    const code = ch.codePointAt(0) ?? 0;
   179→   179→    if (code === 9 || code === 10 || code === 13 || (code >= 28 && code <= 32)) {
   180→   180→      out += " ";
   181→   181→    } else if (code < 32 || (code >= 127 && code <= 159)) {
   182→   182→      continue; // pure control noise — dropped
   183→   183→    } else {
   184→   184→      out += ch;
   185→   185→    }
   186→   186→  }
   187→   187→  return out.replace(/\s+/g, " ").trim();
   188→   188→}

   190→   190→/* ------------------------------------------------------------------ */
   191→   191→/* Devanagari visual→logical repair (pdf-to-powerpoint copy)           */
   192→   192→/* ------------------------------------------------------------------ */
   193→   193→
   194→   194→/* pdf.js concatenates per-glyph ToUnicode in stream order, and shaping
   195→   195→ * puts the three PRE-base matras (ि े ै) BEFORE their base cluster —
   196→   196→ * so text drawn from shaped glyph streams (pdf-lib/Noto, many cheap
   197→   197→ * generators) extracts with "िप्रया" instead of "प्रिया". The inverse
   198→   198→ * move is only applied where it is PROVABLY visual: the matra starts
   199→   199→ * the run, or follows anything that can never end a logical syllable
   200→   200→ * (space, vowel, another sign, virama, Latin…). After a plain
   201→   201→ * CONSONANT the reading is ambiguous and is left untouched — OCR-style
   202→   202→ * logical text is never damaged. */
   203→   203→
   204→   204→const DEV_PRE_MATRA = new Set(["\u093F", "\u0947", "\u0948"]); // ि े ै
   205→   205→const DEV_VIRAMA = "\u094D";
   206→   206→const DEV_NUKTA = "\u093C";
   207→   207→
   208→   208→function isDevConsonant(ch: string | undefined): boolean {
   209→   209→  if (ch === undefined) return false;
   210→   210→  const c = ch.charCodeAt(0);
   211→   211→  return (
   212→   212→    (c >= 0x0915 && c <= 0x0939) || (c >= 0x0958 && c <= 0x095f) ||
   213→   213→    c === 0x0979 || c === 0x097a
   214→   214→  );
   215→   215→}
   216→   216→
   217→   217→/** Greedy match of a base/conjunct cluster: C (nukta?) (virama C (nukta?))*. */
   218→   218→function matchDevCluster(text: string, i: number): string | null {
   219→   219→  let j = i;
   220→   220→  if (!isDevConsonant(text[j])) return null;
   221→   221→  j += 1;
   222→   222→  if (text[j] === DEV_NUKTA) j += 1;
   223→   223→  for (;;) {
   224→   224→    if (text[j] !== DEV_VIRAMA || !isDevConsonant(text[j + 1])) break;
   225→   225→    j += 2;
   226→   226→    if (text[j] === DEV_NUKTA) j += 1;
   227→   227→  }
   228→   228→  return text.slice(i, j);
   229→   229→}
   230→   230→
   231→   231→/** Move provably-visual pre-base matras and dangling rephs to logical order. */
   232→   232→export function fixDevanagariVisual(text: string): string {
   233→   233→  let needsMatra = false;
   234→   234→  let needsReph = false;
   235→   235→  for (let i = 0; i < text.length; i += 1) {
   236→   236→    if (DEV_PRE_MATRA.has(text[i] as string)) needsMatra = true;
   237→   237→    if (text[i] === DEV_RA && text[i + 1] === DEV_VIRAMA) needsReph = true;
   238→   238→  }
   239→   239→  if (!needsMatra && !needsReph) return text;
   240→   240→
   241→   241→  let out = text;
   242→   242→
   243→   243→  if (needsMatra) {
   244→   244→    // Pre-base matras: shaping renders ि े ै BEFORE their cluster, so
   245→   245→    // they extract one position early. Only provably-visual cases move
   246→   246→    // (the matra starts the run or follows something that can never
   247→   247→    // end a logical syllable); after a plain consonant the reading is
   248→   248→    // ambiguous and is left untouched.
   249→   249→    let acc = "";
   250→   250→    let prev = "";
   251→   251→    let i = 0;
   252→   252→    while (i < out.length) {
   253→   253→      const ch = out[i] as string;
   254→   254→      if (DEV_PRE_MATRA.has(ch)) {
   255→   255→        const cluster = matchDevCluster(out, i + 1);
   256→   256→        const ambiguous =
   257→   257→          isDevConsonant(prev) || prev === DEV_NUKTA; // logical reading exists
   258→   258→        if (cluster !== null && !ambiguous) {
   259→   259→          acc += cluster + ch;
   260→   260→          prev = cluster[cluster.length - 1] as string;
   261→   261→          i += 1 + cluster.length;
   262→   262→          continue;
   263→   263→        }
   264→   264→      }
   265→   265→      acc += ch;
   266→   266→      prev = ch;
   267→   267→      i += 1;
   268→   268→    }
   269→   269→    out = acc;
   270→   270→  }
   271→   271→
   272→   272→  if (needsReph) out = fixDevReph(out);
   273→   273→  return out;
   274→   274→}
   275→   275→
   276→   276→/** Post-syllable signs a visual reph may trail behind. */
   277→   277→const DEV_POST_MARK = new Set([
   278→   278→  "\u093E", "\u093F", "\u0940", "\u0941", "\u0942", "\u0943", "\u0944",
   279→   279→  "\u0945", "\u0946", "\u0947", "\u0948", "\u0949", "\u094A", "\u094B",
   280→   280→  "\u094C", "\u0901", "\u0902", "\u0903", "\u093C", "\u0962", "\u0963",
   281→   281→]);
   282→   282→const DEV_RA = "\u0930";
   283→   283→
   284→   284→/**
   285→   285→ * Move a provably-visual reph back in front of its cluster. Shaping
   286→   286→ * puts the syllable-final र ् AFTER the base consonant and its matras
   287→   287→ * ("रिपोटर्" for रिपोर्ट) — a dead consonant that dangles before a
   288→   288→ * non-consonant (end, space, punctuation, Latin, digits, signs) can
   289→   289→ * never be logical, so those cases are provably visual. A र ् directly
   290→   290→ * followed by a consonant is the valid logical ligature form and is
   291→   291→ * never touched, so OCR-style logical text is never damaged.
   292→   292→ */
   293→   293→function fixDevReph(text: string): string {
   294→   294→  let out = text;
   295→   295→  for (let i = 0; i + 1 < out.length; i += 1) {
   296→   296→    if (out[i] !== DEV_RA || out[i + 1] !== DEV_VIRAMA) continue;
   297→   297→    const after = i + 2 < out.length ? (out[i + 2] as string) : undefined;
   298→   298→    if (after !== undefined && (isDevConsonant(after) || after === DEV_VIRAMA)) {
   299→   299→      continue;
   300→   300→    }
   301→   301→    // Provably visual: walk back over post-syllable signs, expect one
   302→   302→    // base consonant, and splice the reph in front of it.
   303→   303→    let j = i - 1;
   304→   304→    while (j >= 0 && DEV_POST_MARK.has(out[j] as string)) j -= 1;
   305→   305→    if (j < 0 || !isDevConsonant(out[j])) continue; // nothing to attach to
   306→   306→    out =
   307→   307→      out.slice(0, j) +
   308→   308→      DEV_RA +
   309→   309→      DEV_VIRAMA +
   310→   310→      out.slice(j, i) +
   311→   311→      out.slice(i + 2);
   312→   312→    i = j + 1; // the moved pair now precedes a consonant — never re-fires
   313→   313→  }
   314→   314→  return out;
   315→   315→}

   361→   361→/** The link whose rect covers a piece's centre (1.5 pt slack). */
   362→   362→function linkForPiece(
   363→   363→  x: number,
   364→   364→  y: number,
   365→   365→  w: number,
   366→   366→  size: number,
   367→   367→  links: readonly MdLinkRect[] | undefined
   368→   368→): string | undefined {
   369→   369→  if (links === undefined || links.length === 0) return undefined;
   370→   370→  const cx = x + w / 2;
   371→   371→  const cy = y + size * 0.45;
   372→   372→  for (const link of links) {
   373→   373→    if (
   374→   374→      cx >= link.x1 - LINK_SLACK &&
   375→   375→      cx <= link.x2 + LINK_SLACK &&
   376→   376→      cy >= link.y1 - LINK_SLACK &&
   377→   377→      cy <= link.y2 + LINK_SLACK
   378→   378→    ) {
   379→   379→      return link.url;
   380→   380→    }
   381→   381→  }
   382→   382→  return undefined;
   383→   383→}

   385→   385→/** Coerces one page of raw pdf.js items into position-tagged pieces. */
   386→   386→function collectPieces(
   387→   387→  items: readonly PdfTextItemInput[],
   388→   388→  styleMap: ReadonlyMap<string, MdFontStyle> | undefined,
   389→   389→  links: readonly MdLinkRect[] | undefined
   390→   390→): Piece[] {
   391→   391→  const pieces: Piece[] = [];
   392→   392→  for (const item of items) {
   393→   393→    if (typeof item !== "object" || item === null) continue;
   394→   394→    const str = typeof item.str === "string" ? item.str : "";
   395→   395→    if (str.length === 0) continue;
   396→   396→    const transform = item.transform;
   397→   397→    if (!Array.isArray(transform) || transform.length < 6) continue;
   398→   398→    const x = Number(transform[4]);
   399→   399→    const y = Number(transform[5]);
   400→   400→    if (!Number.isFinite(x) || !Number.isFinite(y)) continue;
   401→   401→    const size = Math.hypot(
   402→   402→      Number(transform[1]) || 0,
   403→   403→      Number(transform[3]) || 0
   404→   404→    );
   405→   405→    const w = Number(item.width) || 0;
   406→   406→    const fontName = typeof item.fontName === "string" ? item.fontName : "";
   407→   407→    const enriched = fontName.length > 0 ? styleMap?.get(fontName) : undefined;
   408→   408→    const bold =
   409→   409→      item.bold === true ||
   410→   410→      enriched?.bold === true ||
   411→   411→      (fontName.length > 0 && BOLD_NAME_RE.test(fontName));
   412→   412→    const italic =
   413→   413→      item.italic === true ||
   414→   414→      enriched?.italic === true ||
   415→   415→      (fontName.length > 0 && ITALIC_NAME_RE.test(fontName));
   416→   416→    // sanitizeRunText trims the ends — but pdf.js drops its own split
   417→   417→    // items exactly there (it also emits whitespace-ONLY items for the
   418→   418→    // visual spaces it splits away). The raw item's leading/trailing
   419→   419→    // whitespace is restored as a single space; a whitespace-only item
   420→   420→    // becomes exactly ONE space piece (the pinned pdf-to-word lesson:
   421→   421→    // " " + "" + " " would double every visual space).
   422→   422→    const text = sanitizeRunText(str);
   423→   423→    const glued =
   424→   424→      text.length > 0
   425→   425→        ? (/^\s/.test(str) ? " " : "") +
   426→   426→          text +
   427→   427→          (/\s$/.test(str) ? " " : "")
   428→   428→        : " ";
   429→   429→    if (glued.length === 0) continue;
   430→   430→    pieces.push({
   431→   431→      text: glued,
   432→   432→      x,
   433→   433→      y,
   434→   434→      w,
   435→   435→      size: size > 0 ? size : 10,
   436→   436→      bold,
   437→   437→      italic,
   438→   438→      link: linkForPiece(x, y, w, size > 0 ? size : 10, links),
   439→   439→    });
   440→   440→  }
   441→   441→  return pieces;
   442→   442→}
   443→   443→
   444→   444→/** Detects and strips a leading list marker from a line's runs. */
   445→   445→function stripListMarker(runs: LineRun[]): LineMarker | null {
   446→   446→  const joined = runs.map((run) => run.text).join("");
   447→   447→  let length = 0;
   448→   448→  let marker: LineMarker | null = null;
   449→   449→  const ordered = ORDERED_RE.exec(joined);
   450→   450→  if (ordered !== null) {
   451→   451→    marker = { kind: "ordered", number: Number.parseInt(ordered[1], 10) };
   452→   452→    length = ordered[0].length;
   453→   453→  } else {
   454→   454→    const bullet = BULLET_FIRST_RE.exec(joined) ?? BULLET_DASH_RE.exec(joined);
   455→   455→    if (bullet !== null) {
   456→   456→      marker = { kind: "bullet" };
   457→   457→      length = bullet[0].length;
   458→   458→    }
   459→   459→  }
   460→   460→  if (marker === null) return null;
   461→   461→  // Remove the marker's chars from the head of the runs.
   462→   462→  let remaining = length;
   463→   463→  for (const run of runs) {
   464→   464→    if (remaining <= 0) break;
   465→   465→    if (run.text.length <= remaining) {
   466→   466→      remaining -= run.text.length;
   467→   467→      run.text = "";
   468→   468→    } else {
   469→   469→      run.text = run.text.slice(remaining);
   470→   470→      remaining = 0;
   471→   471→    }
   472→   472→  }
   473→   473→  return marker;
   474→   474→}

   476→   476→/** Groups pieces into visual lines with style- and link-aware runs. */
   477→   477→function buildLines(pieces: Piece[]): Line[] {
   478→   478→  if (pieces.length === 0) return [];
   479→   479→  const sorted = [...pieces].sort((a, b) => b.y - a.y || a.x - b.x);
   480→   480→
   481→   481→  const groups: Piece[][] = [];
   482→   482→  let current: Piece[] = [sorted[0] as Piece];
   483→   483→  let groupY = (sorted[0] as Piece).y;
   484→   484→  let groupTol = Math.max(1.5, (sorted[0] as Piece).size * LINE_Y_TOL);
   485→   485→
   486→   486→  for (let i = 1; i < sorted.length; i += 1) {
   487→   487→    const piece = sorted[i] as Piece;
   488→   488→    if (Math.abs(piece.y - groupY) <= groupTol) {
   489→   489→      current.push(piece);
   490→   490→    } else {
   491→   491→      groups.push(current);
   492→   492→      current = [piece];
   493→   493→      groupY = piece.y;
   494→   494→      groupTol = Math.max(1.5, piece.size * LINE_Y_TOL);
   495→   495→    }
   496→   496→  }
   497→   497→  groups.push(current);
   498→   498→
   499→   499→  return groups.map((group) => {
   500→   500→    const ordered = [...group].sort((a, b) => a.x - b.x);
   501→   501→    const size = group[0]?.size ?? 10;
   502→   502→    const runs: LineRun[] = [];
   503→   503→    let endX: number | null = null;
   504→   504→    for (const piece of ordered) {
   505→   505→      const last = runs[runs.length - 1];
   506→   506→      if (
   507→   507→        endX !== null &&
   508→   508→        piece.x - endX > size * SPACE_GAP &&
   509→   509→        !last?.text.endsWith(" ")
   510→   510→      ) {
   511→   511→        // pdf.js items carry no spaces — a real gap becomes one.
   512→   512→        if (last) last.text += " ";
   513→   513→        else {
   514→   514→          runs.push({
   515→   515→            text: " ",
   516→   516→            bold: piece.bold,
   517→   517→            italic: piece.italic,
   518→   518→            size: roundSize(piece.size),
   519→   519→            link: piece.link,
   520→   520→          });
   521→   521→        }
   522→   522→      }
   523→   523→      const sameStyle =
   524→   524→        last !== undefined &&
   525→   525→        last.bold === piece.bold &&
   526→   526→        last.italic === piece.italic &&
   527→   527→        last.size === roundSize(piece.size) &&
   528→   528→        last.link === piece.link;
   529→   529→      if (sameStyle) {
   530→   530→        last.text += piece.text;
   531→   531→      } else {
   532→   532→        runs.push({
   533→   533→          text: piece.text,
   534→   534→          bold: piece.bold,
   535→   535→          italic: piece.italic,
   536→   536→          size: roundSize(piece.size),
   537→   537→          link: piece.link,
   538→   538→        });
   539→   539→      }
   540→   540→      endX = Math.max(endX ?? 0, piece.x + piece.w);
   541→   541→    }
   542→   542→    // Merge identical-style neighbours created by the space glue above.
   543→   543→    const merged: LineRun[] = [];
   544→   544→    for (const run of runs) {
   545→   545→      const prev = merged[merged.length - 1];
   546→   546→      if (
   547→   547→        prev &&
   548→   548→        prev.bold === run.bold &&
   549→   549→        prev.italic === run.italic &&
   550→   550→        prev.size === run.size &&
   551→   551→        prev.link === run.link
   552→   552→      ) {
   553→   553→        prev.text += run.text;
   554→   554→      } else {
   555→   555→        merged.push({ ...run });
   556→   556→      }
   557→   557→    }
   558→   558→    const totalChars = merged.reduce(
   559→   559→      (sum, run) => sum + run.text.replace(/\s/g, "").length,
   560→   560→      0
   561→   561→    );
   562→   562→    const italicChars = merged.reduce(
   563→   563→      (sum, run) =>
   564→   564→        run.italic ? sum + run.text.replace(/\s/g, "").length : sum,
   565→   565→      0
   566→   566→    );
   567→   567→    const marker = stripListMarker(merged);
   568→   568→    return {
   569→   569→      runs: merged.filter((run) => run.text.length > 0),
   570→   570→      y: group[0]?.y ?? 0,
   571→   571→      size,
   572→   572→      x0: ordered[0]?.x ?? 0,
   573→   573→      italicShare: totalChars > 0 ? italicChars / totalChars : 0,
   574→   574→      marker,
   575→   575→    };
   576→   576→  });
   577→   577→}

   583→   583→/** Char-weighted dominant size of a run list (rounded to 0.5 pt). */
   584→   584→function dominantSize(runs: readonly LineRun[]): number {
   585→   585→  let weighted = 0;
   586→   586→  let total = 0;
   587→   587→  for (const run of runs) {
   588→   588→    weighted += run.size * run.text.length;
   589→   589→    total += run.text.length;
   590→   590→  }
   591→   591→  if (total === 0) return 10;
   592→   592→  return Math.round((weighted / total) * 2) / 2;
   593→   593→}

   595→   595→/** Appends one line's runs to a growing block (dehyphenation aware). */
   596→   596→function appendLine(target: LineRun[], line: Line): void {
   597→   597→  if (target.length === 0) {
   598→   598→    target.push(...line.runs.map((run) => ({ ...run })));
   599→   599→    return;
   600→   600→  }
   601→   601→  if (
   602→   602→    SCRIPT_END_RE.test(target[target.length - 1]?.text ?? "") &&
   603→   603→    LOWER_START_RE.test(line.runs[0]?.text ?? "")
   604→   604→  ) {
   605→   605→    // Dehyphenate: "exam-" + "ple" → "example".
   606→   606→    const last = target[target.length - 1] as LineRun;
   607→   607→    const first = line.runs[0] as LineRun;
   608→   608→    last.text = last.text.slice(0, -1);
   609→   609→    if (
   610→   610→      first.bold === last.bold &&
   611→   611→      first.italic === last.italic &&
   612→   612→      first.size === last.size &&
   613→   613→      first.link === last.link
   614→   614→    ) {
   615→   615→      last.text += first.text;
   616→   616→      target.push(...line.runs.slice(1).map((run) => ({ ...run })));
   617→   617→    } else {
   618→   618→      target.push(...line.runs.map((run) => ({ ...run })));
   619→   619→    }
   620→   620→    return;
   621→   621→  }
   622→   622→  const last = target[target.length - 1];
   623→   623→  if (last && !last.text.endsWith(" ")) last.text += " ";
   624→   624→  target.push(...line.runs.map((run) => ({ ...run })));
   625→   625→}

   627→   627→interface RawBlock {
   628→   628→  runs: LineRun[];
   629→   629→  x0: number;
   630→   630→  italicShare: number;
   631→   631→  marker: LineMarker | null;
   632→   632→}

   634→   634→/** Merges one page's lines into blocks (paragraphs + list items). */
   635→   635→function linesToBlocks(lines: readonly Line[]): RawBlock[] {
   636→   636→  const blocks: RawBlock[] = [];
   637→   637→  let runs: LineRun[] = [];
   638→   638→  let x0 = 0;
   639→   639→  let italicChars = 0;
   640→   640→  let plainChars = 0;
   641→   641→  let marker: LineMarker | null = null;
   642→   642→  let prevY: number | null = null;
   643→   643→  let prevSize = 10;
   644→   644→
   645→   645→  const flush = (): void => {
   646→   646→    if (runs.length === 0) return;
   647→   647→    const totalChars = italicChars + plainChars;
   648→   648→    blocks.push({
   649→   649→      runs: runs
   650→   650→        .map((run) => ({ ...run, text: run.text.slice(0, MAX_RUN_CHARS) }))
   651→   651→        .filter((run) => run.text.trim().length > 0),
   652→   652→      x0,
   653→   653→      italicShare: totalChars > 0 ? italicChars / totalChars : 0,
   654→   654→      marker,
   655→   655→    });
   656→   656→    runs = [];
   657→   657→    italicChars = 0;
   658→   658→    plainChars = 0;
   659→   659→    marker = null;
   660→   660→  };
   661→   661→
   662→   662→  const start = (line: Line): void => {
   663→   663→    runs = line.runs.map((run) => ({ ...run }));
   664→   664→    x0 = line.x0;
   665→   665→    marker = line.marker;
   666→   666→    for (const run of runs) {
   667→   667→      const chars = run.text.replace(/\s/g, "").length;
   668→   668→      if (run.italic) italicChars += chars;
   669→   669→      else plainChars += chars;
   670→   670→    }
   671→   671→  };
   672→   672→
   673→   673→  for (const line of lines) {
   674→   674→    if (line.runs.length === 0) continue;
   675→   675→    if (line.marker !== null) {
   676→   676→      // A marked line ALWAYS starts its own item — never merges with
   677→   677→      // the previous paragraph or item.
   678→   678→      flush();
   679→   679→      start(line);
   680→   680→    } else if (runs.length === 0) {
   681→   681→      start(line);
   682→   682→    } else {
   683→   683→      const gap = prevY !== null ? prevY - line.y : 0;
   684→   684→      const tight =
   685→   685→        gap <= prevSize * PARAGRAPH_GAP &&
   686→   686→        Math.abs(line.size - prevSize) / Math.max(prevSize, 1) <= SIZE_CHANGE;
   687→   687→      if (tight) {
   688→   688→        // Soft-wrap continuation: joins the paragraph or list item.
   689→   689→        appendLine(runs, line);
   690→   690→        for (const run of line.runs) {
   691→   691→          const chars = run.text.replace(/\s/g, "").length;
   692→   692→          if (run.italic) italicChars += chars;
   693→   693→          else plainChars += chars;
   694→   694→        }
   695→   695→      } else {
   696→   696→        flush();
   697→   697→        start(line);
   698→   698→      }
   699→   699→    }
   700→   700→    prevY = line.y;
   701→   701→    prevSize = line.size;
   702→   702→  }
   703→   703→  flush();
   704→   704→  return blocks;
   705→   705→}

   711→   711→export function extractMarkdownModel(
   712→   712→  pages: readonly PdfTextItemInput[][],
   713→   713→  opts: ExtractOptions = {}
   714→   714→): MdDocModel {
   715→   715→  const headingsOn = opts.headings !== false;
   716→   716→
   717→   717→  // Pass 1 — everything as lines, plus the char-weighted body size and
   718→   718→  // the document's modal left edge (quote detection).
   719→   719→  const pageLines: Line[][] = [];
   720→   720→  const sizeWeight = new Map<number, number>();
   721→   721→  const leftWeight = new Map<number, number>();
   722→   722→  for (let index = 0; index < pages.length; index += 1) {
   723→   723→    const links = opts.links?.[index];
   724→   724→    const lines = buildLines(
   725→   725→      collectPieces(pages[index], opts.styleMap, links)
   726→   726→    ).filter((line) => line.runs.length > 0);
   727→   727→    pageLines.push(lines);
   728→   728→    for (const line of lines) {
   729→   729→      for (const run of line.runs) {
   730→   730→        const sizeKey = Math.round(run.size * 2) / 2;
   731→   731→        sizeWeight.set(sizeKey, (sizeWeight.get(sizeKey) ?? 0) + run.text.length);
   732→   732→      }
   733→   733→      const leftKey = Math.round(line.x0 * 2) / 2;
   734→   734→      leftWeight.set(
   735→   735→        leftKey,
   736→   736→        (leftWeight.get(leftKey) ?? 0) +
   737→   737→          line.runs.reduce((sum, run) => sum + run.text.trim().length, 0)
   738→   738→      );
   739→   739→    }
   740→   740→  }
   741→   741→
   742→   742→  let bodySize = 11;
   743→   743→  let bestWeight = -1;
   744→   744→  for (const [size, weight] of sizeWeight) {
   745→   745→    if (weight > bestWeight) {
   746→   746→      bestWeight = weight;
   747→   747→      bodySize = size;
   748→   748→    }
   749→   749→  }
   750→   750→  let bodyLeft = Number.NaN;
   751→   751→  let bestLeftWeight = -1;
   752→   752→  for (const [left, weight] of leftWeight) {
   753→   753→    if (weight > bestLeftWeight) {
   754→   754→      bestLeftWeight = weight;
   755→   755→      bodyLeft = left;
   756→   756→    }
   757→   757→  }
   758→   758→
   759→   759→  // Heading tiers: distinct sizes clearly above the body size, largest
   760→   760→  // first. Tier i becomes level i+1, capped at 6 (######).
   761→   761→  const tierMin = Math.max(
   762→   762→    bodySize * HEADING_RATIO,
   763→   763→    bodySize + HEADING_ABS_PT
   764→   764→  );
   765→   765→  const tiers: number[] = [];
   766→   766→  for (const size of sizeWeight.keys()) {
   767→   767→    if (size >= tierMin && (sizeWeight.get(size) ?? 0) >= TIER_MIN_CHARS) {
   768→   768→      tiers.push(size);
   769→   769→    }
   770→   770→  }
   771→   771→  tiers.sort((a, b) => b - a);
   772→   772→  const headingSize =
   773→   773→    headingsOn && tiers.length > 0
   774→   774→      ? new Map<number, 1 | 2 | 3 | 4 | 5 | 6>()
   775→   775→      : null;
   776→   776→  if (headingSize) {
   777→   777→    tiers.forEach((size, index) => {
   778→   778→      const level = Math.min(index + 1, 6) as 1 | 2 | 3 | 4 | 5 | 6;
   779→   779→      headingSize.set(size, level);
   780→   780→    });
   781→   781→  }
   782→   782→
   783→   783→  // Pass 2 — blocks + page marks + classification. The mark is pushed
   784→   784→  // BEFORE the page's blocks, so every page (empty ones included) owns
   785→   785→  // the index its content starts at — pageSlice depends on that.
   786→   786→  const blocks: MdBlock[] = [];
   787→   787→  const pageMarks: number[] = [];
   788→   788→  for (const lines of pageLines) {
   789→   789→    pageMarks.push(blocks.length);
   790→   790→    const pageBlocks = linesToBlocks(lines);
   791→   791→    for (const raw of pageBlocks) {
   792→   792→      const level = headingSize?.get(dominantSize(raw.runs));
   793→   793→      if (level !== undefined) {
   794→   794→        // Heading: plain, anchor-friendly text (inline markers dropped).
   795→   795→        blocks.push({
   796→   796→          kind: "heading",
   797→   797→          level,
   798→   798→          text: fixDevanagariVisual(
   799→   799→            raw.runs.map((run) => run.text).join("").replace(/\s+/g, " ").trim()
   800→   800→          ),
   801→   801→        });
   802→   802→      } else if (
   803→   803→        raw.marker === null &&
   804→   804→        Number.isFinite(bodyLeft) &&
   805→   805→        raw.x0 >= bodyLeft + QUOTE_INDENT_PT &&
   806→   806→        raw.italicShare >= QUOTE_ITALIC_SHARE
   807→   807→      ) {
   808→   808→        blocks.push({ kind: "quote", runs: runsOut(raw.runs) });
   809→   809→      } else if (raw.marker?.kind === "bullet") {
   810→   810→        blocks.push({ kind: "bullet", runs: runsOut(raw.runs) });
   811→   811→      } else if (raw.marker?.kind === "ordered") {
   812→   812→        blocks.push({
   813→   813→          kind: "ordered",
   814→   814→          number: Math.max(1, Math.min(9999, raw.marker.number ?? 1)),
   815→   815→          runs: runsOut(raw.runs),
   816→   816→        });
   817→   817→      } else {
   818→   818→        blocks.push({ kind: "para", runs: runsOut(raw.runs) });
   819→   819→      }
   820→   820→    }
   821→   821→  }
   822→   822→
   823→   823→  return {
   824→   824→    blocks: blocks.slice(0, MAX_BLOCKS),
   825→   825→    pageMarks,
   826→   826→  };
   827→   827→}

   829→   829→/** Internal LineRun list → the exported MdRun shape (Devanagari repaired). */
   830→   830→function runsOut(runs: readonly LineRun[]): MdRun[] {
   831→   831→  const out: MdRun[] = [];
   832→   832→  for (const run of runs) {
   833→   833→    const text = fixDevanagariVisual(run.text);
   834→   834→    if (text.trim().length === 0) continue;
   835→   835→    const last = out[out.length - 1];
   836→   836→    if (
   837→   837→      last &&
   838→   838→      last.bold === run.bold &&
   839→   839→      last.italic === run.italic &&
   840→   840→      last.link === run.link
   841→   841→    ) {
   842→   842→      last.text += text;
   843→   843→    } else {
   844→   844→      out.push({ text, bold: run.bold, italic: run.italic, link: run.link });
   845→   845→    }
   846→   846→  }
   847→   847→  return out;
   848→   848→}

   869→   869→/**
   870→   870→ * Escapes a literal markdown-significant start: the task's * # _ [ set
   871→   871→ * plus + > - and ordered-list markers ("2023. Year" → "2023\. Year").
   872→   872→ * Mid-text characters are left alone — CommonMark needs marker chars to
   873→   873→ * be line-initial (and followed by space) to mean anything.
   874→   874→ */
   875→   875→function escapeLeading(text: string): string {
   876→   876→  if (text.length === 0) return text;
   877→   877→  const first = text[0] as string;
   878→   878→  if ("#*_+[>-".includes(first)) return `\\${text}`;
   879→   879→  const ordered = /^(\d{1,9})([.)])(\s|$)/.exec(text);
   880→   880→  if (ordered !== null) {
   881→   881→    const digits = ordered[1] as string;
   882→   882→    return `${digits}\\${text.slice(digits.length)}`;
   883→   883→  }
   884→   884→  return text;
   885→   885→}

   901→   901→/** Renders a run list as one inline Markdown string. */
   902→   902→function inlineRuns(runs: readonly MdRun[]): string {
   903→   903→  let out = "";
   904→   904→  let pendingSpace = false;
   905→   905→  let emitted = false;
   906→   906→  let firstCore = true;
   907→   907→  for (const run of runs) {
   908→   908→    const core = run.text.replace(/\s+/g, " ").trim();
   909→   909→    if (core.length === 0) {
   910→   910→      if (emitted) pendingSpace = true;
   911→   911→      continue;
   912→   912→    }
   913→   913→    const leadSpace = /^\s/.test(run.text);
   914→   914→    const trailSpace = /\s$/.test(run.text);
   915→   915→    if (emitted && (pendingSpace || leadSpace)) out += " ";
   916→   916→    let body: string;
   917→   917→    if (run.link !== undefined) {
   918→   918→      // Our own [..](..) markup: escape the brackets inside the label.
   919→   919→      body = `[${core.replace(/([[\]])/g, "\\$1")}](${safeUrl(run.link)})`;
   920→   920→    } else {
   921→   921→      // Line-start escaping applies only to the first visible core —
   922→   922→      // our own emphasis/link markers wrap later ones anyway.
   923→   923→      body = firstCore ? escapeLeading(core) : core;
   924→   924→    }
   925→   925→    firstCore = false;
   926→   926→    const emphasis =
   927→   927→      run.bold && run.italic ? "***" : run.bold ? "**" : run.italic ? "*" : "";
   928→   928→    out += emphasis + body + emphasis;
   929→   929→    pendingSpace = trailSpace;
   930→   930→    emitted = true;
   931→   931→  }
   932→   932→  return out;
   933→   933→}
   934→   934→
   935→   935→/** One block → its Markdown line. */
   936→   936→function blockLine(block: MdBlock): string {
   937→   937→  switch (block.kind) {
   938→   938→    case "heading":
   939→   939→      return `${"#".repeat(block.level)} ${escapeLeading(block.text)}`;
   940→   940→    case "quote":
   941→   941→      return `> ${inlineRuns(block.runs)}`;
   942→   942→    case "bullet":
   943→   943→      return `- ${inlineRuns(block.runs)}`;
   944→   944→    case "ordered":
   945→   945→      return `${block.number}. ${inlineRuns(block.runs)}`;
   946→   946→    case "para":
   947→   947→      return inlineRuns(block.runs);
   948→   948→  }
   949→   949→}
   950→   950→
   951→   951→/** True when two consecutive blocks belong to ONE list (single \n join). */
   952→   952→function sameList(a: MdBlock, b: MdBlock): boolean {
   953→   953→  if (a.kind === "bullet" && b.kind === "bullet") return true;
   954→   954→  if (a.kind === "ordered" && b.kind === "ordered") return true;
   955→   955→  return false;
   956→   956→}
   957→   957→
   958→   958→/**
   959→   959→ * Builds the final Markdown document:
   960→   960→ *  - blocks joined with blank lines; consecutive same-kind list items
   961→   961→ *    share single newlines (a tight list);
   962→   962→ *  - optional "---" between PDF pages (single-output mode);
   963→   963→ *  - optional YAML front matter and provenance comment on top.
   964→   964→ */
   965→   965→export function buildMarkdown(
   966→   966→  doc: MdDocModel,
   967→   967→  opts: BuildMarkdownOptions = {}
   968→   968→): string {
   969→   969→  const groups: string[] = [];
   970→   970→  let current: string[] = [];
   971→   971→  let openBlock: MdBlock | null = null;
   972→   972→
   973→   973→  const flushGroup = (): void => {
   974→   974→    if (current.length > 0) groups.push(current.join("\n"));
   975→   975→    current = [];
   976→   976→    openBlock = null;
   977→   977→  };
   978→   978→
   979→   979→  const pageStarts = new Set(doc.pageMarks.slice(1));
   980→   980→  doc.blocks.forEach((block, index) => {
   981→   981→    if (opts.pageRules === true && pageStarts.has(index)) {
   982→   982→      flushGroup();
   983→   983→      groups.push("---");
   984→   984→    }
   985→   985→    if (openBlock !== null && sameList(openBlock, block)) {
   986→   986→      current.push(blockLine(block));
   987→   987→    } else {
   988→   988→      flushGroup();
   989→   989→      current.push(blockLine(block));
   990→   990→      openBlock = block;
   991→   991→    }
   992→   992→  });
   993→   993→  flushGroup();
   994→   994→
   995→   995→  const body = groups.join("\n\n");
   996→   996→
   997→   997→  let head = "";
   998→   998→  if (opts.frontMatter === true) {
   999→   999→    const title = (opts.title ?? "").replace(/\s+/g, " ").trim();
  1000→  1000→    const source = (opts.sourceName ?? "").replace(/\s+/g, " ").trim();
  1001→  1001→    head += "---\n";
  1002→  1002→    if (title.length > 0) head += `title: ${yamlQuote(title)}\n`;
  1003→  1003→    if (source.length > 0) head += `source: ${yamlQuote(source)}\n`;
  1004→  1004→    head += `pages: ${Math.max(1, opts.pages ?? doc.pageMarks.length)}\n`;
  1005→  1005→    head += "converted_by: MyCloudy\n---\n\n";
  1006→  1006→  }
  1007→  1007→  if (opts.metaComment === true) {
  1008→  1008→    head += `${MD_META_COMMENT}\n\n`;
  1009→  1009→  }
  1010→  1010→
  1011→  1011→  if (body.length === 0) {
  1012→  1012→    return head.length > 0 ? `${head.trimEnd()}\n` : "";
  1013→  1013→  }
  1014→  1014→  return `${head}${body}\n`;
  1015→  1015→}
  1016→  1016→
  1017→  1017→/**
  1018→  1018→ * Slices the model to a single page's blocks (per-page output mode).
  1019→  1019→ * Heading tiers stay document-wide — only the emitted blocks narrow.
  1020→  1020→ */
  1021→  1021→export function pageSlice(doc: MdDocModel, pageIndex: number): MdDocModel {
  1022→  1022→  const start = doc.pageMarks[pageIndex] ?? doc.blocks.length;
  1023→  1023→  const end =
  1024→  1024→    pageIndex + 1 < doc.pageMarks.length
  1025→  1025→      ? (doc.pageMarks[pageIndex + 1] as number)
  1026→  1026→      : doc.blocks.length;
  1027→  1027→  return {
  1028→  1028→    blocks: doc.blocks.slice(start, Math.max(start, end)),
  1029→  1029→    pageMarks: start < end ? [0] : [],
  1030→  1030→  };
  1031→  1031→}

  1049→  1049→/**
  1050→  1050→ * coerceRunOptions — the options payload the surface/shell sends.

  1052→  1052→ * switch; the locale must be exactly "hi"; the page spec is sanitized

  1056→  1056→  raw: Record<string, unknown>

  1059→  1059→    typeof v === "string"
  1060→  1060→      ? v.replace(/[\u0000-\u001f]/g, "").trim().slice(0, max)
  1061→  1061→      : "";

  1063→  1063→    pages: str(raw.pages, 200) || "all",
  1064→  1064→    perPage: raw.perPage === true,
  1065→  1065→    pageRules: raw.pageRules === true,
  1066→  1066→    metaComment: raw.metaComment === true,
  1067→  1067→    frontMatter: raw.frontMatter === true,
  1068→  1068→    locale: raw.locale === "hi" ? "hi" : "en",
  1069→  1069→    fileWord: str(raw.fileWord, 40),
  1070→  1070→  };
  1071→  1071→}

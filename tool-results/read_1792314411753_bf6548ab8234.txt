     1→/**
     2→ * Card maker shared model (PRD §"Card" — the 63rd tool) — PURE data +
     3→ * math used by the surface (live preview), the worker (PDF/JPG/PNG
     4→ * export) and the smoke script, so what you preview is exactly what
     5→ * prints. No platform API is touched here; every text-layout routine
     6→ * takes an injected `measure` function so the battery can verify the
     7→ * math with a fake measurer.
     8→ *
     9→ * ── FOLD MODEL (fold-a4) ────────────────────────────────────────────
    10→ * One A4 LANDSCAPE sheet (297×210 mm) per printed side, folded once at
    11→ * the vertical centre → the folded card is A5-portrait-sized. The PDF
    12→ * has TWO pages per card:
    13→ *   Page 1 OUTSIDE — LEFT half = back cover, RIGHT half = front cover.
    14→ *   Page 2 INSIDE  — LEFT half = inside-left (date/venue panel),
    15→ *                    RIGHT half = inside-right (the main message panel,
    16→ *                    i.e. the back of the front cover).
    17→ * Printing advice given in the FAQ (honest): print page 1, re-insert
    18→ * the sheet to print page 2 on the blank side in the same orientation,
    19→ * then fold at the middle line — and do a TEST PRINT first. Which
    20→ * duplex flip mode a printer needs (long-edge vs short-edge) varies by
    21→ * model, so this tool deliberately does NOT assume one.
    22→ * The fold line is exactly W/2 and both halves cover the full sheet
    23→ * (smoke asserts this).
    24→ *
    25→ * ── TEXT-RESET RULES (standard card-maker behaviour) ─────────────────
    26→ * Switching TEMPLATE resets every text field to the new template's
    27→ * defaults for the CURRENT text locale (tracked when the texts were
    28→ * last set — Canva-like). Switching the UI LANGUAGE never touches
    29→ * typed texts; it only re-serializes the option (the output file name
    30→ * is localized). Both rules live in the surface; documented here so
    31→ * the model and the UI agree.
    32→ */
    33→
    34→import type { L10n } from "@/lib/tools/registry";

    36→/* --------------------------- exact geometry --------------------------- */
    37→
    38→/** mm → PDF points, exact factor. */
    39→export const PT_PER_MM = 72 / 25.4;
    40→
    41→/** Worker/export canvases render at 2 px per PDF point (collage policy). */
    42→export const PX_PER_PT = 2;
    43→
    44→export type CardFormat = "flat-a6" | "flat-a5" | "flat-4x6" | "fold-a4";
    45→export type CardOrientation = "portrait" | "landscape";
    46→export type ExportFormat = "pdf" | "jpg" | "png";
    47→
    48→/** Trim sizes in mm. fold-a4 is the A4 LANDSCAPE sheet, folded once. */
    49→export const FORMAT_MM: Record<CardFormat, [w: number, h: number]> = {
    50→  "flat-a6": [105, 148],
    51→  "flat-a5": [148, 210],
    52→  "flat-4x6": [101.6, 152.4],
    53→  "fold-a4": [297, 210],
    54→};
    55→
    56→export const CARD_FORMATS = ["flat-a6", "flat-a5", "flat-4x6", "fold-a4"] as const;
    57→export const CARD_ORIENTATIONS = ["portrait", "landscape"] as const;
    58→export const EXPORT_FORMATS = ["pdf", "jpg", "png"] as const;
    59→
    60→export interface SizePt {
    61→  w: number;
    62→  h: number;
    63→}
    64→
    65→/**
    66→ * Page size in PDF points. fold-a4 is ALWAYS the landscape sheet
    67→ * (297×210 mm) — orientation only matters for the flat formats.
    68→ */
    69→export function pagePtSize(
    70→  format: CardFormat,
    71→  orientation: CardOrientation
    72→): SizePt {
    73→  const [mmW, mmH] = FORMAT_MM[format];
    74→  const ptW = mmW * PT_PER_MM;
    75→  const ptH = mmH * PT_PER_MM;
    76→  const swap = format !== "fold-a4" && orientation === "landscape";
    77→  return swap ? { w: ptH, h: ptW } : { w: ptW, h: ptH };
    78→}
    79→
    80→/** Worker/export canvas size in pixels (pt × PX_PER_PT, rounded). */
    81→export function pagePixelSize(
    82→  format: CardFormat,
    83→  orientation: CardOrientation
    84→): { w: number; h: number } {
    85→  const pt = pagePtSize(format, orientation);
    86→  return {
    87→    w: Math.round(pt.w * PX_PER_PT),
    88→    h: Math.round(pt.h * PX_PER_PT),
    89→  };
    90→}

    92→/* ------------------------------ sheets ------------------------------ */
    93→
    94→export type SheetKind = "flat" | "fold-outside" | "fold-inside";
    95→
    96→export interface CardSheet {
    97→  kind: SheetKind;
    98→  /** 0-based mail-merge row this sheet belongs to. */
    99→  rowIndex: number;
   100→}
   101→
   102→/**
   103→ * Sheets of one export, in print order. Flat formats make one sheet
   104→ * per merge row; fold-a4 makes TWO per row (outside then inside).
   105→ * N rows → flat: N sheets / fold: 2N sheets.
   106→ */
   107→export function computeSheets(
   108→  format: CardFormat,
   109→  mergeRows: number
   110→): CardSheet[] {
   111→  const rows = Math.max(1, Math.floor(mergeRows));
   112→  const sheets: CardSheet[] = [];
   113→  for (let row = 0; row < rows; row += 1) {
   114→    if (format === "fold-a4") {
   115→      sheets.push({ kind: "fold-outside", rowIndex: row });
   116→      sheets.push({ kind: "fold-inside", rowIndex: row });
   117→    } else {
   118→      sheets.push({ kind: "flat", rowIndex: row });
   119→    }
   120→  }
   121→  return sheets;
   122→}
   123→
   124→export interface RectPt {
   125→  x: number;
   126→  y: number;
   127→  w: number;
   128→  h: number;
   129→}
   130→
   131→/**
   132→ * Panels of one printed sheet in PDF points. Folded sheets are EXACT
   133→ * halves: the fold line sits at W/2 and both halves cover the full
   134→ * sheet height. Panel order matches the fold model above:
   135→ *   fold-outside → [back (left), front (right)]
   136→ *   fold-inside  → [inside-left, inside-right]
   137→ *   flat         → [whole page]
   138→ */
   139→export function sheetPanelsPt(
   140→  format: CardFormat,
   141→  orientation: CardOrientation,
   142→  kind: SheetKind
   143→): RectPt[] {
   144→  const { w, h } = pagePtSize(format, orientation);
   145→  if (kind === "flat") return [{ x: 0, y: 0, w, h }];
   146→  const half = w / 2;
   147→  return [
   148→    { x: 0, y: 0, w: half, h },
   149→    { x: half, y: 0, w: half, h },
   150→  ];
   151→}

   459→
   460→const HEX_RE = /^#[0-9a-fA-F]{6}$/;
   461→
   462→/** #rrggbb or the fallback (soft warm paper tone). */
   463→export function sanitizeHexColor(value: unknown, fallback = "#f5f1ea"): string {
   464→  return typeof value === "string" && HEX_RE.test(value.trim())
   465→    ? value.trim().toLowerCase()
   466→    : fallback;
   467→}
   468→
   469→/** Perceived luminance 0..1. */
   470→export function hexLuminance(hex: string): number {
   471→  const clean = sanitizeHexColor(hex, "#ffffff").slice(1);
   472→  const r = Number.parseInt(clean.slice(0, 2), 16) / 255;
   473→  const g = Number.parseInt(clean.slice(2, 4), 16) / 255;
   474→  const b = Number.parseInt(clean.slice(4, 6), 16) / 255;
   475→  return 0.2126 * r + 0.7152 * g + 0.0722 * b;
   476→}
   477→
   478→/** Dark ink on light backgrounds, light ink on dark ones (blank cards). */
   479→export function inkHex(background: string): string {
   480→  return hexLuminance(background) > 0.55 ? "#1c1917" : "#faf9f7";
   481→}

   492→/* --------------------------- text wrapping --------------------------- */
   493→
   494→/**
   495→ * Word-wraps `text` to `maxWidth` using an injected measure function
   496→ * (PURE — the smoke verifies it with a fake measurer).
   497→ *
   498→ * Rules:
   499→ *  - Splits on "\n" first: an explicit newline is always a hard break.
   500→ *  - Each paragraph wraps by spaces (runs of spaces collapse to one).
   501→ *  - A word wider than the line breaks CHARACTER-WISE (Devanagari text
   502→ *    often contains long unbroken strings); only a single character
   503→ *    wider than the whole line is ever emitted over width.
   504→ *  - Trailing whitespace is trimmed per emitted line.
   505→ *  - Whitespace-only text → zero lines. Blank paragraphs at the very
   506→ *    start/end are dropped; blank paragraphs BETWEEN text are kept as
   507→ *    one empty line (a deliberate blank line).
   508→ */
   509→export function wrapText(
   510→  text: string,
   511→  maxWidth: number,
   512→  measure: (t: string) => number
   513→): string[] {
   514→  if (typeof text !== "string" || text.trim() === "") return [];
   515→  if (!(maxWidth > 0)) return [];
   516→  const paragraphs = text.split("\n");
   517→  let start = 0;
   518→  let end = paragraphs.length - 1;
   519→  while (start <= end && paragraphs[start].trim() === "") start += 1;
   520→  while (end >= start && paragraphs[end].trim() === "") end -= 1;
   521→  const lines: string[] = [];
   522→  for (let i = start; i <= end; i += 1) {
   523→    const paragraph = paragraphs[i];
   524→    if (paragraph.trim() === "") {
   525→      lines.push("");
   526→      continue;
   527→    }
   528→    lines.push(...wrapParagraph(paragraph, maxWidth, measure));
   529→  }
   530→  return lines;
   531→}
   532→
   533→function wrapParagraph(
   534→  paragraph: string,
   535→  maxWidth: number,
   536→  measure: (t: string) => number
   537→): string[] {
   538→  const words = paragraph.split(" ").filter((word) => word !== "");
   539→  const lines: string[] = [];
   540→  let line = "";
   541→  for (const word of words) {
   542→    const candidate = line === "" ? word : `${line} ${word}`;
   543→    if (measure(candidate) <= maxWidth) {
   544→      line = candidate;
   545→      continue;
   546→    }
   547→    if (line !== "") {
   548→      lines.push(line);
   549→      line = "";
   550→    }
   551→    if (measure(word) <= maxWidth) {
   552→      line = word;
   553→      continue;
   554→    }
   555→    // Character-break the over-long word; every chunk except the last
   556→    // becomes its own (full) line, the last stays open for more words.
   557→    const chunks = breakWord(word, maxWidth, measure);
   558→    for (let c = 0; c < chunks.length - 1; c += 1) lines.push(chunks[c]);
   559→    line = chunks[chunks.length - 1] ?? "";
   560→  }
   561→  if (line !== "") lines.push(line);
   562→  return lines;
   563→}
   564→
   565→function breakWord(
   566→  word: string,
   567→  maxWidth: number,
   568→  measure: (t: string) => number
   569→): string[] {
   570→  const chunks: string[] = [];
   571→  let chunk = "";
   572→  for (const char of [...word]) {
   573→    const next = chunk + char;
   574→    if (chunk !== "" && measure(next) > maxWidth) {
   575→      chunks.push(chunk);
   576→      chunk = char;
   577→    } else {
   578→      chunk = next;
   579→    }
   580→  }
   581→  if (chunk !== "") chunks.push(chunk);
   582→  return chunks;
   583→}

   585→/**
   586→ * Longest prefix of `text` + "…" that fits `maxWidth` (binary search).
   587→ * "" → ""; nothing fits → the bare ellipsis.
   588→ */
   589→export function truncateWithEllipsis(
   590→  text: string,
   591→  maxWidth: number,
   592→  measure: (t: string) => number
   593→): string {
   594→  if (text === "") return "";
   595→  if (measure(`${text}\u2026`) <= maxWidth) return `${text}\u2026`;
   596→  const chars = [...text];
   597→  let lo = 0;
   598→  let hi = chars.length;
   599→  while (lo < hi) {
   600→    const mid = Math.ceil((lo + hi) / 2);
   601→    if (measure(`${chars.slice(0, mid).join("")}\u2026`) <= maxWidth) lo = mid;
   602→    else hi = mid - 1;
   603→  }
   604→  return `${chars.slice(0, lo).join("")}\u2026`;
   605→}

   609→export interface HeadlineFit {
   610→  /** The chosen font size in pt. */
   611→  sizePt: number;
   612→  /** Wrapped lines at that size (ALL of them — never truncated). */
   613→  lines: string[];
   614→  /** True when even the minimum size does not fit. */
   615→  overflowed: boolean;
   616→}
   617→
   618→export interface HeadlineOptions {
   619→  baseSizePt: number;
   620→  minSizePt: number;
   621→  /** Deterministic step-down (default 2). */
   622→  stepPt: number;
   623→  maxWidthPt: number;
   624→  maxLines: number;
   625→  lineHeightRatio: number;
   626→  maxHeightPt: number;
   627→}
   628→
   629→/**
   630→ * Headline shrink-to-fit: start at the template's base size, step down
   631→ * by `stepPt` (−2pt) until the wrapped headline fits the panel within
   632→ * the allowed line count and height budget. Returns the FIRST size
   633→ * that fits. If nothing fits down to the minimum, keeps the minimum
   634→ * size with its full wrap (never truncates silently) and flags
   635→ * `overflowed`. The loop is bounded by an explicit iteration cap, so
   636→ * it always terminates (smoke verifies with a fake measurer).
   637→ */
   638→export function fitHeadline(
   639→  text: string,
   640→  opts: HeadlineOptions,
   641→  measureAt: (t: string, sizePt: number) => number
   642→): HeadlineFit {
   643→  const step = opts.stepPt > 0 ? opts.stepPt : 2;
   644→  const min = Math.max(1, opts.minSizePt);
   645→  const base = Math.max(min, opts.baseSizePt);
   646→  if (typeof text !== "string" || text.trim() === "") {
   647→    return { sizePt: base, lines: [], overflowed: false };
   648→  }
   649→  const ITERATION_CAP = 512; // (base−min)/step + 1 in practice — hard cap.
   650→  let size = base;
   651→  for (let i = 0; i < ITERATION_CAP; i += 1) {
   652→    const lines = wrapText(text, opts.maxWidthPt, (t) => measureAt(t, size));
   653→    const fits =
   654→      lines.length <= opts.maxLines &&
   655→      lines.length * size * opts.lineHeightRatio <= opts.maxHeightPt;
   656→    if (fits) return { sizePt: size, lines, overflowed: false };
   657→    if (size <= min) return { sizePt: min, lines, overflowed: true };
   658→    size = Math.max(min, size - step);
   659→  }
   660→  const lines = wrapText(text, opts.maxWidthPt, (t) => measureAt(t, min));
   661→  return { sizePt: min, lines, overflowed: true };
   662→}

   664→/* ---------------------------- message body ---------------------------- */
   665→
   666→export interface MessageLayout {
   667→  lines: string[];
   668→  overflowed: boolean;
   669→}
   670→
   671→export interface MessageOptions {
   672→  fontPt: number;
   673→  lineHeightRatio: number;
   674→  maxWidthPt: number;
   675→  maxHeightPt: number;
   676→}
   677→
   678→/**
   679→ * Message body at a FIXED size. Wraps freely; when the measured block
   680→ * overflows the panel bottom, the last fitting line gets an ellipsis
   681→ * "…" and `overflowed` flags it (the surface shows an honest badge
   682→ * from the SAME engine's metrics).
   683→ */
   684→export function layoutMessage(
   685→  text: string,
   686→  opts: MessageOptions,
   687→  measureAt: (t: string, sizePt: number) => number
   688→): MessageLayout {
   689→  if (typeof text !== "string" || text.trim() === "") {
   690→    return { lines: [], overflowed: false };
   691→  }
   692→  const measure = (t: string): number => measureAt(t, opts.fontPt);
   693→  const lines = wrapText(text, opts.maxWidthPt, measure);
   694→  const lineH = opts.fontPt * opts.lineHeightRatio;
   695→  const maxLines = Math.max(1, Math.floor(opts.maxHeightPt / lineH));
   696→  if (lines.length <= maxLines) return { lines, overflowed: false };
   697→  const kept = lines.slice(0, maxLines);
   698→  kept[maxLines - 1] = truncateWithEllipsis(
   699→    kept[maxLines - 1] ?? "",
   700→    opts.maxWidthPt,
   701→    measure
   702→  );
   703→  return { lines: kept, overflowed: true };
   704→}

   708→export const MAX_CSV_ROWS = 200;
   709→/**
   710→ * 2 MB CSV file cap, measured in characters (a UTF-8 file never has
   711→ * more chars than bytes, so chars > cap ⇔ file > 2 MB for valid UTF-8).
   712→ */
   713→export const CSV_MAX_CHARS = 2 * 1024 * 1024;
   714→
   715→export class CsvError extends Error {
   716→  /** i18n error key: card.errorCsvQuote | card.errorCsvRows | card.errorCsvSize. */
   717→  code: string;
   718→  constructor(code: string) {
   719→    super(code);
   720→    this.name = "CsvError";
   721→    this.code = code;
   722→  }
   723→}

   725→export interface ParsedCsv {
   726→  /** First row, EXACTLY as written — case-sensitive, NOT trimmed. */
   727→  columns: string[];
   728→  rows: string[][];
   729→}
   730→
   731→/**
   732→ * RFC-4180 CSV parser (PURE):
   733→ *  - quoted fields with embedded commas and newlines; `""` escapes a
   734→ *    literal quote;
   735→ *  - CRLF, LF and lone CR line endings all work;
   736→ *  - a leading UTF-8 BOM is stripped;
   737→ *  - the first record is the header (column names preserved exactly);
   738→ *  - a SINGLE trailing empty line is skipped (more are kept);
   739→ *  - unterminated quote → CsvError("card.errorCsvQuote");
   740→ *  - more than MAX_CSV_ROWS data rows → CsvError("card.errorCsvRows");
   741→ *  - text larger than the 2 MB cap → CsvError("card.errorCsvSize").
   742→ * Header-only / empty input → zero rows = NO merge (single card).
   743→ */
   744→export function parseCsv(text: string): ParsedCsv {
   745→  if (typeof text !== "string") text = "";
   746→  if (text.startsWith("\uFEFF")) text = text.slice(1);
   747→  if (text.length > CSV_MAX_CHARS) {
   748→    throw new CsvError("card.errorCsvSize");
   749→  }
   750→  const records: string[][] = [];
   751→  let field = "";
   752→  let record: string[] = [];
   753→  let inQuotes = false;
   754→  for (let i = 0; i < text.length; i += 1) {
   755→    const char = text[i];
   756→    if (inQuotes) {
   757→      if (char === '"') {
   758→        if (text[i + 1] === '"') {
   759→          field += '"';
   760→          i += 1;
   761→        } else {
   762→          inQuotes = false;
   763→        }
   764→      } else {
   765→        field += char;
   766→      }
   767→      continue;
   768→    }
   769→    if (char === '"' && field === "") {
   770→      inQuotes = true;
   771→      continue;
   772→    }
   773→    if (char === ",") {
   774→      record.push(field);
   775→      field = "";
   776→      continue;
   777→    }
   778→    if (char === "\n" || char === "\r") {
   779→      if (char === "\r" && text[i + 1] === "\n") i += 1;
   780→      record.push(field);
   781→      field = "";
   782→      records.push(record);
   783→      record = [];
   784→      continue;
   785→    }
   786→    field += char;
   787→  }
   788→  if (inQuotes) throw new CsvError("card.errorCsvQuote");
   789→  if (field !== "" || record.length > 0) {
   790→    record.push(field);
   791→    records.push(record);
   792→  }
   793→  // Skip a single trailing empty line (an empty record with one field).
   794→  if (
   795→    records.length > 0 &&
   796→    records[records.length - 1].length === 1 &&
   797→    records[records.length - 1][0] === ""
   798→  ) {
   799→    records.pop();
   800→  }
   801→  const columns = records[0] ?? [];
   802→  const rows = records.slice(1);
   803→  if (rows.length > MAX_CSV_ROWS) {
   804→    throw new CsvError("card.errorCsvRows");
   805→  }
   806→  return { columns, rows };
   807→}

   809→/**
   810→ * Replaces `{{column}}` placeholders with row values. Exact column
   811→ * name match (case-sensitive). Unknown placeholders and non-matching
   812→ * forms (`{{}}`, `{{ name}}`) are left untouched — they render
   813→ * literally and never crash. Missing row fields become "".
   814→ */
   815→export function mergeText(text: string, row: string[], columns: string[]): string {
   816→  return text.replace(/\{\{([^{}\s]+)\}\}/g, (whole, name: string) => {
   817→    const index = columns.indexOf(name);
   818→    if (index === -1) return whole;
   819→    return row[index] ?? "";
   820→  });
   821→}

   837→/**
   838→ * Collects ALL unknown `{{column}}` names across every text field, in
   839→ * first-appearance order (field order is TEXT_FIELD_ORDER — stable).
   840→ * Empty result = everything merges cleanly.
   841→ */
   842→export function validatePlaceholders(
   843→  texts: CardTexts,
   844→  columns: string[]
   845→): string[] {
   846→  const unknown: string[] = [];
   847→  const seen = new Set<string>();
   848→  const re = /\{\{([^{}\s]+)\}\}/g;
   849→  for (const field of TEXT_FIELD_ORDER) {
   850→    const value = texts[field];
   851→    if (typeof value !== "string") continue;
   852→    for (const match of value.matchAll(re)) {
   853→      const name = match[1];
   854→      if (!columns.includes(name) && !seen.has(name)) {
   855→        seen.add(name);
   856→        unknown.push(name);
   857→      }
   858→    }
   859→  }
   860→  return unknown;
   861→}

   986→/**
   987→ * Cover-crop: the largest centred source rect whose aspect matches the
   988→ * destination (collage's exact math — reused so backgrounds and
   989→ * photos crop identically).
   990→ */
   991→export function coverCrop(sw: number, sh: number, dw: number, dh: number): CropRect {
   992→  if (sw <= 0 || sh <= 0 || dw <= 0 || dh <= 0) return { sx: 0, sy: 0, sw: 0, sh: 0 };
   993→  const scale = Math.max(dw / sw, dh / sh);
   994→  const cropW = dw / scale;
   995→  const cropH = dh / scale;
   996→  return {
   997→    sx: (sw - cropW) / 2,
   998→    sy: (sh - cropH) / 2,
   999→    sw: cropW,
  1000→    sh: cropH,
  1001→  };
  1002→}

   119→
   120→/** Display size (px at `scale`) of a page with the viewer rotation applied.
   121→ *  `base` is the UNROTATED media size + the page's native /Rotate. */
   122→function displaySize(
   123→  base: { w: number; h: number; rot: number },
   124→  scale: number,
   125→  rotation: number
   126→): { w: number; h: number } {
   127→  const swap = (base.rot + rotation) % 180 === 90;
   128→  const w = (swap ? base.h : base.w) * scale;
   129→  const h = (swap ? base.w : base.h) * scale;
   130→  return { w: Math.max(1, Math.round(w)), h: Math.max(1, Math.round(h)) };
   131→}
   132→
   133→function clamp(value: number, lo: number, hi: number): number {
   134→  return Math.min(hi, Math.max(lo, value));
   135→}
   136→
   137→function nextZoomStep(current: number, dir: 1 | -1): number {
   138→  if (dir === 1) {
   139→    for (const step of ZOOM_STEPS) {
   140→      if (step > current + 0.001) return step;
   141→    }
   142→    return MAX_ZOOM;
   143→  }
   144→  for (let i = ZOOM_STEPS.length - 1; i >= 0; i -= 1) {
   145→    if (ZOOM_STEPS[i] < current - 0.001) return ZOOM_STEPS[i];
   146→  }
   147→  return MIN_ZOOM;
   148→}

   152→interface ViewerPageProps {
   153→  pageNumber: number;
   154→  docRef: React.RefObject<PDFDocumentProxy | null>;
   155→  /** Bumped whenever a new document is opened (forces full re-render). */

   166→  /** Register the page slot element for scroll math (stable callback). */
   167→  registerSlot: (page: number, el: HTMLElement | null) => void;
   168→  /** Element that clips + scrolls the pages (IntersectionObserver root). */
   169→  scrollRootRef: React.RefObject<HTMLDivElement | null>;

   171→
   172→/**
   173→ * One lazily rendered page: an IntersectionObserver keeps it mounted only
   174→ * while it is near the viewport (about ±2 pages at typical zooms); the
   175→ * canvas + text layer are rebuilt whenever scale/rotation/doc change.
   176→ */
   177→function ViewerPage({
   178→  pageNumber,
   179→  docRef,
   180→  docEpoch,
   181→  scale,
   182→  rotation,
   183→  base,
   184→  dark,
   185→  matches,
   186→  activeMatchId,
   187→  registerSlot,
   188→  scrollRootRef,
   189→}: ViewerPageProps): React.JSX.Element {
   190→  const { t } = useI18n();
   191→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
   192→  const textLayerHostRef = useRef<HTMLDivElement | null>(null);
   193→  const slotRef = useRef<HTMLDivElement | null>(null);
   194→  const [visible, setVisible] = useState(false);
   195→  /** Text-item geometry of the last completed render (device px, this scale). */
   196→  const [geo, setGeo] = useState<{
   197→    items: GeoItem[];
   198→    renderKey: string;
   199→  } | null>(null);
   200→  const renderKey = `${docEpoch}:${scale.toFixed(4)}:${rotation}`;
   201→  /** The slot is the layout authority: exact display px at this scale. */
   202→  const disp = displaySize(base, scale, rotation);
   203→
   204→  const setSlotRef = useCallback(
   205→    (el: HTMLDivElement | null) => {
   206→      slotRef.current = el;
   207→      registerSlot(pageNumber, el);
   208→    },
   209→    [pageNumber, registerSlot]
   210→  );
   211→
   212→  // Lazy mount/unmount near the viewport (PRD §6.5 "visible ±2 pages").
   213→  useEffect(() => {
   214→    const slot = slotRef.current;
   215→    const root = scrollRootRef.current;
   216→    if (!slot || !root) return;
   217→    const observer = new IntersectionObserver(
   218→      (entries) => {
   219→        for (const entry of entries) setVisible(entry.isIntersecting);
   220→      },
   221→      { root, rootMargin: "200% 0px 200% 0px" }
   222→    );
   223→    observer.observe(slot);
   224→    return () => observer.disconnect();
   225→  }, [scrollRootRef]);
   226→
   227→  // Canvas + text layer render. Cancelled through the `cancelled` flag +
   228→  // renderTask.cancel() whenever deps change or the page unmounts.
   229→  useEffect(() => {
   230→    if (!visible) return;
   231→    const doc = docRef.current;
   232→    if (!doc) return;
   233→    let cancelled = false;
   234→    let renderTask: { cancel: () => void; promise: Promise<unknown> } | null =
   235→      null;
   236→    let textLayer: PdfTextLayer | null = null;
   237→    const key = renderKey;
   238→
   239→    void (async () => {
   240→      const pdfjs = await loadPdfjs();
   241→      const page: PDFPageProxy = await doc.getPage(pageNumber);
   242→      if (cancelled) return;
   243→      const viewport = page.getViewport({
   244→        scale,
   245→        rotation: (page.rotate + rotation) % 360,
   246→      });
   247→
   248→      const canvas = canvasRef.current;
   249→      if (canvas) {
   250→        const dpr = Math.min(
   251→          typeof window === "undefined" ? 1 : window.devicePixelRatio || 1,
   252→          2
   253→        );
   254→        canvas.width = Math.max(1, Math.floor(viewport.width * dpr));
   255→        canvas.height = Math.max(1, Math.floor(viewport.height * dpr));
   256→        canvas.style.width = `${Math.max(1, Math.round(viewport.width))}px`;
   257→        canvas.style.height = `${Math.max(1, Math.round(viewport.height))}px`;
   258→        const context = canvas.getContext("2d");
   259→        if (context) {
   260→          context.fillStyle = "#ffffff";
   261→          context.fillRect(0, 0, canvas.width, canvas.height);
   262→          const transform = dpr !== 1 ? [dpr, 0, 0, dpr, 0, 0] : undefined;
   263→          const task = page.render({
   264→            canvasContext: context,
   265→            viewport,
   266→            transform,
   267→          });
   268→          renderTask = task;
   269→          await task.promise;
   270→          if (cancelled) return;
   271→        }
   272→      }
   273→
   274→      // Text content: geometry for search highlights + the TextLayer.
   275→      // itemIndex keys stay aligned with the RAW content.items order.
   276→      const content = await page.getTextContent();
   277→      if (cancelled) return;
   278→      const items: GeoItem[] = [];
   279→      for (let i = 0; i < content.items.length; i += 1) {
   280→        const raw = content.items[i] as {
   281→          str?: unknown;
   282→          width?: unknown;
   283→          transform?: unknown;
   284→        };
   285→        if (typeof raw.str !== "string" || raw.str.length === 0) continue;
   286→        if (!Array.isArray(raw.transform) || typeof raw.width !== "number") {
   287→          continue;
   288→        }
   289→        const tx = pdfjs.Util.transform(
   290→          viewport.transform,
   291→          raw.transform as number[]
   292→        );
   293→        items.push({
   294→          itemIndex: i,
   295→          str: raw.str,
   296→          left: tx[4],
   297→          top: tx[5],
   298→          width: raw.width * viewport.scale,
   299→          height: Math.hypot(tx[2], tx[3]),
   300→        });
   301→      }
   302→      setGeo({ items, renderKey: key });
   303→
   304→      // Selectable text layer (PRD §6.5 "text select aur copy").
   305→      const host = textLayerHostRef.current;
   306→      if (host) {
   307→        host.replaceChildren();
   308→        host.style.setProperty("--scale-factor", String(viewport.scale));
   309→        const layer = new pdfjs.TextLayer({
   310→          textContentSource: page.streamTextContent(),
   311→          container: host,
   312→          viewport,
   313→        });
   314→        textLayer = layer;
   315→        await layer.render();
   316→        if (cancelled) return;
   317→      }
   318→    })().catch((error: unknown) => {
   319→      const name =
   320→        error !== null && typeof error === "object" && "name" in error
   321→          ? String((error as { name: unknown }).name)
   322→          : "";
   323→      // Cancelled renders resolve quietly; real failures surface as a
   324→      // blank page slot with the placeholder number still showing.
   325→      if (cancelled || name === "RenderingCancelledException") return;
   326→    });
   327→
   328→    return () => {
   329→      cancelled = true;
   330→      renderTask?.cancel();
   331→      textLayer?.cancel();
   332→      const host = textLayerHostRef.current;
   333→      if (host) host.replaceChildren();
   334→    };
   335→  }, [visible, scale, rotation, docEpoch, pageNumber, docRef, renderKey]);
   336→
   337→  // Search highlight boxes for THIS page — a pure derivation from the
   338→  // cached item geometry + the page's matches (proportional char ranges
   339→  // inside each item; the canvas is NOT re-rendered for search).
   340→  const boxes: HighlightBox[] = useMemo(() => {
   341→    if (!geo || geo.renderKey !== renderKey || matches.length === 0) {
   342→      return EMPTY_BOXES;
   343→    }
   344→    const out: HighlightBox[] = [];
   345→    for (const match of matches) {
   346→      const item = geo.items.find((x) => x.itemIndex === match.itemIndex);
   347→      if (!item) continue;
   348→      const len = item.str.length;
   349→      if (len === 0) continue;
   350→      const startFrac = clamp(match.start / len, 0, 1);
   351→      const endFrac = clamp(match.end / len, 0, 1);
   352→      const left = item.left + startFrac * item.width;
   353→      const width = Math.max(4, (endFrac - startFrac) * item.width);
   354→      out.push({
   355→        matchId: match.id,
   356→        left,
   357→        top: item.top - item.height,
   358→        width,
   359→        height: Math.max(item.height, 6),
   360→      });
   361→    }
   362→    return out;
   363→  }, [geo, matches, renderKey]);
   364→
   365→  return (
   366→    <div
   367→      ref={setSlotRef}
   368→      data-viewer-page={pageNumber}
   369→      style={{ width: disp.w, height: disp.h }}
   370→      className="relative shrink-0 overflow-hidden rounded-sm bg-white shadow-md ring-1 ring-border/60 dark:bg-neutral-100"
   371→      aria-label={t("view.a11yPage", {
   372→        page: pageNumber,
   373→      })}
   374→    >
   375→      {visible ? (
   376→        <>
   377→          <canvas
   378→            ref={canvasRef}
   379→            className={cn(
   380→              "block",
   381→              dark && "[filter:invert(0.92)_hue-rotate(180deg)]"
   382→            )}
   383→          />
   384→          <div ref={textLayerHostRef} className="pdf-textLayer" />
   385→          {boxes.map((box) => (
   386→            <div
   387→              key={box.matchId}
   388→              className={cn(
   389→                "pointer-events-none absolute rounded-[2px]",
   390→                box.matchId === activeMatchId
   391→                  ? "bg-primary/45 ring-2 ring-primary"
   392→                  : "bg-primary/25"
   393→              )}
   394→              style={{
   395→                left: box.left,
   396→                top: box.top,
   397→                width: box.width,
   398→                height: box.height,
   399→              }}
   400→            />
   401→          ))}
   402→        </>
   403→      ) : (
   404→        <div className="flex h-full w-full items-center justify-center bg-muted/50">
   405→          <span className="text-xs font-medium text-muted-foreground/70">
   406→            {pageNumber}
   407→          </span>
   408→        </div>
   409→      )}
   410→    </div>
   411→  );
   412→}

   425→/** Lazy sidebar thumbnail rendered straight from the open document. */
   426→function ViewerThumb({
   427→  pageNumber,
   428→  active,
   429→  docRef,
   430→  docEpoch,
   431→  onGo,
   432→  scrollRootRef,
   433→}: ViewerThumbProps): React.JSX.Element {
   434→  const { t } = useI18n();
   435→  const wrapRef = useRef<HTMLButtonElement | null>(null);
   436→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
   437→  const [visible, setVisible] = useState(false);
   438→  const [done, setDone] = useState(-1);
   439→
   440→  useEffect(() => {
   441→    const el = wrapRef.current;
   442→    const root = scrollRootRef.current;
   443→    if (!el || !root) return;
   444→    const observer = new IntersectionObserver(
   445→      (entries) => {
   446→        for (const entry of entries) setVisible(entry.isIntersecting);
   447→      },
   448→      { root, rootMargin: "120% 0px" }
   449→    );
   450→    observer.observe(el);
   451→    return () => observer.disconnect();
   452→  }, [scrollRootRef]);
   453→
   454→  useEffect(() => {
   455→    if (!visible || done === docEpoch) return;
   456→    const doc = docRef.current;
   457→    if (!doc) return;
   458→    let cancelled = false;
   459→    let renderTask: { cancel: () => void; promise: Promise<unknown> } | null =
   460→      null;
   461→    void (async () => {
   462→      const page = await doc.getPage(pageNumber);
   463→      if (cancelled) return;
   464→      const canvas = canvasRef.current;
   465→      if (!canvas) return;
   466→      const dpr = Math.min(
   467→        typeof window === "undefined" ? 1 : window.devicePixelRatio || 1,
   468→        2
   469→      );
   470→      const base = page.getViewport({ scale: 1 });
   471→      const scale = (THUMB_MAX_DIM * dpr) / Math.max(base.width, base.height);
   472→      const viewport = page.getViewport({ scale });
   473→      canvas.width = Math.max(1, Math.round(viewport.width));
   474→      canvas.height = Math.max(1, Math.round(viewport.height));
   475→      canvas.style.width = `${Math.round(viewport.width / dpr)}px`;
   476→      canvas.style.height = `${Math.round(viewport.height / dpr)}px`;
   477→      const context = canvas.getContext("2d");
   478→      if (!context) return;
   479→      context.fillStyle = "#ffffff";
   480→      context.fillRect(0, 0, canvas.width, canvas.height);
   481→      const task = page.render({ canvasContext: context, viewport });
   482→      renderTask = task;
   483→      await task.promise;
   484→      if (cancelled) return;
   485→      setDone(docEpoch);
   486→    })().catch(() => {
   487→      /* thumbnail failure is cosmetic — the page itself still renders */
   488→    });
   489→    return () => {
   490→      cancelled = true;
   491→      renderTask?.cancel();
   492→      const canvas = canvasRef.current;
   493→      if (canvas) {
   494→        canvas.removeAttribute("style");
   495→        canvas.removeAttribute("width");
   496→        canvas.removeAttribute("height");
   497→      }
   498→    };
   499→  }, [visible, done, docEpoch, pageNumber, docRef]);
   500→
   501→  return (
   502→    <button
   503→      type="button"
   504→      ref={wrapRef}
   505→      onClick={() => onGo(pageNumber)}
   506→      aria-label={t("view.a11yThumb", { page: pageNumber })}
   507→      aria-current={active ? "true" : undefined}
   508→      className={cn(
   509→        "group flex w-full flex-col items-center gap-1 rounded-card p-1.5 outline-none transition-colors",
   510→        "hover:bg-muted focus-visible:ring-[3px] focus-visible:ring-ring/50",
   511→        active && "bg-primary/10"
   512→      )}
   513→    >
   514→      <span
   515→        className={cn(
   516→          "relative flex items-center justify-center overflow-hidden rounded-sm bg-white shadow-sm ring-1 ring-border/70",
   517→          active ? "ring-2 ring-primary ring-offset-1 ring-offset-background" : ""
   518→        )}
   519→      >
   520→        <canvas ref={canvasRef} className="block" />
   521→      </span>
   522→      <span
   523→        className={cn(
   524→          "text-[10px] font-medium tabular-nums",
   525→          active ? "text-primary" : "text-muted-foreground"
   526→        )}
   527→      >
   528→        {pageNumber}
   529→      </span>
   530→    </button>
   531→  );
   532→}

   670→                depth,

   950→    (event: React.KeyboardEvent<HTMLDivElement>) => {

   953→        target !== null &&

   984→        case "=":

   988→        case "-":

  1002→        case "r":
  1003→        case "R":
  1004→          event.preventDefault();

  1006→          break;
  1007→        default:
  1008→          break;
  1009→      }
  1010→    },

     1→     1→/**
     2→     2→ * Word to PDF (RB1-a — the 67th tool, rebuild) — pure model. No DOM, no
     3→     3→ * pdf-lib, no worker APIs: the same parseDocx feeds the surface's live
     4→     4→ * file summary AND the worker's render path, so the preview numbers can
     5→     5→ * never disagree with the output.
     6→     6→ *
     7→     7→ *  - parseDocx: fflate unzipSync → word/document.xml (plus optional
     8→     8→ *    word/styles.xml for heading resolution) → a dependency-free SAX-ish

    11→    11→ *       headingLevel?, listLevel?}
    12→    12→ *      {kind:"table", rows: string[][]}
    13→    13→ *      {kind:"pagebreak"}            (w:br w:type="page")
    14→    14→ *  - parseWordRun: the worker-boundary coercion (parseResumeRun pattern)
    15→    15→ *    — clamp arrays, strip control chars, cap every length. A tampered
    16→    16→ *    payload can never reach the renderer as a wrong type.

   104→   104→function docStats(blocks: DocxBlock[]): Omit<ParsedDoc, "blocks" | "truncated"> {
   105→   105→  let paraCount = 0;
   106→   106→  let tableCount = 0;
   107→   107→  let words = 0;
   108→   108→  let chars = 0;
   109→   109→  const count = (s: string): void => {
   110→   110→    chars += s.length;
   111→   111→    words += s.split(/\s+/).filter(Boolean).length;
   112→   112→  };
   113→   113→  for (const b of blocks) {
   114→   114→    if (b.kind === "para") {
   115→   115→      paraCount += 1;
   116→   116→      for (const r of b.runs) count(r.text);
   117→   117→    } else if (b.kind === "table") {
   118→   118→      tableCount += 1;
   119→   119→      for (const row of b.rows) for (const cell of row) count(cell);
   120→   120→    }
   121→   121→  }
   122→   122→  return { paraCount, tableCount, wordCount: words, charCount: chars };
   123→   123→}

   135→   135→/** true when every char is encodable in WinAnsi (CP1252) minus controls. */
   136→   136→export function isWinAnsiText(s: string): boolean {
   137→   137→  for (let i = 0; i < s.length; i++) {
   138→   138→    const c = s.charCodeAt(i);
   139→   139→    if (c >= 0x20 && c <= 0x7e) continue;
   140→   140→    if (c >= 0xa0 && c <= 0xff) continue;
   141→   141→    if (WINANSI_EXTRA.has(c)) continue;
   142→   142→    return false;
   143→   143→  }
   144→   144→  return true;
   145→   145→}

   147→   147→/** true when any drawn text needs the Devanagari (non-WinAnsi) font. */
   148→   148→export function needsNoto(doc: ParsedDoc): boolean {
   149→   149→  for (const b of doc.blocks) {
   150→   150→    if (b.kind === "para") {
   151→   151→      for (const r of b.runs) if (!isWinAnsiText(r.text)) return true;
   152→   152→    } else if (b.kind === "table") {
   153→   153→      for (const row of b.rows) {
   154→   154→        for (const cell of row) if (!isWinAnsiText(cell)) return true;
   155→   155→      }
   156→   156→    }
   157→   157→  }
   158→   158→  return false;
   159→   159→}

   178→   178→/* ------------------------------------------------------------------ */
   179→   179→
   180→   180→const NAMED_ENTITIES: Record<string, string> = {
   181→   181→  amp: "&",
   182→   182→  lt: "<",
   183→   183→  gt: ">",
   184→   184→  quot: '"',
   185→   185→  apos: "'",
   186→   186→  nbsp: "\u00a0",
   187→   187→  tab: "\t",
   188→   188→  newline: "\n",
   189→   189→};
   190→   190→
   191→   191→const ENTITY_RE = /&(#[xX][0-9a-fA-F]+|#[0-9]+|[a-zA-Z][a-zA-Z0-9]*);/g;
   192→   192→
   193→   193→function decodeEntities(s: string): string {
   194→   194→  if (!s.includes("&")) return s;
   195→   195→  return s.replace(ENTITY_RE, (match, body: string) => {
   196→   196→    if (body.charCodeAt(0) === 35 /* # */) {
   197→   197→      const hex = body[1] === "x" || body[1] === "X";
   198→   198→      const code = Number.parseInt(body.slice(hex ? 2 : 1), hex ? 16 : 10);
   199→   199→      if (!Number.isFinite(code) || code < 0 || code > 0x10ffff) return match;
   200→   200→      try {
   201→   201→        return String.fromCodePoint(code);
   202→   202→      } catch {
   203→   203→        return match;
   204→   204→      }
   205→   205→    }
   206→   206→    return NAMED_ENTITIES[body] ?? match;
   207→   207→  });
   208→   208→}

   225→   225→/* Dependency-free XML tokenizer (SAX events; no DOMParser in workers)  */
   226→   226→/* ------------------------------------------------------------------ */
   227→   227→
   228→   228→export interface SaxHandlers {
   229→   229→  open(name: string, attrs: Record<string, string>): void;
   230→   230→  text(value: string): void;
   231→   231→  close(name: string): void;
   232→   232→}
   233→   233→
   234→   234→const WS_RE = /[\s]/;
   235→   235→const NAME_STOP_RE = /[\s/>]/;
   236→   236→const ATTR_NAME_STOP_RE = /[\s=/>]/;
   237→   237→
   238→   238→/**
   239→   239→ * Tolerant linear tokenizer for well-formed Word XML: real elements emit
   240→   240→ * open/close (self-closing emits both), <?…?> / <!--…--> / <![CDATA[…]]>
   241→   241→ * are skipped (CDATA becomes raw text), everything else between tags is
   242→   242→ * text with entity decoding. Malformed input degrades instead of throwing.
   243→   243→ */
   244→   244→export function parseXmlSax(xml: string, h: SaxHandlers): void {
   245→   245→  const n = xml.length;
   246→   246→  let i = 0;
   247→   247→  while (i < n) {
   248→   248→    const lt = xml.indexOf("<", i);
   249→   249→    if (lt < 0) {
   250→   250→      if (i < n) h.text(decodeEntities(xml.slice(i)));
   251→   251→      break;
   252→   252→    }
   253→   253→    if (lt > i) h.text(decodeEntities(xml.slice(i, lt)));
   254→   254→
   255→   255→    if (xml.startsWith("<?", lt)) {
   256→   256→      const end = xml.indexOf("?>", lt);
   257→   257→      i = end < 0 ? n : end + 2;
   258→   258→      continue;
   259→   259→    }
   260→   260→    if (xml.startsWith("<!--", lt)) {
   261→   261→      const end = xml.indexOf("-->", lt);
   262→   262→      i = end < 0 ? n : end + 3;
   263→   263→      continue;
   264→   264→    }
   265→   265→    if (xml.startsWith("<![CDATA[", lt)) {
   266→   266→      const end = xml.indexOf("]]>", lt);
   267→   267→      h.text(xml.slice(lt + 9, end < 0 ? n : end)); // raw — no entities
   268→   268→      i = end < 0 ? n : end + 3;
   269→   269→      continue;
   270→   270→    }
   271→   271→    if (xml.charCodeAt(lt + 1) === 33 /* ! */) {
   272→   272→      const end = xml.indexOf(">", lt);
   273→   273→      i = end < 0 ? n : end + 1;
   274→   274→      continue;
   275→   275→    }
   276→   276→
   277→   277→    let j = lt + 1;
   278→   278→    let closing = false;
   279→   279→    if (xml.charCodeAt(j) === 47 /* / */) {
   280→   280→      closing = true;
   281→   281→      j += 1;
   282→   282→    }
   283→   283→    const nameStart = j;
   284→   284→    while (j < n && !NAME_STOP_RE.test(xml[j])) j += 1;
   285→   285→    const name = xml.slice(nameStart, j);
   286→   286→    if (name === "") {
   287→   287→      // "<" that isn't a tag at all — emit it as text and move on.
   288→   288→      h.text("<");
   289→   289→      i = lt + 1;
   290→   290→      continue;
   291→   291→    }
   292→   292→
   293→   293→    const attrs: Record<string, string> = {};
   294→   294→    let selfClosing = false;
   295→   295→    while (j < n) {
   296→   296→      while (j < n && WS_RE.test(xml[j])) j += 1;
   297→   297→      const c = xml.charCodeAt(j);
   298→   298→      if (c === 47 /* / */) {
   299→   299→        selfClosing = true;
   300→   300→        j += 1;
   301→   301→        continue;
   302→   302→      }
   303→   303→      if (c === 62 /* > */) {
   304→   304→        j += 1;
   305→   305→        break;
   306→   306→      }
   307→   307→      const aStart = j;
   308→   308→      while (j < n && !ATTR_NAME_STOP_RE.test(xml[j])) j += 1;
   309→   309→      const aName = xml.slice(aStart, j);
   310→   310→      while (j < n && WS_RE.test(xml[j])) j += 1;
   311→   311→      let aVal = "";
   312→   312→      if (xml.charCodeAt(j) === 61 /* = */) {
   313→   313→        j += 1;
   314→   314→        while (j < n && WS_RE.test(xml[j])) j += 1;
   315→   315→        const q = xml.charCodeAt(j);
   316→   316→        if (q === 34 /* " */ || q === 39 /* ' */) {
   317→   317→          j += 1;
   318→   318→          const vStart = j;
   319→   319→          while (j < n && xml.charCodeAt(j) !== q) j += 1;
   320→   320→          aVal = xml.slice(vStart, j);
   321→   321→          j += 1;
   322→   322→        } else {
   323→   323→          const vStart = j;
   324→   324→          while (j < n && !WS_RE.test(xml[j]) && xml.charCodeAt(j) !== 62) {
   325→   325→            j += 1;
   326→   326→          }
   327→   327→          aVal = xml.slice(vStart, j);
   328→   328→        }
   329→   329→      }
   330→   330→      if (aName) attrs[aName] = decodeEntities(aVal);
   331→   331→      if (j >= n) break;
   332→   332→    }
   333→   333→
   334→   334→    if (closing) {
   335→   335→      h.close(name);
   336→   336→    } else {
   337→   337→      h.open(name, attrs);
   338→   338→      if (selfClosing) h.close(name);
   339→   339→    }
   340→   340→    i = j;
   341→   341→  }
   342→   342→}

   345→   345→/* styles.xml → styleId → heading level (0 = Title, 1..9)               */
   346→   346→/* ------------------------------------------------------------------ */
   347→   347→
   348→   348→function parseStylesXml(xml: string): Map<string, number> {
   349→   349→  const map = new Map<string, number>();
   350→   350→  let styleId = ""; // active paragraph style id ("" = not tracking)
   351→   351→  let nameVal = ""; // its w:name val
   352→   352→  let outline: number | null = null; // its w:outlineLvl, 1-based
   353→   353→  parseXmlSax(xml, {
   354→   354→    open(name, attrs) {
   355→   355→      if (prefixOf(name) !== "w") return;
   356→   356→      const local = localName(name);
   357→   357→      if (local === "style") {
   358→   358→        const id = attrs["w:styleId"] ?? "";
   359→   359→        styleId =
   360→   360→          id && (attrs["w:type"] === undefined || attrs["w:type"] === "paragraph")
   361→   361→            ? id
   362→   362→            : "";
   363→   363→        nameVal = "";
   364→   364→        outline = null;
   365→   365→      } else if (local === "name" && styleId) {
   366→   366→        nameVal = attrs["w:val"] ?? "";
   367→   367→      } else if (local === "outlineLvl" && styleId) {
   368→   368→        const v = Number.parseInt(attrs["w:val"] ?? "", 10);
   369→   369→        outline = Number.isFinite(v) && v >= 0 && v <= 8 ? v + 1 : null;
   370→   370→      }
   371→   371→    },
   372→   372→    text() {
   373→   373→      /* style names/levels live in attributes, never text */
   374→   374→    },
   375→   375→    close(name) {
   376→   376→      if (prefixOf(name) !== "w" || localName(name) !== "style") return;
   377→   377→      if (styleId) {
   378→   378→        const trim = nameVal.trim();
   379→   379→        let level: number | undefined;
   380→   380→        if (/^title$/i.test(trim)) {
   381→   381→          level = 0;
   382→   382→        } else {
   383→   383→          const m = /^heading[ \-]?([1-9])$/i.exec(trim);
   384→   384→          if (m) level = Number.parseInt(m[1], 10);
   385→   385→          else if (outline !== null) level = outline;
   386→   386→        }
   387→   387→        if (level !== undefined) map.set(styleId, level);
   388→   388→      }
   389→   389→      styleId = "";
   390→   390→    },
   391→   391→  });
   392→   392→  return map;
   393→   393→}

   417→   417→/** Half-point string → pt (4..72), undefined when absent/invalid. */
   418→   418→function halfPointsToPt(val: string | undefined): number | undefined {
   419→   419→  if (val === undefined) return undefined;
   420→   420→  const hp = Number.parseInt(val, 10);
   421→   421→  if (!Number.isFinite(hp)) return undefined;
   422→   422→  const pt = hp / 2;
   423→   423→  return pt >= 4 && pt <= 72 ? pt : undefined;
   424→   424→}

   478→   478→  private textTarget = false;
   479→   479→
   480→   480→  constructor(styles: Map<string, number>) {
   481→   481→    this.styles = styles;
   482→   482→  }
   483→   483→
   484→   484→  private pushBlock(b: DocxBlock): void {
   485→   485→    if (this.blocks.length >= MAX_BLOCKS) {
   486→   486→      this.truncated = true;
   487→   487→      return;
   488→   488→    }
   489→   489→    if (b.kind === "para") {
   490→   490→      if (this.paraBlocks >= MAX_PARAGRAPHS) {
   491→   491→        this.truncated = true;
   492→   492→        return;
   493→   493→      }
   494→   494→      this.paraBlocks += 1;
   495→   495→    }
   496→   496→    this.blocks.push(b);
   497→   497→  }
   498→   498→
   499→   499→  readonly handlers: SaxHandlers = {
   500→   500→    open: (name, attrs) => {
   501→   501→      this.depth += 1;
   502→   502→      if (this.skipDepth !== -1) return;
   503→   503→      const local = localName(name);
   504→   504→      if (local === "sectPr") {
   505→   505→        this.skipDepth = this.depth;
   506→   506→        return;
   507→   507→      }
   508→   508→      // drawings / VML pictures: skipped wholesale (images are never
   509→   509→      // rendered — the honest limit; also drops text-box duplicates).
   510→   510→      if (local === "drawing" || local === "pict") {
   511→   511→        this.skipDepth = this.depth;
   512→   512→        return;
   513→   513→      }
   514→   514→      if (prefixOf(name) !== "w") return;
   515→   515→      switch (local) {
   516→   516→        case "body":
   517→   517→          this.inBody = true;
   518→   518→          break;
   519→   519→        case "tbl":
   520→   520→          this.tableStack.push({ rows: [] });
   521→   521→          break;
   522→   522→        case "tr":
   523→   523→          this.rowStack.push([]);
   524→   524→          break;
   525→   525→        case "tc":
   526→   526→          this.cellStack.push({ texts: [] });
   527→   527→          break;
   528→   528→        case "p":
   529→   529→          this.para = {
   530→   530→            runs: [],
   531→   531→            run: null,
   532→   532→            styleId: "",
   533→   533→            align: undefined,
   534→   534→            listLevel: -1,
   535→   535→            pageBreak: false,
   536→   536→            firstRunDone: false,
   537→   537→            textLen: 0,
   538→   538→          };
   539→   539→          break;
   540→   540→        case "r": {
   541→   541→          const para = this.para;
   542→   542→          if (para !== null && para.run === null) {
   543→   543→            para.run = { text: "", bold: false, italic: false, sizePt: undefined };
   544→   544→          }
   545→   545→          break;
   546→   546→        }
   547→   547→        case "pStyle":
   548→   548→          if (this.para !== null) this.para.styleId = attrs["w:val"] ?? "";
   549→   549→          break;
   550→   550→        case "jc":
   551→   551→          if (this.para !== null) {
   552→   552→            this.para.align = ALIGN_MAP[attrs["w:val"] ?? ""] ?? undefined;
   553→   553→          }
   554→   554→          break;
   555→   555→        case "numPr":
   556→   556→          if (this.para !== null) {
   557→   557→            this.inNumPr = true;
   558→   558→            this.para.listLevel = 0;
   559→   559→          }
   560→   560→          break;
   561→   561→        case "ilvl":
   562→   562→          if (this.inNumPr && this.para !== null) {
   563→   563→            const v = Number.parseInt(attrs["w:val"] ?? "0", 10);
   564→   564→            this.para.listLevel = Number.isFinite(v) && v >= 0 ? Math.min(v, 3) : 0;
   565→   565→          }
   566→   566→          break;
   567→   567→        case "numId":
   568→   568→          // numId 0 explicitly CANCELS a list inherited from the style.
   569→   569→          if (this.inNumPr && this.para !== null && (attrs["w:val"] ?? "") === "0") {
   570→   570→            this.para.listLevel = -1;
   571→   571→          }
   572→   572→          break;
   573→   573→        case "b":
   574→   574→        case "bCs": {
   575→   575→          const run = this.para?.run;
   576→   576→          if (run) run.bold = run.bold || toggleVal(attrs);
   577→   577→          break;
   578→   578→        }
   579→   579→        case "i":
   580→   580→        case "iCs": {
   581→   581→          const run = this.para?.run;
   582→   582→          if (run) run.italic = run.italic || toggleVal(attrs);
   583→   583→          break;
   584→   584→        }
   585→   585→        case "sz": {
   586→   586→          const run = this.para?.run;
   587→   587→          if (run && run.sizePt === undefined) {
   588→   588→            run.sizePt = halfPointsToPt(attrs["w:val"]);
   589→   589→          }
   590→   590→          break;
   591→   591→        }
   592→   592→        case "szCs": {
   593→   593→          const run = this.para?.run;
   594→   594→          if (run && run.sizePt === undefined) {
   595→   595→            run.sizePt = halfPointsToPt(attrs["w:val"]);
   596→   596→          }
   597→   597→          break;
   598→   598→        }
   599→   599→        case "t":
   600→   600→          this.textTarget = true;
   601→   601→          break;
   602→   602→        case "tab": {
   603→   603→          const run = this.para?.run;
   604→   604→          if (run) run.text += "\t";
   605→   605→          break;
   606→   606→        }
   607→   607→        case "br": {
   608→   608→          if (attrs["w:type"] === "page") {
   609→   609→            if (this.para !== null) this.para.pageBreak = true;
   610→   610→          } else {
   611→   611→            const run = this.para?.run;
   612→   612→            if (run) run.text += "\n";
   613→   613→          }
   614→   614→          break;
   615→   615→        }
   616→   616→        default:
   617→   617→          break;
   618→   618→      }
   619→   619→    },
   620→   620→
   621→   621→    text: (value) => {
   622→   622→      if (!this.textTarget) return;
   623→   623→      const run = this.para?.run;
   624→   624→      if (run) run.text += hindiNorm(value);
   625→   625→    },
   626→   626→
   627→   627→    close: (name) => {
   628→   628→      const closingDepth = this.depth;
   629→   629→      this.depth = Math.max(0, this.depth - 1);
   630→   630→      if (this.skipDepth !== -1) {
   631→   631→        if (closingDepth === this.skipDepth) this.skipDepth = -1;
   632→   632→        return;
   633→   633→      }
   634→   634→      const local = localName(name);
   635→   635→      if (prefixOf(name) !== "w") return;
   636→   636→      switch (local) {
   637→   637→        case "body":
   638→   638→          this.inBody = false;
   639→   639→          break;
   640→   640→        case "t":
   641→   641→          this.textTarget = false;
   642→   642→          break;
   643→   643→        case "numPr":
   644→   644→          this.inNumPr = false;
   645→   645→          break;
   646→   646→        case "r":
   647→   647→          this.endRun();
   648→   648→          break;
   649→   649→        case "p":
   650→   650→          this.endPara();
   651→   651→          break;
   652→   652→        case "tc": {
   653→   653→          const cell = this.cellStack.pop();
   654→   654→          if (cell && this.rowStack.length > 0) {
   655→   655→            const row = this.rowStack[this.rowStack.length - 1];
   656→   656→            if (row.length < MAX_TABLE_COLS) {
   657→   657→              row.push(cell.texts.join("\n").slice(0, MAX_CELL_TEXT));
   658→   658→            } else {
   659→   659→              this.truncated = true;
   660→   660→            }
   661→   661→          }
   662→   662→          break;
   663→   663→        }
   664→   664→        case "tr": {
   665→   665→          const row = this.rowStack.pop();
   666→   666→          if (row && this.tableStack.length > 0) {
   667→   667→            const table = this.tableStack[this.tableStack.length - 1];
   668→   668→            if (table.rows.length < MAX_TABLE_ROWS) table.rows.push(row);
   669→   669→            else this.truncated = true;
   670→   670→          }
   671→   671→          break;
   672→   672→        }
   673→   673→        case "tbl": {
   674→   674→          const table = this.tableStack.pop();
   675→   675→          if (!table) break;
   676→   676→          if (this.tableStack.length === 0) {
   677→   677→            if (table.rows.length > 0) {
   678→   678→              this.pushBlock({ kind: "table", rows: table.rows });
   679→   679→            }
   680→   680→          } else if (this.cellStack.length > 0) {
   681→   681→            // nested table inside a cell — flatten its text honestly.
   682→   682→            const cell = this.cellStack[this.cellStack.length - 1];
   683→   683→            const flat = table.rows.map((r) => r.join(" ")).join(" ");
   684→   684→            if (flat) cell.texts.push(flat.slice(0, MAX_CELL_TEXT));
   685→   685→          }
   686→   686→          break;
   687→   687→        }
   688→   688→        default:
   689→   689→          break;
   690→   690→      }
   691→   691→    },
   692→   692→  };
   693→   693→
   694→   694→  /** Non-empty text runs become blocks; the first one carries the bullet. */
   695→   695→  private endRun(): void {
   696→   696→    const para = this.para;
   697→   697→    if (!para || !para.run) return;
   698→   698→    const run = para.run;
   699→   699→    para.run = null;
   700→   700→    if (run.text === "") return;
   701→   701→    const text = run.text.slice(0, MAX_RUN_TEXT);
   702→   702→    if (text === "") return;
   703→   703→    if (para.runs.length >= MAX_RUNS_PER_PARA || para.textLen >= MAX_PARA_TEXT) {
   704→   704→      this.truncated = true;
   705→   705→      return;
   706→   706→    }
   707→   707→    para.textLen += text.length;
   708→   708→    para.runs.push({
   709→   709→      text,
   710→   710→      bold: run.bold,
   711→   711→      italic: run.italic,
   712→   712→      sizePt: run.sizePt,
   713→   713→      bullet: para.listLevel >= 0 && !para.firstRunDone ? true : undefined,
   714→   714→    });
   715→   715→    para.firstRunDone = true;
   716→   716→  }
   717→   717→
   718→   718→  private endPara(): void {
   719→   719→    const para = this.para;
   720→   720→    this.para = null;
   721→   721→    this.inNumPr = false;
   722→   722→    if (!para) return;
   723→   723→
   724→   724→    if (this.cellStack.length > 0) {
   725→   725→      // paragraph inside a table cell — contributes to the cell text
   726→   726→      const cell = this.cellStack[this.cellStack.length - 1];
   727→   727→      cell.texts.push(para.runs.map((r) => r.text).join("").slice(0, MAX_CELL_TEXT));
   728→   728→      return;
   729→   729→    }
   730→   730→    if (!this.inBody) return;
   731→   731→
   732→   732→    const text = para.runs.map((r) => r.text).join("");
   733→   733→    const isList = para.listLevel >= 0;
   734→   734→    const hasText = text.trim().length > 0;
   735→   735→
   736→   736→    if (!hasText && para.pageBreak) {
   737→   737→      this.pushBlock({ kind: "pagebreak" });
   738→   738→      return;
   739→   739→    }
   740→   740→    if (!hasText && !isList) {
   741→   741→      // Word's empty spacing paragraph — keep one honest blank line.
   742→   742→      this.pushBlock({ kind: "para", runs: [] });
   743→   743→      return;
   744→   744→    }
   745→   745→
   746→   746→    let headingLevel: 0 | 1 | 2 | 3 | undefined;
   747→   747→    const direct = directHeadingLevel(para.styleId);
   748→   748→    const viaStyles = direct === undefined ? this.styles.get(para.styleId) : direct;
   749→   749→    if (viaStyles !== undefined) {
   750→   750→      headingLevel = viaStyles === 0 ? 0 : (Math.min(3, viaStyles) as 1 | 2 | 3);
   751→   751→    }
   752→   752→
   753→   753→    this.pushBlock({
   754→   754→      kind: "para",
   755→   755→      runs: para.runs,
   756→   756→      align: para.align,
   757→   757→      headingLevel,
   758→   758→      listLevel: isList ? para.listLevel : undefined,
   759→   759→    });
   760→   760→    if (para.pageBreak) this.pushBlock({ kind: "pagebreak" });
   761→   761→  }
   762→   762→
   763→   763→  finish(): ParsedDoc {
   764→   764→    return {
   765→   765→      blocks: this.blocks,
   766→   766→      ...docStats(this.blocks),
   767→   767→      truncated: this.truncated,
   768→   768→    };
   769→   769→  }
   770→   770→}

   773→   773→/* parseDocx                                                            */
   774→   774→/* ------------------------------------------------------------------ */
   775→   775→
   776→   776→/**
   777→   777→ * Unzips a .docx (OOXML ZIP) and extracts the body flow. Any failure —
   778→   778→ * not a ZIP, no word/document.xml, empty body — returns an EMPTY doc
   779→   779→ * (the caller decides whether that is an error for the user).
   780→   780→ */
   781→   781→export function parseDocx(bytes: Uint8Array): ParsedDoc {
   782→   782→  let entries: Record<string, Uint8Array> | null = null;
   783→   783→  try {
   784→   784→    entries = unzipSync(bytes) as Record<string, Uint8Array>;
   785→   785→  } catch {
   786→   786→    return emptyDoc();
   787→   787→  }
   788→   788→  const docBytes = entries?.["word/document.xml"];
   789→   789→  if (!docBytes || docBytes.length === 0) return emptyDoc();
   790→   790→
   791→   791→  const decode = (b: Uint8Array): string => {
   792→   792→    let s = new TextDecoder("utf-8", { fatal: false }).decode(b);
   793→   793→    if (s.charCodeAt(0) === 0xfeff) s = s.slice(1);
   794→   794→    return s;
   795→   795→  };
   796→   796→
   797→   797→  let styles = new Map<string, number>();
   798→   798→  const styleBytes = entries?.["word/styles.xml"];
   799→   799→  if (styleBytes && styleBytes.length > 0) {
   800→   800→    try {
   801→   801→      styles = parseStylesXml(decode(styleBytes));
   802→   802→    } catch {
   803→   803→      styles = new Map();
   804→   804→    }
   805→   805→  }
   806→   806→
   807→   807→  const asm = new DocxAssembler(styles);
   808→   808→  try {
   809→   809→    parseXmlSax(decode(docBytes), asm.handlers);
   810→   810→  } catch {
   811→   811→    return emptyDoc();
   812→   812→  }
   813→   813→  return asm.finish();
   814→   814→}

   817→   817→/* parseWordRun — worker-boundary coercion (parseResumeRun pattern)     */

   819→   819→
   820→   820→function cleanText(v: unknown, max: number): string {
   821→   821→  if (typeof v !== "string") return "";
   822→   822→  return v
   823→   823→    .replace(/[\u0000-\u0008\u000B\u000C\u000E-\u001F\u007F]/g, "")
   824→   824→    .slice(0, max);
   825→   825→}
   826→   826→
   827→   827→function coerceAlign(v: unknown): DocxAlign | undefined {
   828→   828→  return v === "left" || v === "center" || v === "right" || v === "justify"
   829→   829→    ? v
   830→   830→    : undefined;
   831→   831→}
   832→   832→
   833→   833→function coerceHeading(v: unknown): 0 | 1 | 2 | 3 | undefined {
   834→   834→  const n = typeof v === "number" ? v : Number.parseInt(String(v), 10);
   835→   835→  if (!Number.isFinite(n)) return undefined;
   836→   836→  if (n === 0) return 0;
   837→   837→  if (n >= 1 && n <= 3) return n as 1 | 2 | 3;
   838→   838→  return undefined;
   839→   839→}
   840→   840→
   841→   841→function coerceLevel(v: unknown): number | undefined {
   842→   842→  const n = typeof v === "number" ? v : Number.parseInt(String(v), 10);
   843→   843→  if (!Number.isFinite(n) || n < 0) return undefined;
   844→   844→  return Math.min(3, Math.floor(n));
   845→   845→}
   846→   846→
   847→   847→function coerceSize(v: unknown): number | undefined {
   848→   848→  const n = typeof v === "number" ? v : Number.NaN;
   849→   849→  if (!Number.isFinite(n)) return undefined;
   850→   850→  return n >= 4 && n <= 72 ? n : undefined;
   851→   851→}
   852→   852→
   853→   853→/**
   854→   854→ * Full defensive coercion of a serialized ParsedDoc at the worker
   855→   855→ * boundary: garbage → empty doc, arrays clamped, every text capped,
   856→   856→ * unknown enum values dropped. Also re-derives listLevel from run-level
   857→   857→ * bullet flags when the block forgot it.
   858→   858→ */
   859→   859→export function parseWordRun(raw: string): ParsedDoc {
   860→   860→  let parsed: unknown;
   861→   861→  try {
   862→   862→    parsed = JSON.parse(raw);
   863→   863→  } catch {
   864→   864→    return emptyDoc();
   865→   865→  }
   866→   866→  const o = (typeof parsed === "object" && parsed !== null ? parsed : {}) as Record<
   867→   867→    string,
   868→   868→    unknown
   869→   869→  >;
   870→   870→  if (!Array.isArray(o.blocks)) return emptyDoc();
   871→   871→
   872→   872→  let truncated = o.truncated === true;
   873→   873→  const blocks: DocxBlock[] = [];
   874→   874→  let paraBlocks = 0;
   875→   875→
   876→   876→  outer: for (const bo of o.blocks) {
   877→   877→    if (blocks.length >= MAX_BLOCKS) {
   878→   878→      truncated = true;
   879→   879→      break;
   880→   880→    }
   881→   881→    if (typeof bo !== "object" || bo === null) continue;
   882→   882→    const b = bo as Record<string, unknown>;
   883→   883→    const kind = b.kind;
   884→   884→
   885→   885→    if (kind === "pagebreak") {
   886→   886→      blocks.push({ kind: "pagebreak" });
   887→   887→      continue;
   888→   888→    }
   889→   889→
   890→   890→    if (kind === "table") {
   891→   891→      const rawRows = Array.isArray(b.rows) ? b.rows : [];
   892→   892→      const rows: string[][] = [];
   893→   893→      for (const ro of rawRows) {
   894→   894→        if (rows.length >= MAX_TABLE_ROWS) {
   895→   895→          truncated = true;
   896→   896→          break;
   897→   897→        }
   898→   898→        if (!Array.isArray(ro)) continue;
   899→   899→        const cells: string[] = [];
   900→   900→        for (const co of ro) {
   901→   901→          if (cells.length >= MAX_TABLE_COLS) {
   902→   902→            truncated = true;
   903→   903→            break;
   904→   904→          }
   905→   905→          cells.push(hindiNorm(cleanText(co, MAX_CELL_TEXT)));
   906→   906→        }
   907→   907→        if (cells.length > 0) rows.push(cells);
   908→   908→      }
   909→   909→      if (rows.length > 0) blocks.push({ kind: "table", rows });
   910→   910→      continue;
   911→   911→    }
   912→   912→
   913→   913→    if (kind !== "para") continue;
   914→   914→    if (paraBlocks >= MAX_PARAGRAPHS) {
   915→   915→      truncated = true;
   916→   916→      break outer;
   917→   917→    }
   918→   918→    const rawRuns = Array.isArray(b.runs) ? b.runs : [];
   919→   919→    const runs: DocxRun[] = [];
   920→   920→    let textLen = 0;
   921→   921→    let bulletSeen = false;
   922→   922→    for (const ro of rawRuns) {
   923→   923→      if (runs.length >= MAX_RUNS_PER_PARA || textLen >= MAX_PARA_TEXT) {
   924→   924→        truncated = true;
   925→   925→        break;
   926→   926→      }
   927→   927→      if (typeof ro !== "object" || ro === null) continue;
   928→   928→      const r = ro as Record<string, unknown>;
   929→   929→      let text = hindiNorm(cleanText(r.text, MAX_RUN_TEXT));
   930→   930→      if (textLen + text.length > MAX_PARA_TEXT) {
   931→   931→        text = text.slice(0, Math.max(0, MAX_PARA_TEXT - textLen));
   932→   932→      }
   933→   933→      if (text === "") continue;
   934→   934→      textLen += text.length;
   935→   935→      const bullet = r.bullet === true;
   936→   936→      if (bullet) bulletSeen = true;
   937→   937→      runs.push({
   938→   938→        text,
   939→   939→        bold: r.bold === true,
   940→   940→        italic: r.italic === true,
   941→   941→        sizePt: coerceSize(r.sizePt),
   942→   942→        bullet: bullet || undefined,
   943→   943→      });
   944→   944→    }
   945→   945→    const headingLevel = coerceHeading(b.headingLevel);
   946→   946→    const align = coerceAlign(b.align);
   947→   947→    let listLevel = coerceLevel(b.listLevel);
   948→   948→    if (listLevel === undefined && bulletSeen) listLevel = 0;
   949→   949→    if (runs.length === 0 && listLevel === undefined && headingLevel === undefined && align === undefined) {
   950→   950→      // preserve honest blank lines, drop fully-empty husks
   951→   951→      blocks.push({ kind: "para", runs: [] });
   952→   952→      paraBlocks += 1;
   953→   953→      continue;
   954→   954→    }
   955→   955→    blocks.push({
   956→   956→      kind: "para",
   957→   957→      runs,
   958→   958→      align,
   959→   959→      headingLevel,
   960→   960→      listLevel,
   961→   961→    });
   962→   962→    paraBlocks += 1;
   963→   963→  }

   965→   965→  return { blocks, ...docStats(blocks), truncated };
   966→   966→}

   968→   968→/* ------------------------------------------------------------------ */
   969→   969→/* Output file name (house pdf-to-txt convention + localized fallback)  */
   970→   970→/* ------------------------------------------------------------------ */
   971→   971→
   972→   972→/**
   973→   973→ * "Report v2.docx" → "Report v2.pdf"; a nameless file falls back to the
   974→   974→ * localized word ("document" / "दस्तावेज़"). Unicode survives; path
   975→   975→ * separators and unsafe chars do not.
   976→   976→ */
   977→   977→export function outputPdfName(fileName: string, fileWord: string): string {
   978→   978→  const base = fileName
   979→   979→    .replace(/\.(docx|doc)$/i, "")
   980→   980→    .replace(/[\\/:*?"<>|\u0000-\u001f]/g, "_")
   981→   981→    .trim()
   982→   982→    .slice(0, 120);
   983→   983→  const fallback = cleanText(fileWord, 60).trim() || "document";
   984→   984→  return `${base || fallback}.pdf`;
   985→   985→}

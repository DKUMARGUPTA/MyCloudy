   800→  1750→function arrowHeadPoints(
   801→  1751→  x1: number,
   802→  1752→  y1: number,
   803→  1753→  x2: number,
   804→  1754→  y2: number,
   805→  1755→  size: number
   806→  1756→): string {
   807→  1757→  const angle = Math.atan2(y2 - y1, x2 - x1);
   808→  1758→  const a1 = angle + Math.PI - 0.45;
   809→  1759→  const a2 = angle + Math.PI + 0.45;
   810→  1760→  return [
   811→  1761→    `${x2},${y2}`,
   812→  1762→    `${x2 + size * Math.cos(a1)},${y2 + size * Math.sin(a1)}`,
   813→  1763→    `${x2 + size * Math.cos(a2)},${y2 + size * Math.sin(a2)}`,
   814→  1764→  ].join(" ");
   815→  1765→}

     1→"use client";
     2→
     3→import {
     4→  Fragment,
     5→  useEffect,
     6→  useId,
     7→  useMemo,
     8→  useRef,
     9→  useState,
    10→  type KeyboardEvent,
    11→} from "react";
    12→import {
    13→  Bookmark,
    14→  Files,
    15→  Loader2,
    16→  Scissors,
    17→  TriangleAlert,
    18→} from "lucide-react";
    19→
    20→import { Input } from "@/components/ui/input";
    21→import { Label } from "@/components/ui/label";
    22→import { Switch } from "@/components/ui/switch";
    23→import { useI18n } from "@/components/providers/locale-provider";
    24→import type { ToolPanelProps } from "@/lib/tools/panel-types";
    25→import {
    26→  detectBlankPages,
    27→  getPdfPageCount,
    28→  getPdfTopLevelOutline,
    29→  renderPdfThumbnail,
    30→  type PdfOutlineEntry,
    31→} from "@/lib/pdf/thumbnails";
    32→import {
    33→  computeSplitPlan,
    34→  fileKey,
    35→  parseNumberListOption,
    36→  parseRangeList,
    37→  rangesOverlap,
    38→  type SplitMode,
    39→  type SplitPart,
    40→} from "./plan";
    41→import { cn } from "@/lib/utils";
    42→
    43→/**
    44→ * Split PDF custom panel (PRD §8.1): mode tabs (7 modes), per-mode inputs
    45→ * and the LIVE preview strip — cut markers, color-coded parts and the
    46→ * "N files banengi" count — all computed with the same pure plan module the
    47→ * worker executes. Bookmark + blank-page detection run on the main thread
    48→ * with the self-hosted pdf.js and reach the worker as plain options.
    49→ */
    50→
    51→const MODE_IDS: { id: SplitMode; labelKey: string }[] = [
    52→  { id: "range", labelKey: "split.modeRange" },
    53→  { id: "everyN", labelKey: "split.modeEveryN" },
    54→  { id: "bookmarks", labelKey: "split.modeBookmarks" },
    55→  { id: "size", labelKey: "split.modeSize" },
    56→  { id: "visual", labelKey: "split.modeVisual" },
    57→  { id: "oddEven", labelKey: "split.modeOddEven" },
    58→  { id: "blanks", labelKey: "split.modeBlanks" },
    59→];
    60→
    61→const PART_STYLES = [
    62→  "border-primary/30 bg-primary/10 text-primary dark:text-primary",
    63→  "border-purple-500/30 bg-purple-500/10 text-purple-600 dark:text-purple-400",
    64→  "border-emerald-500/30 bg-emerald-500/10 text-emerald-600 dark:text-emerald-400",
    65→  "border-amber-500/30 bg-amber-500/10 text-amber-600 dark:text-amber-400",
    66→];
    67→
    68→function partStyle(index: number): string {
    69→  return PART_STYLES[index % PART_STYLES.length];
    70→}
    71→
    72→export function SplitPanel({
    73→  files,
    74→  values,
    75→  onChange,
    76→}: ToolPanelProps) {
    77→  const { t } = useI18n();
    78→  // Instance-unique ids: this panel is mounted in BOTH the desktop aside and
    79→  // the mobile bottom sheet (one hidden, one visible — both in the DOM), so
    80→  // hardcoded ids would duplicate and break label→input associations.
    81→  const uid = useId();
    82→  const ids = {
    83→    ranges: `${uid}-split-ranges`,
    84→    onePerRange: `${uid}-split-one-per-range`,
    85→    everyN: `${uid}-split-every-n`,
    86→    maxSize: `${uid}-split-max-size`,
    87→  } as const;
    88→  const file = files.length > 0 ? files[0].file : null;
    89→  const tabRefs = useRef<(HTMLButtonElement | null)[]>([]);
    90→
    91→  const mode: SplitMode = useMemo(() => {
    92→    const raw = values.mode;
    93→    return typeof raw === "string" &&
    94→      MODE_IDS.some((entry) => entry.id === raw)
    95→      ? (raw as SplitMode)
    96→      : "range";
    97→  }, [values.mode]);
    98→
    99→  const ranges = typeof values.ranges === "string" ? values.ranges : "";
   100→  const oneFilePerRange = values.oneFilePerRange !== false;
   101→  const everyN = typeof values.everyN === "number" ? values.everyN : 5;
   102→  const maxSizeMb =
   103→    typeof values.maxSizeMb === "number" ? values.maxSizeMb : 5;
   104→  const cutAfter = useMemo(
   105→    () => parseNumberListOption(values.cutAfter),
   106→    [values.cutAfter]
   107→  );
   108→
   109→  /* ---------------- Page count + outline detection ---------------- */
   110→  // Async results are keyed by their File and only applied to state inside
   111→  // promise callbacks (never synchronously in the effect body).
   112→
   113→  const [pageCountState, setPageCountState] = useState<{
   114→    file: File;
   115→    count: number;
   116→  } | null>(null);
   117→  const pageCount =
   118→    pageCountState !== null && pageCountState.file === file
   119→      ? pageCountState.count
   120→      : null;
   121→
   122→  const [outlineState, setOutlineState] = useState<{
   123→    file: File;
   124→    entries: PdfOutlineEntry[];
   125→  } | null>(null);
   126→  const outline =
   127→    outlineState !== null && outlineState.file === file
   128→      ? outlineState.entries
   129→      : null;
   130→
   131→  useEffect(() => {
   132→    if (!file) return;
   133→    let cancelled = false;
   134→    void getPdfPageCount(file)
   135→      .then((count) => {
   136→        if (!cancelled) setPageCountState({ file, count });
   137→      })
   138→      .catch(() => {
   139→        if (!cancelled) setPageCountState({ file, count: 0 });
   140→      });
   141→    return () => {
   142→      cancelled = true;
   143→    };
   144→  }, [file]);
   145→
   146→  // Outline runs for every file (it decides whether the bookmarks tab is
   147→  // usable) and mirrors its result into options for the worker.
   148→  useEffect(() => {
   149→    if (!file) return;
   150→    let cancelled = false;
   151→    const key = fileKey(file.name, file.size);
   152→    void getPdfTopLevelOutline(file)
   153→      .then((entries) => {
   154→        if (cancelled) return;
   155→        setOutlineState({ file, entries });
   156→        onChange("bookmarkStatus", "done");
   157→        onChange("bookmarkFileKey", key);
   158→        onChange(
   159→          "bookmarkCuts",
   160→          JSON.stringify([...new Set(entries.map((entry) => entry.page))].sort((a, b) => a - b))
   161→        );
   162→      })
   163→      .catch(() => {
   164→        if (cancelled) return;
   165→        setOutlineState({ file, entries: [] });
   166→        onChange("bookmarkStatus", "done");
   167→        onChange("bookmarkFileKey", key);
   168→        onChange("bookmarkCuts", "[]");
   169→      });
   170→    return () => {
   171→      cancelled = true;
   172→    };
   173→  }, [file, onChange]);
   174→
   175→  /* ---------------- Blank-page detection (mode 7) ---------------- */
   176→
   177→  const [blankProgress, setBlankProgress] = useState<{
   178→    file: File;
   179→    done: number;
   180→    total: number;
   181→  } | null>(null);
   182→
   183→  const blankDetectionDone =
   184→    file !== null &&
   185→    values.blankStatus === "done" &&
   186→    values.blankFileKey === fileKey(file.name, file.size);
   187→  const blankPages = blankDetectionDone
   188→    ? parseNumberListOption(values.blankPages)
   189→    : [];
   190→  const blanksDetecting =
   191→    file !== null && mode === "blanks" && !blankDetectionDone;
   192→
   193→  useEffect(() => {
   194→    if (!file || mode !== "blanks") return;
   195→    if (
   196→      values.blankStatus === "done" &&
   197→      values.blankFileKey === fileKey(file.name, file.size)
   198→    ) {
   199→      return; // already detected for this exact file
   200→    }
   201→    let cancelled = false;
   202→    void detectBlankPages(file, (progress) => {
   203→      if (!cancelled) {
   204→        setBlankProgress({ file, done: progress.done, total: progress.total });
   205→      }
   206→    })
   207→      .then((pages) => {
   208→        if (cancelled) return;
   209→        setBlankProgress(null);
   210→        onChange("blankStatus", "done");
   211→        onChange("blankFileKey", fileKey(file.name, file.size));
   212→        onChange("blankPages", JSON.stringify(pages));
   213→      })
   214→      .catch(() => {
   215→        if (cancelled) return;
   216→        setBlankProgress(null);
   217→        onChange("blankStatus", "done");
   218→        onChange("blankFileKey", fileKey(file.name, file.size));
   219→        onChange("blankPages", "[]");
   220→      });
   221→    return () => {
   222→      cancelled = true;
   223→    };
   224→  }, [file, mode, onChange, values.blankStatus, values.blankFileKey]);
   225→
   226→  /* ---------------- Live plan preview ---------------- */
   227→
   228→  const rangeItems =
   229→    mode === "range" && pageCount !== null
   230→      ? parseRangeList(ranges, pageCount)
   231→      : null;
   232→  const rangesValid = mode !== "range" || rangeItems !== null;
   233→  const overlapping =
   234→    rangeItems !== null && rangesOverlap(rangeItems) ? true : false;
   235→
   236→  const plan = useMemo(() => {
   237→    if (!file || pageCount === null || pageCount < 1) return null;
   238→    return computeSplitPlan({
   239→      mode,
   240→      pageCount,
   241→      fileSize: file.size,
   242→      ranges,
   243→      oneFilePerRange,
   244→      everyN,
   245→      maxSizeMb,
   246→      cutAfter,
   247→      bookmarkCuts: outline ? [...new Set(outline.map((e) => e.page))] : [],
   248→      blankPages,
   249→    });
   250→  }, [
   251→    file,
   252→    pageCount,
   253→    mode,
   254→    ranges,
   255→    oneFilePerRange,
   256→    everyN,
   257→    maxSizeMb,
   258→    cutAfter,
   259→    outline,
   260→    blankPages,
   261→  ]);
   262→
   263→  const bookmarkTabDisabled = outline !== null && outline.length === 0;
   264→  const canShowCount =
   265→    plan !== null && !(mode === "blanks" && blankDetectionDone && blankPages.length === 0);
   266→
   267→  const toggleCut = (page: number) => {
   268→    const next = cutAfter.includes(page)
   269→      ? cutAfter.filter((item) => item !== page)
   270→      : [...cutAfter, page].sort((a, b) => a - b);
   271→    onChange("cutAfter", JSON.stringify(next));
   272→  };
   273→
   274→  const selectMode = (next: SplitMode) => {
   275→    if (next === "bookmarks" && bookmarkTabDisabled) return;
   276→    onChange("mode", next);
   277→  };
   278→
   279→  const onTablistKeyDown = (event: KeyboardEvent<HTMLDivElement>) => {
   280→    if (event.key !== "ArrowRight" && event.key !== "ArrowLeft") return;
   281→    event.preventDefault();
   282→    const direction = event.key === "ArrowRight" ? 1 : -1;
   283→    const enabledIndexes = MODE_IDS.map((entry, index) => ({
   284→      entry,
   285→      index,
   286→    }))
   287→      .filter(({ entry }) => !(entry.id === "bookmarks" && bookmarkTabDisabled))
   288→      .map(({ index }) => index);
   289→    if (enabledIndexes.length === 0) return;
   290→    const currentIndex = MODE_IDS.findIndex((entry) => entry.id === mode);
   291→    const position = enabledIndexes.indexOf(currentIndex);
   292→    const nextPosition =
   293→      (position === -1 ? 0 : position + direction + enabledIndexes.length) %
   294→      enabledIndexes.length;
   295→    const nextIndex = enabledIndexes[nextPosition];
   296→    const nextEntry = MODE_IDS[nextIndex];
   297→    tabRefs.current[nextIndex]?.focus();
   298→    selectMode(nextEntry.id);
   299→  };
   300→
   301→  /* ---------------- Render ---------------- */
   302→
   303→  return (
   304→    <section
   305→      aria-label={t("split.panelTitle")}
   306→      className="flex flex-col gap-3"
   307→    >
   308→      <div className="flex items-center justify-between gap-2">
   309→        <h3 className="flex items-center gap-1.5 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   310→          <Scissors className="size-3.5" aria-hidden="true" />
   311→          {t("split.panelTitle")}
   312→        </h3>
   313→        {file ? (
   314→          <span className="text-[11px] text-muted-foreground">
   315→            {pageCount !== null
   316→              ? t("split.pagesInFile", { count: pageCount })
   317→              : t("split.detecting")}
   318→          </span>
   319→        ) : null}
   320→      </div>
   321→
   322→      {!file ? (
   323→        <p className="rounded-btn border border-dashed border-border bg-muted/30 px-3 py-2 text-[11px] text-muted-foreground">
   324→          {t("split.emptyHint")}
   325→        </p>
   326→      ) : (
   327→        <>
   328→          {/* Mode tabs (PRD §8.1 "Modes (tabs)") */}
   329→          <div
   330→            role="tablist"
   331→            aria-label={t("split.modeLabel")}
   332→            className="flex flex-wrap gap-1.5"
   333→            onKeyDown={onTablistKeyDown}
   334→          >
   335→            {MODE_IDS.map((entry, index) => {
   336→              const disabled = entry.id === "bookmarks" && bookmarkTabDisabled;
   337→              const selected = mode === entry.id;
   338→              return (
   339→                <button
   340→                  key={entry.id}
   341→                  ref={(node) => {
   342→                    tabRefs.current[index] = node;
   343→                  }}
   344→                  type="button"
   345→                  role="tab"
   346→                  aria-selected={selected}
   347→                  aria-disabled={disabled}
   348→                  disabled={disabled}
   349→                  title={
   350→                    disabled ? t("split.bookmarksDisabled") : undefined
   351→                  }
   352→                  onClick={() => selectMode(entry.id)}
   353→                  className={cn(
   354→                    "min-h-9 rounded-btn border px-2.5 text-xs font-medium outline-none transition-colors",
   355→                    selected
   356→                      ? "border-primary bg-primary text-primary-foreground"
   357→                      : "border-border bg-background text-foreground hover:border-primary/40 hover:bg-accent",
   358→                    disabled && "cursor-not-allowed opacity-50",
   359→                    "focus-visible:ring-[3px] focus-visible:ring-ring/50"
   360→                  )}
   361→                >
   362→                  {t(entry.labelKey)}
   363→                </button>
   364→              );
   365→            })}
   366→          </div>
   367→
   368→          {/* Per-mode controls */}
   369→          {mode === "range" ? (
   370→            <div className="flex flex-col gap-2">
   371→              <div className="flex flex-col gap-1.5">
   372→                <Label htmlFor={ids.ranges} className="text-xs font-medium">
   373→                  {t("split.rangeLabel")}
   374→                </Label>
   375→                <Input
   376→                  id={ids.ranges}
   377→                  type="text"
   378→                  spellCheck={false}
   379→                  autoComplete="off"
   380→                  value={ranges}
   381→                  placeholder={t("split.rangePlaceholder")}
   382→                  onChange={(event) => onChange("ranges", event.target.value)}
   383→                  aria-invalid={!rangesValid && ranges.trim() !== ""}
   384→                  className={cn(
   385→                    "h-10 rounded-btn text-xs font-medium",
   386→                    !rangesValid &&
   387→                      ranges.trim() !== "" &&
   388→                      "border-destructive focus-visible:ring-destructive/30"
   389→                  )}
   390→                />
   391→                {ranges.trim() === "" ? (
   392→                  <p className="text-[11px] leading-relaxed text-muted-foreground">
   393→                    {t("split.rangeHint")}
   394→                  </p>
   395→                ) : rangesValid ? (
   396→                  overlapping ? (
   397→                    <p className="flex items-start gap-1.5 rounded-btn border border-amber-500/40 bg-amber-500/10 px-2 py-1.5 text-[11px] font-medium leading-relaxed text-amber-600 dark:text-amber-400">
   398→                      <TriangleAlert className="mt-0.5 size-3.5 shrink-0" aria-hidden="true" />
   399→                      {t("split.overlapWarning")}
   400→                    </p>
   401→                  ) : (
   402→                    <p className="text-[11px] leading-relaxed text-muted-foreground">
   403→                      {t("split.rangeHint")}
   404→                    </p>
   405→                  )
   406→                ) : (
   407→                  <p className="text-[11px] font-medium leading-relaxed text-destructive">
   408→                    {t("split.invalidRanges")}
   409→                  </p>
   410→                )}
   411→              </div>
   412→              <div className="flex items-center justify-between gap-3 rounded-btn border border-border bg-background px-2.5 py-2">
   413→                <Label
   414→                  htmlFor={ids.onePerRange}
   415→                  className="text-xs font-medium leading-snug"
   416→                >
   417→                  {t("split.oneFilePerRangeLabel")}
   418→                  <span className="block text-[11px] font-normal text-muted-foreground">
   419→                    {t("split.oneFilePerRangeHint")}
   420→                  </span>
   421→                </Label>
   422→                <Switch
   423→                  id={ids.onePerRange}
   424→                  checked={oneFilePerRange}
   425→                  onCheckedChange={(next) => onChange("oneFilePerRange", next)}
   426→                  aria-label={t("split.oneFilePerRangeLabel")}
   427→                />
   428→              </div>
   429→            </div>
   430→          ) : null}
   431→
   432→          {mode === "everyN" ? (
   433→            <div className="flex flex-col gap-1.5">
   434→              <Label htmlFor={ids.everyN} className="text-xs font-medium">
   435→                {t("split.everyNLabel")}
   436→              </Label>
   437→              <div className="flex items-center gap-2">
   438→                <Input
   439→                  id={ids.everyN}
   440→                  type="number"
   441→                  inputMode="numeric"
   442→                  min={1}
   443→                  max={pageCount ?? 9999}
   444→                  step={1}
   445→                  value={everyN}
   446→                  onChange={(event) =>
   447→                    onChange("everyN", event.target.valueAsNumber || 1)
   448→                  }
   449→                  className="h-10 w-24 rounded-btn text-xs font-medium"
   450→                />
   451→                <span className="text-xs text-muted-foreground">
   452→                  {t("split.pagesSuffix")}
   453→                </span>
   454→              </div>
   455→              <p className="text-[11px] leading-relaxed text-muted-foreground">
   456→                {t("split.everyNHint")}
   457→              </p>
   458→            </div>
   459→          ) : null}
   460→
   461→          {mode === "bookmarks" ? (
   462→            outline === null ? (
   463→              <p className="flex items-center gap-2 rounded-btn border border-dashed border-border bg-muted/30 px-3 py-2 text-[11px] text-muted-foreground">
   464→                <Loader2
   465→                  className="size-3.5 animate-spin"
   466→                  aria-hidden="true"
   467→                />
   468→                {t("split.detecting")}
   469→              </p>
   470→            ) : outline.length === 0 ? (
   471→              <p className="flex items-start gap-1.5 rounded-btn border border-dashed border-border bg-muted/30 px-3 py-2 text-[11px] text-muted-foreground">
   472→                <Bookmark className="mt-0.5 size-3.5 shrink-0" aria-hidden="true" />
   473→                {t("split.bookmarksNone")}
   474→              </p>
   475→            ) : (
   476→              <div className="flex flex-col gap-1.5">
   477→                <p className="text-[11px] text-muted-foreground">
   478→                  {t("split.bookmarksCount", { count: outline.length })} —{" "}
   479→                  {t("split.bookmarksList")}
   480→                </p>
   481→                <ul className="scrollbar-thin flex max-h-36 flex-col gap-1 overflow-y-auto pr-1">
   482→                  {outline.map((entry, index) => (
   483→                    <li
   484→                      key={`${entry.page}-${index}`}
   485→                      className="flex items-center gap-2 rounded-btn border border-border bg-background px-2 py-1.5"
   486→                    >
   487→                      <span
   488→                        className={cn(
   489→                          "flex size-5 shrink-0 items-center justify-center rounded-full text-[10px] font-bold",
   490→                          partStyle(index)
   491→                        )}
   492→                        aria-hidden="true"
   493→                      >
   494→                        {index + 1}
   495→                      </span>
   496→                      <span className="min-w-0 flex-1 truncate text-xs text-foreground" title={entry.title}>
   497→                        {entry.title}
   498→                      </span>
   499→                      <span className="shrink-0 rounded-chip bg-muted px-1.5 py-0.5 text-[10px] font-semibold text-muted-foreground">
   500→                        {t("split.pageChipLabel", { page: entry.page })}
   501→                      </span>
   502→                    </li>
   503→                  ))}
   504→                </ul>
   505→              </div>
   506→            )
   507→          ) : null}
   508→
   509→          {mode === "size" ? (
   510→            <div className="flex flex-col gap-1.5">
   511→              <Label htmlFor={ids.maxSize} className="text-xs font-medium">
   512→                {t("split.sizeLabel")}
   513→              </Label>
   514→              <div className="flex items-center gap-2">
   515→                <Input
   516→                  id={ids.maxSize}
   517→                  type="number"
   518→                  inputMode="decimal"
   519→                  min={0.1}
   520→                  max={2000}
   521→                  step={0.5}
   522→                  value={maxSizeMb}
   523→                  onChange={(event) =>
   524→                    onChange(
   525→                      "maxSizeMb",
   526→                      event.target.valueAsNumber > 0
   527→                        ? event.target.valueAsNumber
   528→                        : 0.1
   529→                    )
   530→                  }
   531→                  className="h-10 w-24 rounded-btn text-xs font-medium"
   532→                />
   533→                <span className="text-xs text-muted-foreground">MB</span>
   534→              </div>
   535→              <p className="text-[11px] leading-relaxed text-muted-foreground">
   536→                {t("split.sizeHint")}
   537→              </p>
   538→            </div>
   539→          ) : null}
   540→
   541→          {mode === "visual" ? (
   542→            <div className="flex flex-col gap-2">
   543→              <p className="text-[11px] leading-relaxed text-muted-foreground">
   544→                {t("split.visualHint")}
   545→              </p>
   546→              {cutAfter.length > 0 ? (
   547→                <button
   548→                  type="button"
   549→                  onClick={() => onChange("cutAfter", "[]")}
   550→                  className="flex min-h-9 w-fit items-center gap-1.5 rounded-btn border border-border bg-background px-2.5 text-xs font-medium text-muted-foreground outline-none transition-colors hover:border-destructive/40 hover:text-destructive focus-visible:ring-[3px] focus-visible:ring-ring/50"
   551→                >
   552→                  <Scissors className="size-3.5" aria-hidden="true" />
   553→                  {t("split.visualClear", { count: cutAfter.length })}
   554→                </button>
   555→              ) : null}
   556→            </div>
   557→          ) : null}
   558→
   559→          {mode === "oddEven" ? (
   560→            <p className="text-[11px] leading-relaxed text-muted-foreground">
   561→              {t("split.oddEvenHint")}
   562→            </p>
   563→          ) : null}
   564→
   565→          {mode === "blanks" ? (
   566→            blanksDetecting ? (
   567→              <p className="flex items-center gap-2 rounded-btn border border-dashed border-border bg-muted/30 px-3 py-2 text-[11px] text-muted-foreground">
   568→                <Loader2 className="size-3.5 animate-spin" aria-hidden="true" />
   569→                {blankProgress !== null && blankProgress.file === file
   570→                  ? t("split.blanksDetectingPct", {
   571→                      percent: Math.round(
   572→                        (blankProgress.done / Math.max(1, blankProgress.total)) * 100
   573→                      ),
   574→                    })
   575→                  : t("split.blanksDetecting")}
   576→              </p>
   577→            ) : blankPages.length === 0 ? (
   578→              <p className="flex items-start gap-1.5 rounded-btn border border-dashed border-border bg-muted/30 px-3 py-2 text-[11px] text-muted-foreground">
   579→                <TriangleAlert className="mt-0.5 size-3.5 shrink-0" aria-hidden="true" />
   580→                {t("split.blanksNone")}
   581→              </p>
   582→            ) : (
   583→              <div className="flex flex-col gap-2">
   584→                <p className="text-[11px] font-medium leading-relaxed text-foreground">
   585→                  {t("split.blanksFound", { count: blankPages.length })}
   586→                </p>
   587→                <div className="flex flex-wrap gap-1" aria-label={t("split.blanksChipLabel")}>
   588→                  {blankPages.map((page) => (
   589→                    <span
   590→                      key={page}
   591→                      className="rounded-md border border-border bg-muted px-1.5 py-0.5 text-[11px] font-semibold text-muted-foreground line-through"
   592→                    >
   593→                      {page}
   594→                    </span>
   595→                  ))}
   596→                </div>
   597→              </div>
   598→            )
   599→          ) : null}
   600→
   601→          {/* Live preview (PRD §8.1: cut markers + color-coded parts +
   602→              live count) */}
   603→          <div className="flex flex-col gap-2.5 rounded-card border border-border bg-background p-3">
   604→            <div className="flex items-center justify-between gap-2">
   605→              <h4 className="flex items-center gap-1.5 text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">
   606→                <Files className="size-3.5" aria-hidden="true" />
   607→                {t("split.previewTitle")}
   608→              </h4>
   609→              <div className="flex items-center gap-1.5">
   610→                {plan?.estimated ? (
   611→                  <span
   612→                    className="rounded-chip border border-amber-500/40 bg-amber-500/10 px-1.5 py-0.5 text-[10px] font-semibold text-amber-600 dark:text-amber-400"
   613→                    title={t("split.sizeHint")}
   614→                  >
   615→                    ≈ {t("split.estimated")}
   616→                  </span>
   617→                ) : null}
   618→                {canShowCount && plan !== null ? (
   619→                  <span className="rounded-chip bg-primary/10 px-2 py-0.5 text-[11px] font-semibold text-primary">
   620→                    {plan.parts.length === 1
   621→                      ? t("split.willCreateOne")
   622→                      : t("split.willCreate", { count: plan.parts.length })}
   623→                  </span>
   624→                ) : null}
   625→              </div>
   626→            </div>
   627→
   628→            {mode === "visual" && pageCount !== null ? (
   629→              <VisualStrip
   630→                file={file}
   631→                pageCount={pageCount}
   632→                cutAfter={cutAfter}
   633→                parts={plan?.parts ?? []}
   634→                onToggleCut={toggleCut}
   635→              />
   636→            ) : pageCount === null ? (
   637→              <p className="flex items-center gap-2 px-1 py-2 text-[11px] text-muted-foreground">
   638→                <Loader2 className="size-3.5 animate-spin" aria-hidden="true" />
   639→                {t("split.detecting")}
   640→              </p>
   641→            ) : plan === null ? (
   642→              ranges.trim() === "" ? (
   643→                <p className="px-1 py-2 text-[11px] text-muted-foreground">
   644→                  {t("split.rangePrompt")}
   645→                </p>
   646→              ) : (
   647→                <p className="px-1 py-2 text-[11px] font-medium text-destructive">
   648→                  {t("split.invalidRanges")}
   649→                </p>
   650→              )
   651→            ) : plan.parts.length === 0 ? (
   652→              <p className="px-1 py-2 text-[11px] text-muted-foreground">
   653→                {t("split.noParts")}
   654→              </p>
   655→            ) : (
   656→              <CompactStrip parts={plan.parts} />
   657→            )}
   658→          </div>
   659→        </>
   660→      )}
   661→    </section>
   662→  );
   663→}

   724→/** Thumbnail picker: real page thumbs + scissors buttons between pages. */
   725→function VisualStrip({
   726→  file,
   727→  pageCount,
   728→  cutAfter,
   729→  parts,
   730→  onToggleCut,
   731→}: {
   732→  file: File;
   733→  pageCount: number;
   734→  cutAfter: number[];
   735→  parts: SplitPart[];
   736→  onToggleCut: (page: number) => void;
   737→}) {
   738→  const { t } = useI18n();
   739→  const partIndexOf = new Map<number, number>();
   740→  parts.forEach((part, partIndex) => {
   741→    for (const pageIndex of part.pages) partIndexOf.set(pageIndex, partIndex);
   742→  });
   743→
   744→  return (
   745→    <div
   746→      role="group"
   747→      aria-label={t("split.stripLabel")}
   748→      className="scrollbar-thin flex max-h-72 flex-wrap items-center gap-1 overflow-y-auto pr-1"
   749→    >
   750→      {Array.from({ length: pageCount }, (_, index) => index + 1).map(
   751→        (pageNumber) => {
   752→          const partIndex = partIndexOf.get(pageNumber - 1) ?? 0;
   753→          const cut = cutAfter.includes(pageNumber);
   754→          return (
   755→            <Fragment key={pageNumber}>
   756→              <div className="flex w-14 flex-col items-center gap-1">
   757→                <VisualThumb file={file} pageIndex={pageNumber - 1} />
   758→                <span
   759→                  className={cn(
   760→                    "rounded-md border px-1 text-[10px] font-bold",
   761→                    partStyle(partIndex)
   762→                  )}
   763→                >
   764→                  {pageNumber}
   765→                </span>
   766→              </div>
   767→              {pageNumber < pageCount ? (
   768→                <button
   769→                  type="button"
   770→                  aria-pressed={cut}
   771→                  aria-label={t("split.cutAfter", { page: pageNumber })}
   772→                  title={t("split.cutAfter", { page: pageNumber })}
   773→                  onClick={() => onToggleCut(pageNumber)}
   774→                  className={cn(
   775→                    "flex size-11 shrink-0 items-center justify-center rounded-full border outline-none transition-colors focus-visible:ring-[3px] focus-visible:ring-ring/50",
   776→                    cut
   777→                      ? "border-primary bg-primary text-primary-foreground"
   778→                      : "border-dashed border-border bg-background text-muted-foreground hover:border-primary/50 hover:text-primary"
   779→                  )}
   780→                >
   781→                  <Scissors className="size-4" aria-hidden="true" />
   782→                </button>
   783→              ) : null}
   784→            </Fragment>
   785→          );
   786→        }
   787→      )}
   788→    </div>
   789→  );
   790→}

   792→/** One queued thumbnail (render queue caps concurrency at 2). */
   793→function VisualThumb({
   794→  file,
   795→  pageIndex,
   796→}: {
   797→  file: File;
   798→  pageIndex: number;
   799→}) {
   800→  const [url, setUrl] = useState<string | null>(null);
   801→
   802→  useEffect(() => {
   803→    let cancelled = false;
   804→    let objectUrl: string | null = null;
   805→    void renderPdfThumbnail(file, { pageIndex, maxDim: 112 })
   806→      .then((result) => {
   807→        if (cancelled) return;
   808→        objectUrl = URL.createObjectURL(result.blob);
   809→        setUrl(objectUrl);
   810→      })
   811→      .catch(() => undefined);
   812→    return () => {
   813→      cancelled = true;
   814→      if (objectUrl) URL.revokeObjectURL(objectUrl);
   815→    };
   816→  }, [file, pageIndex]);
   817→
   818→  return (
   819→    <div className="relative flex aspect-[3/4] w-full items-center justify-center overflow-hidden rounded-md border border-border bg-muted/40">
   820→      {url ? (
   821→        <img
   822→          src={url}
   823→          alt=""
   824→          loading="lazy"
   825→          decoding="async"
   826→          className="size-full object-contain"
   827→        />
   828→      ) : (
   829→        <div
   830→          aria-hidden="true"
   831→          className="size-full animate-pulse bg-gradient-to-br from-muted to-muted/40"
   832→        />
   833→      )}
   834→    </div>
   835→  );
   836→}

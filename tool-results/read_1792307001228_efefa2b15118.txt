   132→   132→/** Keeps the edited list aligned with the live workspace files. */
   133→   133→function reconcileEntries(
   134→   134→  entries: CollageEntry[],
   135→   135→  files: ToolSurfaceProps["files"],
   136→   136→  previousCount: number
   137→   137→): CollageEntry[] {
   138→   138→  const count = files.length;
   139→   139→  if (count === previousCount) return entries;
   140→   140→  const next: CollageEntry[] = [];
   141→   141→  const seen = new Set<number>();
   142→   142→  if (count < previousCount) {
   143→   143→    for (const entry of entries) {
   144→   144→      if (entry.source < count && !seen.has(entry.source)) {
   145→   145→        next.push(entry);
   146→   146→        seen.add(entry.source);
   147→   147→      }
   148→   148→    }
   149→   149→  } else {
   150→   150→    for (const entry of entries) {
   151→   151→      if (!seen.has(entry.source)) {
   152→   152→        next.push(entry);
   153→   153→        seen.add(entry.source);
   154→   154→      }
   155→   155→    }
   156→   156→  }
   157→   157→  for (let source = 0; source < count; source += 1) {
   158→   158→    if (!seen.has(source)) {
   159→   159→      next.push({
   160→   160→        uid: nextUid(),
   161→   161→        source,
   162→   162→        rotation: 0,
   163→   163→        caption: defaultCaptionFor(files[source].file.name),
   164→   164→      });
   165→   165→    }
   166→   166→  }
   167→   167→  return next;
   168→   168→}

   172→   172→/** Cheap TIFF check without reading bytes (type or extension). */
   173→   173→function looksTiff(file: File): boolean {
   174→   174→  return file.type === "image/tiff" || /\.(tif|tiff)$/i.test(file.name);
   175→   175→}

   177→   177→/** Full-resolution first page of a TIFF (clean-room reader, main thread). */
   178→   178→async function tiffBitmap(file: File): Promise<ImageBitmap | null> {
   179→   179→  try {
   180→   180→    const bytes = new Uint8Array(await file.arrayBuffer());
   181→   181→    const { TiffFile } = await import("../imgconv/tiff");
   182→   182→    const tiff = TiffFile.parse(bytes);
   183→   183→    if (tiff.pageCount === 0) return null;
   184→   184→    const page = tiff.decodePage(0);
   185→   185→    const canvas = document.createElement("canvas");
   186→   186→    canvas.width = page.width;
   187→   187→    canvas.height = page.height;
   188→   188→    const ctx = canvas.getContext("2d");
   189→   189→    if (!ctx) return null;
   190→   190→    ctx.putImageData(
   191→   191→      new ImageData(page.rgba, page.width, page.height) as unknown as ImageData,
   192→   192→      0,
   193→   193→      0
   194→   194→    );
   195→   195→    return await createImageBitmap(canvas);
   196→   196→  } catch {
   197→   197→    return null;
   198→   198→  }
   199→   199→}

   201→   201→/** PNG blob-URL card preview for TIFF first pages. */
   202→   202→async function tiffPreviewUrl(
   203→   203→  file: File
   204→   204→): Promise<{ url: string; width: number; height: number } | null> {
   205→   205→  const bitmap = await tiffBitmap(file);
   206→   206→  if (!bitmap) return null;
   207→   207→  try {
   208→   208→    const canvas = document.createElement("canvas");
   209→   209→    canvas.width = bitmap.width;
   210→   210→    canvas.height = bitmap.height;
   211→   211→    const ctx = canvas.getContext("2d");
   212→   212→    if (!ctx) return null;
   213→   213→    ctx.drawImage(bitmap, 0, 0);
   214→   214→    const blob = await new Promise<Blob | null>((resolve) =>
   215→   215→      canvas.toBlob(resolve, "image/png")
   216→   216→    );
   217→   217→    if (!blob) return null;
   218→   218→    return {
   219→   219→      url: URL.createObjectURL(blob),
   220→   220→      width: canvas.width,
   221→   221→      height: canvas.height,
   222→   222→    };
   223→   223→  } finally {
   224→   224→    bitmap.close();
   225→   225→  }
   226→   226→}

   228→   228→/** Small capped preview source for the live canvas (HEIC uses the queued thumbnail). */
   229→   229→async function decodePreviewSource(
   230→   230→  file: File,
   231→   231→  rotation: number
   232→   232→): Promise<CollageSource> {
   233→   233→  let bitmap: ImageBitmap | null = null;
   234→   234→  if (looksHeic(file)) {
   235→   235→    const preview = await resolveHeicPreview(file);
   236→   236→    if (preview) {
   237→   237→      try {
   238→   238→        bitmap = await createImageBitmap(preview.blob);
   239→   239→      } catch {
   240→   240→        bitmap = null;
   241→   241→      }
   242→   242→    }
   243→   243→  }
   244→   244→  if (bitmap === null && looksTiff(file)) {
   245→   245→    bitmap = await tiffBitmap(file);
   246→   246→  }
   247→   247→  if (bitmap === null) {
   248→   248→    bitmap = await createImageBitmap(file);
   249→   249→  }
   250→   250→  const oriented = orientedSource(bitmap, rotation, (w, h) => {
   251→   251→    const canvas = document.createElement("canvas");
   252→   252→    canvas.width = w;
   253→   253→    canvas.height = h;
   254→   254→    const ctx = canvas.getContext("2d");
   255→   255→    if (!ctx) throw new Error("canvas-2d-unavailable");
   256→   256→    return { canvas, ctx };
   257→   257→  });
   258→   258→  if (oriented !== bitmap) bitmap.close();
   259→   259→  return oriented;
   260→   260→}

   262→   262→/** Full-resolution source for the quick downloads (HEIC via libheif). */
   263→   263→async function decodeExportSource(
   264→   264→  file: File,
   265→   265→  rotation: number
   266→   266→): Promise<CollageSource> {
   267→   267→  let bitmap: ImageBitmap | null = null;
   268→   268→  try {
   269→   269→    bitmap = await createImageBitmap(file);
   270→   270→  } catch {
   271→   271→    bitmap = null;
   272→   272→  }
   273→   273→  if (bitmap === null && looksHeic(file)) {
   274→   274→    bitmap = await decodeHeicToBitmap(new Uint8Array(await file.arrayBuffer()));
   275→   275→  }
   276→   276→  if (bitmap === null && looksTiff(file)) {
   277→   277→    bitmap = await tiffBitmap(file);
   278→   278→  }
   279→   279→  if (bitmap === null) {
   280→   280→    throw new Error("decode-failed");
   281→   281→  }
   282→   282→  const oriented = orientedSource(bitmap, rotation, (w, h) => {
   283→   283→    const canvas = document.createElement("canvas");
   284→   284→    canvas.width = w;
   285→   285→    canvas.height = h;
   286→   286→    const ctx = canvas.getContext("2d");
   287→   287→    if (!ctx) throw new Error("canvas-2d-unavailable");
   288→   288→    return { canvas, ctx };
   289→   289→  });
   290→   290→  if (oriented !== bitmap) bitmap.close();
   291→   291→  return oriented;
   292→   292→}

   296→   296→export function CollageSurface({
   297→   297→  files,
   298→   298→  onChange,
   299→   299→  addFiles,
   300→   300→  setRunBlocked,
   301→   301→}: ToolSurfaceProps) {
   302→   302→  const { t, locale } = useI18n();
   303→   303→  const { toast } = useToast();
   304→   304→
   305→   305→  const [state, setState] = useState<SurfaceState>(() => ({
   306→   306→    // Cards start from the files already in the workspace (the surface
   307→   307→    // can mount with photos present — a chain handoff or a re-entry),
   308→   308→    // with captions defaulting to the file stems.
   309→   309→    entries: files.map((ref, source) => ({
   310→   310→      uid: nextUid(),
   311→   311→      source,
   312→   312→      rotation: 0,
   313→   313→      caption: defaultCaptionFor(ref.file.name),
   314→   314→    })),
   315→   315→    style: { ...DEFAULT_STYLE, captions: [] },
   316→   316→    exportFormat: "pdf",
   317→   317→  }));
   318→   318→  const [prevFileCount, setPrevFileCount] = useState(files.length);
   319→   319→  const [previewPage, setPreviewPage] = useState(0);
   320→   320→  const [quickBusy, setQuickBusy] = useState<"jpg" | "png" | null>(null);
   321→   321→
   322→   322→  const { entries, style, exportFormat } = state;
   323→   323→  const pages = useMemo(
   324→   324→    () => computePages(entries.length, style.layout),
   325→   325→    [entries.length, style.layout]
   326→   326→  );
   327→   327→  const page = Math.min(previewPage, Math.max(0, pages.length - 1));
   328→   328→
   329→   329→  const fileNameOf = useCallback(
   330→   330→    (source: number): string => files[source]?.file.name ?? "",
   331→   331→    [files]
   332→   332→  );
   333→   333→
   334→   334→  /* ---------------- option sync (event-context only) ---------------- */
   335→   335→
   336→   336→  const syncOption = useCallback(
   337→   337→    (next: SurfaceState): void => {
   338→   338→      onChange(
   339→   339→        "collage",
   340→   340→        serializeCollage({
   341→   341→          photos: next.entries.map(
   342→   342→            (entry): CollagePhoto => [entry.source, entry.rotation]
   343→   343→          ),
   344→   344→          style: {
   345→   345→            ...next.style,
   346→   346→            captions: next.entries.map((entry) => entry.caption),
   347→   347→          },
   348→   348→          exportFormat: next.exportFormat,
   349→   349→          locale,
   350→   350→        })
   351→   351→      );
   352→   352→    },
   353→   353→    [locale, onChange]
   354→   354→  );
   355→   355→
   356→   356→  const apply = useCallback(
   357→   357→    (partial: Partial<SurfaceState>): void => {
   358→   358→      const next = { ...state, ...partial };
   359→   359→      setState(next);
   360→   360→      syncOption(next);
   361→   361→      setRunBlocked(next.entries.length === 0 ? "collage.ctaNoPhotos" : null);
   362→   362→    },
   363→   363→    [state, setRunBlocked, syncOption]
   364→   364→  );
   365→   365→
   366→   366→  const patchStyle = useCallback(
   367→   367→    (partial: Partial<CollageStyle>): void => {
   368→   368→      apply({ style: { ...style, ...partial } });
   369→   369→    },
   370→   370→    [apply, style]
   371→   371→  );
   372→   372→
   373→   373→  // Mount sync: the serialized option always mirrors the freshly
   374→   374→  // initialized state (a remount from the result view resets edits — the
   375→   375→  // worker must never receive a stale recipe). Microtask, like the
   376→   376→  // reconcile path — never a synchronous write inside the effect body.
   377→   377→  const initialSync = useRef(false);
   378→   378→  useEffect(() => {
   379→   379→    if (initialSync.current) return;
   380→   380→    initialSync.current = true;
   381→   381→    void Promise.resolve().then(() => {
   382→   382→      syncOption(state);
   383→   383→      setRunBlocked(state.entries.length === 0 ? "collage.ctaNoPhotos" : null);
   384→   384→    });
   385→   385→    // Mount-only by design (state is the lazy initializer's value).
   386→   386→  }, []);
   387→   387→
   388→   388→  // Language switch re-sync: `locale` rides inside the serialized option
   389→   389→  // (it names the output file), so switching EN ↔ HI after the last edit
   390→   390→  // must re-serialize — otherwise a Hindi UI exports with the English name
   391→   391→  // (and vice versa). Microtask, like the paths above.
   392→   392→  const stateRef = useRef(state);
   393→   393→  stateRef.current = state;
   394→   394→  const localeSeen = useRef(locale);
   395→   395→  useEffect(() => {
   396→   396→    if (localeSeen.current === locale) return;
   397→   397→    localeSeen.current = locale;
   398→   398→    void Promise.resolve().then(() => {
   399→   399→      syncOption(stateRef.current);
   400→   400→    });
   401→   401→  }, [locale, syncOption]);
   402→   402→
   403→   403→  /* ---------------- reconcile with the shell file list ---------------- */
   404→   404→
   405→   405→  useEffect(() => {
   406→   406→    if (files.length === prevFileCount) return;
   407→   407→    let cancelled = false;
   408→   408→    void Promise.resolve().then(() => {
   409→   409→      if (cancelled) return;
   410→   410→      const next: SurfaceState = {
   411→   411→        ...state,
   412→   412→        entries: reconcileEntries(state.entries, files, prevFileCount),
   413→   413→      };
   414→   414→      setState(next);
   415→   415→      setPrevFileCount(files.length);
   416→   416→      syncOption(next);
   417→   417→      setRunBlocked(next.entries.length === 0 ? "collage.ctaNoPhotos" : null);
   418→   418→    });
   419→   419→    return () => {
   420→   420→      cancelled = true;
   421→   421→    };
   422→   422→  }, [files, prevFileCount, state, setRunBlocked, syncOption]);
   423→   423→
   424→   424→  // Card mutations
   425→   425→  const rotateAll = (delta: number) => {
   426→   426→    apply({
   427→   427→      entries: entries.map((entry) => ({
   428→   428→        ...entry,
   429→   429→        rotation: normalizeRotation(entry.rotation + delta),
   430→   430→      })),
   431→   431→    });
   432→   432→  };
   433→   433→
   434→   434→  const sortAz = () => {
   435→   435→    apply({
   436→   436→      entries: [...entries].sort((a, b) =>
   437→   437→        fileNameOf(a.source).localeCompare(fileNameOf(b.source), undefined, {
   438→   438→          numeric: true,
   439→   439→          sensitivity: "base",
   440→   440→        })
   441→   441→      ),
   442→   442→    });
   443→   443→  };
   444→   444→
   445→   445→  /* ---------------- drag reorder ---------------- */
   446→   446→
   447→   447→  const sensors = useSensors(
   448→   448→    useSensor(PointerSensor, { activationConstraint: { distance: 6 } }),
   449→   449→    useSensor(TouchSensor, { activationConstraint: { delay: 200, tolerance: 8 } }),
   450→   450→    useSensor(KeyboardSensor, { coordinateGetter: sortableKeyboardCoordinates })
   451→   451→  );
   452→   452→
   453→   453→  const handleDragEnd = (event: DragEndEvent) => {
   454→   454→    const { active, over } = event;
   455→   455→    if (!over || active.id === over.id) return;
   456→   456→    const from = entries.findIndex((entry) => entry.uid === active.id);
   457→   457→    const to = entries.findIndex((entry) => entry.uid === over.id);
   458→   458→    if (from < 0 || to < 0) return;
   459→   459→    const next = [...entries];
   460→   460→    const [moved] = next.splice(from, 1);
   461→   461→    next.splice(to, 0, moved);
   462→   462→    apply({ entries: next });
   463→   463→  };
   464→   464→
   465→   465→  /* ---------------- live preview (shared engine) ---------------- */
   466→   466→
   467→   467→  const wrapRef = useRef<HTMLDivElement | null>(null);
   468→   468→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
   469→   469→  const [previewW, setPreviewW] = useState(0);
   470→   470→  const bitmapCache = useRef(new Map<string, CollageSource>());
   471→   471→  const renderToken = useRef(0);
   472→   472→
   473→   473→  useEffect(() => {
   474→   474→    const element = wrapRef.current;
   475→   475→    if (!element) return;
   476→   476→    const observer = new ResizeObserver((observed) => {
   477→   477→      const width = observed[0]?.contentRect.width ?? 0;
   478→   478→      if (width > 0) setPreviewW(Math.round(Math.min(560, width)));
   479→   479→    });
   480→   480→    observer.observe(element);
   481→   481→    return () => observer.disconnect();
   482→   482→  }, []);
   483→   483→
   484→   484→  const styleAll = useMemo<CollageStyle>(
   485→   485→    () => ({ ...style, captions: entries.map((entry) => entry.caption) }),
   486→   486→    [style, entries]
   487→   487→  );
   488→   488→
   489→   489→  useEffect(() => {
   490→   490→    if (entries.length === 0 || previewW === 0 || pages.length === 0) return;
   491→   491→    const token = renderToken.current + 1;
   492→   492→    renderToken.current = token;
   493→   493→    const timer = setTimeout(() => {
   494→   494→      void (async () => {
   495→   495→        const canvas = canvasRef.current;
   496→   496→        if (!canvas || renderToken.current !== token) return;
   497→   497→        const plan = pages[page];
   498→   498→        if (!plan) return;
   499→   499→        const pt = pagePtSize(style);
   500→   500→        const dpr = Math.min(2, window.devicePixelRatio || 1);
   501→   501→        const width = Math.round(previewW * dpr);
   502→   502→        const height = Math.round((previewW * pt.h) / pt.w * dpr);
   503→   503→        canvas.width = width;

  1245→  1245→  onRotate: (delta: number) => void;
  1246→  1246→  onRemove: () => void;

  1249→  1249→}) {

  1322→  1322→    >
  1323→  1323→      <div
  1324→  1324→        className={cn(
  1325→  1325→          "group relative flex flex-col overflow-hidden rounded-card border border-border bg-card transition-all duration-200",
  1326→  1326→          "hover:-translate-y-0.5 hover:border-primary/30 hover:shadow-md focus-within:ring-[3px] focus-within:ring-ring/50",

  1331→  1331→        <div className="relative aspect-square w-full overflow-hidden bg-muted/40">

  1333→  1333→            <img
  1334→  1334→              src={url}
  1335→  1335→              alt={file?.name ?? ""}
  1336→  1336→              loading="lazy"
  1337→  1337→              decoding="async"
  1338→  1338→              className="size-full object-cover transition-transform duration-300 group-hover:[--s:1.04]"
  1339→  1339→              style={{
  1340→  1340→                transform: `rotate(${entry.rotation}deg) scale(var(--s,1))`,
  1341→  1341→              }}
  1342→  1342→            />
  1343→  1343→          ) : (
  1344→  1344→            <div
  1345→  1345→              aria-hidden="true"
  1346→  1346→              className="size-full animate-pulse bg-gradient-to-br from-muted to-muted/40"
  1347→  1347→            />

  1349→  1349→          <span

  1358→  1358→            </span>

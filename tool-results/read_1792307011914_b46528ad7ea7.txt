   374→interface ShimBitmap {

   381→function isShimBitmap(source: ShimBitmap | ShimCanvas): source is ShimBitmap {

   385→interface ShimState {
   386→  tx: number;
   387→  ty: number;
   388→  rotDeg: number;
   389→}
   390→
   391→function fontSizeOf(font: string): number {
   392→  const match = /(\d+(?:\.\d+)?)px/.exec(font);
   393→  return match ? Number(match[1]) : 16;
   394→}
   395→
   396→class ShimCtx {
   397→  canvas: ShimCanvas;
   398→  fillStyle = "#000000";
   399→  strokeStyle = "#000000";
   400→  lineWidth = 1;
   401→  font = "16px sans-serif";
   402→  textAlign = "start";
   403→  textBaseline = "alphabetic";
   404→  imageSmoothingEnabled = true;
   405→  imageSmoothingQuality: "high" | "low" | "medium" = "high";
   406→  private stack: ShimState[] = [];
   407→  private state: ShimState = { tx: 0, ty: 0, rotDeg: 0 };
   408→
   409→  constructor(canvas: ShimCanvas) {
   410→    this.canvas = canvas;
   411→  }
   412→
   413→  save(): void {
   414→    this.stack.push({ ...this.state });
   415→  }
   416→
   417→  restore(): void {
   418→    const prev = this.stack.pop();
   419→    if (prev) this.state = prev;
   420→  }
   421→
   422→  translate(x: number, y: number): void {
   423→    this.state.tx += x;
   424→    this.state.ty += y;
   425→  }
   426→
   427→  rotate(radians: number): void {
   428→    this.state.rotDeg = (this.state.rotDeg + Math.round((radians * 180) / Math.PI)) % 360;
   429→  }
   430→
   431→  beginPath(): void { /* recorded via ops; clips are approximated away */ }
   432→  moveTo(): void { /* path node */ }
   433→  lineTo(): void { /* path node */ }
   434→  arcTo(): void { /* path node */ }
   435→  closePath(): void { /* path node */ }
   436→  clip(): void { /* rounded clips are cosmetic — pixel fidelity is QA'd in-browser */ }
   437→  stroke(): void { /* hairlines are cosmetic */ }
   438→
   439→  fillRect(x: number, y: number, w: number, h: number): void {
   440→    this.canvas.ops.push({
   441→      kind: "rect",
   442→      x,
   443→      y,
   444→      w,
   445→      h,
   446→      fill: this.fillStyle,
   447→      ...this.state,
   448→    });
   449→  }
   450→
   451→  fillText(text: string, x: number, y: number): void {
   452→    this.canvas.ops.push({
   453→      kind: "text",
   454→      text,
   455→      x,
   456→      y,
   457→      font: this.font,
   458→      fill: this.fillStyle,
   459→      align: this.textAlign,
   460→      baseline: this.textBaseline,
   461→      ...this.state,
   462→    });
   463→  }
   464→
   465→  measureText(text: string): { width: number } {
   466→    return { width: [...text].length * fontSizeOf(this.font) * 0.55 };
   467→  }
   468→
   469→  drawImage(source: ShimBitmap | ShimCanvas, ...args: number[]): void {
   470→    this.canvas.ops.push({
   471→      kind: "image",
   472→      source,
   473→      args,
   474→      ...this.state,
   475→    });
   476→  }
   477→}

   479→class ShimCanvas {
   480→  width: number;
   481→  height: number;
   482→  ops: ShimOp[] = [];
   483→  private raw: Promise<Buffer> | null = null;
   484→
   485→  constructor(width: number, height: number) {
   486→    this.width = width;
   487→    this.height = height;
   488→  }
   489→
   490→  getContext(kind: string): ShimCtx | null {
   491→    return kind === "2d" ? new ShimCtx(this) : null;
   492→  }
   493→
   494→  /** Raw RGBA pixels of every recorded op, composited in order. */
   495→  _ensureRaw(): Promise<Buffer> {
   496→    if (!this.raw) this.raw = this._rasterize();
   497→    return this.raw;
   498→  }
   499→
   500→  private async _rasterize(): Promise<Buffer> {
   501→    // Background: the first full-canvas rect (if any), else transparent.
   502→    let background = { r: 0, g: 0, b: 0, alpha: 0 };
   503→    for (const op of this.ops) {
   504→      if (
   505→        op.kind === "rect" &&
   506→        op.x === 0 && op.y === 0 &&
   507→        Math.round(op.w) === this.width && Math.round(op.h) === this.height
   508→      ) {
   509→        background = parseColor(op.fill);
   510→        break;
   511→      }
   512→    }
   513→    const base = await sharp({
   514→      create: {
   515→        width: this.width,
   516→        height: this.height,
   517→        channels: 4,
   518→        background,
   519→      },
   520→    })
   521→      .raw()
   522→      .toBuffer();
   523→
   524→    const layers: sharp.OverlayOptions[] = [];
   525→    const place = async (op: RectOp | TextOp): Promise<void> => {
   526→      if (op.kind === "rect") {
   527→        const svg = Buffer.from(
   528→          `<svg width="${this.width}" height="${this.height}" xmlns="http://www.w3.org/2000/svg"><rect x="${roundSafe(op.x + op.tx)}" y="${roundSafe(op.y + op.ty)}" width="${Math.max(1, Math.round(op.w))}" height="${Math.max(1, Math.round(op.h))}" fill="${op.fill}"/></svg>`
   529→        );
   530→        layers.push({ input: await sharp(svg).png().toBuffer() });
   531→      } else {
   532→        const size = fontSizeOf(op.font);
   533→        const anchor =
   534→          op.align === "center" ? "middle" : op.align === "right" || op.align === "end" ? "end" : "start";
   535→        const dominant = op.baseline === "middle" ? "central" : "auto";
   536→        const escaped = op.text
   537→          .replace(/&/g, "&amp;")
   538→          .replace(/</g, "&lt;")
   539→          .replace(/>/g, "&gt;");
   540→        const family = "sans-serif";
   541→        const svg = Buffer.from(
   542→          `<svg width="${this.width}" height="${this.height}" xmlns="http://www.w3.org/2000/svg"><text x="${roundSafe(op.x + op.tx)}" y="${roundSafe(op.y + op.ty)}" font-family="${family}" font-size="${size}" fill="${op.fill}" text-anchor="${anchor}" dominant-baseline="${dominant}">${escaped}</text></svg>`
   543→        );
   544→        layers.push({ input: await sharp(svg).png().toBuffer() });
   545→      }
   546→    };
   547→
   548→    for (const op of this.ops) {
   549→      if (op.kind === "rect" || op.kind === "text") {
   550→        if (op.rotDeg !== 0) continue; // rotated text is not used by the engine
   551→        await place(op);
   552→        continue;
   553→      }
   554→      // Image op
   555→      const sourceRaw = await sourcePixels(op.source);
   556→      const srcW = op.source.width;
   557→      const srcH = op.source.height;
   558→      let layer: Buffer;
   559→      let left: number;
   560→      let top: number;
   561→      if (op.args.length >= 5) {
   562→        // 9-arg cover-crop form
   563→        const [sx, sy, sw, sh, dx, dy, dw, dh] = op.args;
   564→        const region = await sharp(sourceRaw, {
   565→          raw: { width: srcW, height: srcH, channels: 4 },
   566→        })
   567→          .extract({
   568→            left: Math.max(0, Math.round(sx)),
   569→            top: Math.max(0, Math.round(sy)),
   570→            width: Math.max(1, Math.min(Math.round(sw), srcW - Math.max(0, Math.round(sx)))),
   571→            height: Math.max(1, Math.min(Math.round(sh), srcH - Math.max(0, Math.round(sy)))),
   572→          })
   573→          .resize(Math.max(1, Math.round(dw)), Math.max(1, Math.round(dh)), { fit: "fill" })
   574→          .png()
   575→          .toBuffer();
   576→        layer = region;
   577→        left = Math.round(dx + op.tx);
   578→        top = Math.round(dy + op.ty);
   579→      } else if (op.args.length === 2) {
   580→        // 3-arg natural size (the rotation canvas path)
   581→        const [dx, dy] = op.args;
   582→        if (op.rotDeg !== 0) {
   583→          const rotated = await sharp(sourceRaw, {
   584→            raw: { width: srcW, height: srcH, channels: 4 },
   585→          })
   586→            .rotate(op.rotDeg)
   587→            .png()
   588→            .toBuffer();
   589→          layer = rotated;
   590→          // Rotated around the canvas centre → covers [0..W,0..H] exactly.
   591→          left = Math.round(dx + op.tx + srcW / 2 - (op.rotDeg % 180 === 90 ? srcH : srcW) / 2);
   592→          top = Math.round(dy + op.ty + srcH / 2 - (op.rotDeg % 180 === 90 ? srcW : srcH) / 2);
   593→        } else {
   594→          layer = await sharp(sourceRaw, {
   595→            raw: { width: srcW, height: srcH, channels: 4 },
   596→          })
   597→            .png()
   598→            .toBuffer();
   599→          left = Math.round(dx + op.tx);
   600→          top = Math.round(dy + op.ty);
   601→        }
   602→      } else {
   603→        continue;
   604→      }
   605→      layers.push({ input: layer, left, top });
   606→    }
   607→
   608→    if (layers.length === 0) return base;
   609→    return await sharp(base, {
   610→      raw: { width: this.width, height: this.height, channels: 4 },
   611→    })
   612→      .composite(layers)
   613→      .raw()
   614→      .toBuffer();
   615→  }
   616→
   617→  async convertToBlob(options: { type: string; quality?: number }): Promise<Blob> {
   618→    const raw = await this._ensureRaw();
   619→    const pipeline = sharp(raw, {
   620→      raw: { width: this.width, height: this.height, channels: 4 },
   621→    });
   622→    const bytes =
   623→      options.type === "image/png"
   624→        ? await pipeline.png().toBuffer()
   625→        : await pipeline.jpeg({ quality: Math.round((options.quality ?? 0.92) * 100) }).toBuffer();
   626→    return new Blob([new Uint8Array(bytes)], { type: options.type });
   627→  }
   628→}

   635→function parseColor(fill: string): { r: number; g: number; b: number; alpha: number } {
   636→  const hex = /^#([0-9a-fA-F]{6})$/.exec(fill.trim());
   637→  if (hex) {
   638→    return {
   639→      r: Number.parseInt(hex[1].slice(0, 2), 16),
   640→      g: Number.parseInt(hex[1].slice(2, 4), 16),
   641→      b: Number.parseInt(hex[1].slice(4, 6), 16),
   642→      alpha: 1,
   643→    };
   644→  }
   645→  const rgba = /rgba?\((\d+)[,\s]+(\d+)[,\s]+(\d+)(?:[,\s]+([\d.]+))?\)/.exec(fill);
   646→  if (rgba) {
   647→    return {
   648→      r: Number(rgba[1]),
   649→      g: Number(rgba[2]),
   650→      b: Number(rgba[3]),
   651→      alpha: rgba[4] === undefined ? 1 : Number(rgba[4]),
   652→    };
   653→  }
   654→  return { r: 0, g: 0, b: 0, alpha: 1 };
   655→}

   691→      height: info.height,
   692→      _raw: data,
   693→      close(): void {
   694→        /* no-op in the shim */
   695→      },
   696→    };

   732→interface RunOutcome {
   733→  result:
   734→    | (Posted & { type: "result"; detail?: Record<string, unknown> })
   735→    | null;
   736→  error: (Posted & { type: "error"; code?: string }) | null;
   737→  outputs: { name: string; mime: string; blob: Blob }[];
   738→  progress: number;
   739→}

   741→async function runWorker(
   742→  files: FakeFile[],
   743→  options: Record<string, string | number | boolean>
   744→): Promise<RunOutcome> {
   745→  inbox.length = 0;
   746→  workerGlobal.onmessage!({
   747→    data: { type: "run", runId: "smoke", files, options, assets: [] },
   748→  });
   749→  for (let i = 0; i < 600; i += 1) {
   750→    const terminal = inbox.find((m) => m.type === "result" || m.type === "error");
   751→    if (terminal) {
   752→      await new Promise((r) => setTimeout(r, 0));
   753→      const result =
   754→        (inbox.find((m) => m.type === "result") as RunOutcome["result"]) ?? null;
   755→      const error =
   756→        (inbox.find((m) => m.type === "error") as RunOutcome["error"]) ?? null;
   757→      const outputs = (result?.outputs as RunOutcome["outputs"]) ?? [];
   758→      return {
   759→        result,
   760→        error,
   761→        outputs,
   762→        progress: inbox.filter((m) => m.type === "progress").length,
   763→      };
   764→    }
   765→    await new Promise((r) => setTimeout(r, 2));
   766→  }
   767→  throw new Error("worker never posted a terminal message");
   768→}
   769→
   770→async function bytesOf(outcome: RunOutcome): Promise<Uint8Array> {
   771→  if (!outcome.result) throw new Error("no result message");
   772→  return new Uint8Array(await outcome.outputs[0].blob.arrayBuffer());
   773→}

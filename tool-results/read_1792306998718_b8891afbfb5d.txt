   122→let uidSeq = 0;
   123→function nextUid(): string {
   124→  uidSeq += 1;
   125→  return `pc-${Date.now().toString(36)}-${uidSeq}`;
   126→}
   127→
   128→function normalizeRotation(rotation: number): number {
   129→  return ((rotation % 360) + 360) % 360;
   130→}

   132→/** Keeps the edited list aligned with the live workspace files. */
   133→function reconcileEntries(
   134→  entries: CollageEntry[],
   135→  files: ToolSurfaceProps["files"],
   136→  previousCount: number
   137→): CollageEntry[] {
   138→  const count = files.length;
   139→  if (count === previousCount) return entries;
   140→  const next: CollageEntry[] = [];
   141→  const seen = new Set<number>();
   142→  if (count < previousCount) {
   143→    for (const entry of entries) {
   144→      if (entry.source < count && !seen.has(entry.source)) {
   145→        next.push(entry);
   146→        seen.add(entry.source);
   147→      }
   148→    }
   149→  } else {
   150→    for (const entry of entries) {
   151→      if (!seen.has(entry.source)) {
   152→        next.push(entry);
   153→        seen.add(entry.source);
   154→      }
   155→    }
   156→  }
   157→  for (let source = 0; source < count; source += 1) {
   158→    if (!seen.has(source)) {
   159→      next.push({
   160→        uid: nextUid(),
   161→        source,
   162→        rotation: 0,
   163→        caption: defaultCaptionFor(files[source].file.name),
   164→      });
   165→    }
   166→  }
   167→  return next;
   168→}

   172→/** Cheap TIFF check without reading bytes (type or extension). */
   173→function looksTiff(file: File): boolean {
   174→  return file.type === "image/tiff" || /\.(tif|tiff)$/i.test(file.name);
   175→}
   176→
   177→/** Full-resolution first page of a TIFF (clean-room reader, main thread). */
   178→async function tiffBitmap(file: File): Promise<ImageBitmap | null> {
   179→  try {
   180→    const bytes = new Uint8Array(await file.arrayBuffer());
   181→    const { TiffFile } = await import("../imgconv/tiff");
   182→    const tiff = TiffFile.parse(bytes);
   183→    if (tiff.pageCount === 0) return null;
   184→    const page = tiff.decodePage(0);
   185→    const canvas = document.createElement("canvas");
   186→    canvas.width = page.width;
   187→    canvas.height = page.height;
   188→    const ctx = canvas.getContext("2d");
   189→    if (!ctx) return null;
   190→    ctx.putImageData(
   191→      new ImageData(page.rgba, page.width, page.height) as unknown as ImageData,
   192→      0,
   193→      0
   194→    );
   195→    return await createImageBitmap(canvas);
   196→  } catch {
   197→    return null;
   198→  }
   199→}
   200→
   201→/** PNG blob-URL card preview for TIFF first pages. */
   202→async function tiffPreviewUrl(
   203→  file: File
   204→): Promise<{ url: string; width: number; height: number } | null> {
   205→  const bitmap = await tiffBitmap(file);
   206→  if (!bitmap) return null;
   207→  try {
   208→    const canvas = document.createElement("canvas");
   209→    canvas.width = bitmap.width;
   210→    canvas.height = bitmap.height;
   211→    const ctx = canvas.getContext("2d");
   212→    if (!ctx) return null;
   213→    ctx.drawImage(bitmap, 0, 0);
   214→    const blob = await new Promise<Blob | null>((resolve) =>
   215→      canvas.toBlob(resolve, "image/png")
   216→    );
   217→    if (!blob) return null;
   218→    return {
   219→      url: URL.createObjectURL(blob),
   220→      width: canvas.width,
   221→      height: canvas.height,
   222→    };
   223→  } finally {
   224→    bitmap.close();
   225→  }
   226→}

   228→/** Small capped preview source for the live canvas (HEIC uses the queued thumbnail). */
   229→async function decodePreviewSource(
   230→  file: File,
   231→  rotation: number
   232→): Promise<CollageSource> {
   233→  let bitmap: ImageBitmap | null = null;
   234→  if (looksHeic(file)) {
   235→    const preview = await resolveHeicPreview(file);
   236→    if (preview) {
   237→      try {
   238→        bitmap = await createImageBitmap(preview.blob);
   239→      } catch {
   240→        bitmap = null;
   241→      }
   242→    }
   243→  }
   244→  if (bitmap === null && looksTiff(file)) {
   245→    bitmap = await tiffBitmap(file);
   246→  }
   247→  if (bitmap === null) {
   248→    bitmap = await createImageBitmap(file);
   249→  }
   250→  const oriented = orientedSource(bitmap, rotation, (w, h) => {
   251→    const canvas = document.createElement("canvas");
   252→    canvas.width = w;
   253→    canvas.height = h;
   254→    const ctx = canvas.getContext("2d");
   255→    if (!ctx) throw new Error("canvas-2d-unavailable");
   256→    return { canvas, ctx };
   257→  });
   258→  if (oriented !== bitmap) bitmap.close();
   259→  return oriented;
   260→}
   261→
   262→/** Full-resolution source for the quick downloads (HEIC via libheif). */
   263→async function decodeExportSource(
   264→  file: File,
   265→  rotation: number
   266→): Promise<CollageSource> {
   267→  let bitmap: ImageBitmap | null = null;
   268→  try {
   269→    bitmap = await createImageBitmap(file);
   270→  } catch {
   271→    bitmap = null;
   272→  }
   273→  if (bitmap === null && looksHeic(file)) {
   274→    bitmap = await decodeHeicToBitmap(new Uint8Array(await file.arrayBuffer()));
   275→  }
   276→  if (bitmap === null && looksTiff(file)) {
   277→    bitmap = await tiffBitmap(file);
   278→  }
   279→  if (bitmap === null) {
   280→    throw new Error("decode-failed");
   281→  }
   282→  const oriented = orientedSource(bitmap, rotation, (w, h) => {
   283→    const canvas = document.createElement("canvas");
   284→    canvas.width = w;
   285→    canvas.height = h;
   286→    const ctx = canvas.getContext("2d");
   287→    if (!ctx) throw new Error("canvas-2d-unavailable");
   288→    return { canvas, ctx };
   289→  });
   290→  if (oriented !== bitmap) bitmap.close();
   291→  return oriented;
   292→}

   295→
   296→export function CollageSurface({
   297→  files,
   298→  onChange,
   299→  addFiles,
   300→  setRunBlocked,
   301→}: ToolSurfaceProps) {
   302→  const { t, locale } = useI18n();
   303→  const { toast } = useToast();
   304→
   305→  const [state, setState] = useState<SurfaceState>(() => ({
   306→    // Cards start from the files already in the workspace (the surface
   307→    // can mount with photos present — a chain handoff or a re-entry),
   308→    // with captions defaulting to the file stems.
   309→    entries: files.map((ref, source) => ({
   310→      uid: nextUid(),
   311→      source,
   312→      rotation: 0,
   313→      caption: defaultCaptionFor(ref.file.name),
   314→    })),
   315→    style: { ...DEFAULT_STYLE, captions: [] },
   316→    exportFormat: "pdf",
   317→  }));
   318→  const [prevFileCount, setPrevFileCount] = useState(files.length);
   319→  const [previewPage, setPreviewPage] = useState(0);
   320→  const [quickBusy, setQuickBusy] = useState<"jpg" | "png" | null>(null);
   321→
   322→  const { entries, style, exportFormat } = state;
   323→  const pages = useMemo(
   324→    () => computePages(entries.length, style.layout),
   325→    [entries.length, style.layout]
   326→  );
   327→  const page = Math.min(previewPage, Math.max(0, pages.length - 1));
   328→
   329→  const fileNameOf = useCallback(
   330→    (source: number): string => files[source]?.file.name ?? "",
   331→    [files]
   332→  );
   333→
   334→  /* ---------------- option sync (event-context only) ---------------- */
   335→
   336→  const syncOption = useCallback(
   337→    (next: SurfaceState): void => {
   338→      onChange(
   339→        "collage",
   340→        serializeCollage({
   341→          photos: next.entries.map(
   342→            (entry): CollagePhoto => [entry.source, entry.rotation]
   343→          ),
   344→          style: {
   345→            ...next.style,
   346→            captions: next.entries.map((entry) => entry.caption),
   347→          },
   348→          exportFormat: next.exportFormat,
   349→          locale,
   350→        })
   351→      );
   352→    },
   353→    [locale, onChange]
   354→  );
   355→
   356→  const apply = useCallback(
   357→    (partial: Partial<SurfaceState>): void => {
   358→      const next = { ...state, ...partial };
   359→      setState(next);
   360→      syncOption(next);
   361→      setRunBlocked(next.entries.length === 0 ? "collage.ctaNoPhotos" : null);
   362→    },
   363→    [state, setRunBlocked, syncOption]
   364→  );
   365→
   366→  const patchStyle = useCallback(
   367→    (partial: Partial<CollageStyle>): void => {
   368→      apply({ style: { ...style, ...partial } });
   369→    },
   370→    [apply, style]
   371→  );
   372→
   373→  // Mount sync: the serialized option always mirrors the freshly
   374→  // initialized state (a remount from the result view resets edits — the
   375→  // worker must never receive a stale recipe). Microtask, like the
   376→  // reconcile path — never a synchronous write inside the effect body.
   377→  const initialSync = useRef(false);
   378→  useEffect(() => {
   379→    if (initialSync.current) return;
   380→    initialSync.current = true;
   381→    void Promise.resolve().then(() => {
   382→      syncOption(state);
   383→      setRunBlocked(state.entries.length === 0 ? "collage.ctaNoPhotos" : null);
   384→    });
   385→    // Mount-only by design (state is the lazy initializer's value).
   386→  }, []);
   387→
   388→  // Language switch re-sync: `locale` rides inside the serialized option
   389→  // (it names the output file), so switching EN ↔ HI after the last edit
   390→  // must re-serialize — otherwise a Hindi UI exports with the English name
   391→  // (and vice versa). Microtask, like the paths above.
   392→  const stateRef = useRef(state);
   393→  stateRef.current = state;
   394→  const localeSeen = useRef(locale);
   395→  useEffect(() => {
   396→    if (localeSeen.current === locale) return;
   397→    localeSeen.current = locale;
   398→    void Promise.resolve().then(() => {
   399→      syncOption(stateRef.current);
   400→    });
   401→  }, [locale, syncOption]);
   402→
   403→  /* ---------------- reconcile with the shell file list ---------------- */
   404→
   405→  useEffect(() => {
   406→    if (files.length === prevFileCount) return;
   407→    let cancelled = false;
   408→    void Promise.resolve().then(() => {
   409→      if (cancelled) return;
   410→      const next: SurfaceState = {
   411→        ...state,
   412→        entries: reconcileEntries(state.entries, files, prevFileCount),
   413→      };
   414→      setState(next);
   415→      setPrevFileCount(files.length);
   416→      syncOption(next);
   417→      setRunBlocked(next.entries.length === 0 ? "collage.ctaNoPhotos" : null);
   418→    });
   419→    return () => {
   420→      cancelled = true;
   421→    };
   422→  }, [files, prevFileCount, state, setRunBlocked, syncOption]);
   423→
   424→  // Card mutations
   425→  const rotateAll = (delta: number) => {
   426→    apply({
   427→      entries: entries.map((entry) => ({
   428→        ...entry,
   429→        rotation: normalizeRotation(entry.rotation + delta),
   430→      })),
   431→    });
   432→  };
   433→
   434→  const sortAz = () => {
   435→    apply({
   436→      entries: [...entries].sort((a, b) =>
   437→        fileNameOf(a.source).localeCompare(fileNameOf(b.source), undefined, {
   438→          numeric: true,
   439→          sensitivity: "base",
   440→        })
   441→      ),
   442→    });
   443→  };
   444→
   445→  /* ---------------- drag reorder ---------------- */
   446→
   447→  const sensors = useSensors(
   448→    useSensor(PointerSensor, { activationConstraint: { distance: 6 } }),
   449→    useSensor(TouchSensor, { activationConstraint: { delay: 200, tolerance: 8 } }),
   450→    useSensor(KeyboardSensor, { coordinateGetter: sortableKeyboardCoordinates })
   451→  );
   452→
   453→  const handleDragEnd = (event: DragEndEvent) => {
   454→    const { active, over } = event;
   455→    if (!over || active.id === over.id) return;
   456→    const from = entries.findIndex((entry) => entry.uid === active.id);
   457→    const to = entries.findIndex((entry) => entry.uid === over.id);
   458→    if (from < 0 || to < 0) return;
   459→    const next = [...entries];
   460→    const [moved] = next.splice(from, 1);
   461→    next.splice(to, 0, moved);
   462→    apply({ entries: next });
   463→  };
   464→
   465→  /* ---------------- live preview (shared engine) ---------------- */
   466→
   467→  const wrapRef = useRef<HTMLDivElement | null>(null);
   468→  const canvasRef = useRef<HTMLCanvasElement | null>(null);
   469→  const [previewW, setPreviewW] = useState(0);
   470→  const bitmapCache = useRef(new Map<string, CollageSource>());
   471→  const renderToken = useRef(0);
   472→
   473→  useEffect(() => {
   474→    const element = wrapRef.current;
   475→    if (!element) return;
   476→    const observer = new ResizeObserver((observed) => {
   477→      const width = observed[0]?.contentRect.width ?? 0;
   478→      if (width > 0) setPreviewW(Math.round(Math.min(560, width)));
   479→    });
   480→    observer.observe(element);
   481→    return () => observer.disconnect();
   482→  }, []);
   483→
   484→  const styleAll = useMemo<CollageStyle>(
   485→    () => ({ ...style, captions: entries.map((entry) => entry.caption) }),
   486→    [style, entries]
   487→  );
   488→
   489→  useEffect(() => {
   490→    if (entries.length === 0 || previewW === 0 || pages.length === 0) return;
   491→    const token = renderToken.current + 1;
   492→    renderToken.current = token;
   493→    const timer = setTimeout(() => {
   494→      void (async () => {
   495→        const canvas = canvasRef.current;
   496→        if (!canvas || renderToken.current !== token) return;
   497→        const plan = pages[page];
   498→        if (!plan) return;
   499→        const pt = pagePtSize(style);
   500→        const dpr = Math.min(2, window.devicePixelRatio || 1);
   501→        const width = Math.round(previewW * dpr);
   502→        const height = Math.round((previewW * pt.h) / pt.w * dpr);
   503→        canvas.width = width;
   504→        canvas.height = height;
   505→        canvas.style.width = `${previewW}px`;
   506→        canvas.style.height = `${Math.round((previewW * pt.h) / pt.w)}px`;
   507→        const ctx = canvas.getContext("2d");
   508→        if (!ctx) return;
   509→        // Skeleton first (background + polaroid frames + captions), so
   510→        // the layout is visible while the photos decode.
   511→        drawCollagePage(ctx, plan, [], styleAll, page, pages.length);
   512→
   513→        const images: CollageSource[] = [];
   514→        const missing: Array<{ key: string; file: File; rotation: number; slot: number }> = [];
   515→        const needed = new Set<string>();
   516→        for (const cell of plan.cells) {
   517→          if (cell.photo === null) continue;
   518→          const entry = entries[cell.photo];
   519→          const ref = files[entry.source];
   520→          if (!ref) continue;
   521→          const key = `${ref.id}#${entry.rotation}`;
   522→          needed.add(key);
   523→          const cached = bitmapCache.current.get(key);
   524→          if (cached) images[cell.photo] = cached;
   525→          else missing.push({ key, file: ref.file, rotation: entry.rotation, slot: cell.photo });
   526→        }
   527→        // Keep the cache bounded to what the preview needs (+slack).
   528→        for (const key of [...bitmapCache.current.keys()]) {
   529→          if (bitmapCache.current.size <= 18) break;
   530→          if (!needed.has(key)) {
   531→            const dropped = bitmapCache.current.get(key);
   532→            bitmapCache.current.delete(key);
   533→            if (dropped && "close" in dropped && typeof dropped.close === "function") {
   534→              dropped.close();
   535→            }
   536→          }
   537→        }
   538→        // Decode the misses (an empty list resolves immediately), then
   539→        // ALWAYS repaint with whatever `images` holds — the skeleton draw
   540→        // above only ran with [] and a fully-cached warm pass must still
   541→        // paint the photos, not leave the frames empty.
   542→        if (missing.length > 0) {
   543→          await Promise.all(
   544→            missing.map(async (item) => {
   545→              try {
   546→                const source = await decodePreviewSource(item.file, item.rotation);
   547→                if (renderToken.current !== token) {
   548→                  if ("close" in source && typeof source.close === "function") source.close();
   549→                  return;
   550→                }
   551→                bitmapCache.current.set(item.key, source);
   552→                images[item.slot] = source;
   553→              } catch {
   554→                // The card keeps its placeholder; the run raises the
   555→                // honest decode error for this file.
   556→              }
   557→            })
   558→          );
   559→          if (renderToken.current !== token) return;
   560→        }
   561→        drawCollagePage(ctx, plan, images, styleAll, page, pages.length);
   562→      })().catch(() => undefined);
   563→    }, 140);
   564→    return () => clearTimeout(timer);
   565→  }, [entries, files, page, pages, previewW, styleAll]);
   566→
   567→  // Release cached bitmaps on unmount.
   568→  useEffect(() => {
   569→    const cache = bitmapCache.current;
   570→    return () => {
   571→      for (const source of cache.values()) {
   572→        if ("close" in source && typeof source.close === "function") source.close();
   573→      }
   574→      cache.clear();
   575→    };
   576→  }, []);
   577→
   578→  /* ---------------- quick downloads (page 1, main thread) ---------------- */
   579→
   580→  const downloadImage = useCallback(
   581→    async (format: "jpg" | "png"): Promise<void> => {
   582→      if (entries.length === 0 || pages.length === 0) return;
   583→      setQuickBusy(format);
   584→      try {
   585→        const plan = pages[0];
   586→        const images: CollageSource[] = [];
   587→        await Promise.all(
   588→          plan.cells
   589→            .filter((cell) => cell.photo !== null)
   590→            .map(async (cell) => {
   591→              const entry = entries[cell.photo as number];
   592→              const ref = files[entry.source];
   593→              if (!ref) return;
   594→              images[cell.photo as number] = await decodeExportSource(
   595→                ref.file,
   596→                entry.rotation
   597→              );
   598→            })
   599→        );
   600→        const px = pagePixelSize(style);
   601→        const canvas = document.createElement("canvas");
   602→        canvas.width = px.w;
   603→        canvas.height = px.h;
   604→        const ctx = canvas.getContext("2d");

  1121→          <div className="flex min-w-0 flex-1 flex-col gap-1.5">

  1134→                  <Loader2 className="size-4 animate-spin" aria-hidden="true" />
  1135→                ) : (
  1136→                  <ImageDown className="size-4" aria-hidden="true" />

  1138→                {t("collage.quickJpg")}
  1139→              </Button>

  1144→                disabled={entries.length === 0 || quickBusy !== null}
  1145→                onClick={() => void downloadImage("png")}
  1146→              >
  1147→                {quickBusy === "png" ? (
  1148→                  <Loader2 className="size-4 animate-spin" aria-hidden="true" />
  1149→                ) : (
  1150→                  <ImageDown className="size-4" aria-hidden="true" />
  1151→                )}
  1152→                {t("collage.quickPng")}
  1153→              </Button>

  1156→        </div>
  1157→      </div>
  1158→
  1159→      {/* Card grid */}
  1160→      <DndContext
  1161→        sensors={sensors}
  1162→        collisionDetection={closestCenter}
  1163→        modifiers={[restrictToParentElement]}
  1164→        onDragEnd={handleDragEnd}
  1165→      >
  1166→        <SortableContext
  1167→          items={entries.map((entry) => entry.uid)}
  1168→          strategy={rectSortingStrategy}
  1169→        >
  1170→          <ul className="grid list-none grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-4">
  1171→            {entries.map((entry, index) => (
  1172→              <CollageCard
  1173→                key={entry.uid}
  1174→                entry={entry}
  1175→                index={index}
  1176→                total={entries.length}
  1177→                file={files[entry.source]?.file ?? null}
  1178→                polaroid={style.polaroid}
  1179→                onRotate={(delta) =>
  1180→                  apply({
  1181→                    entries: entries.map((item, i) =>
  1182→                      i === index
  1183→                        ? { ...item, rotation: normalizeRotation(item.rotation + delta) }
  1184→                        : item
  1185→                    ),
  1186→                  })
  1187→                }
  1188→                onRemove={() =>
  1189→                  apply({ entries: entries.filter((_, i) => i !== index) })
  1190→                }
  1191→                onMove={(from, to) => {
  1192→                  if (from === to || from < 0 || to < 0 || from >= entries.length || to >= entries.length) return;
  1193→                  const next = [...entries];
  1194→                  const [moved] = next.splice(from, 1);
  1195→                  next.splice(to, 0, moved);
  1196→                  apply({ entries: next });
  1197→                }}
  1198→                onCaption={(caption) =>
  1199→                  apply({
  1200→                    entries: entries.map((item, i) =>
  1201→                      i === index ? { ...item, caption } : item
  1202→                    ),
  1203→                  })
  1204→                }
  1205→              />
  1206→            ))}
  1207→          </ul>
  1208→        </SortableContext>
  1209→      </DndContext>
  1210→
  1211→      {/* Add more (compact) — same shell pipeline as the landing drop */}
  1212→      <DropZone
  1213→        labelKey="collage.addMore"
  1214→        multiple
  1215→        maxFiles={50}
  1216→        acceptKind="image"
  1217→        onAccept={addFiles}
  1218→        currentCount={files.length}
  1219→        className="py-2"
  1220→      />
  1221→    </section>
  1222→  );
  1223→}

  1229→function CollageCard({
  1230→  entry,
  1231→  index,
  1232→  total,
  1233→  file,
  1234→  polaroid,
  1235→  onRotate,
  1236→  onRemove,
  1237→  onMove,
  1238→  onCaption,
  1239→}: {
  1240→  entry: CollageEntry;
  1241→  index: number;
  1242→  total: number;
  1243→  file: File | null;
  1244→  polaroid: boolean;
  1245→  onRotate: (delta: number) => void;
  1246→  onRemove: () => void;
  1247→  onMove: (from: number, to: number) => void;
  1248→  onCaption: (caption: string) => void;
  1249→}) {
  1250→  const { t } = useI18n();
  1251→  const {
  1252→    attributes,
  1253→    listeners,
  1254→    setNodeRef,
  1255→    transform,
  1256→    transition,
  1257→    isDragging,
  1258→    isOver,
  1259→  } = useSortable({ id: entry.uid });
  1260→
  1261→  const [dims, setDims] = useState<{ w: number; h: number } | null>(null);
  1262→  const [url, setUrl] = useState<string | null>(null);
  1263→
  1264→  // Card preview (imagegrid pattern): object URL for regular images, a
  1265→  // queued libheif preview for HEIC (cache-owned URL), a first-page PNG
  1266→  // for TIFF. Promise callbacks only — never sync setState in effects.
  1267→  useEffect(() => {
  1268→    if (!file) return;
  1269→    let cancelled = false;
  1270→    let objectUrl: string | null = null;
  1271→    void Promise.resolve().then(async () => {
  1272→      if (cancelled) return;
  1273→      if (looksHeic(file)) {
  1274→        const preview = await resolveHeicPreview(file);
  1275→        if (cancelled) return;
  1276→        if (preview) {
  1277→          setUrl(preview.url);
  1278→          setDims({ w: preview.width, h: preview.height });
  1279→          return;
  1280→        }
  1281→        // Byte-sniff says not HEIC after all — fall through to <img>.
  1282→      }
  1283→      if (looksTiff(file)) {
  1284→        const preview = await tiffPreviewUrl(file);
  1285→        if (cancelled) {
  1286→          if (preview) URL.revokeObjectURL(preview.url);
  1287→          return;
  1288→        }
  1289→        if (preview) {
  1290→          objectUrl = preview.url;
  1291→          setUrl(preview.url);
  1292→          setDims({ w: preview.width, h: preview.height });
  1293→          return;
  1294→        }
  1295→        // Unreadable TIFF — keep the pulse placeholder; the run gives
  1296→        // the honest error.
  1297→        return;
  1298→      }
  1299→      objectUrl = URL.createObjectURL(file);
  1300→      setUrl(objectUrl);
  1301→      const img = new Image();
  1302→      img.onload = () => {
  1303→        if (!cancelled) setDims({ w: img.naturalWidth, h: img.naturalHeight });
  1304→      };
  1305→      img.src = objectUrl;
  1306→    });
  1307→    return () => {
  1308→      cancelled = true;
  1309→      if (objectUrl) URL.revokeObjectURL(objectUrl);
  1310→    };
  1311→  }, [file]);
  1312→
  1313→  return (
  1314→    <li
  1315→      ref={setNodeRef}
  1316→      style={{ transform: CSS.Transform.toString(transform), transition }}
  1317→      className={cn(
  1318→        "relative list-none",

  1323→      <div

  1325→          "group relative flex flex-col overflow-hidden rounded-card border border-border bg-card transition-all duration-200",
  1326→          "hover:-translate-y-0.5 hover:border-primary/30 hover:shadow-md focus-within:ring-[3px] focus-within:ring-ring/50",

  1329→      >
  1330→        {/* Preview with LIVE CSS rotation */}
  1331→        <div className="relative aspect-square w-full overflow-hidden bg-muted/40">
  1332→          {url ? (
  1333→            <img
  1334→              src={url}
  1335→              alt={file?.name ?? ""}
  1336→              loading="lazy"
  1337→              decoding="async"
  1338→              className="size-full object-cover transition-transform duration-300 group-hover:[--s:1.04]"
  1339→              style={{

  1341→              }}
  1342→            />
  1343→          ) : (
  1344→            <div
  1345→              aria-hidden="true"
  1346→              className="size-full animate-pulse bg-gradient-to-br from-muted to-muted/40"

  1349→          <span

  1352→          >
  1353→            {index + 1}
  1354→          </span>
  1355→          {entry.rotation !== 0 ? (
  1356→            <span className="pointer-events-none absolute right-1.5 top-1.5 rounded-chip bg-amber-500/95 px-1.5 py-0.5 text-[10px] font-bold text-white shadow-sm">
  1357→              {entry.rotation}°

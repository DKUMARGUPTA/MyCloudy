     6→/**
     7→ * PDF → SVG model (RB4-a) — the operator-list → SVG translator.
     8→ *
     9→ * 100% pure TypeScript: no DOM, no pdf.js, no pdf-lib. It consumes the
    10→ * pdf.js operator list (fnArray/argsArray — the same shape
    11→ * tools/make-pdf-fillable/detect.ts pioneered) plus a viewport-derived
    12→ * base transform, and emits one standalone SVG per page with the viewBox
    13→ * in PDF points (y-down, exactly how the page renders).
    14→ *
    15→ * Coordinate recipe (all verified against pdfjs-dist 4.10.38 source):
    16→ *  - `baseTransform` = page.getViewport({scale: 1}).transform — maps PDF
    17→ *    user space (y-up) to the SVG viewBox (y-down) and folds /Rotate in.
    18→ *  - The device matrix for a shape is D = baseTransform × CTM; the CTM
    19→ *    starts as identity and is tracked through save/restore/transform
    20→ *    (+ paintFormXObjectBegin/End — the way pdf.js inlines Form XObjects).
    21→ *  - Text: the glyph matrix is CTM × Tm × translate(penX, penY + rise) ×
    22→ *    scale(hScale, -1) × diag(fontSize) — pdf.js's own composition —
    23→ *    premultiplied by D. The SVG font-size is the PDF font size; anything
    24→ *    left over (rotation, squeeze, mirrored text) rides a `transform`.
    25→ *  - Pen advance per glyph (pdf.js semantics): width × fontMatrix0 ×
    26→ *    fontSize (+ wordSpacing on spaces, + charSpacing), × hScale; TJ
    27→ *    numbers move the pen by -n × fontSize/1000.
    28→ *  - Images map the unit square: local box = CTM × [1,0,0,-1,0,1] — the
    29→ *    [0,1]×[0,1] box with row 0 at user y=1 that pdf.js draws
    30→ *    (canvas.js paintInlineImageXObject: scale(1/w, -1/h), draw at -h).
    31→ *  - Colors: the evaluator normalizes every color op to
    32→ *    setFillRGBColor/setStrokeRGBColor with 0-255 int triples.
    33→ *
    34→ * HONEST subset (documented in the tool FAQ): gradients/shadings and
    35→ * patterns are omitted (shadingFill skipped), clipping is approximated
    36→ * away (clipped paths still paint), Type3 glyph procedures are
    37→ * approximated by their font-matrix-scaled advances, stroked-text render
    38→ * modes render filled, image masks are skipped, fonts are substituted by
    39→ * family name — so text stays selectable but the page is not
    40→ * pixel-identical to the PDF.
    41→ */
    42→

   259→/** XML-escape for text and attribute content: & < > " '. */
   260→export function xmlEscape(text: string): string {

   263→    .replace(/</g, "&lt;")

   265→    .replace(/"/g, "&quot;")
   266→    .replace(/'/g, "&apos;");

   269→/** Strips characters that are illegal inside XML 1.0 text nodes. */
   270→export function sanitizeSvgText(text: string): string {
   271→  let out = "";
   272→  for (const ch of text) {
   273→    const code = ch.codePointAt(0) ?? 0;
   274→    if (code === 9 || code === 10 || code === 13) {
   275→      out += " ";
   276→    } else if (code < 32 || (code >= 127 && code <= 159)) {
   277→      continue;
   278→    } else {
   279→      out += ch;
   280→    }
   281→  }
   282→  return out;
   283→}

   292→/** Matrix product (PDF composition order: mul(m, n) = m × n). */
   293→export function mul(m: Mat, n: Mat): Mat {
   294→  return [
   295→    m[0] * n[0] + m[2] * n[1],
   296→    m[1] * n[0] + m[3] * n[1],
   297→    m[0] * n[2] + m[2] * n[3],
   298→    m[1] * n[2] + m[3] * n[3],
   299→    m[0] * n[4] + m[2] * n[5] + m[4],
   300→    m[1] * n[4] + m[3] * n[5] + m[5],
   301→  ];
   302→}
   303→
   304→/** Apply a matrix to a point. */
   305→function apply(m: Mat, x: number, y: number): [number, number] {
   306→  return [m[0] * x + m[2] * y + m[4], m[1] * x + m[3] * y + m[5]];
   307→}

   309→/** True when the matrix is axis-aligned (no rotation/skew). */
   310→function isAxisAligned(m: Mat): boolean {
   311→  return Math.abs(m[1]) < 1e-6 && Math.abs(m[2]) < 1e-6;
   312→}
   313→
   314→/** True when the matrix is identity within tolerance. */
   315→function isIdentityish(m: Mat): boolean {
   316→  return (
   317→    Math.abs(m[0] - 1) < 1e-3 &&
   318→    Math.abs(m[1]) < 1e-3 &&
   319→    Math.abs(m[2]) < 1e-3 &&
   320→    Math.abs(m[3] - 1) < 1e-3
   321→  );
   322→}

   324→function finiteMat(m: Mat): boolean {
   325→  return m.every((v) => Number.isFinite(v));

   328→function num(value: unknown, fallback = 0): number {
   329→  const n = typeof value === "number" ? value : Number(value);

   333→/**
   334→ * pdf.js color op args (setFillRGBColor/setStrokeRGBColor) arrive as a
   335→ * 3-element Uint8ClampedArray of 0-255 ints (evaluator getRgb output —
   336→ * verified against 4.10.38). Clamp + hex them; null when unusable.
   337→ */
   338→export function colorToHex(args: unknown): string | null {
   339→  if (args === null || typeof args !== "object") return null;
   340→  const parts = args as { length?: unknown } & Record<number, unknown>;
   341→  if (parts.length !== 3) return null;
   342→  const rgb: number[] = [];
   343→  for (let i = 0; i < 3; i += 1) {
   344→    const n = num(parts[i], -1);
   345→    if (n < 0) return null;
   346→    rgb.push(Math.min(255, Math.round(n)));
   347→  }
   348→  return `#${rgb.map((v) => v.toString(16).padStart(2, "0")).join("")}`;
   349→}

   351→/** Font-object → SvgFontInfo coercion (worker + smoke share it). */
   352→export function toFontInfo(raw: unknown): SvgFontInfo | null {
   353→  if (raw === null || typeof raw !== "object") return null;
   354→  const obj = raw as Record<string, unknown>;
   355→  const name = typeof obj.name === "string" ? obj.name.trim() : "";
   356→  const fallback =
   357→    typeof obj.fallbackName === "string" ? obj.fallbackName.trim() : "";
   358→  const useName = name.length > 0 ? name : fallback || "sans-serif";
   359→  const fontMatrix = obj.fontMatrix;
   360→  const fontMatrix0 =
   361→    Array.isArray(fontMatrix) &&
   362→    typeof fontMatrix[0] === "number" &&
   363→    fontMatrix[0] > 0
   364→      ? fontMatrix[0]
   365→      : 0.001;
   366→  const generic = ["serif", "sans-serif", "monospace"].includes(fallback)
   367→    ? fallback
   368→    : /courier|mono/i.test(useName)
   369→      ? "monospace"
   370→      : /times|serif|georgia|garamond|roman/i.test(useName)
   371→        ? "serif"
   372→        : "sans-serif";
   373→  return {
   374→    family: useName,
   375→    generic,
   376→    bold: obj.bold === true || BOLD_NAME_RE.test(useName),
   377→    italic: obj.italic === true || ITALIC_NAME_RE.test(useName),
   378→    fontMatrix0,
   379→  };
   380→}

   386→function bytesToBase64(bytes: Uint8Array | Uint8ClampedArray): string {
   387→  let binary = "";
   388→  const CHUNK = 0x8000;
   389→  for (let i = 0; i < bytes.length; i += CHUNK) {
   390→    binary += String.fromCharCode(
   391→      ...bytes.subarray(i, Math.min(i + CHUNK, bytes.length))
   392→    );
   393→  }
   394→  return btoa(binary);
   395→}

   397→/**
   398→ * Raw image payload (pdf.js ImageKind data) → PNG data URI. Reuses the
   399→ * extract-images RGB expansion + the imgconv clean-room PNG encoder, so
   400→ * the worker and the bun smoke produce byte-identical embeds.
   401→ */
   402→export async function imageToDataUri(
   403→  payload: SvgImagePayload
   404→): Promise<string | null> {
   405→  const { width, height, kind, data } = payload;
   406→  if (
   407→    !Number.isInteger(width) ||
   408→    !Number.isInteger(height) ||
   409→    width <= 0 ||
   410→    height <= 0 ||
   411→    width > MAX_IMAGE_SIDE ||
   412→    height > MAX_IMAGE_SIDE
   413→  ) {
   414→    return null;
   415→  }
   416→  const rgba = expandToRGBA(data, kind, width, height);
   417→  if (rgba === null) return null;
   418→  const blob = pngEncode({ width, height, data: rgba });
   419→  const png = new Uint8Array(await blob.arrayBuffer());
   420→  if (png[0] !== 0x89 || png[1] !== 0x50) return null;
   421→  return `data:image/png;base64,${bytesToBase64(png)}`;
   422→}

   488→/** Parses constructPath args [ops, coords, minMax] into a pending path. */
   489→export function parseConstructPath(
   490→  ops: SvgOpsTable,
   491→  args: readonly unknown[]
   492→): PendingPath | null {
   493→  const pathOps = args[0];
   494→  const coords = args[1];
   495→  if (!Array.isArray(pathOps) || !Array.isArray(coords)) return null;
   496→  let ci = 0;
   497→  let cx = 0;
   498→  let cy = 0;
   499→  let started = false;
   500→  const cmds: PathCmd[] = [];
   501→  const pureRects: { x: number; y: number; w: number; h: number }[] = [];
   502→  let onlyRects = true;
   503→
   504→  for (const op of pathOps) {
   505→    if (op === ops.moveTo) {
   506→      cx = num(coords[ci]);
   507→      cy = num(coords[ci + 1]);
   508→      ci += 2;
   509→      started = true;
   510→      onlyRects = false;
   511→      cmds.push({ cmd: "M", pts: [cx, cy] });
   512→    } else if (op === ops.lineTo) {
   513→      cx = num(coords[ci]);
   514→      cy = num(coords[ci + 1]);
   515→      ci += 2;
   516→      started = true;
   517→      onlyRects = false;
   518→      cmds.push({ cmd: "L", pts: [cx, cy] });
   519→    } else if (op === ops.curveTo) {
   520→      const pts = [
   521→        num(coords[ci]),
   522→        num(coords[ci + 1]),
   523→        num(coords[ci + 2]),
   524→        num(coords[ci + 3]),
   525→        num(coords[ci + 4]),
   526→        num(coords[ci + 5]),
   527→      ];
   528→      ci += 6;
   529→      cx = pts[4];
   530→      cy = pts[5];
   531→      started = true;
   532→      onlyRects = false;
   533→      cmds.push({ cmd: "C", pts });
   534→    } else if (op === ops.curveTo2) {
   535→      // x2 y2 x3 y3 — the current point doubles as the first control.
   536→      const x2 = num(coords[ci]);
   537→      const y2 = num(coords[ci + 1]);
   538→      const x3 = num(coords[ci + 2]);
   539→      const y3 = num(coords[ci + 3]);
   540→      ci += 4;
   541→      cmds.push({ cmd: "C", pts: [cx, cy, x2, y2, x3, y3] });
   542→      cx = x3;
   543→      cy = y3;
   544→      started = true;
   545→      onlyRects = false;
   546→    } else if (op === ops.curveTo3) {
   547→      // x1 y1 x2 y2 — the endpoint doubles as the second control.
   548→      const x1 = num(coords[ci]);
   549→      const y1 = num(coords[ci + 1]);
   550→      const x2 = num(coords[ci + 2]);
   551→      const y2 = num(coords[ci + 3]);
   552→      ci += 4;
   553→      cmds.push({ cmd: "C", pts: [x1, y1, x2, y2, x2, y2] });
   554→      cx = x2;
   555→      cy = y2;
   556→      started = true;
   557→      onlyRects = false;
   558→    } else if (op === ops.closePath) {
   559→      cmds.push({ cmd: "Z", pts: [] });
   560→      onlyRects = false;
   561→    } else if (op === ops.rectangle) {
   562→      const x = num(coords[ci]);
   563→      const y = num(coords[ci + 1]);
   564→      const w = num(coords[ci + 2]);
   565→      const h = num(coords[ci + 3]);
   566→      ci += 4;
   567→      pureRects.push({
   568→        x: Math.min(x, x + w),
   569→        y: Math.min(y, y + h),
   570→        w: Math.abs(w),
   571→        h: Math.abs(h),
   572→      });
   573→      // A rectangle subpath (counterclockwise, per the PDF `re` operator).
   574→      cmds.push({ cmd: "M", pts: [x, y] });
   575→      cmds.push({ cmd: "L", pts: [x + w, y] });
   576→      cmds.push({ cmd: "L", pts: [x + w, y + h] });
   577→      cmds.push({ cmd: "L", pts: [x, y + h] });
   578→      cmds.push({ cmd: "Z", pts: [] });
   579→      cx = x;
   580→      cy = y;
   581→      started = true;
   582→    } else {
   583→      return null; // unknown path op — drop the whole path honestly
   584→    }
   585→  }
   586→
   587→  if (!started && pureRects.length === 0) return null;
   588→  return {
   589→    cmds,
   590→    rect: pureRects.length === 1 && onlyRects ? pureRects[0] : null,
   591→  };
   592→}
   593→
   594→/** Serializes path commands; each point optionally transformed by `m`. */
   595→function pathData(cmds: readonly PathCmd[], m: Mat | null): string {
   596→  const pt = (x: number, y: number): string => {
   597→    const p = m === null ? ([x, y] as [number, number]) : apply(m, x, y);
   598→    return `${fmt(p[0])} ${fmt(p[1])}`;
   599→  };
   600→  let d = "";
   601→  for (const c of cmds) {
   602→    if (c.cmd === "Z") {
   603→      d += "Z";
   604→      continue;
   605→    }
   606→    const step = c.cmd === "C" ? 6 : 2;
   607→    const parts: string[] = [];
   608→    for (let i = 0; i + 1 < c.pts.length; i += step) {
   609→      if (c.cmd === "C") {
   610→        parts.push(
   611→          `${pt(c.pts[i], c.pts[i + 1])} ${pt(c.pts[i + 2], c.pts[i + 3])} ${pt(
   612→            c.pts[i + 4],
   613→            c.pts[i + 5]
   614→          )}`
   615→        );
   616→      } else {
   617→        parts.push(pt(c.pts[i], c.pts[i + 1]));
   618→      }
   619→    }
   620→    if (parts.length > 0) d += `${c.cmd}${parts.join(" ")}`;
   621→  }
   622→  return d;
   623→}

   647→/** glyph.unicode → display string ("" for skipped glyphs). */
   648→function glyphUnicode(glyph: GlyphObject): string {
   649→  if (typeof glyph.unicode !== "string") return "";
   650→  return sanitizeSvgText(glyph.unicode);

   653→/**
   654→ * Serializes one <text> element — one per showText run ("group runs into
   655→ * <text> elements"). Kerning numbers inside the glyph array split
   656→ * tspans so explicit x positions stay faithful.
   657→ */
   658→function textToSvg(el: TextElement): string {
   659→  const attrs: string[] = [
   660→    `font-family="${xmlEscape(`${el.family}, ${el.generic}`)}"`,
   661→    `font-size="${fmt(el.size)}"`,
   662→  ];
   663→  if (el.bold) attrs.push('font-weight="bold"');
   664→  if (el.italic) attrs.push('font-style="italic"');
   665→  attrs.push(`fill="${el.fill}"`);
   666→  attrs.push('xml:space="preserve"');
   667→  if (el.transform !== null) {
   668→    attrs.unshift(`transform="matrix(${el.transform.map((v) => fmt(v)).join(" ")})"`);
   669→  }
   670→  const live = el.pieces.filter((p) => p.text.length > 0);
   671→  if (live.length === 0) return "";
   672→  if (live.length === 1 && el.transform === null) {
   673→    return `<text x="${fmt(live[0].x)}" y="${fmt(live[0].y)}" ${attrs.join(" ")}>${xmlEscape(live[0].text)}</text>`;
   674→  }
   675→  const tspans = live
   676→    .map(
   677→      (p) =>
   678→        `<tspan x="${fmt(p.x)}" y="${fmt(p.y)}">${xmlEscape(p.text)}</tspan>`
   679→    )
   680→    .join("");
   681→  return `<text ${attrs.join(" ")}>${tspans}</text>`;
   682→}

   688→/**
   689→ * Stroke width in viewBox units for baked coordinates — lineWidth scaled
   690→ * by the average axis scale (exact when |a| = |d|; an approximation that
   691→ * matches how the canvas pen distorts under non-uniform scales).

   693→function strokeWidthFor(state: GState, device: Mat): number {
   694→  if (state.lineWidth <= 0) return 0;
   695→  return (state.lineWidth * (Math.abs(device[0]) + Math.abs(device[3]))) / 2;
   696→}

   706→/** Guards + runs the async font resolution for one showText op. */
   707→async function showTextRun(
   708→  state: GState,
   709→  ctx: SvgPageContext,
   710→  glyphs: unknown,
   711→  stats: SvgBuildStats,
   712→  out: string[]
   713→): Promise<void> {
   714→  if (!Array.isArray(glyphs)) return;
   715→  if (state.fontSize <= 0) return;
   716→  // Invisible (3) and clip-only (7) text render nothing.
   717→  if (state.renderMode === 3 || state.renderMode === 7) return;
   718→
   719→  const font = await ctx.resolveFont(state.fontRef);
   720→  if (font === null) return; // unknown font — skip the run honestly
   721→
   722→  // Glyph origin matrix (pdf.js composition):
   723→  // CTM × Tm × translate(penX, penY + rise) × scale(h, -1) × diag(fs).
   724→  const base = mul(
   725→    mul(state.ctm, state.tm),
   726→    [1, 0, 0, 1, state.penX, state.penY + state.rise]
   727→  );
   728→  const device = mul(
   729→    mul(ctx.baseTransform, base),
   730→    [state.fontSize * state.hScale, 0, 0, -state.fontSize, 0, 0]
   731→  );
   732→  if (!finiteMat(device)) return;
   733→
   734→  const size = state.fontSize;
   735→  // Residual transform: what is left after the SVG font-size carries the
   736→  // em scale (identity for ordinary horizontal text).
   737→  const residual: Mat = [
   738→    device[0] / size,
   739→    device[1] / size,
   740→    device[2] / size,
   741→    device[3] / size,
   742→    device[4],
   743→    device[5],
   744→  ];
   745→  const plain =
   746→    isAxisAligned(residual) &&
   747→    residual[0] > 0 &&
   748→    residual[3] > 0 &&
   749→    isIdentityish(residual);
   750→
   751→  const el: TextElement = {
   752→    family: font.family,
   753→    generic: font.generic,
   754→    bold: font.bold,
   755→    italic: font.italic,
   756→    size,
   757→    fill: state.fill,
   758→    pieces: [],
   759→    transform: plain ? null : residual,
   760→  };
   761→
   762→  const advance = (width: number, isSpace: boolean): number =>
   763→    (width * size * font.fontMatrix0 +
   764→      (isSpace ? state.wordSpacing : 0) +
   765→      state.charSpacing) *
   766→    state.hScale;
   767→
   768→  let piece: TextPiece | null = null;
   769→  for (const entry of glyphs as SvgGlyph[]) {
   770→    if (typeof entry === "number") {
   771→      // TJ adjustment: the pen moves by -n/1000 × fontSize (pdf.js keeps
   772→      // spacingDir at -1 for horizontal fonts). Breaks the piece.
   773→      piece = null;
   774→      state.penX += (-entry * size) / 1000;
   775→      continue;
   776→    }
   777→    if (entry === null || entry === undefined) continue;
   778→    const glyph = entry as GlyphObject;
   779→    const text = glyphUnicode(glyph);
   780→    const width = num(glyph.width, 0);
   781→
   782→    if (piece === null) {
   783→      const originDevice = mul(ctx.baseTransform, base);
   784→      piece = { x: originDevice[4], y: originDevice[5], text: "" };
   785→      el.pieces.push(piece);
   786→    }
   787→    piece.text += text;
   788→
   789→    state.penX += advance(width, glyph.isSpace === true);
   790→  }
   791→
   792→  const svg = textToSvg(el);
   793→  if (svg.length === 0) return;
   794→  if (stats.texts + stats.rects + stats.paths + stats.images >= MAX_ELEMENTS) {
   795→    stats.truncated = true;
   796→    return;
   797→  }
   798→  out.push(svg);
   799→  stats.texts += 1;
   800→}

   806→    `transform="matrix(${device.map((v) => fmt(v)).join(" ")})" ` +

   811→/** Emits a rect or path for the pending path + paint op combination. */
   812→function paintPath(
   813→  state: GState,
   814→  ctx: SvgPageContext,
   815→  pending: PendingPath | null,
   816→  mode: "fill" | "stroke" | "fillStroke",
   817→  evenOdd: boolean,
   818→  stats: SvgBuildStats,
   819→  out: string[]
   820→): void {
   821→  if (pending === null || pending.cmds.length === 0) return;
   822→  if (stats.texts + stats.rects + stats.paths + stats.images >= MAX_ELEMENTS) {
   823→    stats.truncated = true;
   824→    return;
   825→  }
   826→  const device = mul(ctx.baseTransform, state.ctm);
   827→  if (!finiteMat(device)) return;
   828→
   829→  const attrs: string[] = [];
   830→  if (mode !== "stroke") attrs.push(`fill="${state.fill}"`);
   831→  else attrs.push('fill="none"');
   832→  if (mode !== "fill") {
   833→    attrs.push(`stroke="${state.stroke}"`);
   834→    attrs.push(`stroke-width="${fmt(strokeWidthFor(state, device))}"`);
   835→    if (state.dash.length > 0) {
   836→      attrs.push(
   837→        `stroke-dasharray="${state.dash.map((v) => fmt(v)).join(" ")}"`
   838→      );
   839→    }
   840→  }
   841→  if (evenOdd && mode !== "stroke") attrs.push('fill-rule="evenodd"');
   842→
   843→  // Pure rectangle fast path — an exact baked <rect> when axis-aligned.
   844→  if (
   845→    pending.rect !== null &&
   846→    isAxisAligned(device) &&
   847→    device[0] !== 0 &&
   848→    device[3] !== 0
   849→  ) {
   850→    const r = pending.rect;
   851→    const corners: [number, number][] = [
   852→      [r.x, r.y],
   853→      [r.x + r.w, r.y],
   854→      [r.x, r.y + r.h],
   855→      [r.x + r.w, r.y + r.h],
   856→    ];
   857→    let minX = Infinity;
   858→    let minY = Infinity;
   859→    let maxX = -Infinity;
   860→    let maxY = -Infinity;
   861→    for (const [x, y] of corners) {
   862→      const [dx, dy] = apply(device, x, y);
   863→      if (dx < minX) minX = dx;
   864→      if (dy < minY) minY = dy;
   865→      if (dx > maxX) maxX = dx;
   866→      if (dy > maxY) maxY = dy;
   867→    }
   868→    if (![minX, minY, maxX, maxY].every((v) => Number.isFinite(v))) return;
   869→    out.push(
   870→      `<rect x="${fmt(minX)}" y="${fmt(minY)}" width="${fmt(
   871→        maxX - minX
   872→      )}" height="${fmt(maxY - minY)}" ${attrs.join(" ")}/>`
   873→    );
   874→    stats.rects += 1;
   875→    return;
   876→  }
   877→
   878→  // General path: bake coordinates when axis-aligned (the readable,
   879→  // smoke-checkable form), else carry the matrix and keep PDF-space data
   880→  // (stroke geometry then distorts exactly like the pdf.js canvas pen).
   881→  if (isAxisAligned(device) && device[0] > 0 && device[3] < 0) {
   882→    out.push(`<path d="${pathData(pending.cmds, device)}" ${attrs.join(" ")}/>`);
   883→    stats.paths += 1;
   884→    return;
   885→  }
   886→  out.push(
   887→    `<path transform="matrix(${device
   888→      .map((v) => fmt(v))
   889→      .join(" ")})" d="${pathData(pending.cmds, null)}" ${attrs.join(" ")}/>`
   890→  );
   891→  stats.paths += 1;
   892→}

   894→/** Inline image payloads live directly in the operator args. */
   895→function rawToPayload(raw: unknown): SvgImagePayload | null {
   896→  if (raw === null || typeof raw !== "object") return null;
   897→  const obj = raw as Record<string, unknown>;
   898→  const width = obj.width;
   899→  const height = obj.height;
   900→  if (typeof width !== "number" || typeof height !== "number") return null;
   901→  const data = obj.data;
   902→  if (!(data instanceof Uint8Array) && !(data instanceof Uint8ClampedArray)) {
   903→    return null;
   904→  }
   905→  return {
   906→    width,
   907→    height,
   908→    kind: typeof obj.kind === "number" ? obj.kind : 0,
   909→    data,
   910→  };
   911→}

   913→/** setGState entries: LW / D / Font (the rest is approximated or ignored). */
   914→function applyGState(state: GState, entries: unknown): void {
   915→  if (!Array.isArray(entries)) return;
   916→  for (const entry of entries) {
   917→    if (!Array.isArray(entry) || entry.length < 2) continue;
   918→    const key = entry[0];
   919→    const value = entry[1];
   920→    if (key === "LW") {
   921→      state.lineWidth = Math.max(0, num(value, 1));
   922→    } else if (key === "D" && Array.isArray(value)) {
   923→      const rawDash = Array.isArray(value[0]) ? value[0] : [];
   924→      state.dash = rawDash
   925→        .map((v) => num(v))
   926→        .filter((v) => Number.isFinite(v) && v >= 0);
   927→    } else if (key === "Font" && Array.isArray(value)) {
   928→      state.fontRef = value[0];
   929→      state.fontSize = num(value[1]);
   930→    }
   931→  }
   932→}

   934→/** Builds the full SVG for one page from its operator list. */
   935→export async function buildPageSvg(
   936→  ops: SvgOpsTable,
   937→  list: SvgOperatorList,
   938→  ctx: SvgPageContext
   939→): Promise<SvgBuildResult> {
   940→  const stats: SvgBuildStats = {
   941→    texts: 0,
   942→    rects: 0,
   943→    paths: 0,
   944→    images: 0,
   945→    truncated: false,
   946→  };
   947→  const out: string[] = [];
   948→
   949→  let state = freshState();
   950→  const stack: GState[] = [];
   951→  const formStack: Mat[] = [];
   952→  let pending: PendingPath | null = null;
   953→
   954→  const fnArray = list.fnArray;
   955→  const argsArray = list.argsArray;
   956→
   957→  for (let i = 0; i < fnArray.length; i += 1) {
   958→    const fn = fnArray[i];
   959→    const args = argsArray[i] ?? [];
   960→
   961→    if (fn === ops.save) {
   962→      stack.push(state);
   963→      state = {
   964→        ...state,
   965→        ctm: [...state.ctm] as Mat,
   966→        tm: [...state.tm] as Mat,
   967→        dash: [...state.dash],
   968→      };
   969→    } else if (fn === ops.restore) {
   970→      const restored = stack.pop();
   971→      if (restored !== undefined) state = restored;
   972→    } else if (fn === ops.transform) {
   973→      const t: Mat = [
   974→        num(args[0], 1),
   975→        num(args[1]),
   976→        num(args[2]),
   977→        num(args[3], 1),
   978→        num(args[4]),
   979→        num(args[5]),
   980→      ];
   981→      state.ctm = mul(state.ctm, t);
   982→    } else if (fn === ops.constructPath) {
   983→      pending = parseConstructPath(ops, args);
   984→    } else if (fn === ops.setFillRGBColor) {
   985→      const hex = colorToHex(args);
   986→      if (hex !== null) state.fill = hex;
   987→    } else if (fn === ops.setStrokeRGBColor) {
   988→      const hex = colorToHex(args);
   989→      if (hex !== null) state.stroke = hex;
   990→    } else if (fn === ops.setLineWidth) {
   991→      state.lineWidth = Math.max(0, num(args[0], 1));
   992→    } else if (fn === ops.setDash) {
   993→      const rawDash = Array.isArray(args[0]) ? args[0] : [];
   994→      state.dash = rawDash
   995→        .map((v) => num(v))
   996→        .filter((v) => Number.isFinite(v) && v >= 0);
   997→    } else if (fn === ops.setGState) {
   998→      applyGState(state, args[0]);
   999→    } else if (fn === ops.beginText) {
  1000→      state.tm = IDENTITY_MATRIX;
  1001→      state.penX = state.penY = state.lineX = state.lineY = 0;
  1002→    } else if (fn === ops.endText) {
  1003→      // Text state survives ET (matches pdf.js).
  1004→    } else if (fn === ops.setCharSpacing) {
  1005→      state.charSpacing = num(args[0]);
  1006→    } else if (fn === ops.setWordSpacing) {
  1007→      state.wordSpacing = num(args[0]);
  1008→    } else if (fn === ops.setHScale) {
  1009→      const scale = num(args[0], 100) / 100;
  1010→      state.hScale = scale > 0 && Number.isFinite(scale) ? scale : 1;
  1011→    } else if (fn === ops.setLeading) {
  1012→      state.leading = -num(args[0]);
  1013→    } else if (fn === ops.setFont) {
  1014→      state.fontRef = args[0];
  1015→      state.fontSize = num(args[1]);
  1016→    } else if (fn === ops.setTextRenderingMode) {
  1017→      state.renderMode = num(args[0]);
  1018→    } else if (fn === ops.setTextRise) {
  1019→      state.rise = num(args[0]);
  1020→    } else if (fn === ops.setTextMatrix) {
  1021→      state.tm = [
  1022→        num(args[0], 1),
  1023→        num(args[1]),
  1024→        num(args[2]),
  1025→        num(args[3], 1),
  1026→        num(args[4]),
  1027→        num(args[5]),
  1028→      ];
  1029→      state.penX = state.penY = state.lineX = state.lineY = 0;
  1030→    } else if (fn === ops.moveText) {
  1031→      state.penX = state.lineX += num(args[0]);
  1032→      state.penY = state.lineY += num(args[1]);
  1033→    } else if (fn === ops.setLeadingMoveText) {
  1034→      state.leading = -num(args[1]);
  1035→      state.penX = state.lineX += num(args[0]);
  1036→      state.penY = state.lineY += num(args[1]);
  1037→    } else if (fn === ops.nextLine) {
  1038→      state.penX = state.lineX;
  1039→      state.penY = state.lineY += state.leading;
  1040→    } else if (
  1041→      fn === ops.showText ||
  1042→      fn === ops.showSpacedText ||
  1043→      fn === ops.nextLineShowText
  1044→    ) {
  1045→      if (fn === ops.nextLineShowText) {
  1046→        state.penX = state.lineX;
  1047→        state.penY = state.lineY += state.leading;
  1048→      }
  1049→      await showTextRun(state, ctx, args[0], stats, out);
  1050→    } else if (fn === ops.nextLineSetSpacingShowText) {
  1051→      state.wordSpacing = num(args[0]);
  1052→      state.charSpacing = num(args[1]);
  1053→      state.penX = state.lineX;
  1054→      state.penY = state.lineY += state.leading;
  1055→      await showTextRun(state, ctx, args[2], stats, out);
  1056→    } else if (
  1057→      fn === ops.fill ||
  1058→      fn === ops.eoFill ||
  1059→      fn === ops.stroke ||
  1060→      fn === ops.closeStroke ||
  1061→      fn === ops.fillStroke ||
  1062→      fn === ops.eoFillStroke ||
  1063→      fn === ops.closeFillStroke ||
  1064→      fn === ops.closeEOFillStroke
  1065→    ) {
  1066→      const isStroke =
  1067→        fn === ops.stroke ||
  1068→        fn === ops.closeStroke ||
  1069→        fn === ops.fillStroke ||
  1070→        fn === ops.eoFillStroke ||
  1071→        fn === ops.closeFillStroke ||
  1072→        fn === ops.closeEOFillStroke;
  1073→      const isFill =
  1074→        fn === ops.fill ||
  1075→        fn === ops.eoFill ||
  1076→        fn === ops.fillStroke ||
  1077→        fn === ops.eoFillStroke ||
  1078→        fn === ops.closeFillStroke ||
  1079→        fn === ops.closeEOFillStroke;
  1080→      paintPath(
  1081→        state,
  1082→        ctx,
  1083→        pending,
  1084→        isFill && isStroke ? "fillStroke" : isStroke ? "stroke" : "fill",
  1085→        fn === ops.eoFill ||
  1086→          fn === ops.eoFillStroke ||
  1087→          fn === ops.closeEOFillStroke,
  1088→        stats,
  1089→        out
  1090→      );
  1091→      pending = null;
  1092→    } else if (fn === ops.endPath || fn === ops.clip || fn === ops.eoClip) {
  1093→      // endPath: clip-only or discarded — nothing paints (clipping is
  1094→      // approximated away). clip/eoClip keep the path for a FOLLOWING
  1095→      // paint op (PDF `W f` semantics), so only endPath consumes it.
  1096→      if (fn === ops.endPath) pending = null;
  1097→    } else if (fn === ops.paintFormXObjectBegin) {
  1098→      formStack.push(state.ctm);
  1099→      const matrix = args[0];
  1100→      if (Array.isArray(matrix) && matrix.length === 6) {
  1101→        state.ctm = mul(
  1102→          state.ctm,
  1103→          [
  1104→            num(matrix[0], 1),
  1105→            num(matrix[1]),
  1106→            num(matrix[2]),
  1107→            num(matrix[3], 1),
  1108→            num(matrix[4]),
  1109→            num(matrix[5]),
  1110→          ] as Mat
  1111→        );
  1112→      }
  1113→    } else if (fn === ops.paintFormXObjectEnd) {
  1114→      const restored = formStack.pop();
  1115→      if (restored !== undefined) state.ctm = restored;
  1116→    } else if (fn === ops.shadingFill) {
  1117→      // Gradient fills — honestly omitted (the FAQ documents this).
  1118→    } else if (fn === ops.paintImageXObject) {
  1119→      if (ctx.includeImages) {
  1120→        const payload = await ctx.resolveImage(args[0]);
  1121→        const dataUri =
  1122→          payload === null ? null : await imageToDataUri(payload);
  1123→        if (dataUri !== null) {
  1124→          if (stats.texts + stats.rects + stats.paths + stats.images < MAX_ELEMENTS) {
  1125→            out.push(
  1126→              emitImage(dataUri, mul(mul(ctx.baseTransform, state.ctm), IMAGE_FLIP))
  1127→            );
  1128→            stats.images += 1;
  1129→          } else {
  1130→            stats.truncated = true;
  1131→          }
  1132→        }
  1133→      }
  1134→    } else if (fn === ops.paintImageXObjectRepeat) {
  1135→      if (ctx.includeImages) {
  1136→        const payload = await ctx.resolveImage(args[0]);
  1137→        const dataUri =
  1138→          payload === null ? null : await imageToDataUri(payload);
  1139→        if (dataUri !== null) {
  1140→          const sx = num(args[1], 1);
  1141→          const sy = num(args[2], 1);
  1142→          const positions = Array.isArray(args[3]) ? args[3] : [];
  1143→          for (let p = 0; p + 1 < positions.length; p += 2) {
  1144→            if (
  1145→              stats.texts + stats.rects + stats.paths + stats.images >=
  1146→              MAX_ELEMENTS
  1147→            ) {
  1148→              stats.truncated = true;
  1149→              break;
  1150→            }
  1151→            const m = mul(
  1152→              mul(ctx.baseTransform, state.ctm),
  1153→              mul(
  1154→                [sx, 0, 0, sy, num(positions[p]), num(positions[p + 1])],
  1155→                IMAGE_FLIP
  1156→              )
  1157→            );
  1158→            out.push(emitImage(dataUri, m));
  1159→            stats.images += 1;
  1160→          }
  1161→        }
  1162→      }
  1163→    } else if (
  1164→      fn === ops.paintInlineImageXObject ||
  1165→      fn === ops.paintInlineImageXObjectGroup
  1166→    ) {
  1167→      if (ctx.includeImages) {
  1168→        const payload = rawToPayload(args[0]);
  1169→        const dataUri =
  1170→          payload === null ? null : await imageToDataUri(payload);
  1171→        if (dataUri !== null) {
  1172→          if (stats.texts + stats.rects + stats.paths + stats.images < MAX_ELEMENTS) {
  1173→            out.push(
  1174→              emitImage(
  1175→                dataUri,
  1176→                mul(mul(ctx.baseTransform, state.ctm), IMAGE_FLIP)
  1177→              )
  1178→            );
  1179→            stats.images += 1;
  1180→          } else {
  1181→            stats.truncated = true;
  1182→          }
  1183→        }
  1184→      }
  1185→    }
  1186→    // Everything else (dependency, marked content, masks, …) is skipped.
  1187→  }
  1188→
  1189→  const svg =
  1190→    `<?xml version="1.0" encoding="UTF-8"?>\n` +
  1191→    `<svg xmlns="http://www.w3.org/2000/svg" xmlns:xlink="http://www.w3.org/1999/xlink" ` +
  1192→    `width="${fmt(ctx.width)}pt" height="${fmt(ctx.height)}pt" ` +
  1193→    `viewBox="0 0 ${fmt(ctx.width)} ${fmt(ctx.height)}" version="1.1">\n` +
  1194→    `<rect x="0" y="0" width="${fmt(ctx.width)}" height="${fmt(
  1195→      ctx.height
  1196→    )}" fill="#ffffff"/>\n` +
  1197→    `${out.join("\n")}\n` +
  1198→    `</svg>\n`;
  1199→  return { svg, stats };
  1200→}

  1211→/** Coerces the raw options payload (pages spec + images switch). */
  1212→export function parseSvgOptions(
  1213→  options: OptionValues | Record<string, unknown>
  1214→): PdfToSvgRunOptions {
  1215→  const pages =
  1216→    typeof options.pages === "string" && options.pages.trim().length > 0
  1217→      ? options.pages
  1218→      : "all";
  1219→  return {
  1220→    pages,
  1221→    includeImages: options.includeImages !== false,
  1222→  };
  1223→}

  1225→/** Shared download-name sanitizer (pdf-to-images rule). */
  1226→export function sanitizeSvgBase(fileName: string): string {
  1227→  return (
  1228→    fileName
  1229→      .replace(/\.pdf$/i, "")
  1230→      .replace(/[\\/:*?"<>|\u0000-\u001f]/g, "_")
  1231→      .trim()
  1232→      .slice(0, 120) || "document"
  1233→  );
  1234→}

  1236→/**
  1237→ * One output name: `base.svg` when a single page is exported, else
  1238→ * `base_page<N>.svg` (the pdf-to-images multi-page convention).
  1239→ */
  1240→export function svgOutputName(
  1241→  base: string,
  1242→  pageNumber: number,
  1243→  single: boolean
  1244→): string {
  1245→  return single ? `${base}.svg` : `${base}_page${pageNumber}.svg`;
  1246→}

  1248→/**

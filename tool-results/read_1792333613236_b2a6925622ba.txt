   301→   301→  const base = String(s ?? "")
   302→   302→    .toLowerCase()
   303→   303→    .replace(/[^a-z0-9]+/g, "_")
   304→   304→    .replace(/^_+|_+$/g, "")
   305→   305→    .slice(0, 40);
   306→   306→  if (base === "") return "field";
   307→   307→  if (/^[0-9]/.test(base)) return "f_" + base;
   308→   308→  return base;
   309→   309→}

   311→   311→/** Dedupe a slug against taken names: "qty", "qty_2", "qty_3"… */
   312→   312→export function uniqueName(base: string, taken: Set<string>): string {
   313→   313→  const clean = base.slice(0, 40) || "field";
   314→   314→  if (!taken.has(clean)) {
   315→   315→    taken.add(clean);
   316→   316→    return clean;
   317→   317→  }
   318→   318→  for (let n = 2; n < 500; n += 1) {
   319→   319→    const candidate = `${clean}_${n}`;
   320→   320→    if (!taken.has(candidate)) {
   321→   321→      taken.add(candidate);
   322→   322→      return candidate;
   323→   323→    }
   324→   324→  }
   325→   325→  const stamped = `${clean}_${Date.now()}`;
   326→   326→  taken.add(stamped);
   327→   327→  return stamped;
   328→   328→}

   360→   360→  a: { x: number; y: number; w: number; h: number },

   362→   362→): boolean {

   368→   368→ * 8px steps (column-flow). Falls back to the margin origin when the page
   369→   369→ * is crowded — the user can drag from there.
   370→   370→ */
   371→   371→export function nextSlot(
   372→   372→  elements: FormElement[],
   373→   373→  w: number,
   374→   374→  h: number,
   375→   375→): { x: number; y: number } {
   376→   376→  const maxX = PAGE_W_PX - MARGIN_PX;
   377→   377→  const maxY = PAGE_H_PX - MARGIN_PX;
   378→   378→  for (let y = MARGIN_PX; ; y += GRID_PX) {
   379→   379→    const clampedY = Math.min(y, maxY);
   380→   380→    for (let x = MARGIN_PX; ; x += GRID_PX) {
   381→   381→      const clampedX = Math.min(x, maxX);
   382→   382→      const probe = {
   383→   383→        x: clampedX,
   384→   384→        y: clampedY,
   385→   385→        w: Math.min(w, maxX - clampedX),
   386→   386→        h,
   387→   387→      };
   388→   388→      if (!elements.some((el) => overlaps(probe, el))) {
   389→   389→        return { x: clampedX, y: clampedY };
   390→   390→      }
   391→   391→      if (x + GRID_PX > maxX) break;
   392→   392→    }
   393→   393→    if (y + GRID_PX > maxY) break;
   394→   394→  }
   395→   395→  return { x: MARGIN_PX, y: MARGIN_PX };
   396→   396→}

   480→   480→/* ------------------------------------------------------------------ */
   481→   481→/* Formula compiler — tokenizer                                         */
   482→   482→/* ------------------------------------------------------------------ */
   483→   483→
   484→   484→export type Tok =
   485→   485→  | { t: "num"; v: string }
   486→   486→  | { t: "id"; v: string }
   487→   487→  | { t: "op"; v: "+" | "-" | "*" | "/" | "(" | ")" };
   488→   488→
   489→   489→export interface TokenizeResult {
   490→   490→  toks: Tok[];
   491→   491→  error: boolean;
   492→   492→}
   493→   493→
   494→   494→export function tokenize(src: string): TokenizeResult {
   495→   495→  const toks: Tok[] = [];
   496→   496→  let i = 0;
   497→   497→  while (i < src.length) {
   498→   498→    const ch = src[i];
   499→   499→    if (/\s/.test(ch)) {
   500→   500→      i += 1;
   501→   501→      continue;
   502→   502→    }
   503→   503→    if (/[0-9]/.test(ch) || (ch === "." && /[0-9]/.test(src[i + 1] ?? ""))) {
   504→   504→      let j = i;
   505→   505→      while (j < src.length && /[0-9]/.test(src[j])) j += 1;
   506→   506→      if (src[j] === ".") {
   507→   507→        j += 1;
   508→   508→        while (j < src.length && /[0-9]/.test(src[j])) j += 1;
   509→   509→      }
   510→   510→      toks.push({ t: "num", v: src.slice(i, j) });
   511→   511→      i = j;
   512→   512→      continue;
   513→   513→    }
   514→   514→    if (/[A-Za-z_]/.test(ch)) {
   515→   515→      let j = i;
   516→   516→      while (j < src.length && /[A-Za-z0-9_]/.test(src[j])) j += 1;
   517→   517→      toks.push({ t: "id", v: src.slice(i, j) });
   518→   518→      i = j;
   519→   519→      continue;
   520→   520→    }
   521→   521→    if (
   522→   522→      ch === "+" ||
   523→   523→      ch === "-" ||
   524→   524→      ch === "*" ||
   525→   525→      ch === "/" ||
   526→   526→      ch === "(" ||
   527→   527→      ch === ")"
   528→   528→    ) {
   529→   529→      toks.push({ t: "op", v: ch });
   530→   530→      i += 1;
   531→   531→      continue;
   532→   532→    }
   533→   533→    return { toks, error: true };
   534→   534→  }
   535→   535→  return { toks, error: false };
   536→   536→}
   537→   537→
   538→   538→/* ------------------------------------------------------------------ */
   539→   539→/* Formula compiler — parser (recursive descent)                        */
   540→   540→/* ------------------------------------------------------------------ */
   541→   541→
   542→   542→/**
   543→   543→ * Grammar:
   544→   544→ *   expr   := term (("+" | "-") term)*
   545→   545→ *   term   := unary (("*" | "/") unary)*
   546→   546→ *   unary  := ("-" | "+") unary | primary
   547→   547→ *   primary:= NUM | ID | "(" expr ")"
   548→   548→ * Unary minus IS supported and survives into the emitted JS verbatim
   549→   549→ * (e.g. `-qty * 2` → `- g("qty") * 2`).
   550→   550→ */
   551→   551→export function parseTokens(toks: Tok[]): boolean {
   552→   552→  let pos = 0;
   553→   553→  const peek = (): Tok | null => toks[pos] ?? null;
   554→   554→  const eat = (): Tok | null => toks[pos++] ?? null;
   555→   555→
   556→   556→  const parsePrimary = (): boolean => {
   557→   557→    const t = peek();
   558→   558→    if (t === null) return false;
   559→   559→    if (t.t === "num" || t.t === "id") {
   560→   560→      eat();
   561→   561→      return true;
   562→   562→    }
   563→   563→    if (t.t === "op" && t.v === "(") {
   564→   564→      eat();
   565→   565→      if (!parseExpr()) return false;
   566→   566→      const close = peek();
   567→   567→      if (close === null || close.t !== "op" || close.v !== ")") return false;
   568→   568→      eat();
   569→   569→      return true;
   570→   570→    }
   571→   571→    return false;
   572→   572→  };
   573→   573→
   574→   574→  const parseUnary = (): boolean => {
   575→   575→    const t = peek();
   576→   576→    if (t !== null && t.t === "op" && (t.v === "-" || t.v === "+")) {
   577→   577→      eat();
   578→   578→      return parseUnary();
   579→   579→    }
   580→   580→    return parsePrimary();
   581→   581→  };
   582→   582→
   583→   583→  const parseTerm = (): boolean => {
   584→   584→    if (!parseUnary()) return false;
   585→   585→    for (;;) {
   586→   586→      const t = peek();
   587→   587→      if (t !== null && t.t === "op" && (t.v === "*" || t.v === "/")) {
   588→   588→        eat();
   589→   589→        if (!parseUnary()) return false;
   590→   590→      } else {
   591→   591→        return true;
   592→   592→      }
   593→   593→    }
   594→   594→  };
   595→   595→
   596→   596→  function parseExpr(): boolean {
   597→   597→    if (!parseTerm()) return false;
   598→   598→    for (;;) {
   599→   599→      const t = peek();
   600→   600→      if (t !== null && t.t === "op" && (t.v === "+" || t.v === "-")) {
   601→   601→        eat();
   602→   602→        if (!parseTerm()) return false;
   603→   603→      } else {
   604→   604→        return true;
   605→   605→      }
   606→   606→    }
   607→   607→  }
   608→   608→
   609→   609→  if (toks.length === 0) return false;
   610→   610→  if (!parseExpr()) return false;
   611→   611→  return pos === toks.length; // trailing operator = syntax error
   612→   612→}

   614→   614→/** Identifiers referenced by a token stream (order preserved, deduped). */
   615→   615→export function collectIds(toks: Tok[]): string[] {
   616→   616→  const out: string[] = [];
   617→   617→  for (const t of toks) if (t.t === "id" && !out.includes(t.v)) out.push(t.v);
   618→   618→  return out;
   619→   619→}
   620→   620→
   621→   621→/**
   622→   622→ * Emit JS from the ORIGINAL token stream: every identifier becomes
   623→   623→ * g("name"), everything else is verbatim — parens and precedence are
   624→   624→ * preserved by construction.
   625→   625→ */
   626→   626→export function emitJs(toks: Tok[]): string {
   627→   627→  return toks
   628→   628→    .map((t) => (t.t === "id" ? `g(${JSON.stringify(t.v)})` : t.v))
   629→   629→    .join(" ");
   630→   630→}

   633→   633→/* Formula compiler — the emitted action script                        */
   634→   634→/* ------------------------------------------------------------------ */
   635→   635→
   636→   636→/**
   637→   637→ * The per-target calculation script written into the field's /AA /C
   638→   638→ * action (ISO 32000-1 §12.6.4). `this` at the outer level is the Doc in
   639→   639→ * Acrobat's calculation context; the IIFE captures it once as `d` so the
   640→   640→ * nested `g` helper can call `d.getField(name)` for every reference —
   641→   641→ * missing fields read as 0 and non-numeric values read as 0, exactly like
   642→   642→ * the pinned template. (Capturing `d` instead of using `this` inside `g`
   643→   643→ * is the one deliberate deviation: a plainly-called `g` would otherwise

   647→   647→export const CALC_PREFIX =
   648→   648→  'event.value = (function(){var d=this;var g=function(n){var f=d.getField(n);if(!f){return 0;}var v=parseFloat(f.value);return isNaN(v)?0:v;};return (';
   649→   649→export const CALC_SUFFIX = ');}).call(this);';
   650→   650→

   652→   652→  return CALC_PREFIX + exprJs + CALC_SUFFIX;

   682→   682→/** Levenshtein ≤ 2 (case-insensitive) → the closest known name. */
   683→   683→export function nearMatch(name: string, known: readonly string[]): string {
   684→   684→  const lower = name.toLowerCase();
   685→   685→  let best = "";
   686→   686→  let bestDist = 3;
   687→   687→  for (const k of known) {
   688→   688→    const d = editDistance(lower, k.toLowerCase());
   689→   689→    if (d < bestDist) {
   690→   690→      bestDist = d;
   691→   691→      best = k;
   692→   692→    }
   693→   693→  }
   694→   694→  return best;
   695→   695→}
   696→   696→
   697→   697→function editDistance(a: string, b: string): number {
   698→   698→  const m = a.length;
   699→   699→  const n = b.length;
   700→   700→  if (Math.abs(m - n) > 2) return 9;
   701→   701→  const dp: number[][] = Array.from({ length: m + 1 }, () => Array(n + 1).fill(0));
   702→   702→  for (let i = 0; i <= m; i++) dp[i][0] = i;
   703→   703→  for (let j = 0; j <= n; j++) dp[0][j] = j;
   704→   704→  for (let i = 1; i <= m; i++) {
   705→   705→    for (let j = 1; j <= n; j++) {
   706→   706→      dp[i][j] = Math.min(
   707→   707→        dp[i - 1][j] + 1,
   708→   708→        dp[i][j - 1] + 1,
   709→   709→        dp[i - 1][j - 1] + (a[i - 1] === b[j - 1] ? 0 : 1),
   710→   710→      );
   711→   711→    }
   712→   712→  }
   713→   713→  return dp[m][n];
   714→   714→}

   716→   716→/** DFS cycle detection; returns the exact cycle path "a → b → a" or null. */
   717→   717→export function findCycle(edges: Map<string, string[]>): string[] | null {
   718→   718→  const state = new Map<string, 1 | 2>(); // 1 = on stack, 2 = done
   719→   719→  const stack: string[] = [];
   720→   720→  const visit = (node: string): string[] | null => {
   721→   721→    const st = state.get(node);
   722→   722→    if (st === 2) return null;
   723→   723→    if (st === 1) {
   724→   724→      const at = stack.indexOf(node);
   725→   725→      return [...stack.slice(at), node];
   726→   726→    }
   727→   727→    state.set(node, 1);
   728→   728→    stack.push(node);
   729→   729→    for (const next of edges.get(node) ?? []) {
   730→   730→      const cycle = visit(next);
   731→   731→      if (cycle) return cycle;
   732→   732→    }
   733→   733→    stack.pop();
   734→   734→    state.set(node, 2);
   735→   735→    return null;
   736→   736→  };
   737→   737→  for (const node of edges.keys()) {
   738→   738→    const cycle = visit(node);
   739→   739→    if (cycle) return cycle;
   740→   740→  }
   741→   741→  return null;
   742→   742→}

   744→   744→/**
   745→   745→ * Compile the rules list. Referencable names: number fields and other
   746→   746→ * rule targets. Output order is a stable topological sort (definition
   747→   747→ * order among ready nodes).
   748→   748→ */
   749→   749→export function compileRules(
   750→   750→  rules: CalcRule[],
   751→   751→  fields: readonly { name: string; kind: FieldKind }[],
   752→   752→): CalcCompile {
   753→   753→  const errors: CalcError[] = [];
   754→   754→  const known = new Set(fields.map((f) => f.name));
   755→   755→  const numeric = new Set(
   756→   756→    fields.filter((f) => f.kind === "number").map((f) => f.name),
   757→   757→  );
   758→   758→
   759→   759→  interface Parsed {
   760→   760→    idx: number;
   761→   761→    target: string;
   762→   762→    exprJs: string;
   763→   763→    ids: string[];
   764→   764→  }
   765→   765→  const parsed: Parsed[] = [];
   766→   766→
   767→   767→  rules.forEach((rule, idx) => {
   768→   768→    const target = String(rule.target ?? "").trim();
   769→   769→    if (target === "") {
   770→   770→      errors.push({ rule: idx, key: "form.calcErrNoTarget" });
   771→   771→      return;
   772→   772→    }
   773→   773→    if (!numeric.has(target)) {
   774→   774→      errors.push({
   775→   775→        rule: idx,
   776→   776→        key: known.has(target) ? "form.calcErrTargetKind" : "form.calcErrTarget",
   777→   777→        name: target,
   778→   778→      });
   779→   779→      return;
   780→   780→    }
   781→   781→    const { toks, error } = tokenize(String(rule.expr ?? ""));
   782→   782→    if (error) {
   783→   783→      errors.push({ rule: idx, key: "form.calcErrChar" });
   784→   784→      return;
   785→   785→    }

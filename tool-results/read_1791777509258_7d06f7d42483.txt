    99→    99→function initialsOf(name: string): string {
   100→   100→  const parts = name.trim().split(/\s+/).filter(Boolean);
   101→   101→  if (parts.length === 0) return "";
   102→   102→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase();
   103→   103→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
   104→   104→}

   376→   376→    if (!drag) return;

   378→   378→    const { orig, mode } = drag;

   380→   380→      case "move": {
   381→   381→        const cx = Math.min(Math.max(fx - drag.grabDx, 0), 1);
   382→   382→        const cy = Math.min(Math.max(fy - drag.grabDy, 0), 1);
   383→   383→        setDrag({ ...drag, orig: { ...orig, cx, cy } });
   384→   384→        break;
   385→   385→      }
   386→   386→      case "resize": {
   387→   387→        const leftEdge = orig.cx - orig.w / 2;
   388→   388→        const w = Math.min(Math.max((fx - leftEdge) / 1, 0.05), 1.2);
   389→   389→        setDrag({ ...drag, orig: { ...orig, w } });
   390→   390→        break;
   391→   391→      }
   392→   392→      case "rotate": {
   393→   393→        const dx = fx - orig.cx;
   394→   394→        const dy = fy - orig.cy;
   395→   395→        let deg = (Math.atan2(dy, dx) * 180) / Math.PI + 90;
   396→   396→        // Snap to 15° steps when close — signatures usually sit straight.
   397→   397→        const snapped = Math.round(deg / 15) * 15;
   398→   398→        if (Math.abs(deg - snapped) < 5) deg = snapped;
   399→   399→        deg = Math.round(deg * 10) / 10;
   400→   400→        setDrag({ ...drag, orig: { ...orig, rot: Math.min(Math.max(deg, -180), 180) } });

   402→   402→        break;
   403→   403→      }

   510→   510→  const aspectOf = (sigId: string): number => {

   516→   516→  const renderPlacementBox = (placement: SignPlacement): React.JSX.Element | null => {
   517→   517→    const sig = sigById.get(placement.sigId);
   518→   518→    if (!sig) return null;
   519→   519→    const live = drag !== null && drag.id === placement.id ? drag.orig : placement;
   520→   520→    const aspectW = aspectOf(placement.sigId); // w/h
   521→   521→    const hFrac =
   522→   522→      (live.w * (current ? current.dispW / current.dispH : 1)) / Math.max(aspectW, 0.05);
   523→   523→    const isSel = selectedId === placement.id;

   525→   525→      <div

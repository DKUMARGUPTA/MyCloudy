    79→/**
    80→ * ToolShell (PRD §6.3): the universal 4-state tool template.
    81→ * Landing (upload + SEO copy) → Workspace (preview + options) →
    82→ * Processing (real worker progress) → Result (download/share/chain).
    83→ * Every enabled tool renders from its registry entry + config — this
    84→ * component is the ONLY tool page in the app.
    85→ */
    86→
    87→type ShellPhase = "landing" | "workspace" | "processing" | "result";
    88→
    89→let fileSeq = 0;
    90→function nextFileId(): string {
    91→  fileSeq += 1;
    92→  return `wf-${Date.now().toString(36)}-${fileSeq}`;
    93→}
    94→
    95→export function ToolShell({ tool, config }: { tool: ToolDef; config: ToolConfig }) {
    96→  const { t, locale } = useI18n();
    97→  const { toast } = useToast();
    98→  const category = getCategory(tool.category);
    99→  // Viewer tools (PRD §6.5) never process anything: no run CTA, no options
   100→  // aside, no processing/result phases — the workspace surface IS the tool.
   101→  const viewerOnly = config.viewerOnly === true;
   102→
   103→  // One-time chain handoff peek — set by a previous tool's result view or
   104→  // the home Smart Drop Zone right before navigating here. Consumed via
   105→  // the lazy initializer (pure read); cleared in the effect below.
   106→  const [chainIn] = useState<{
   107→    fromToolId: string;
   108→    items: WorkspaceFile[];
   109→  } | null>(() => {
   110→    const handoff = peekChainHandoff();
   111→    if (!handoff || handoff.items.length === 0) return null;
   112→    return {
   113→      fromToolId: handoff.fromToolId,
   114→      items: handoff.items.map((item) => ({
   115→        id: item.id,
   116→        file: item.file,
   117→      })),
   118→    };
   119→  });
   120→
   121→  const [phase, setPhase] = useState<ShellPhase>(() => (chainIn ? "workspace" : "landing"));
   122→  const [files, setFiles] = useState<WorkspaceFile[]>(() => chainIn?.items ?? []);
   123→  const [options, setOptions] = useState<OptionValues>(() =>
   124→    defaultOptionValues(config.options)
   125→  );
   126→  const [progress, setProgress] = useState<RunProgress>({
   127→    pct: 0,
   128→    stage: "reading",
   129→  });
   130→  const [outputs, setOutputs] = useState<ToolOutput[]>([]);
   131→  const [inputBytes, setInputBytes] = useState(0);
   132→  const [runSummary, setRunSummary] = useState<RunDetail | null>(null);
   133→  const [optionsOpen, setOptionsOpen] = useState(false);
   134→  const [helpOpen, setHelpOpen] = useState(false);
   135→
   136→  // Chain handoff consumed once at mount (lazy initializer = pure read);
   137→  // clearing the module store is an external-system sync in this effect.
   138→  const chainFrom: ToolDef | null = chainIn
   139→    ? (getTool(chainIn.fromToolId) ?? null)
   140→    : null;
   141→  useEffect(() => {
   142→    if (chainIn) clearChainHandoff();
   143→  }, [chainIn]);
   144→
   145→  // Input kind drives the DropZone sniffing + accept filter + labels.
   146→  const acceptKind =
   147→    tool.accepts === "image" ? "image" : tool.accepts === "pdf" ? "pdf" : "any";
   148→  const selectLabelKey =
   149→    tool.accepts === "image" ? "tool.selectImageFiles" : "tool.selectPdfFiles";
   150→  const addMoreLabelKey =
   151→    tool.accepts === "image" ? "tool.addMoreImages" : "tool.addMoreFiles";
   152→  const recentKind = tool.accepts === "image" ? "image" : "pdf";
   153→
   154→  // Chain/SmartDrop files get the same treatment as dropped files (PRD
   155→  // §6.1): OPFS working copy + a metadata-only recent entry with thumbnail.
   156→  // External-system sync only — no React state is touched here.
   157→  useEffect(() => {
   158→    if (!chainIn) return;
   159→    for (const item of chainIn.items) {
   160→      void writeWorkingFile(item.id, item.file).catch(() => undefined);
   161→      void makeFileThumbnail(item.file, 160)
   162→        .then((blob) =>
   163→          addRecentFile({
   164→            id: item.id,
   165→            kind: recentKind as "pdf" | "image",
   166→            label: item.file.name,
   167→            size: item.file.size,
   168→            toolId: tool.id,
   169→            thumbnail: blob ?? undefined,
   170→          })
   171→        )
   172→        .catch(() => undefined);
   173→    }
   174→  }, [chainIn, recentKind, tool.id]);
   175→
   176→  const runHandleRef = useRef<RunHandle | null>(null);
   177→
   178→  // Optional custom options panel (PRD §2.6 Panel.tsx pattern). Direct
   179→  // map lookup — a stable module-level reference, never created in render.
   180→  const CustomPanel = TOOL_PANELS[tool.id];
   181→
   182→  // Optional workspace surface (PageGrid family): rendered in the main
   183→  // column instead of the FileList; can gate the CTA via runBlocked.
   184→  const WorkspaceSurface = TOOL_SURFACES[tool.id];
   185→
   186→  // Surface-driven CTA gating (PRD §8.1 safety: "sab pages delete ho to
   187→  // Apply disabled" / extract needs a selection). Event-context only —
   188→  // surfaces call this from their action handlers, never during render.
   189→  const [runBlockedKey, setRunBlockedKey] = useState<string | null>(null);
   190→  const handleRunBlocked = useCallback((reasonKey: string | null) => {
   191→    setRunBlockedKey(reasonKey);
   192→  }, []);
   193→  const runDisabled =
   194→    files.length < config.minFiles || runBlockedKey !== null;
   195→
   196→  const handleOptionChange = useCallback(
   197→    (id: string, value: string | number | boolean) => {
   198→      setOptions((prev) => ({ ...prev, [id]: value }));
   199→    },
   200→    []
   201→  );
   202→
   203→  const relatedTools = useMemo(
   204→    () =>
   205→      toolsByCategory(tool.category)
   206→        .filter((item) => item.id !== tool.id)
   207→        .slice(0, 6),
   208→    [tool.category, tool.id]
   209→  );
   210→
   211→  const chainTargets = useMemo(() => {
   212→    // PRD §6.4 "Ab aage kya?" — natural next steps. Image outputs chain
   213→    // to image-accepting tools (the PDF default set would be a type lie);
   214→    // configs can override with their own curated list.
   215→    const suggestionIds =
   216→      config.chainToolIds ?? [
   217→        "compress-pdf",
   218→        "protect-pdf",
   219→        "sign-pdf",
   220→        "split-pdf",
   221→        "edit-pdf",
   222→      ];
   223→    const resolved = suggestionIds
   224→      .map((id) => getTool(id))
   225→      .filter((item): item is ToolDef => item !== undefined && item.id !== tool.id);
   226→    return resolved.slice(0, 5);
   227→  }, [config.chainToolIds, tool.id]);
   228→
   229→  // Working copies are cleared when the tool session ends (PRD §2.4).
   230→  useEffect(
   231→    () => () => {
   232→      void clearWorkingFiles().catch(() => undefined);
   233→    },
   234→    []
   235→  );
   236→
   237→  const addFiles = useCallback(
   238→    (incoming: File[]) => {
   239→      if (incoming.length === 0) return;
   240→      const room = config.maxFiles - files.length;
   241→      if (room <= 0) {
   242→        toast({
   243→          title: t("dropzone.errTooManyTitle"),
   244→          description: t("dropzone.errTooManyDesc", {
   245→            count: config.maxFiles,
   246→          }),
   247→          variant: "destructive",
   248→        });
   249→        return;
   250→      }
   251→      const accepted = incoming.slice(0, room);
   252→      const items = accepted.map((file) => ({ id: nextFileId(), file }));
   253→
   254→      // Side effects run ONCE here — never inside the setFiles updater
   255→      // (React StrictMode double-invokes updaters; they must stay pure).
   256→      for (const item of items) {
   257→        void writeWorkingFile(item.id, item.file).catch(() => undefined);
   258→        // Recent entry: metadata + thumbnail only, never the bytes.
   259→        void makeFileThumbnail(item.file, 160)
   260→          .then(async (blob) => {
   261→            await addRecentFile({
   262→              id: item.id,
   263→              kind: recentKind as "pdf" | "image",
   264→              label: item.file.name,
   265→              size: item.file.size,
   266→              toolId: tool.id,
   267→              thumbnail: blob ?? undefined,
   268→            });
   269→          })
   270→          .catch(() => undefined);
   271→      }
   272→
   273→      setFiles((prev) => [...prev, ...items]);
   274→      setPhase("workspace");
   275→    },
   276→    [config.maxFiles, files.length, recentKind, t, toast, tool.id]
   277→  );
   278→
   279→  const removeFile = useCallback((id: string) => {
   280→    setFiles((prev) => prev.filter((item) => item.id !== id));
   281→  }, []);
   282→
   283→  const moveFile = useCallback((from: number, to: number) => {
   284→    setFiles((prev) => {
   285→      if (to < 0 || to >= prev.length) return prev;
   286→      const next = [...prev];
   287→      const [moved] = next.splice(from, 1);
   288→      next.splice(to, 0, moved);
   289→      return next;
   290→    });
   291→  }, []);
   292→
   293→  const clearAll = useCallback(() => {
   294→    setFiles([]);
   295→    void clearWorkingFiles().catch(() => undefined);
   296→    setPhase("landing");
   297→  }, []);
   298→
   299→  // Stable function ref from the config (module constant) — keeps the
   300→  // React Compiler's dependency inference happy.
   301→  const collectAssets = config.collectAssets;
   302→
   303→  const startRun = useCallback(() => {
   304→    if (files.length < config.minFiles || runBlockedKey !== null) return;
   305→    trackToolUse(tool.id);
   306→    setInputBytes(files.reduce((sum, item) => sum + item.file.size, 0));
   307→    setProgress({ pct: 0, stage: "reading" });
   308→    setPhase("processing");
   309→    // Extra in-memory assets (watermark logo etc.) ride the worker message.
   310→    const assets = collectAssets?.() ?? undefined;
   311→    runHandleRef.current = toolRunner.run(
   312→      tool.id,
   313→      options,
   314→      files.map((item) => item.file),
   315→      setProgress,
   316→      files.map((item) => item.id),
   317→      assets
   318→    );
   319→    runHandleRef.current.promise
   320→      .then((result) => {
   321→        runHandleRef.current = null;
   322→        // Working copies of the INPUTS die with the run (PRD §2.4);
   323→        // results live in memory + can chain into the next tool.
   324→        setOutputs(result.outputs);
   325→        setRunSummary(result.detail ?? null);
   326→        setPhase("result");
   327→        void clearWorkingFiles().catch(() => undefined);
   328→      })
   329→      .catch((error: unknown) => {
   330→        runHandleRef.current = null;
   331→        if (error instanceof RunCancelledError) {
   332→          setPhase("workspace");
   333→          toast({ title: t("tool.cancelledToast") });
   334→          return;
   335→        }
   336→        setPhase("workspace");
   337→        const code =
   338→          error instanceof ToolRunError ? error.code : "tool.errorGeneric";
   339→        toast({
   340→          title: t("tool.errorTitle"),
   341→          description: t(code, {
   342→            detail: error instanceof ToolRunError ? error.detail ?? "" : "",
   343→          }),
   344→          variant: "destructive",
   345→        });
   346→      });
   347→  }, [collectAssets, config.minFiles, files, options, runBlockedKey, t, toast, tool.id]);
   348→
   349→  const cancelRun = useCallback(() => {
   350→    runHandleRef.current?.cancel();
   351→  }, []);
   352→
   353→  const handleChain = useCallback(
   354→    (target: ToolDef, resultFiles: File[]) => {
   355→      if (!target.enabled) {
   356→        toast({
   357→          title: t("tools.comingToastTitle", { tool: target.name[locale] }),
   358→          description: t("tools.comingToastDesc", {
   359→            phase: target.phase,
   360→          }),
   361→        });
   362→        return;
   363→      }
   364→      setChainHandoff({
   365→        fromToolId: tool.id,
   366→        items: resultFiles.map((file) => ({ id: nextFileId(), file })),
   367→      });
   368→      navigateToHash(ROUTES.tool(target.id));
   369→    },
   370→    [locale, t, toast, tool.id]
   371→  );
   372→
   373→  // Viewer surfaces chain the current file straight into another tool
   374→  // (they have no result phase of their own). Same enabled/coming-soon
   375→  // behavior as result-view chains.
   376→  const surfaceChain = useCallback(
   377→    (targetId: string) => {
   378→      const target = getTool(targetId);
   379→      if (!target) return;
   380→      handleChain(target, files.map((item) => item.file));
   381→    },
   382→    [files, handleChain]
   383→  );
   384→
   385→  const removeFromDevice = useCallback(() => {
   386→    void clearWorkingFiles().catch(() => undefined);
   387→    // Remove this run's recent entries by their file metadata (label+size).
   388→    void removeRecentFilesByMeta(
   389→      files.map((item) => ({ label: item.file.name, size: item.file.size }))
   390→    ).catch(() => undefined);
   391→    setOutputs([]);
   392→    setFiles([]);
   393→    setPhase("landing");
   394→    toast({ title: t("result.removedToast") });
   395→  }, [files, t, toast]);
   396→
   397→  /**
   398→   * "Start over with a new file" (result view): a clean slate — outputs,
   399→   * workspace files AND options (per-file marks, placements, per-file
   400→   * ranges) all go, so nothing from the previous document can leak into
   401→   * the next one. Stays on this tool's page.
   402→   */
   403→  const startOver = useCallback(() => {
   404→    setOutputs([]);
   405→    setRunSummary(null);
   406→    setFiles([]);
   407→    setOptions(defaultOptionValues(config.options));
   408→    setRunBlockedKey(null);
   409→    void clearWorkingFiles().catch(() => undefined);
   410→    setPhase("landing");
   411→  }, [config.options]);
   412→
   413→  const totalSize = files.reduce((sum, item) => sum + item.file.size, 0);
   414→  const ctaLabel = `${tool.name[locale]} ${t("tool.ctaArrow")}`;
   415→  const Icon = tool.icon;
   416→
   417→  return (
   418→    <div className="mx-auto w-full max-w-6xl px-4 pb-28 sm:pb-24">
   419→      {/* Breadcrumb (PRD §6.3): Home > Category > Tool */}
   420→      <nav aria-label={t("a11y.breadcrumb")} className="pt-4">
   421→        <ol className="flex flex-wrap items-center gap-1.5 text-xs text-muted-foreground">
   422→          <li>
   423→            <a
   424→              href={ROUTES.home}
   425→              className="inline-flex items-center gap-1 rounded-btn outline-none hover:text-foreground focus-visible:ring-[3px] focus-visible:ring-ring/50"
   426→            >
   427→              <Home className="size-3" aria-hidden="true" />
   428→              {t("nav.home")}
   429→            </a>
   430→          </li>
   431→          <li aria-hidden="true">/</li>
   432→          <li>
   433→            <a
   434→              href={ROUTES.category(tool.category)}
   435→              className="rounded-btn outline-none hover:text-foreground focus-visible:ring-[3px] focus-visible:ring-ring/50"
   436→            >
   437→              {category.name[locale]}
   438→            </a>
   439→          </li>
   440→          <li aria-hidden="true">/</li>
   441→          <li aria-current="page" className="font-medium text-foreground">
   442→            {tool.name[locale]}
   443→          </li>
   444→        </ol>
   445→      </nav>
   446→
   447→      {phase === "landing" ? (
   448→        <section className="flex flex-col gap-8 pt-6">
   449→          <header className="flex flex-col gap-3">
   450→            <div className="flex items-center gap-3">
   451→              <span
   452→                className={cn(
   453→                  "flex size-11 items-center justify-center rounded-btn",
   454→                  "bg-cat-organize/10 text-cat-organize"
   455→                )}
   456→                aria-hidden="true"
   457→              >
   458→                <Icon className="size-5" />
   459→              </span>
   460→              <h1 className="text-2xl font-bold tracking-tight text-foreground sm:text-3xl">
   461→                {tool.name[locale]}
   462→              </h1>
   463→            </div>
   464→            <p className="max-w-2xl text-sm text-muted-foreground sm:text-base">
   465→              {tool.description[locale]}
   466→            </p>
   467→            <div className="flex flex-wrap gap-2">
   468→              <Badge variant={tool.mode === "D" ? "private" : "server"} className="rounded-chip">
   469→                <Lock className="size-3" aria-hidden="true" />
   470→                {tool.mode === "D"
   471→                  ? t("tools.privateBadge")
   472→                  : t("tools.serverBadge")}
   473→              </Badge>
   474→              <Badge variant="outline" className="rounded-chip">
   475→                {t("tool.freeBadge")}
   476→              </Badge>
   477→            </div>
   478→          </header>
   479→
   480→          <DropZone
   481→            labelKey={selectLabelKey}
   482→            multiple={config.maxFiles > 1}
   483→            maxFiles={config.maxFiles}
   484→            acceptKind={acceptKind}
   485→            onAccept={addFiles}
   486→            currentCount={files.length}
   487→          />
   488→
   489→          <div className="grid gap-8 lg:grid-cols-[2fr_1fr]">
   490→            <div className="flex flex-col gap-8">
   491→              {/* How it works */}
   492→              <section aria-labelledby="how-heading">
   493→                <h2 id="how-heading" className="flex items-center gap-2 text-sm font-semibold text-foreground">
   494→                  <ListChecks className="size-4 text-primary" aria-hidden="true" />
   495→                  {t("tool.howTitle")}
   496→                </h2>
   497→                <ol className="mt-3 grid gap-3 sm:grid-cols-3">
   498→                  {(viewerOnly
   499→                    ? ["view.how1", "view.how2", "view.how3"]
   500→                    : ["tool.how1", "tool.how2", "tool.how3"]
   501→                  ).map((key, index) => (
   502→                    <li
   503→                      key={key}
   504→                      className="rounded-card border border-border bg-card p-4"
   505→                    >
   506→                      <span className="flex size-7 items-center justify-center rounded-full bg-primary/10 text-xs font-bold text-primary">
   507→                        {index + 1}
   508→                      </span>
   509→                      <p className="mt-2 text-xs leading-relaxed text-muted-foreground">
   510→                        {t(key)}
   511→                      </p>
   512→                    </li>
   513→                  ))}
   514→                </ol>
   515→              </section>
   516→
   517→              {/* FAQ */}
   518→              <section aria-labelledby="faq-heading">
   519→                <h2 id="faq-heading" className="text-sm font-semibold text-foreground">
   520→                  {t("tool.faqTitle")}
   521→                </h2>
   522→                <Accordion type="single" collapsible className="mt-3">
   523→                  {config.faq.map((item, index) => (
   524→                    <AccordionItem key={index} value={`faq-${index}`}>
   525→                      <AccordionTrigger className="text-left text-sm">
   526→                        {item.q[locale]}
   527→                      </AccordionTrigger>
   528→                      <AccordionContent className="text-xs leading-relaxed text-muted-foreground">
   529→                        {item.a[locale]}
   530→                      </AccordionContent>
   531→                    </AccordionItem>
   532→                  ))}
   533→                </Accordion>
   534→              </section>
   535→            </div>
   536→
   537→            <aside className="flex flex-col gap-8">
   538→              {/* Features */}
   539→              <section aria-labelledby="features-heading">
   540→                <h2 id="features-heading" className="flex items-center gap-2 text-sm font-semibold text-foreground">
   541→                  <Sparkles className="size-4 text-primary" aria-hidden="true" />
   542→                  {t("tool.featuresTitle")}
   543→                </h2>
   544→                <ul className="mt-3 flex flex-col gap-2.5">
   545→                  {config.features.map((feature, index) => (
   546→                    <li
   547→                      key={index}
   548→                      className="flex gap-2 rounded-card border border-border bg-card p-3 text-xs leading-relaxed text-muted-foreground"
   549→                    >
   550→                      <span
   551→                        aria-hidden="true"
   552→                        className="mt-0.5 size-1.5 shrink-0 rounded-full bg-primary"
   553→                      />
   554→                      {feature[locale]}
   555→                    </li>
   556→                  ))}
   557→                </ul>
   558→              </section>
   559→
   560→              {/* Related */}
   561→              <section aria-labelledby="related-heading">
   562→                <h2 id="related-heading" className="text-sm font-semibold text-foreground">
   563→                  {t("tool.relatedTitle")}
   564→                </h2>
   565→                <ul className="mt-3 flex flex-col gap-1.5">
   566→                  {relatedTools.map((item) => {
   567→                    const RelatedIcon = item.icon;
   568→                    return (
   569→                      <li key={item.id}>
   570→                        <a
   571→                          href={item.enabled ? ROUTES.tool(item.id) : ROUTES.category(item.category)}
   572→                          className="flex min-h-11 items-center gap-2.5 rounded-btn px-2 text-sm outline-none transition-colors hover:bg-accent focus-visible:ring-[3px] focus-visible:ring-ring/50"
   573→                        >
   574→                          <RelatedIcon className="size-4 text-muted-foreground" aria-hidden="true" />
   575→                          <span className="truncate text-foreground">
   576→                            {item.name[locale]}
   577→                          </span>
   578→                          {!item.enabled ? (
   579→                            <Badge variant="outline" className="ml-auto rounded-chip text-[10px] text-muted-foreground">
   580→                              {t("tools.soonBadge")}
   581→                            </Badge>
   582→                          ) : null}
   583→                        </a>
   584→                      </li>
   585→                    );
   586→                  })}
   587→                </ul>
   588→              </section>
   589→            </aside>
   590→          </div>
   591→        </section>
   592→      ) : null}
   593→
   594→      {phase === "workspace" ? (
   595→        <section className="pt-6">
   596→          {/* Workspace toolbar (PRD §6.3 state 2) */}
   597→          <div className="flex flex-wrap items-center gap-2">
   598→            <Button
   599→              type="button"
   600→              variant="ghost"
   601→              size="sm"
   602→              onClick={() => setPhase("landing")}
   603→              className="rounded-btn"
   604→            >
   605→              <ArrowLeft className="size-4" aria-hidden="true" />
   606→              {t("tool.back")}
   607→            </Button>
   608→            <span className="ml-auto flex items-center gap-2">
   609→              <Button
   610→                type="button"
   611→                variant="outline"
   612→                size="sm"
   613→                className="rounded-btn"
   614→                onClick={() => setHelpOpen(true)}
   615→                aria-label={t("tool.helpLabel")}
   616→              >
   617→                <CircleHelp className="size-4" aria-hidden="true" />
   618→                <span className="hidden sm:inline">{t("tool.helpLabel")}</span>
   619→              </Button>
   620→              <Button
   621→                type="button"
   622→                variant="outline"
   623→                size="sm"
   624→                className="rounded-btn"
   625→                onClick={clearAll}
   626→              >
   627→                <Trash2 className="size-4" aria-hidden="true" />
   628→                {t("tool.clearAll")}
   629→              </Button>
   630→            </span>
   631→          </div>
   632→
   633→          <div
   634→            className={cn(
   635→              "mt-4 grid grid-cols-1 gap-6",
   636→              !viewerOnly && "lg:grid-cols-[minmax(0,1fr)_320px]"
   637→            )}
   638→          >
   639→            {/* Main column: the surface owns file presentation when one is
   640→                registered (PageGrid family); otherwise the classic
   641→                FileList + add-more DropZone (PRD §6.3). Viewer tools go
   642→                full-width — their surface is the entire product (§6.5). */}
   643→            <div className="flex flex-col gap-3">
   644→              {WorkspaceSurface && files.length > 0 ? (
   645→                <WorkspaceSurface
   646→                  files={files}
   647→                  values={options}
   648→                  onChange={handleOptionChange}
   649→                  setFileOrder={setFiles}
   650→                  setRunBlocked={handleRunBlocked}
   651→                  addFiles={addFiles}
   652→                  onChain={surfaceChain}
   653→                />
   654→              ) : (
   655→                <>
   656→                  <FileList
   657→                    items={files}
   658→                    onMove={moveFile}
   659→                    onRemove={removeFile}
   660→                  />
   661→                  <DropZone
   662→                    labelKey={addMoreLabelKey}
   663→                    multiple={config.maxFiles > 1}
   664→                    maxFiles={config.maxFiles}
   665→                    acceptKind={acceptKind}
   666→                    onAccept={addFiles}
   667→                    currentCount={files.length}
   668→                    className="border-dashed py-4"
   669→                  />
   670→                </>
   671→              )}
   672→            </div>
   673→
   674→            {/* Options: desktop side panel / mobile bottom sheet.
   675→                Height budget: sticky header (top-20) + sticky CTA bar at the
   676→                bottom — so the panel never slides under the CTA bar.
   677→                Viewer tools have no options panel — full-width surface. */}
   678→            {!viewerOnly ? (
   679→            <aside className="hidden lg:block">
   680→              <div className="sticky top-20 mb-20 flex max-h-[calc(100vh-11.5rem)] flex-col gap-5 overflow-y-auto scrollbar-thin rounded-card border border-border bg-card p-5">
   681→                <div>
   682→                  <h2 className="text-sm font-semibold text-foreground">
   683→                    {t("tool.optionsTitle")}
   684→                  </h2>
   685→                  {CustomPanel ? (
   686→                    <div className="mt-4">
   687→                      <CustomPanel
   688→                        files={files}
   689→                        values={options}
   690→                        onChange={handleOptionChange}
   691→                        setFileOrder={setFiles}
   692→                      />
   693→                    </div>
   694→                  ) : null}
   695→                  <OptionsPanel
   696→                    fields={config.options}
   697→                    values={options}
   698→                    onChange={handleOptionChange}
   699→                    className="mt-4"
   700→                  />
   701→                </div>
   702→                <p className="border-t border-border pt-3 text-[11px] text-muted-foreground">
   703→                  {files.length > 0 && files.length < config.minFiles
   704→                    ? t("tool.summaryMinFiles", { count: config.minFiles })
   705→                    : t("tool.summary", {
   706→                        count: files.length,
   707→                        size: formatBytes(totalSize),
   708→                      })}
   709→                </p>
   710→              </div>
   711→            </aside>
   712→            ) : null}
   713→          </div>
   714→
   715→          {/* Sticky CTA (PRD §6.3): count + estimated size + action.
   716→              Hidden for viewer tools — the surface owns its actions. */}
   717→          {!viewerOnly ? (
   718→          <div className="fixed inset-x-0 bottom-0 z-40 border-t border-border bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/85">
   719→            <div className="mx-auto flex max-w-6xl items-center gap-3 px-4 py-3">
   720→              <div className="min-w-0 flex-1">
   721→                <p
   722→                  className={cn(
   723→                    "flex items-center gap-1.5 truncate text-xs",
   724→                    runBlockedKey !== null
   725→                      ? "font-medium text-amber-600 dark:text-amber-400"
   726→                      : files.length > 0 && files.length < config.minFiles
   727→                        ? "font-medium text-primary"
   728→                        : "text-muted-foreground"
   729→                  )}
   730→                >
   731→                  {runBlockedKey !== null ? (
   732→                    <TriangleAlert
   733→                      className="size-3.5 shrink-0"
   734→                      aria-hidden="true"
   735→                    />
   736→                  ) : null}
   737→                  <span className="truncate">
   738→                    {runBlockedKey !== null
   739→                      ? t(runBlockedKey)
   740→                      : files.length > 0 && files.length < config.minFiles
   741→                        ? t("tool.summaryMinFiles", { count: config.minFiles })
   742→                        : t("tool.summary", {
   743→                            count: files.length,
   744→                            size: formatBytes(totalSize),
   745→                          })}
   746→                  </span>
   747→                </p>
   748→                {chainFrom ? (
   749→                  <p className="truncate text-[11px] text-primary">
   750→                    {t("tool.chainFrom", { tool: chainFrom.name[locale] })}
   751→                  </p>
   752→                ) : null}
   753→              </div>
   754→              <Button
   755→                type="button"
   756→                variant="outline"
   757→                size="touch"
   758→                className="rounded-btn lg:hidden"
   759→                onClick={() => setOptionsOpen(true)}
   760→              >
   761→                {t("tool.optionsButton")}
   762→              </Button>
   763→              <Button
   764→                type="button"
   765→                size="touch"
   766→                className="rounded-btn font-semibold"
   767→                onClick={startRun}
   768→                disabled={runDisabled}
   769→              >
   770→                {ctaLabel}
   771→                <ArrowRight className="size-4" aria-hidden="true" />
   772→              </Button>
   773→            </div>
   774→          </div>
   775→          ) : null}
   776→
   777→          {!viewerOnly ? (
   778→          <BottomSheet
   779→            open={optionsOpen}
   780→            onOpenChange={setOptionsOpen}
   781→            title={t("tool.optionsTitle")}
   782→            description={tool.description[locale]}
   783→            footer={
   784→              <Button
   785→                type="button"
   786→                size="touch"
   787→                className="w-full rounded-btn font-semibold"
   788→                onClick={() => {
   789→                  setOptionsOpen(false);
   790→                  startRun();
   791→                }}
   792→                disabled={runDisabled}
   793→              >
   794→                {ctaLabel}
   795→                <ArrowRight className="size-4" aria-hidden="true" />
   796→              </Button>
   797→            }
   798→          >
   799→            {CustomPanel ? (
   800→              <div className="mb-4">
   801→                <CustomPanel
   802→                  files={files}
   803→                  values={options}
   804→                  onChange={handleOptionChange}
   805→                  setFileOrder={setFiles}
   806→                />
   807→              </div>
   808→            ) : null}
   809→            <OptionsPanel
   810→              fields={config.options}
   811→              values={options}
   812→              onChange={handleOptionChange}
   813→            />
   814→          </BottomSheet>
   815→          ) : null}
   816→        </section>
   817→      ) : null}
   818→
   819→      {phase === "processing" ? (
   820→        <ProcessingView
   821→          toolName={tool.name[locale]}
   822→          progress={progress}
   823→          onCancel={cancelRun}
   824→        />
   825→      ) : null}
   826→
   827→      {phase === "result" ? (
   828→        <ResultView
   829→          tool={tool}
   830→          outputs={outputs}
   831→          inputBytes={inputBytes}
   832→          runSummary={runSummary}
   833→          chainFromName={chainFrom ? chainFrom.name[locale] : undefined}
   834→          chainTargets={chainTargets}
   835→          onChain={handleChain}
   836→          onRemoveFromDevice={removeFromDevice}
   837→          onStartOver={startOver}
   838→        />
   839→      ) : null}
   840→
   841→      {/* Help dialog */}
   842→      <Modal
   843→        open={helpOpen}
   844→        onOpenChange={setHelpOpen}
   845→        title={t("tool.helpTitle")}
   846→        footer={
   847→          <Button
   848→            type="button"
   849→            variant="default"
   850→            size="touch"
   851→            className="w-full rounded-btn sm:w-auto"
   852→            onClick={() => setHelpOpen(false)}
   853→          >
   854→            {t("pro.action")}
   855→          </Button>
   856→        }
   857→      >
   858→        <div className="flex flex-col gap-3 text-sm text-muted-foreground">
   859→          <p className="flex gap-2">
   860→            <FilePlus2 className="mt-0.5 size-4 shrink-0 text-primary" aria-hidden="true" />
   861→            {t("tool.how2")}
   862→          </p>
   863→          <p>{t("tool.helpPrivacy")}</p>
   864→        </div>
   865→      </Modal>
   866→    </div>
   867→  );
   868→}

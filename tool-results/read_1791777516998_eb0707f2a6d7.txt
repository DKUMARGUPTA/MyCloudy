    79→    79→/**
    80→    80→ * ToolShell (PRD §6.3): the universal 4-state tool template.
    81→    81→ * Landing (upload + SEO copy) → Workspace (preview + options) →
    82→    82→ * Processing (real worker progress) → Result (download/share/chain).
    83→    83→ * Every enabled tool renders from its registry entry + config — this
    84→    84→ * component is the ONLY tool page in the app.
    85→    85→ */
    86→    86→
    87→    87→type ShellPhase = "landing" | "workspace" | "processing" | "result";
    88→    88→
    89→    89→let fileSeq = 0;
    90→    90→function nextFileId(): string {
    91→    91→  fileSeq += 1;
    92→    92→  return `wf-${Date.now().toString(36)}-${fileSeq}`;
    93→    93→}
    94→    94→
    95→    95→export function ToolShell({ tool, config }: { tool: ToolDef; config: ToolConfig }) {
    96→    96→  const { t, locale } = useI18n();
    97→    97→  const { toast } = useToast();
    98→    98→  const category = getCategory(tool.category);
    99→    99→  // Viewer tools (PRD §6.5) never process anything: no run CTA, no options
   100→   100→  // aside, no processing/result phases — the workspace surface IS the tool.
   101→   101→  const viewerOnly = config.viewerOnly === true;
   102→   102→
   103→   103→  // One-time chain handoff peek — set by a previous tool's result view or
   104→   104→  // the home Smart Drop Zone right before navigating here. Consumed via
   105→   105→  // the lazy initializer (pure read); cleared in the effect below.
   106→   106→  const [chainIn] = useState<{
   107→   107→    fromToolId: string;
   108→   108→    items: WorkspaceFile[];
   109→   109→  } | null>(() => {
   110→   110→    const handoff = peekChainHandoff();
   111→   111→    if (!handoff || handoff.items.length === 0) return null;
   112→   112→    return {
   113→   113→      fromToolId: handoff.fromToolId,
   114→   114→      items: handoff.items.map((item) => ({
   115→   115→        id: item.id,
   116→   116→        file: item.file,
   117→   117→      })),
   118→   118→    };
   119→   119→  });
   120→   120→
   121→   121→  const [phase, setPhase] = useState<ShellPhase>(() => (chainIn ? "workspace" : "landing"));
   122→   122→  const [files, setFiles] = useState<WorkspaceFile[]>(() => chainIn?.items ?? []);
   123→   123→  const [options, setOptions] = useState<OptionValues>(() =>
   124→   124→    defaultOptionValues(config.options)
   125→   125→  );
   126→   126→  const [progress, setProgress] = useState<RunProgress>({
   127→   127→    pct: 0,
   128→   128→    stage: "reading",
   129→   129→  });
   130→   130→  const [outputs, setOutputs] = useState<ToolOutput[]>([]);
   131→   131→  const [inputBytes, setInputBytes] = useState(0);
   132→   132→  const [runSummary, setRunSummary] = useState<RunDetail | null>(null);
   133→   133→  const [optionsOpen, setOptionsOpen] = useState(false);
   134→   134→  const [helpOpen, setHelpOpen] = useState(false);
   135→   135→
   136→   136→  // Chain handoff consumed once at mount (lazy initializer = pure read);
   137→   137→  // clearing the module store is an external-system sync in this effect.
   138→   138→  const chainFrom: ToolDef | null = chainIn
   139→   139→    ? (getTool(chainIn.fromToolId) ?? null)
   140→   140→    : null;
   141→   141→  useEffect(() => {
   142→   142→    if (chainIn) clearChainHandoff();
   143→   143→  }, [chainIn]);
   144→   144→
   145→   145→  // Input kind drives the DropZone sniffing + accept filter + labels.
   146→   146→  const acceptKind =
   147→   147→    tool.accepts === "image" ? "image" : tool.accepts === "pdf" ? "pdf" : "any";
   148→   148→  const selectLabelKey =
   149→   149→    tool.accepts === "image" ? "tool.selectImageFiles" : "tool.selectPdfFiles";
   150→   150→  const addMoreLabelKey =
   151→   151→    tool.accepts === "image" ? "tool.addMoreImages" : "tool.addMoreFiles";
   152→   152→  const recentKind = tool.accepts === "image" ? "image" : "pdf";
   153→   153→
   154→   154→  // Chain/SmartDrop files get the same treatment as dropped files (PRD
   155→   155→  // §6.1): OPFS working copy + a metadata-only recent entry with thumbnail.
   156→   156→  // External-system sync only — no React state is touched here.
   157→   157→  useEffect(() => {
   158→   158→    if (!chainIn) return;
   159→   159→    for (const item of chainIn.items) {
   160→   160→      void writeWorkingFile(item.id, item.file).catch(() => undefined);
   161→   161→      void makeFileThumbnail(item.file, 160)
   162→   162→        .then((blob) =>
   163→   163→          addRecentFile({
   164→   164→            id: item.id,
   165→   165→            kind: recentKind as "pdf" | "image",
   166→   166→            label: item.file.name,
   167→   167→            size: item.file.size,
   168→   168→            toolId: tool.id,
   169→   169→            thumbnail: blob ?? undefined,
   170→   170→          })
   171→   171→        )
   172→   172→        .catch(() => undefined);
   173→   173→    }
   174→   174→  }, [chainIn, recentKind, tool.id]);
   175→   175→
   176→   176→  const runHandleRef = useRef<RunHandle | null>(null);
   177→   177→
   178→   178→  // Optional custom options panel (PRD §2.6 Panel.tsx pattern). Direct
   179→   179→  // map lookup — a stable module-level reference, never created in render.
   180→   180→  const CustomPanel = TOOL_PANELS[tool.id];
   181→   181→
   182→   182→  // Optional workspace surface (PageGrid family): rendered in the main
   183→   183→  // column instead of the FileList; can gate the CTA via runBlocked.
   184→   184→  const WorkspaceSurface = TOOL_SURFACES[tool.id];
   185→   185→
   186→   186→  // Surface-driven CTA gating (PRD §8.1 safety: "sab pages delete ho to
   187→   187→  // Apply disabled" / extract needs a selection). Event-context only —
   188→   188→  // surfaces call this from their action handlers, never during render.
   189→   189→  const [runBlockedKey, setRunBlockedKey] = useState<string | null>(null);
   190→   190→  const handleRunBlocked = useCallback((reasonKey: string | null) => {
   191→   191→    setRunBlockedKey(reasonKey);
   192→   192→  }, []);
   193→   193→  const runDisabled =
   194→   194→    files.length < config.minFiles || runBlockedKey !== null;
   195→   195→
   196→   196→  const handleOptionChange = useCallback(
   197→   197→    (id: string, value: string | number | boolean) => {
   198→   198→      setOptions((prev) => ({ ...prev, [id]: value }));
   199→   199→    },
   200→   200→    []
   201→   201→  );
   202→   202→
   203→   203→  const relatedTools = useMemo(
   204→   204→    () =>
   205→   205→      toolsByCategory(tool.category)
   206→   206→        .filter((item) => item.id !== tool.id)
   207→   207→        .slice(0, 6),
   208→   208→    [tool.category, tool.id]
   209→   209→  );
   210→   210→
   211→   211→  const chainTargets = useMemo(() => {
   212→   212→    // PRD §6.4 "Ab aage kya?" — natural next steps. Image outputs chain
   213→   213→    // to image-accepting tools (the PDF default set would be a type lie);
   214→   214→    // configs can override with their own curated list.
   215→   215→    const suggestionIds =
   216→   216→      config.chainToolIds ?? [
   217→   217→        "compress-pdf",
   218→   218→        "protect-pdf",
   219→   219→        "sign-pdf",
   220→   220→        "split-pdf",
   221→   221→        "edit-pdf",
   222→   222→      ];
   223→   223→    const resolved = suggestionIds
   224→   224→      .map((id) => getTool(id))
   225→   225→      .filter((item): item is ToolDef => item !== undefined && item.id !== tool.id);
   226→   226→    return resolved.slice(0, 5);
   227→   227→  }, [config.chainToolIds, tool.id]);
   228→   228→
   229→   229→  // Working copies are cleared when the tool session ends (PRD §2.4).
   230→   230→  useEffect(
   231→   231→    () => () => {
   232→   232→      void clearWorkingFiles().catch(() => undefined);
   233→   233→    },
   234→   234→    []
   235→   235→  );
   236→   236→
   237→   237→  const addFiles = useCallback(
   238→   238→    (incoming: File[]) => {
   239→   239→      if (incoming.length === 0) return;
   240→   240→      const room = config.maxFiles - files.length;
   241→   241→      if (room <= 0) {
   242→   242→        toast({
   243→   243→          title: t("dropzone.errTooManyTitle"),
   244→   244→          description: t("dropzone.errTooManyDesc", {
   245→   245→            count: config.maxFiles,
   246→   246→          }),
   247→   247→          variant: "destructive",
   248→   248→        });
   249→   249→        return;
   250→   250→      }
   251→   251→      const accepted = incoming.slice(0, room);
   252→   252→      const items = accepted.map((file) => ({ id: nextFileId(), file }));
   253→   253→
   254→   254→      // Side effects run ONCE here — never inside the setFiles updater
   255→   255→      // (React StrictMode double-invokes updaters; they must stay pure).
   256→   256→      for (const item of items) {
   257→   257→        void writeWorkingFile(item.id, item.file).catch(() => undefined);
   258→   258→        // Recent entry: metadata + thumbnail only, never the bytes.
   259→   259→        void makeFileThumbnail(item.file, 160)
   260→   260→          .then(async (blob) => {
   261→   261→            await addRecentFile({
   262→   262→              id: item.id,
   263→   263→              kind: recentKind as "pdf" | "image",
   264→   264→              label: item.file.name,
   265→   265→              size: item.file.size,
   266→   266→              toolId: tool.id,
   267→   267→              thumbnail: blob ?? undefined,
   268→   268→            });
   269→   269→          })
   270→   270→          .catch(() => undefined);
   271→   271→      }
   272→   272→
   273→   273→      setFiles((prev) => [...prev, ...items]);
   274→   274→      setPhase("workspace");
   275→   275→    },
   276→   276→    [config.maxFiles, files.length, recentKind, t, toast, tool.id]
   277→   277→  );
   278→   278→
   279→   279→  const removeFile = useCallback((id: string) => {
   280→   280→    setFiles((prev) => prev.filter((item) => item.id !== id));
   281→   281→  }, []);
   282→   282→
   283→   283→  const moveFile = useCallback((from: number, to: number) => {
   284→   284→    setFiles((prev) => {
   285→   285→      if (to < 0 || to >= prev.length) return prev;
   286→   286→      const next = [...prev];
   287→   287→      const [moved] = next.splice(from, 1);
   288→   288→      next.splice(to, 0, moved);
   289→   289→      return next;
   290→   290→    });
   291→   291→  }, []);
   292→   292→
   293→   293→  const clearAll = useCallback(() => {
   294→   294→    setFiles([]);
   295→   295→    void clearWorkingFiles().catch(() => undefined);
   296→   296→    setPhase("landing");
   297→   297→  }, []);
   298→   298→
   299→   299→  // Stable function ref from the config (module constant) — keeps the
   300→   300→  // React Compiler's dependency inference happy.
   301→   301→  const collectAssets = config.collectAssets;
   302→   302→
   303→   303→  const startRun = useCallback(() => {
   304→   304→    if (files.length < config.minFiles || runBlockedKey !== null) return;
   305→   305→    trackToolUse(tool.id);
   306→   306→    setInputBytes(files.reduce((sum, item) => sum + item.file.size, 0));
   307→   307→    setProgress({ pct: 0, stage: "reading" });
   308→   308→    setPhase("processing");
   309→   309→    // Extra in-memory assets (watermark logo etc.) ride the worker message.
   310→   310→    const assets = collectAssets?.() ?? undefined;
   311→   311→    runHandleRef.current = toolRunner.run(
   312→   312→      tool.id,
   313→   313→      options,
   314→   314→      files.map((item) => item.file),
   315→   315→      setProgress,
   316→   316→      files.map((item) => item.id),
   317→   317→      assets
   318→   318→    );
   319→   319→    runHandleRef.current.promise
   320→   320→      .then((result) => {
   321→   321→        runHandleRef.current = null;
   322→   322→        // Working copies of the INPUTS die with the run (PRD §2.4);
   323→   323→        // results live in memory + can chain into the next tool.
   324→   324→        setOutputs(result.outputs);
   325→   325→        setRunSummary(result.detail ?? null);
   326→   326→        setPhase("result");
   327→   327→        void clearWorkingFiles().catch(() => undefined);
   328→   328→      })
   329→   329→      .catch((error: unknown) => {
   330→   330→        runHandleRef.current = null;
   331→   331→        if (error instanceof RunCancelledError) {
   332→   332→          setPhase("workspace");
   333→   333→          toast({ title: t("tool.cancelledToast") });
   334→   334→          return;
   335→   335→        }
   336→   336→        setPhase("workspace");
   337→   337→        const code =
   338→   338→          error instanceof ToolRunError ? error.code : "tool.errorGeneric";
   339→   339→        toast({
   340→   340→          title: t("tool.errorTitle"),
   341→   341→          description: t(code, {
   342→   342→            detail: error instanceof ToolRunError ? error.detail ?? "" : "",
   343→   343→          }),
   344→   344→          variant: "destructive",
   345→   345→        });
   346→   346→      });
   347→   347→  }, [collectAssets, config.minFiles, files, options, runBlockedKey, t, toast, tool.id]);
   348→   348→
   349→   349→  const cancelRun = useCallback(() => {
   350→   350→    runHandleRef.current?.cancel();
   351→   351→  }, []);
   352→   352→
   353→   353→  const handleChain = useCallback(
   354→   354→    (target: ToolDef, resultFiles: File[]) => {
   355→   355→      if (!target.enabled) {
   356→   356→        toast({
   357→   357→          title: t("tools.comingToastTitle", { tool: target.name[locale] }),
   358→   358→          description: t("tools.comingToastDesc", {
   359→   359→            phase: target.phase,
   360→   360→          }),
   361→   361→        });
   362→   362→        return;
   363→   363→      }
   364→   364→      setChainHandoff({
   365→   365→        fromToolId: tool.id,
   366→   366→        items: resultFiles.map((file) => ({ id: nextFileId(), file })),
   367→   367→      });
   368→   368→      navigateToHash(ROUTES.tool(target.id));
   369→   369→    },
   370→   370→    [locale, t, toast, tool.id]
   371→   371→  );
   372→   372→
   373→   373→  // Viewer surfaces chain the current file straight into another tool
   374→   374→  // (they have no result phase of their own). Same enabled/coming-soon
   375→   375→  // behavior as result-view chains.
   376→   376→  const surfaceChain = useCallback(
   377→   377→    (targetId: string) => {
   378→   378→      const target = getTool(targetId);
   379→   379→      if (!target) return;
   380→   380→      handleChain(target, files.map((item) => item.file));
   381→   381→    },
   382→   382→    [files, handleChain]
   383→   383→  );
   384→   384→
   385→   385→  const removeFromDevice = useCallback(() => {
   386→   386→    void clearWorkingFiles().catch(() => undefined);
   387→   387→    // Remove this run's recent entries by their file metadata (label+size).
   388→   388→    void removeRecentFilesByMeta(
   389→   389→      files.map((item) => ({ label: item.file.name, size: item.file.size }))
   390→   390→    ).catch(() => undefined);
   391→   391→    setOutputs([]);
   392→   392→    setFiles([]);
   393→   393→    setPhase("landing");
   394→   394→    toast({ title: t("result.removedToast") });
   395→   395→  }, [files, t, toast]);
   396→   396→
   397→   397→  const totalSize = files.reduce((sum, item) => sum + item.file.size, 0);
   398→   398→  const ctaLabel = `${tool.name[locale]} ${t("tool.ctaArrow")}`;
   399→   399→  const Icon = tool.icon;
   400→   400→
   401→   401→  return (
   402→   402→    <div className="mx-auto w-full max-w-6xl px-4 pb-28 sm:pb-24">
   403→   403→      {/* Breadcrumb (PRD §6.3): Home > Category > Tool */}
   404→   404→      <nav aria-label={t("a11y.breadcrumb")} className="pt-4">
   405→   405→        <ol className="flex flex-wrap items-center gap-1.5 text-xs text-muted-foreground">
   406→   406→          <li>
   407→   407→            <a
   408→   408→              href={ROUTES.home}
   409→   409→              className="inline-flex items-center gap-1 rounded-btn outline-none hover:text-foreground focus-visible:ring-[3px] focus-visible:ring-ring/50"
   410→   410→            >
   411→   411→              <Home className="size-3" aria-hidden="true" />
   412→   412→              {t("nav.home")}
   413→   413→            </a>
   414→   414→          </li>
   415→   415→          <li aria-hidden="true">/</li>
   416→   416→          <li>
   417→   417→            <a
   418→   418→              href={ROUTES.category(tool.category)}
   419→   419→              className="rounded-btn outline-none hover:text-foreground focus-visible:ring-[3px] focus-visible:ring-ring/50"
   420→   420→            >
   421→   421→              {category.name[locale]}
   422→   422→            </a>
   423→   423→          </li>
   424→   424→          <li aria-hidden="true">/</li>
   425→   425→          <li aria-current="page" className="font-medium text-foreground">
   426→   426→            {tool.name[locale]}
   427→   427→          </li>
   428→   428→        </ol>
   429→   429→      </nav>
   430→   430→
   431→   431→      {phase === "landing" ? (
   432→   432→        <section className="flex flex-col gap-8 pt-6">
   433→   433→          <header className="flex flex-col gap-3">
   434→   434→            <div className="flex items-center gap-3">
   435→   435→              <span
   436→   436→                className={cn(
   437→   437→                  "flex size-11 items-center justify-center rounded-btn",
   438→   438→                  "bg-cat-organize/10 text-cat-organize"
   439→   439→                )}
   440→   440→                aria-hidden="true"
   441→   441→              >
   442→   442→                <Icon className="size-5" />
   443→   443→              </span>
   444→   444→              <h1 className="text-2xl font-bold tracking-tight text-foreground sm:text-3xl">
   445→   445→                {tool.name[locale]}
   446→   446→              </h1>
   447→   447→            </div>
   448→   448→            <p className="max-w-2xl text-sm text-muted-foreground sm:text-base">
   449→   449→              {tool.description[locale]}
   450→   450→            </p>
   451→   451→            <div className="flex flex-wrap gap-2">
   452→   452→              <Badge variant={tool.mode === "D" ? "private" : "server"} className="rounded-chip">
   453→   453→                <Lock className="size-3" aria-hidden="true" />
   454→   454→                {tool.mode === "D"
   455→   455→                  ? t("tools.privateBadge")
   456→   456→                  : t("tools.serverBadge")}
   457→   457→              </Badge>
   458→   458→              <Badge variant="outline" className="rounded-chip">
   459→   459→                {t("tool.freeBadge")}
   460→   460→              </Badge>
   461→   461→            </div>
   462→   462→          </header>
   463→   463→
   464→   464→          <DropZone
   465→   465→            labelKey={selectLabelKey}
   466→   466→            multiple={config.maxFiles > 1}
   467→   467→            maxFiles={config.maxFiles}
   468→   468→            acceptKind={acceptKind}
